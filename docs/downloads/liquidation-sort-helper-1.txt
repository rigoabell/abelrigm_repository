(function () {
  'use strict';

  /* ─── Constants ─────────────────────────────────────────────────────────── */

  const SCRIPT_VERSION = '2.2.7';
  const PAGE_BRIDGE = 'lsh-page-bridge-v1';
  let pageBridgeFailed = true;
  const STORAGE_KEY = 'liquidation-sort-helper-v2';
  const TRACE_MAX = 200;
  const DEFAULT_MAX_TOTE_UNITS = 399;
  const MAX_TOTE_UNITS = DEFAULT_MAX_TOTE_UNITS;
  const COMPLETED_LPN_MAX = 5000;
  const ERROR_LOG_MAX = 500;

  /* Timing tuned for speed while keeping React submit + verify behavior unchanged. */
  const TIMING = {
    maxWaitMs: 2000,
    preSubmitMs: 240,
    postSubmitMs: 240,
    typeDurationMs: 280,
    settleMs: 50,
    postUpdateMs: 120,
    pollMinMs: 24,
    pollMaxMs: 48,
    stuckRunningMs: 15000,
    stuckStepMs: 12000,
    initDelayMs: 300,
    bridgeTimeoutMs: 500,
  };
  const MIN_CHAR_DELAY_MS = 15;

  const SORT_TYPE = 'WDSorting';

  const SEL = {
    sortType: '#sort-type-textfield',
    sourceContainer: '#source-container-textfield',
    barcode: '#barcode-scan-textfield',
    sortCode: '#sort-code-id',
    scanTote: '#scan-tote-textfield',
    updateLink: '#change-sort-type',
    sortTypeDisplay: '#sort-type',
    exceptionBtn: '#exception-btn',
    exceptionInfo: '#exception-info',
    exceptionContainer: '#exception-container',
    exceptionPage: '#page-exception-component',
    updateBtn: 'button[data-testid="update-button"], #update-button, button.update-button, input[value="Update"]',
  };

  const ROW_STATUS = {
    PENDING: 'pending',
    ACTIVE: 'active',
    COMPLETE: 'complete',
    ERROR: 'error',
  };

  /* ─── Trace / Logging ───────────────────────────────────────────────────── */

  const traceBuffer = [];

  function trace(event, detail) {
    const entry = {
      t: Date.now(),
      iso: new Date().toISOString(),
      event,
      ...(detail !== undefined ? { detail } : {}),
    };
    traceBuffer.push(entry);
    if (traceBuffer.length > TRACE_MAX) traceBuffer.shift();
    console.log(`[LSH] ${event}`, detail !== undefined ? detail : '');
  }

  /* ─── Persistence ───────────────────────────────────────────────────────── */

  function defaultState() {
    return {
      queue: [],
      queueIndex: 0,
      rowStatuses: [],
      results: [],
      completedLpns: {},
      errorLog: [],
      toteCounts: {},
      toteDisplayIds: {},
      sortSlots: {},
      toteMaxLimits: {},
      sessionArmed: false,
      running: false,
      exceptionFlowActive: false,
      pageResetInProgress: false,
      jobStartupInProgress: false,
      sourceContainerDone: false,
      barcodeDone: false,
      postBarcodeHandled: false,
      pendingLpn: null,
      lastSubmittedLpn: null,
      lastSubmittedContainer: null,
      lastSourceSubmitAt: 0,
      currentContainer: null,
      autoCloseErrors: true,
      stopClickUpdateOnStop: true,
      lastRunningAt: 0,
      jobStartedAt: 0,
      lastUpdateReason: null,
      resumeRequired: false,
      stepProgressAt: 0,
      currentStep: 'idle',
      recoveryUpdateUsed: false,
      sessionStartedAt: 0,
      sessionJobsCompleted: 0,
      sessionEtaDeadline: 0,
      toteMappingPrompt: null,
      toteFullPrompt: null,
      version: SCRIPT_VERSION,
    };
  }

  let stepProgressAt = Date.now();
  let currentStep = 'idle';

  let state = loadState();
  let persistTimer = null;

  function loadState() {
    try {
      const raw = GM_getValue(STORAGE_KEY, null);
      if (!raw) return defaultState();
      const parsed = typeof raw === 'string' ? JSON.parse(raw) : raw;
      const merged = { ...defaultState(), ...parsed, running: false, pageResetInProgress: false, jobStartupInProgress: false };
      normalizeSortSlots(merged);
      migrateToteCounts(merged);
      migrateCompletedLpns(merged);
      migrateErrorLog(merged);
      return merged;
    } catch (e) {
      trace('persist.loadError', { error: String(e) });
      return defaultState();
    }
  }

  function buildPersistPayload() {
    return { ...state, running: false, pageResetInProgress: false, jobStartupInProgress: false };
  }

  function persistStateNow() {
    if (persistTimer) {
      clearTimeout(persistTimer);
      persistTimer = null;
    }
    try {
      GM_setValue(STORAGE_KEY, JSON.stringify(buildPersistPayload()));
    } catch (e) {
      trace('persist.saveError', { error: String(e) });
    }
  }

  function persistState() {
    if (persistTimer) clearTimeout(persistTimer);
    persistTimer = setTimeout(() => {
      persistTimer = null;
      persistStateNow();
    }, 100);
  }

  function syncRowStatuses() {
    while (state.rowStatuses.length < state.queue.length) {
      state.rowStatuses.push(ROW_STATUS.PENDING);
    }
    if (state.rowStatuses.length > state.queue.length) {
      state.rowStatuses = state.rowStatuses.slice(0, state.queue.length);
    }
  }

  function getRowLpn(row) {
    return (row?.lpn || row?.barcode || '').trim();
  }

  function getRowContainer(row) {
    return (row?.container || '').trim();
  }

  function normalizeLpnKey(lpn) {
    return (lpn || '').trim().toUpperCase();
  }

  function normalizeContainerKey(container) {
    return (container || '').trim().toUpperCase();
  }

  function makePairKey(container, lpn) {
    const c = normalizeContainerKey(container);
    const l = normalizeLpnKey(lpn);
    if (!c || !l) return '';
    return `${c}|${l}`;
  }

  function isRowCompleted(row) {
    const key = makePairKey(getRowContainer(row), getRowLpn(row));
    return !!(key && state.completedLpns && state.completedLpns[key]);
  }

  function isRowErrored(row) {
    const key = makePairKey(getRowContainer(row), getRowLpn(row));
    if (!key || !Array.isArray(state.errorLog)) return false;
    return state.errorLog.some((e) => makePairKey(e.container, e.lpn) === key);
  }

  function getRowSkipKind(row, index) {
    if (isRowCompleted(row)) return 'complete';
    if (isRowErrored(row)) return 'error';
    if (index >= 0 && state.rowStatuses[index] === ROW_STATUS.COMPLETE) return 'complete';
    if (index >= 0 && state.rowStatuses[index] === ROW_STATUS.ERROR) return 'error';
    return null;
  }

  function shouldSkipRow(row, index = state.queueIndex) {
    return getRowSkipKind(row, index) !== null;
  }

  function trimCompletedLpns(target = state) {
    const map = target.completedLpns || {};
    const keys = Object.keys(map);
    if (keys.length <= COMPLETED_LPN_MAX) return;
    keys
      .sort((a, b) => (map[a].at || '').localeCompare(map[b].at || ''))
      .slice(0, keys.length - COMPLETED_LPN_MAX)
      .forEach((key) => delete map[key]);
  }

  function migrateCompletedLpns(target) {
    if (!target.completedLpns || typeof target.completedLpns !== 'object') {
      target.completedLpns = {};
    }
    const legacy = { ...target.completedLpns };
    target.completedLpns = {};
    for (const [key, val] of Object.entries(legacy)) {
      if (key.includes('|')) {
        target.completedLpns[key] = val;
        continue;
      }
      const lpn = normalizeLpnKey(key);
      const container = normalizeContainerKey(val?.container || '');
      const pairKey = container ? makePairKey(container, lpn) : '';
      if (pairKey) target.completedLpns[pairKey] = val;
    }
    for (const entry of target.results || []) {
      if (entry.status !== 'complete') continue;
      const row = entry.row || {};
      const pairKey = makePairKey(row.container || entry.container, entry.lpn || getRowLpn(row));
      if (!pairKey || target.completedLpns[pairKey]) continue;
      target.completedLpns[pairKey] = {
        at: entry.at || new Date().toISOString(),
        container: row.container || entry.container || '',
        lpn: getRowLpn(row) || entry.lpn || '',
        sortCode: entry.sortCode || '',
        toteId: entry.toteId || '',
      };
    }
    trimCompletedLpns(target);
  }

  function recordCompletedRow(row, meta) {
    const key = makePairKey(
      getRowContainer(row) || meta.container,
      getRowLpn(row) || meta.lpn
    );
    if (!key) return;
    if (!state.completedLpns) state.completedLpns = {};
    state.completedLpns[key] = {
      at: new Date().toISOString(),
      container: getRowContainer(row) || meta.container || '',
      lpn: getRowLpn(row) || meta.lpn || '',
      sortCode: meta.sortCode || '',
      toteId: meta.toteId || '',
    };
    trimCompletedLpns();
  }

  function applyKnownRowsToQueue() {
    syncRowStatuses();
    let markedComplete = 0;
    let markedError = 0;
    state.queue.forEach((row, i) => {
      if (isRowCompleted(row)) {
        if (state.rowStatuses[i] !== ROW_STATUS.COMPLETE) markedComplete += 1;
        state.rowStatuses[i] = ROW_STATUS.COMPLETE;
      } else if (isRowErrored(row)) {
        if (state.rowStatuses[i] !== ROW_STATUS.ERROR) markedError += 1;
        state.rowStatuses[i] = ROW_STATUS.ERROR;
      }
    });
    state.queueIndex = findFirstPendingIndex();
    trace('pair.historyApplied', {
      queueLength: state.queue.length,
      markedComplete,
      markedError,
      nextIndex: state.queueIndex,
      knownComplete: Object.keys(state.completedLpns || {}).length,
      knownError: (state.errorLog || []).length,
    });
    return markedComplete + markedError;
  }

  function applyCompletedLpnsToQueue() {
    return applyKnownRowsToQueue();
  }

  function findFirstPendingIndex() {
    syncRowStatuses();
    for (let i = 0; i < state.queue.length; i++) {
      const row = state.queue[i];
      const kind = getRowSkipKind(row, i);
      if (kind === 'complete') {
        state.rowStatuses[i] = ROW_STATUS.COMPLETE;
        continue;
      }
      if (kind === 'error') {
        state.rowStatuses[i] = ROW_STATUS.ERROR;
        continue;
      }
      return i;
    }
    return state.queue.length;
  }

  function skipKnownRows(row) {
    if (!shouldSkipRow(row, state.queueIndex)) return false;
    let skipped = 0;
    let skippedComplete = 0;
    let skippedError = 0;
    while (state.queueIndex < state.queue.length) {
      const current = getCurrentRow();
      const kind = getRowSkipKind(current, state.queueIndex);
      if (!kind) break;
      if (kind === 'error') {
        markRowStatus(state.queueIndex, ROW_STATUS.ERROR);
        skippedError += 1;
      } else {
        markRowStatus(state.queueIndex, ROW_STATUS.COMPLETE);
        skippedComplete += 1;
      }
      skipped += 1;
      state.queueIndex += 1;
    }
    if (state.queueIndex >= state.queue.length) {
      trace('pair.skipKnown', { skipped, skippedComplete, skippedError, queueIndex: state.queueIndex });
      stopRunning('queue complete (known pairs skipped)');
      return true;
    }
    if (skipped) {
      resetJobFlags();
      persistState();
      renderAll();
      trace('pair.skipKnown', { skipped, skippedComplete, skippedError, queueIndex: state.queueIndex });
    }
    return skipped > 0;
  }

  function migrateErrorLog(target) {
    if (!Array.isArray(target.errorLog)) target.errorLog = [];
    for (const entry of target.results || []) {
      if (entry.status !== 'error') continue;
      const row = entry.row || {};
      appendErrorLogEntry(
        {
          container: row.container || '',
          lpn: getRowLpn(row) || entry.lpn || '',
          sortCode: row.sortCode || '',
          message: entry.message || 'Unknown error',
          reason: entry.reason || '',
          at: entry.at,
        },
        target
      );
    }
    trimErrorLog(target);
  }

  function appendErrorLogEntry(entry, target = state) {
    if (!Array.isArray(target.errorLog)) target.errorLog = [];
    const normalized = {
      container: (entry.container || '').trim(),
      lpn: (entry.lpn || '').trim(),
      sortCode: (entry.sortCode || '').trim(),
      message: (entry.message || 'Unknown error').trim(),
      reason: (entry.reason || '').trim(),
      at: entry.at || new Date().toISOString(),
    };
    if (!normalized.lpn && !normalized.container && !normalized.message) return;
    const dupe = target.errorLog.find(
      (e) =>
        e.lpn === normalized.lpn &&
        e.container === normalized.container &&
        e.message === normalized.message &&
        e.at === normalized.at
    );
    if (dupe) return;
    target.errorLog.unshift(normalized);
    trimErrorLog(target);
  }

  function trimErrorLog(target = state) {
    if (!Array.isArray(target.errorLog)) return;
    if (target.errorLog.length > ERROR_LOG_MAX) {
      target.errorLog.length = ERROR_LOG_MAX;
    }
  }

  function recordError(row, message, meta = {}) {
    appendErrorLogEntry({
      container: row?.container || meta.container || '',
      lpn: getRowLpn(row) || meta.lpn || '',
      sortCode: row?.sortCode || meta.sortCode || '',
      message: message || 'Unknown error',
      reason: meta.reason || '',
    });
    trace('errorLog.record', {
      container: row?.container || meta.container,
      lpn: getRowLpn(row) || meta.lpn,
      message,
    });
  }

  function clearErrorLog() {
    state.errorLog = [];
    persistState();
    renderErrorLogModal();
    renderAll();
    trace('errorLog.cleared');
  }

  function formatErrorLogTime(iso) {
    if (!iso) return '—';
    try {
      return new Date(iso).toLocaleString();
    } catch (e) {
      return iso;
    }
  }

  function escapeHtml(text) {
    return String(text || '')
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;');
  }

  /* ─── DOM Helpers ───────────────────────────────────────────────────────── */

  function $(sel) {
    return document.querySelector(sel);
  }

  function isVisible(el) {
    if (!el) return false;
    const style = window.getComputedStyle(el);
    if (style.display === 'none' || style.visibility === 'hidden' || style.opacity === '0') return false;
    const rect = el.getBoundingClientRect();
    return rect.width > 0 && rect.height > 0;
  }

  function fieldSnapshot() {
    const f = (sel) => {
      const el = $(sel);
      if (!el) return { exists: false };
      return {
        exists: true,
        visible: isVisible(el),
        value: (el.value || '').trim(),
        effective: getEffectiveFieldValue(sel),
        placeholder: el.placeholder || '',
        disabled: el.disabled,
      };
    };
    return {
      sortType: f(SEL.sortType),
      sourceContainer: f(SEL.sourceContainer),
      barcode: f(SEL.barcode),
      sortCode: { text: getSortCodeText() },
      scanTote: f(SEL.scanTote),
    };
  }

  function pageSnapshot() {
    return {
      url: location.href,
      title: document.title,
      fields: fieldSnapshot(),
      flags: {
        sessionArmed: state.sessionArmed,
        running: state.running,
        exceptionFlowActive: state.exceptionFlowActive,
        pageResetInProgress: state.pageResetInProgress,
        jobStartupInProgress: state.jobStartupInProgress,
        sourceContainerDone: state.sourceContainerDone,
        barcodeDone: state.barcodeDone,
        postBarcodeHandled: state.postBarcodeHandled,
        resumeRequired: state.resumeRequired,
        currentStep: state.currentStep,
        stepProgressAt: state.stepProgressAt,
      },
      queueIndex: state.queueIndex,
      currentRow: getCurrentRow(),
      workflowStep: getWorkflowStep(),
      exceptionVisible: isExceptionPanelVisible(),
      exceptionMessage: getExceptionMessage(),
    };
  }

  function getSortCodeText() {
    const el = $(SEL.sortCode);
    return el ? (el.textContent || el.innerText || '').trim() : '';
  }

  const PLACEHOLDER_VALUE_RE = /^scan\s+/i;

  function isPlaceholderScanText(val) {
    if (!val) return true;
    const t = val.trim();
    if (!t) return true;
    if (PLACEHOLDER_VALUE_RE.test(t)) return true;
    if (/^enter\s+/i.test(t)) return true;
    return false;
  }

  function getFieldValue(sel) {
    const el = $(sel);
    return el ? (el.value || '').trim() : '';
  }

  function getEffectiveFieldValue(sel) {
    const val = getFieldValue(sel);
    return isPlaceholderScanText(val) ? '' : val;
  }

  function getSortTypeDisplayText() {
    const el = $(SEL.sortTypeDisplay);
    if (!el) return '';
    return (el.textContent || '').replace(/\s+/g, ' ').trim();
  }

  function isSortTypeAcceptedOnPage() {
    if (isExceptionPanelVisible()) return false;
    const sortEl = $(SEL.sortType);
    if (!sortEl) return false;
    if (sortEl.disabled || sortEl.readOnly) return true;

    const displayText = getSortTypeDisplayText();
    if (displayText && new RegExp('\\b' + SORT_TYPE + '\\b', 'i').test(displayText)) {
      return true;
    }

    const activeId = document.activeElement?.id || '';
    if (activeId === 'source-container-textfield' || activeId === 'barcode-scan-textfield') {
      const displayText = getSortTypeDisplayText();
      if (displayText && new RegExp('\\b' + SORT_TYPE + '\\b', 'i').test(displayText)) {
        return true;
      }
      if (state.jobStartedAt) return true;
    }

    const sortEff = getEffectiveFieldValue(SEL.sortType);
    if (sortEff.toUpperCase() === SORT_TYPE.toUpperCase() && document.activeElement !== sortEl) {
      return true;
    }
    return false;
  }

  function isSourceAcceptedOnPage(expectedContainer) {
    const activeId = document.activeElement?.id || '';
    if (activeId === 'barcode-scan-textfield') return true;

    const pageContainer = getEffectiveFieldValue(SEL.sourceContainer);
    if (pageContainer) {
      if (expectedContainer && pageContainer.toUpperCase() !== expectedContainer.toUpperCase()) return false;
      if (getSortCodeText()) return true;
      if (isToteFieldActive()) return true;
      return activeId === 'barcode-scan-textfield';
    }

    if (
      isSortTypeSubmitted() &&
      state.lastSourceSubmitAt &&
      Date.now() - state.lastSourceSubmitAt >= TIMING.postSubmitMs &&
      Date.now() - state.lastSourceSubmitAt < TIMING.maxWaitMs + TIMING.typeDurationMs + TIMING.preSubmitMs * 3
    ) {
      if (
        expectedContainer &&
        state.lastSubmittedContainer &&
        state.lastSubmittedContainer.toUpperCase() === expectedContainer.toUpperCase()
      ) {
        trace('source.accept.clearedField', { expectedContainer });
        return true;
      }
    }
    return false;
  }

  function getWorkflowStep() {
    if (!isSortTypeSubmitted()) return 'sortType';

    const sourceEff = getEffectiveFieldValue(SEL.sourceContainer);
    if (!sourceEff && !state.sourceContainerDone) return 'sourceContainer';

    if (!state.barcodeDone) return 'barcode';

    const sortCode = getSortCodeText();
    if (sortCode) {
      if (isToteFieldActive() && !getEffectiveFieldValue(SEL.scanTote)) return 'tote';
      return 'postBarcode';
    }
    if (isToteFieldActive() && !getEffectiveFieldValue(SEL.scanTote)) return 'tote';
    return 'postBarcode';
  }

  function findUpdateButton() {
    const updateLink = $(SEL.updateLink);
    if (updateLink && isVisible(updateLink)) {
      const text = (updateLink.textContent || '').replace(/\s+/g, ' ').trim();
      if (/update/i.test(text)) return updateLink;

    const candidates = document.querySelectorAll(

    for (const btn of candidates) {

    }
    const bySel = document.querySelector(SEL.updateBtn);
    if (bySel && isVisible(bySel) && !bySel.disabled) return bySel;

    for (const btn of candidates) {

  function getExceptionMessage() {
    const info = $(SEL.exceptionInfo);
    if (info) {
      let text = (info.textContent || '').replace(/\s+/g, ' ').trim();
      const btn = $(SEL.exceptionBtn);
      if (btn) {
        const btnText = (btn.textContent || '').replace(/\s+/g, ' ').trim();
        text = text.replace(btnText, '').replace(/^ERROR\s*/i, '').trim();
      }
      if (text.length > 5) return text;
    }
    const container = $(SEL.exceptionContainer);
    if (container) {
      let text = (container.textContent || '').replace(/\s+/g, ' ').trim();
      text = text.replace(/^ERROR\s*/i, '').replace(/Continue.*$/i, '').trim();
      if (text.length > 5) return text;
    }
    return '';
  }

  function detectExceptionDialog() {
    if (isExceptionPanelVisible()) {
      return { el: $(SEL.exceptionPage), text: getExceptionMessage() || 'Exception panel visible' };
    }
    const dialogs = document.querySelectorAll('[role="dialog"], .modal, .exception-dialog, .a-modal');
    for (const d of dialogs) {
      if (!isVisible(d)) continue;
      const text = (d.textContent || '').replace(/\s+/g, ' ').trim();
      if (text.length > 10) return { el: d, text };
    }
    const bodyText = document.body.innerText || '';
    if (/item could not be moved/i.test(bodyText)) {
      return { el: null, text: 'Item could not be moved' };
    }
    if (/destination mismatch/i.test(bodyText)) {
      return { el: null, text: 'Destination mismatch' };
    }
    if (/does not correspond to any items/i.test(bodyText)) {
      return { el: null, text: getExceptionMessage() || 'Container/item mismatch' };
    }
    if (/encountered error while contacting backend/i.test(bodyText)) {
      return { el: null, text: getExceptionMessage() || 'Backend service error' };
    }
    return null;
  }

  function setNativeInputValue(el, value) {
    const proto = window.HTMLInputElement.prototype;
    const desc = Object.getOwnPropertyDescriptor(proto, 'value');
    if (desc && desc.set) {
      desc.set.call(el, value);
    } else {
      el.value = value;
    }
  }

      key: 'Enter',

      charCode: 13,
      bubbles: true,

    el.dispatchEvent(new KeyboardEvent('keypress', opts));
    el.dispatchEvent(new KeyboardEvent('keyup', opts));
  }

  async function clearScanField(el) {
    el.focus();
    await sleep(50);
    setNativeInputValue(el, '');
    try {
      el.dispatchEvent(new InputEvent('input', { bubbles: true, cancelable: true, inputType: 'deleteContentBackward' }));

      dispatchInputEvents(el);

  async function typeLikeScanner(el, value, label) {
    if (!el) return false;
    await clearScanField(el);
    const charDelay = Math.max(MIN_CHAR_DELAY_MS, Math.floor(TIMING.typeDurationMs / Math.max(value.length, 1)));
    for (let i = 0; i < value.length; i++) {
      const char = value[i];
      const partial = value.slice(0, i + 1);
      setNativeInputValue(el, partial);
      try {
        el.dispatchEvent(new InputEvent('input', {
          bubbles: true,
          cancelable: true,
          inputType: 'insertText',
          data: char,
        }));
      } catch (e) {
        dispatchInputEvents(el);
      }
      await sleep(charDelay);
    }
    dispatchInputEvents(el);
    trace('field.scanType', { label, value, final: (el.value || '').trim() });
    await settle();
    return true;
  }

  function reactSetInputValue(el, value) {

    if (el._valueTracker) {
      el._valueTracker.setValue(lastValue);

    dispatchInputEvents(el);
  }

  function runInPageContext(pageFnBody, payload) {
    return new Promise((resolve) => {
      const requestId = 'req-' + Date.now() + '-' + Math.floor(Math.random() * 1e6);
      let settled = false;
      function onMsg(ev) {
        if (ev.source !== window) return;
        if (!ev.data || ev.data.channel !== PAGE_BRIDGE || ev.data.requestId !== requestId) return;
        window.removeEventListener('message', onMsg);
        settled = true;
        resolve(ev.data.result || { ok: false, reason: 'empty-result' });
      }
      window.addEventListener('message', onMsg);
      const script = document.createElement('script');
      script.textContent = `
        (function() {
          var requestId = ${JSON.stringify(requestId)};
          var payload = ${JSON.stringify(payload)};
          var result = { ok: false };
          try {
            result = (${pageFnBody})(payload);
          } catch (err) {
            result = { ok: false, error: String(err) };
          }
          window.postMessage({ channel: ${JSON.stringify(PAGE_BRIDGE)}, requestId: requestId, result: result }, '*');
        })();
      `;
      (document.documentElement || document.head).appendChild(script);
      script.remove();
      setTimeout(() => {
        if (!settled) {
          window.removeEventListener('message', onMsg);
          pageBridgeFailed = true;
          trace('pageBridge.disabled', { reason: 'timeout' });
          resolve({ ok: false, reason: 'page-bridge-timeout' });
        }
      }, TIMING.bridgeTimeoutMs);
    });
  }

  const PAGE_FN_TYPE_VALUE = `function(payload) {
    var el = document.querySelector(payload.selector);
    if (!el) return { ok: false, reason: 'missing-element' };
    el.focus();
    el.click();
    var method = '';
    try {
      document.execCommand('selectAll', false, null);
      if (document.execCommand('insertText', false, payload.value)) method = 'execCommand';
    } catch (e1) {}
    if ((el.value || '').trim().toUpperCase() !== payload.value.toUpperCase()) {
      var desc = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value');
      var prev = el.value;
      desc.set.call(el, payload.value);
      if (el._valueTracker) el._valueTracker.setValue(prev);
      el.dispatchEvent(new Event('input', { bubbles: true }));
      el.dispatchEvent(new Event('change', { bubbles: true }));
      method = method || 'reactSet';
    }
    if ((el.value || '').trim().toUpperCase() !== payload.value.toUpperCase()) {
      el.value = '';
      for (var i = 0; i < payload.value.length; i++) {
        var ch = payload.value[i];
        el.value += ch;
        el.dispatchEvent(new InputEvent('input', { bubbles: true, inputType: 'insertText', data: ch }));
      }
      method = method || 'charLoop';
    }
    return {
      ok: true,
      method: method || 'unknown',
      value: (el.value || '').trim(),
      activeId: document.activeElement ? (document.activeElement.id || '') : '',
    };
  }`;

  async function clickAutocompleteMatch(text) {
    await sleep(40);
    const target = text.trim().toUpperCase();
    const selectors = [
      '[role="option"]',
      '[role="listbox"] li',
      '.a-dropdown-item',
      '.autocomplete-item',
      'ul.a-unordered-list li',
      'li.a-dropdown-item',
    ].join(', ');
    for (const opt of document.querySelectorAll(selectors)) {
      if (!isVisible(opt)) continue;
      const t = (opt.textContent || '').trim();
      if (!t) continue;
      if (t.toUpperCase() === target || t.toUpperCase().includes(target)) {
        trace('autocomplete.click', { text: t, match: text });
        opt.click();
        await settle();
        return true;
      }
    }
    trace('autocomplete.none', { match: text });
    return false;
  }

  async function unsafeTypeSlow(selector, value) {
    if (!value) return { ok: false, reason: 'empty-value' };
    if (!state.running) return { ok: false, reason: 'not-running' };
    const charDelay = Math.max(MIN_CHAR_DELAY_MS, Math.floor(TIMING.typeDurationMs / value.length));
    let last = { ok: false, reason: 'no-chars' };
    for (let i = 0; i < value.length; i++) {
      if (!state.running) return { ok: false, reason: 'not-running' };
      const partial = value.slice(0, i + 1);
      last = unsafeSetPartialValue(selector, partial);
      if (!last.ok) return last;
      touchStepProgress();
      if (i < value.length - 1) await sleep(charDelay);
    }
    if (last.value && last.value.toUpperCase() !== value.toUpperCase()) {
      trace('field.unsafeTypeMismatch', { selector, expected: value, got: last.value });
      last = unsafeSetFullValue(selector, value);
    }
    trace('field.unsafeTypeDone', { selector, value, charDelay, result: last });
    return last;
  }

  async function isolatedFieldValue(el, value, label) {
    if (!el) return false;
    await sleep(TIMING.preSubmitMs);
    el.focus();
    await settle();
    await typeLikeScanner(el, value, label);
    trace('field.isolatedType', { label, value, final: (el.value || '').trim() });
    await clickAutocompleteMatch(value);
    await sleep(TIMING.preSubmitMs);
    pressEnter(el);
    trace('field.enter', { label, value: (el.value || '').trim() });
    await sleep(50);
    pressEnter(el);
    trace('field.enterRetry', { label });
    await sleep(TIMING.postSubmitMs);
    await settle();
    return true;
  }

  async function submitFieldEnter(el, label) {
    if (!el) return false;
    await sleep(TIMING.preSubmitMs);
    const selector = el.id ? '#' + el.id : null;
    if (selector) {
      const enterResult = unsafePressEnter(selector);
      trace('field.unsafeEnter', { label, result: enterResult });
      await sleep(TIMING.postSubmitMs);
      await settle();
      return !!enterResult.ok;
    }
    el.focus();
    await settle();
    pressEnter(el);
    trace('field.enter', { label, value: (el.value || '').trim() });
    await sleep(50);
    pressEnter(el);
    trace('field.enterRetry', { label });
    await sleep(TIMING.postSubmitMs);
    await settle();
    return true;
  }

  async function submitFieldValue(el, value, label) {
    if (!el) return false;
    if (!state.running) return false;
    const selector = el.id ? '#' + el.id : null;
    await sleep(TIMING.preSubmitMs);
    if (!state.running) return false;
    touchStepProgress();

    if (selector) {
      const typeResult = await unsafeTypeSlow(selector, value);
      trace('field.unsafeType', { label, value, result: typeResult });
      if (!state.running) return false;
      if (!typeResult.ok || (typeResult.value || '').toUpperCase() !== value.toUpperCase()) {
        trace('field.unsafeTypeAbort', { label, value, result: typeResult });
        return false;
      }
      await clickAutocompleteMatch(value);
      await sleep(TIMING.preSubmitMs);
      if (!state.running) return false;
      touchStepProgress();
      const enterResult = unsafePressEnter(selector);
      trace('field.unsafeEnter', { label, result: enterResult });
      await sleep(TIMING.postSubmitMs);
      if (!state.running) return false;
      await settle();
      return true;
    }

    if (!state.running) return false;
    await isolatedFieldValue(el, value, label);
    return !!state.running;
  }

  function parseToteMaxInput(el, fallback) {
    const raw = el?.value ?? fallback ?? DEFAULT_MAX_TOTE_UNITS;
    const n = parseInt(String(raw), 10);
    if (!Number.isFinite(n)) return fallback ?? DEFAULT_MAX_TOTE_UNITS;
    return Math.max(1, Math.min(9999, n));
  }

    const key = normalizeToteKey(toteId);
    if (!key) return false;
    if (!state.toteMaxLimits) state.toteMaxLimits = {};
    state.toteMaxLimits[key] = parseToteMaxInput({ value: max }, DEFAULT_MAX_TOTE_UNITS);
    persistState();
    renderToteMapping();
    return true;
  }

  function normalizeSortSlots(target = state) {

      if (key && val) merged[key] = val;
    }
    target.sortSlots = merged;
  }

  }

  function getToteCount(toteId) {
    const key = normalizeToteKey(toteId);
    if (!key) return 0;
    return state.toteCounts[key] || 0;
  }

  function ensureToteCountEntry(toteId) {
    const key = normalizeToteKey(toteId);
    if (!key) return '';
    if (!state.toteCounts) state.toteCounts = {};
    if (state.toteCounts[key] == null) state.toteCounts[key] = 0;
    if (!state.toteDisplayIds) state.toteDisplayIds = {};
    if (!state.toteDisplayIds[key]) state.toteDisplayIds[key] = (toteId || '').trim() || key;
    return key;
  }

  function migrateToteCounts(target = state) {
    if (!target.toteCounts) target.toteCounts = {};
    if (!target.toteDisplayIds) target.toteDisplayIds = {};
    const merged = {};
    for (const [rawKey, rawCount] of Object.entries(target.toteCounts)) {
      const key = normalizeToteKey(rawKey);
      if (!key) continue;
      merged[key] = (merged[key] || 0) + (rawCount || 0);
      if (!target.toteDisplayIds[key]) target.toteDisplayIds[key] = (rawKey || '').trim() || key;
    }
    target.toteCounts = merged;
    for (const tote of Object.values(target.sortSlots || {})) {
      const key = normalizeToteKey(tote);
      if (!key) continue;
      if (target.toteCounts[key] == null) target.toteCounts[key] = 0;
      if (!target.toteDisplayIds[key]) target.toteDisplayIds[key] = (tote || '').trim() || key;
    }
  }

  function getKnownToteKeys() {
    const keys = new Set([
      ...Object.keys(state.toteCounts || {}),
      ...Object.values(state.sortSlots || {}).map((t) => normalizeToteKey(t)).filter(Boolean),
    ]);
    return [...keys].filter(Boolean).sort((a, b) => getToteDisplayId(a).localeCompare(getToteDisplayId(b)));
  }

  function saveSortSlotMapping(sortCode, toteId) {
    const trimmedCode = (sortCode || '').trim();
    const trimmedTote = (toteId || '').trim();
    if (!trimmedCode || !trimmedTote) return false;
    normalizeSortSlots();
    state.sortSlots[trimmedCode.toUpperCase()] = trimmedTote;
    ensureToteCountEntry(trimmedTote);
    persistState();
    renderToteMapping();
    return true;
  }

  function incrementToteCount(toteId) {
    const key = ensureToteCountEntry(toteId);
    if (!key) return;
    state.toteCounts[key] = (state.toteCounts[key] || 0) + 1;
    persistState();
    renderToteMapping();
    renderToteHistory();
  }

  function markRowStatus(index, status) {
    syncRowStatuses();
    if (index >= 0 && index < state.rowStatuses.length) {
      state.rowStatuses[index] = status;
      persistState();
      renderQueue();
    }
  }

  function advanceQueue() {
    if (state.queueIndex < state.queue.length) {
      markRowStatus(state.queueIndex, ROW_STATUS.COMPLETE);
    }
    state.queueIndex += 1;
    resetJobFlags();
    persistState();
    renderAll();
    trace('queue.advance', { queueIndex: state.queueIndex });
    if (state.queueIndex >= state.queue.length) {
      trace('queue.done');
      stopRunning('queue complete');
    }
  }

  function resetJobFlags() {
    state.sourceContainerDone = false;
    state.barcodeDone = false;
    state.postBarcodeHandled = false;
    state.pendingLpn = null;
    state.lastSubmittedLpn = null;
    state.lastSubmittedContainer = null;
    state.lastSourceSubmitAt = 0;
    state.currentContainer = null;
    state.jobStartedAt = 0;
    state.exceptionFlowActive = false;
    state.recoveryUpdateUsed = false;
    setStep('idle');
  }

  function setStep(step) {
    if (currentStep !== step) {
      currentStep = step;
      stepProgressAt = Date.now();
      state.currentStep = step;
      state.stepProgressAt = stepProgressAt;
      trace('step.change', { step });
    }
  }

  function touchStepProgress() {
    stepProgressAt = Date.now();
    state.stepProgressAt = stepProgressAt;
  }

  function resetErrorStatuses() {
    syncRowStatuses();
    for (let i = 0; i < state.rowStatuses.length; i++) {
      if (state.rowStatuses[i] === ROW_STATUS.ERROR && !isRowErrored(state.queue[i])) {
        state.rowStatuses[i] = ROW_STATUS.PENDING;
      }
    }
    state.queueIndex = findFirstPendingIndex();
    persistState();
    renderAll();
    trace('errors.reset');
  }

  /* ─── Re-entrancy Lock ──────────────────────────────────────────────────── */

  let automationLock = false;
  let watcherTimer = null;

  async function withLock(fn, label) {
    if (automationLock) {
      trace('lock.skip', { label });
      return false;
    }
    automationLock = true;
    try {
      return await fn();
    } finally {
      automationLock = false;
    }
  }

  /* ─── Verification ──────────────────────────────────────────────────────── */

  async function waitFor(conditionFn, timeoutMs, pollMs, label) {
    const start = Date.now();
    while (Date.now() - start < timeoutMs) {
      if (!state.running) {
        trace('wait.abort', { label, reason: 'not running' });
        return false;
      }
      try {
        if (await conditionFn()) {
          trace('wait.ok', { label, ms: Date.now() - start });
          return true;
        }
      } catch (e) {
        trace('wait.error', { label, error: String(e) });
      }
      await sleep(pollMs || randomPollMs());
    }
    trace('wait.timeout', { label, timeoutMs });
    return false;
  }

  async function verifySortTypeAccepted() {
    if (isExceptionPanelVisible()) return false;
    return isSortTypeAcceptedOnPage();
  }

  async function verifySourceAccepted(expectedContainer) {
    if (isExceptionPanelVisible()) return false;
    return isSourceAcceptedOnPage(expectedContainer);
  }

  async function verifyBarcodeAccepted(expectedLpn) {
    if (isExceptionPanelVisible()) return false;
    const pageLpn = getEffectiveFieldValue(SEL.barcode);
    if (getSortCodeText()) return true;
    const step = getWorkflowStep();
    if (step === 'tote' || step === 'postBarcode') return true;
    if (pageLpn && expectedLpn && pageLpn.toUpperCase() === expectedLpn.toUpperCase()) {
      return step !== 'barcode';
    }
    return false;
  }

  async function verifyToteMoveSuccess() {
    const sortCodeVisible = getSortCodeText();
    const toteActive = isToteFieldActive() && !getEffectiveFieldValue(SEL.scanTote);
    const barcodeEmpty = !getEffectiveFieldValue(SEL.barcode);
    return !toteActive && (barcodeEmpty || sortCodeVisible === '');
  }

  /* ─── Update Click (guarded) ────────────────────────────────────────────── */

  function isUpdateAllowed(reason) {
    const allowedPrefixes = ['job.end:', 'recovery:', 'stop:'];
    return allowedPrefixes.some((p) => reason.startsWith(p));
  }

  async function clickUpdate(reason) {
    if (!isUpdateAllowed(reason)) {
      trace('update.denied', { reason, policy: 'Update only for job.end, recovery, or stop' });
      return false;
    }
    if (state.pageResetInProgress) {
      trace('update.skip', { reason: 'pageResetInProgress', requested: reason });
      return false;
    }
    if (reason.startsWith('recovery:') && state.recoveryUpdateUsed) {
      trace('update.skip', { reason: 'recoveryAlreadyUsed', requested: reason });
      return false;
    }
    const btn = findUpdateButton();
    if (!btn) {
      trace('update.missingButton', { reason });
      return false;
    }
    trace('update.clickSingle', { reason });
    state.lastUpdateReason = reason;
    state.pageResetInProgress = true;
    if (reason.startsWith('recovery:')) state.recoveryUpdateUsed = true;
    persistStateNow();
    btn.click();
    await sleep(TIMING.postUpdateMs);
    await waitForPageReadyAfterUpdate();
    state.pageResetInProgress = false;
    trace('update.clickPageReset', { reason, done: true });
    resetJobFlags();
    persistState();
    return true;
  }

  async function waitForPageReadyAfterUpdate() {
    await waitFor(
      () => isVisible($(SEL.sortType)) || isVisible($(SEL.sourceContainer)),
      TIMING.maxWaitMs,
      randomPollMs(),
      'page.afterUpdate'
    );
  }

  /* ─── Step Actions ──────────────────────────────────────────────────────── */

  async function submitSortType() {
    setStep('sortType');
    trace('sortType.begin');
    const el = $(SEL.sortType);
    if (!el) {
      trace('sortType.missing');
      return false;
    }
    if (isSortTypeSubmitted()) {
      trace('sortType.alreadySubmitted', { workflowStep: getWorkflowStep() });
      state.jobStartedAt = Date.now();
      trace('job.start', { sortType: SORT_TYPE });
      touchStepProgress();
      return true;
    }
    const sortEff = getEffectiveFieldValue(SEL.sortType);
    if (sortEff.toUpperCase() === SORT_TYPE.toUpperCase()) {
      trace('sortType.resubmit', { reason: 'value present but workflow still on sortType' });
      await submitFieldValue(el, SORT_TYPE, 'sortType-resubmit');
    } else {
      await submitFieldValue(el, SORT_TYPE, 'sortType');
    }
    let ok = await waitFor(verifySortTypeAccepted, TIMING.maxWaitMs, randomPollMs(), 'sortType.accept');
    if (!ok) {
      trace('sortType.retryEnter', { reason: 'first submit did not advance workflow' });
      await submitFieldValue(el, SORT_TYPE, 'sortType-retry');
      ok = await waitFor(verifySortTypeAccepted, TIMING.maxWaitMs, randomPollMs(), 'sortType.acceptRetry');
    }
    if (ok) {
      state.jobStartedAt = Date.now();
      trace('job.start', { sortType: SORT_TYPE, workflowStep: getWorkflowStep() });
      touchStepProgress();
    } else {
      trace('sortType.submitFailed', {
        raw: getFieldValue(SEL.sortType),
        effective: getEffectiveFieldValue(SEL.sortType),
        workflowStep: getWorkflowStep(),
      });
    }
    return ok;
  }

  async function submitSourceContainer(container) {
    setStep('sourceContainer');
    trace('source.type', { container });
    const el = $(SEL.sourceContainer);
    if (!el) {
      trace('source.missing');
      return false;
    }
    const pageVal = getEffectiveFieldValue(SEL.sourceContainer);
    if (pageVal && pageVal.toUpperCase() === container.toUpperCase()) {
      if (await verifySourceAccepted(container)) {
        trace('source.alreadyOnPage', { container: pageVal });
        state.sourceContainerDone = true;
        state.currentContainer = container;
        touchStepProgress();
        return true;
      }
      if (isVisible(el)) {
        trace('source.resubmitEnter', { container: pageVal, reason: 'value on page but step not advanced' });
        state.lastSubmittedContainer = container;
        state.lastSourceSubmitAt = Date.now();
        await submitFieldEnter(el, 'sourceContainer');
        const resubmitOk = await waitFor(
          () => verifySourceAccepted(container),
          TIMING.maxWaitMs,
          randomPollMs(),
          'source.resubmit.accept'
        );
        if (resubmitOk) {
          state.sourceContainerDone = true;
          state.currentContainer = container;
          trace('source.accepted', { container, via: 'resubmitEnter' });
          touchStepProgress();
          return true;
        }
      }
    }
    state.lastSubmittedContainer = container;
    state.lastSourceSubmitAt = Date.now();
    const submitted = await submitFieldValue(el, container, 'sourceContainer');
    if (!submitted) return false;
    const ok = await waitFor(
      () => verifySourceAccepted(container),
      TIMING.maxWaitMs,
      randomPollMs(),
      'source.accept'
    );
    if (ok) {
      state.sourceContainerDone = true;
      state.currentContainer = container;
      trace('source.accepted', { container, workflowStep: getWorkflowStep() });
      touchStepProgress();
    }
    return ok;
  }

  async function submitBarcode(lpn) {
    setStep('barcode');
    trace('barcode.submit', { lpn });
    state.pendingLpn = lpn;
    state.lastSubmittedLpn = lpn;
    const el = $(SEL.barcode);
    if (!el) {
      trace('barcode.missing');
      return false;
    }
    await submitFieldValue(el, lpn, 'barcode');
    const ok = await waitFor(
      () => verifyBarcodeAccepted(lpn),
      TIMING.maxWaitMs,
      randomPollMs(),
      'barcode.accept'
    );
    if (ok) {
      state.barcodeDone = true;
      state.pendingLpn = null;
      trace('barcode.accepted', { lpn });
      touchStepProgress();
    }
    return ok;
  }

  async function submitTote(toteId) {
    setStep('tote');
    trace('tote.submit', { toteId });
    const el = $(SEL.scanTote);
    if (!el) {
      trace('tote.missing');
      return false;
    }
    await submitFieldValue(el, toteId, 'scanTote');
    const ok = await waitFor(verifyToteMoveSuccess, TIMING.maxWaitMs, randomPollMs(), 'tote.move');
    if (ok) {
      incrementToteCount(toteId);
      trace('tote.success', { toteId });
      touchStepProgress();
    } else {
      trace('tote.failOrTimeout', { toteId });
    }
    return ok;
  }

  /* ─── Exception Handling ────────────────────────────────────────────────── */

  async function handleRecoverableException(message, reason) {
    trace('exception.recover', { message, reason, autoClose: state.autoCloseErrors });
    if (!state.autoCloseErrors) {
      markCurrentRowError(message, { reason });
      stopRunning(`${message} (auto-close off)`);
      return true;
    }
    await clickExceptionContinue($(SEL.exceptionBtn));
    await settle();
    markCurrentRowError(message, { reason });
    await recoverAfterError(reason);
    return true;
  }

  async function handleExceptions() {
    const exBtn = $(SEL.exceptionBtn);
    const panelVisible = isExceptionPanelVisible();
    if (exBtn && isVisible(exBtn)) {
      trace('exception.btnVisible', { text: (exBtn.textContent || '').trim(), panelVisible });
    }

    const ex = detectExceptionDialog();
    if (!ex && !(exBtn && isVisible(exBtn))) return false;

    const exText = (ex && ex.text) || getExceptionMessage() || 'Unknown exception';
    trace('exception.detected', { text: exText.slice(0, 300) });
    state.exceptionFlowActive = true;

    if (/stop program/i.test(exText)) {
      trace('exception.blocker', { action: 'halt' });
      markCurrentRowError(exText.slice(0, 200), { reason: 'stop-program' });
      stopRunning('STOP PROGRAM blocker');
      return true;
    }

    if (/item could not be moved/i.test(exText)) {
      return handleRecoverableException('Item could not be moved', 'item-not-moved');
    }

    if (/destination mismatch/i.test(exText)) {
      return handleRecoverableException('Destination mismatch', 'dest-mismatch');
    }

    if (/does not correspond to any items/i.test(exText)) {
      return handleRecoverableException(exText.slice(0, 200), 'container-item-mismatch');
    }

    if (/encountered error while contacting backend/i.test(exText)) {
      return handleRecoverableException(exText.slice(0, 200), 'backend-error');
    }

    if (/^error$/i.test(exText.trim()) || exText === 'Exception panel visible') {
      const msg = getExceptionMessage() || 'Sort tool exception';
      return handleRecoverableException(msg.slice(0, 200), 'exception-generic');
    }

    if (state.autoCloseErrors && panelVisible && getExceptionMessage()) {
      return handleRecoverableException(exText.slice(0, 200), 'exception-auto');
    }

    return false;
  }

  async function clickExceptionContinue(exBtn) {
    if (exBtn && isVisible(exBtn)) {
      exBtn.click();
      await settle();
      return true;
    }
    const cont = findContinueButton();
    if (cont) {
      cont.click();
      await settle();
      return true;
    }
    return false;
  }

  function markCurrentRowError(message, meta = {}) {
    const row = getCurrentRow();
    trace('row.error', { index: state.queueIndex, message, row });
    markRowStatus(state.queueIndex, ROW_STATUS.ERROR);
    recordError(row, message, meta);
    state.results.push({
      index: state.queueIndex,
      row,
      status: 'error',
      message,
      reason: meta.reason || '',
      at: new Date().toISOString(),
    });
    persistState();
    renderAll();
  }

  async function recoverAfterError(reason) {
    trace('reset.begin', { reason });
    state.exceptionFlowActive = false;
    state.queueIndex += 1;
    syncRowStatuses();
    persistState();
    renderAll();
    if (state.queueIndex >= state.queue.length) {
      stopRunning('queue complete after error skip');
      trace('reset.end', { reason, action: 'queue done' });
      return;
    }
    if (state.running) {
      const updated = await clickUpdate(`recovery:${reason}`);
      if (!updated) {
        trace('reset.updateRetry', { reason });
        state.recoveryUpdateUsed = false;
        await sleep(200);
        await clickUpdate(`recovery:${reason}-retry`);
      }
    }
    trace('reset.end', { reason, nextIndex: state.queueIndex });
  }

  /* ─── Stale State Detection ─────────────────────────────────────────────── */

  function isStaleBarcodeOnPage(expectedLpn) {
    const pageLpn = getEffectiveFieldValue(SEL.barcode);
    if (!pageLpn || !expectedLpn) return false;
    return pageLpn.toUpperCase() !== expectedLpn.toUpperCase() && state.barcodeDone === false;
  }

  function needsBarcodeSubmit(expectedLpn) {
    const effective = getEffectiveFieldValue(SEL.barcode);
    if (!effective) return isBarcodeFieldActive();
    return effective.toUpperCase() !== expectedLpn.toUpperCase();
  }

  function isStaleSortLabelBeforeBarcode() {
    const sortText = getSortCodeText();
    if (!sortText) return false;
    if (state.barcodeDone) return false;
    if (isSourceContainerFieldActive()) return false;
    if (!state.sourceContainerDone) return false;
    return isBarcodeFieldActive();
  }

  function isFreshWorkflowEntry() {
    return getWorkflowStep() === 'sortType' && !state.barcodeDone;
  }

  function shouldDeferStaleSortReset() {
    if (isSourceContainerFieldActive()) {
      trace('barcode.deferStaleSort', { reason: 'sourceContainerFieldActive' });
      return true;
    }
    if (isFreshWorkflowEntry()) {
      trace('barcode.deferStaleSort', { reason: 'freshWorkflowEntry' });
      return true;
    }
    return false;
  }

  /* ─── Main Watcher / State Machine ──────────────────────────────────────── */

  async function watcherTick() {
    if (!state.running) return;
    ensureProgressTicker();
    if (state.resumeRequired) {
      trace('watcher.skip', { reason: 'resumeRequired' });
      return;
    }

    await withLock(async () => {
      state.lastRunningAt = Date.now();
      trace('watcher.poll', { queueIndex: state.queueIndex, workflowStep: getWorkflowStep(), flags: {
        sourceContainerDone: state.sourceContainerDone,
        barcodeDone: state.barcodeDone,
        postBarcodeHandled: state.postBarcodeHandled,
        pageResetInProgress: state.pageResetInProgress,
      }});

      if (checkStuckRunning()) return;
      if (await attemptStuckStepRecovery()) return;

      if (await handleExceptions()) return;

      const row = getCurrentRow();
      if (!row) {
        trace('watcher.noRow');
        stopRunning('no current row');
        return;
      }

      if (skipKnownRows(row)) return;

      markRowStatus(state.queueIndex, ROW_STATUS.ACTIVE);

      if (state.pageResetInProgress || state.jobStartupInProgress) {
        trace('watcher.skip', { reason: 'resetOrStartupInProgress' });
        return;
      }

      if (!state.jobStartedAt) {
        state.jobStartupInProgress = true;
        const sortOk = await submitSortType();
        state.jobStartupInProgress = false;
        if (!sortOk) {
          trace('watcher.wait', { step: 'sortType' });
          return;
        }
      }

      const expectedContainer = (row.container || '').trim();
      const expectedLpn = getRowLpn(row);
      const expectedSort = (row.sortCode || row.sort || '').trim();

      if (expectedContainer) {
        const pageContainer = getEffectiveFieldValue(SEL.sourceContainer);
        const containerMismatch = pageContainer && pageContainer.toUpperCase() !== expectedContainer.toUpperCase();
        if (containerMismatch && state.sourceContainerDone) {
          trace('watcher.containerMismatch', { page: pageContainer, expected: expectedContainer });
          state.sourceContainerDone = false;
        }
        if (!state.sourceContainerDone) {
          const wfStep = getWorkflowStep();
          if (wfStep === 'sortType') {
            trace('watcher.skipSourceNeedSortType', { expectedContainer });
            state.jobStartedAt = 0;
            return;
          }
          if (wfStep === 'sourceContainer' || !pageContainer || containerMismatch) {
            trace('watcher.skipBarcodeNeedSource', { expectedContainer, workflowStep: wfStep });
            await submitSourceContainer(expectedContainer);
            return;
          }
          if (pageContainer.toUpperCase() === expectedContainer.toUpperCase()) {
            if (await verifySourceAccepted(expectedContainer)) {
              trace('source.alreadyOnPage', { container: pageContainer, sameContainerRow: true });
              state.sourceContainerDone = true;
              state.currentContainer = expectedContainer;
            } else {
              trace('watcher.skipBarcodeNeedSource', { expectedContainer, reason: 'container value present but step not verified' });
              await submitSourceContainer(expectedContainer);
              return;
            }
          }
        }
      } else {
        state.sourceContainerDone = true;
      }

      if (!state.barcodeDone) {
        if (isStaleBarcodeOnPage(expectedLpn)) {
          trace('barcode.staleOnPage', { page: getEffectiveFieldValue(SEL.barcode), raw: getFieldValue(SEL.barcode), expected: expectedLpn });
        }

        if (isStaleSortLabelBeforeBarcode() && !shouldDeferStaleSortReset()) {
          trace('barcode.staleSortReset', { sortCode: getSortCodeText() });
        } else if (shouldDeferStaleSortReset() && getSortCodeText()) {
          trace('barcode.deferStaleSort', { sortCode: getSortCodeText() });
        }

        if (needsBarcodeSubmit(expectedLpn)) {
          trace('barcode.readyToSubmit', { expected: expectedLpn, raw: getFieldValue(SEL.barcode) });
          await submitBarcode(expectedLpn);
          return;
        }
        if (getEffectiveFieldValue(SEL.barcode).toUpperCase() === expectedLpn.toUpperCase()) {
          state.barcodeDone = true;
          touchStepProgress();
        }
      }

      if (state.barcodeDone && !state.postBarcodeHandled) {
        if (isExceptionPanelVisible()) {
          trace('watcher.wait', { step: 'exceptionPanelBeforeTote' });
          return;
        }
        const sortText = getSortCodeText();
        const lookupCode = expectedSort || sortText;
        if (!lookupCode) {
          trace('watcher.wait', { step: 'sortCodeFromPage' });
          return;
        }
        const toteId = getToteForSortCode(lookupCode);
        if (!toteId) {
          const needsMappingPrompt =
            !state.toteMappingPrompt ||
            state.toteMappingPrompt.sortCode !== lookupCode ||
            !isExecutionModalOpen(TOTE_MODAL_SEL);
          if (needsMappingPrompt) {
            trace('tote.noMapping', {
              sortCode: lookupCode,
              reshow: !!state.toteMappingPrompt,
              modalOpen: isExecutionModalOpen(TOTE_MODAL_SEL),
            });
            showToteMappingModal(lookupCode);
          } else {
            touchStepProgress();
          }
          trace('watcher.wait', { step: 'toteMappingPrompt' });
          return;
        }
        if (state.toteMappingPrompt) hideToteMappingModal();
        if (state.toteFullPrompt) hideToteFullModal();
        const count = getToteCount(toteId);
        const toteMax = getToteMaxLimit(toteId);
        if (count >= toteMax) {
          const needsFullPrompt =
            !state.toteFullPrompt ||
            state.toteFullPrompt.sortCode !== lookupCode ||
            !isExecutionModalOpen(TOTE_FULL_MODAL_SEL);
          if (needsFullPrompt) {
            trace('tote.maxReached', {
              toteId,
              count,
              toteMax,
              sortCode: lookupCode,
              reshow: !!state.toteFullPrompt,
              modalOpen: isExecutionModalOpen(TOTE_FULL_MODAL_SEL),
            });
            showToteFullModal(lookupCode, toteId, count);
          } else {
            touchStepProgress();
          }
          trace('watcher.wait', { step: 'toteFullPrompt' });
          return;
        }
        if (isToteFieldActive()) {
          const moved = await submitTote(toteId);
          if (moved) {
            state.postBarcodeHandled = true;
            persistState();
            const finished = await finishJob(expectedLpn, toteId, expectedSort || sortText);
            if (!finished) state.postBarcodeHandled = false;
          }
          return;
        }
        trace('watcher.wait', { step: 'toteField' });
        return;
      }

      if (state.postBarcodeHandled) {
        const sortText = getSortCodeText();
        const lookupCode = expectedSort || sortText;
        const toteId = lookupCode ? getToteForSortCode(lookupCode) : null;
        trace('watcher.jobCompleteRetry', { lpn: expectedLpn, toteId });
        if (toteId) {
          const finished = await finishJob(expectedLpn, toteId, lookupCode);
          if (!finished) state.postBarcodeHandled = false;
        }
        return;
      }
    }, 'watcherTick');
  }

  async function finishJob(lpn, toteId, sortCode) {
    trace('job.end.pending', { lpn, toteId, sortCode });
    const prevCompleted = getSessionJobsCompleted();
    const prevDeadline = state.sessionEtaDeadline || 0;
    state.sessionJobsCompleted = prevCompleted + 1;
    const progressBeforeAdvance = getQueueProgress();
    refreshSessionEtaDeadline(Math.max(0, progressBeforeAdvance.remaining - 1));
    persistStateNow();

    let updated = await clickUpdate('job.end:post-tote-success');
    if (!updated) {
      await sleep(400);
      updated = await clickUpdate('job.end:post-tote-success-retry');
    }
    if (!updated) {
      state.sessionJobsCompleted = prevCompleted;
      state.sessionEtaDeadline = prevDeadline;
      persistStateNow();
      trace('job.end.updateFailed');
      return false;
    }
    state.results.push({
      index: state.queueIndex,
      row: getCurrentRow(),
      status: 'complete',
      lpn,
      toteId,
      sortCode,
      at: new Date().toISOString(),
    });
    trace('job.end', { lpn, toteId, sortCode, sessionJobsCompleted: state.sessionJobsCompleted });
    recordCompletedRow(getCurrentRow(), {
      container: getCurrentRow()?.container || '',
      lpn,
      sortCode,
      toteId,
    });
    advanceQueue();
    if (state.running && state.queueIndex < state.queue.length) {
      trace('job.nextAutoStart', { queueIndex: state.queueIndex });
    }
    return true;
  }

  function checkStuckRunning() {
    const stepIdleMs = Date.now() - stepProgressAt;
    const runIdleMs = Date.now() - (state.lastRunningAt || Date.now());
    if (stepIdleMs > TIMING.stuckRunningMs) {
      trace('stuck.timeout', { step: currentStep, stepIdleMs, runIdleMs });
      state.pageResetInProgress = false;
      state.jobStartupInProgress = false;
      state.exceptionFlowActive = false;
      automationLock = false;
      touchStepProgress();
      return true;
    }
    return false;
  }

  async function attemptStuckStepRecovery() {
    if (state.toteMappingPrompt || state.toteFullPrompt) return false;
    const stepIdleMs = Date.now() - stepProgressAt;
    if (stepIdleMs < TIMING.stuckStepMs) return false;
    if (state.recoveryUpdateUsed) return false;
    if (state.pageResetInProgress) return false;

    const betweenWdAndSource =
      state.jobStartedAt &&
      !state.sourceContainerDone &&
      (currentStep === 'sortType' || currentStep === 'sourceContainer');

    if (betweenWdAndSource) {
      trace('watcher.staleRecover', {
        action: 'deferUpdate',
        reason: 'between WDSorting and source — prefer container scan',
        step: currentStep,
        stepIdleMs,
      });
      const row = getCurrentRow();
      if (row && row.container && isSourceContainerFieldActive()) {
        await submitSourceContainer(row.container.trim());
      }
      touchStepProgress();
      return true;
    }

    if (currentStep === 'idle' || currentStep === 'sortType') return false;

    trace('watcher.staleRecover', {
      action: 'recoveryUpdate',
      step: currentStep,
      stepIdleMs,
    });
    await clickUpdate(`recovery:stuck-step:${currentStep}`);
    touchStepProgress();
    return true;
  }

  function startWatcher() {
    if (watcherTimer) clearInterval(watcherTimer);
    watcherTimer = setInterval(() => {
      watcherTick().catch((e) => trace('watcher.error', { error: String(e) }));
    }, TIMING.pollMaxMs);
  }

  function stopWatcher() {
    if (watcherTimer) {
      clearInterval(watcherTimer);
      watcherTimer = null;
    }
  }

  /* ─── Start / Stop / Resume ─────────────────────────────────────────────── */

  function armSession(fromTop) {
    if (fromTop) {
      state.queueIndex = 0;
      state.rowStatuses = state.queue.map((row) => {
        if (isRowCompleted(row)) return ROW_STATUS.COMPLETE;
        if (isRowErrored(row)) return ROW_STATUS.ERROR;
        return ROW_STATUS.PENDING;
      });
    }
    applyKnownRowsToQueue();
    state.queueIndex = findFirstPendingIndex();
    state.sessionArmed = true;
    state.resumeRequired = false;
    persistState();
    trace('session.armed', { fromTop, queueIndex: state.queueIndex });
    renderAll();
  }

  function startRunning(fromTop) {
    if (state.queue.length === 0) {
      trace('start.abort', { reason: 'empty queue' });
      alert('Queue is empty. Paste queue data first.');
      return;
    }
    armSession(fromTop);
    if (state.queueIndex >= state.queue.length) {
      trace('start.abort', { reason: 'all rows already completed or errored' });
      alert('All rows in this queue are already completed or logged as errors.');
      return;
    }
    resetJobFlags();
    state.running = true;
    state.lastRunningAt = Date.now();
    state.sessionStartedAt = Date.now();
    state.sessionJobsCompleted = 0;
    state.sessionEtaDeadline = 0;
    state.resumeRequired = false;
    touchStepProgress();
    persistState();
    trace('start', { fromTop, queueIndex: state.queueIndex });
    startWatcher();
    startProgressTicker();
    renderAll();
  }

  async function stopRunning(reason) {
    trace('stop', { reason });
    state.running = false;
    stopProgressTicker();
    hideToteMappingModal();
    hideToteFullModal();
    stopWatcher();
    const row = getCurrentRow();
    if (row && state.rowStatuses[state.queueIndex] === ROW_STATUS.ACTIVE) {
      markRowStatus(state.queueIndex, ROW_STATUS.PENDING);
      trace('stop.rowPending', { index: state.queueIndex });
    }
    if (state.stopClickUpdateOnStop && state.sessionArmed) {
      await clickUpdate('stop:reset-page');
    }
    state.sessionArmed = false;
    persistState();
    renderAll();
  }

  function resumeAfterRefresh() {
    if (!state.sessionArmed) {
      trace('resume.abort', { reason: 'not armed' });
      return;
    }
    trace('resume.enter', { queueIndex: state.queueIndex });
    applyKnownRowsToQueue();
    if (state.queueIndex >= state.queue.length) {
      trace('resume.abort', { reason: 'all rows already completed or errored' });
      state.running = false;
      persistState();
      renderAll();
      return;
    }
    state.resumeRequired = false;
    state.running = true;
    state.lastRunningAt = Date.now();
    if (!state.sessionStartedAt) state.sessionStartedAt = Date.now();
    if (getSessionJobsCompleted() >= 1) {
      refreshSessionEtaDeadline(getQueueProgress().remaining);
    }
    resetJobFlags();
    persistState();
    startWatcher();
    startProgressTicker();
    renderAll();
  }

  function ensureAlertAudio() {
    try {
      if (!alertAudioCtx) {
        alertAudioCtx = new (window.AudioContext || window.webkitAudioContext)();
      }
      if (alertAudioCtx.state === 'suspended') {
        alertAudioCtx.resume().catch(() => {});
      }
      return alertAudioCtx;
    } catch (e) {
      trace('alert.audioInitFailed', { error: String(e) });
      return null;
    }
  }

  function playAlertBeep() {
    try {
      const ctx = ensureAlertAudio();
      if (!ctx) return;
      const now = ctx.currentTime;
      const beepSec = 0.22;
      const gapSec = 0.1;
      const pattern = [880, 1100, 880, 1100];

      pattern.forEach((freq, i) => {
        const start = now + i * (beepSec + gapSec);
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        const shaper = ctx.createWaveShaper();
        const curve = new Float32Array(256);
        for (let j = 0; j < 256; j++) {
          const x = (j / 128) - 1;
          curve[j] = Math.tanh(x * 4);
        }
        shaper.curve = curve;
        osc.type = 'square';
        osc.frequency.value = freq;
        gain.gain.setValueAtTime(0.0001, start);
        gain.gain.exponentialRampToValueAtTime(1, start + 0.008);
        gain.gain.setValueAtTime(1, start + beepSec - 0.02);
        gain.gain.exponentialRampToValueAtTime(0.0001, start + beepSec);
        osc.connect(gain);
        gain.connect(shaper);
        shaper.connect(ctx.destination);
        osc.start(start);
        osc.stop(start + beepSec + 0.02);
      });
      trace('alert.beep');
    } catch (e) {
      trace('alert.beepFailed', { error: String(e) });
    }
  }

  function armAlertAudioFromUserGesture() {
    ensureAlertAudio();
  }

  async function skipCurrentRowAndContinue(reason) {
    hideToteMappingModal();
    hideToteFullModal();
    if (!state.running) {
      trace('row.skip.aborted', { reason, running: false });
      return;
    }
    const skippedIndex = state.queueIndex;
    const row = getCurrentRow();
    trace('row.skip', { index: skippedIndex, reason, row });
    markRowStatus(skippedIndex, ROW_STATUS.PENDING);
    state.queueIndex += 1;
    resetJobFlags();
    state.recoveryUpdateUsed = false;
    persistState();
    renderAll();
    if (state.queueIndex >= state.queue.length) {
      stopRunning('queue complete after skip');
      trace('row.skip.end', { reason, skippedIndex });
      return;
    }
    const updated = await clickUpdate(`recovery:skip:${reason}`);
    if (!updated) {
      state.recoveryUpdateUsed = false;
      await sleep(200);
      await clickUpdate(`recovery:skip:${reason}-retry`);
    }
    touchStepProgress();
    trace('row.skip.continue', { reason, skippedIndex, nextIndex: state.queueIndex });
  }

  function cancelExecutionPrompt(reason) {
    trace('executionPrompt.cancel', { reason, running: state.running });
    if (state.running) {
      stopRunning(reason);
      return;
    }
    hideToteMappingModal();
    hideToteFullModal();
  }

    });
    cancelBtn.insertAdjacentElement('afterend', btn);
  }

  function submitToteFullRaiseLimit() {
    const fullToteId = state.toteFullPrompt?.fullToteId || '';
    const count = state.toteFullPrompt?.count || 0;
    const maxInput = $('#lsh-full-modal-tote-max');
    const newMax = parseToteMaxInput(maxInput, getToteMaxLimit(fullToteId));
    if (newMax <= count) {
      alert(`Max must be greater than the current count (${count}).`);
      maxInput?.focus();
      maxInput?.select();
      return;
    }
    saveToteMaxLimit(fullToteId, newMax);
    trace('tote.maxRaised', { toteId: fullToteId, count, newMax });
    hideToteFullModal();
    touchStepProgress();
  }

  function submitToteFullModal() {
    const input = $('#lsh-full-modal-tote-input');
    const maxInput = $('#lsh-full-modal-tote-max');
    const sortCode = state.toteFullPrompt?.sortCode || ($('#lsh-full-modal-sort-code') || {}).textContent || '';
    const oldToteId = state.toteFullPrompt?.fullToteId || '';
    const newToteId = (input?.value || '').trim();
    if (!newToteId) {
      input?.focus();
      return;
    }
    if (newToteId.toUpperCase() === oldToteId.toUpperCase()) {
      trace('tote.fullSameTote', { toteId: newToteId });
      input?.focus();
      input?.select();
      return;
    }
    if (!saveSortSlotMapping(sortCode, newToteId)) return;
    saveToteMaxLimit(newToteId, parseToteMaxInput(maxInput, DEFAULT_MAX_TOTE_UNITS));
    trace('tote.fullReassigned', { sortCode, oldToteId, newToteId, max: getToteMaxLimit(newToteId) });
    hideToteFullModal();
    touchStepProgress();
  }

      summary.textContent = errors.length
        ? `${errors.length} error${errors.length === 1 ? '' : 's'} logged (newest first)`
        : 'No errors logged yet.';
    }
    if (!errors.length) {
      list.innerHTML = '<div class="lsh-err-empty">No container → LPN errors recorded.</div>';
      return;

    const rows = errors
      .map(
        (e) => `

          <td>${escapeHtml(e.container || '—')}</td>
          <td>${escapeHtml(e.lpn || '—')}</td>
          <td class="lsh-err-msg">${escapeHtml(e.message || '—')}</td>

      .join('');

        <thead>
          <tr>
            <th>Container</th>
            <th>LPN</th>
            <th>Error</th>
            <th>When</th>
          </tr>
        </thead>
        <tbody>${rows}</tbody>
      </table>
    `;
  }

  function copyErrorLog() {
    const lines = (state.errorLog || []).map((e) =>
      [e.container || '', e.lpn || '', e.message || '', e.at || ''].join('\t')
    );
    const header = 'Container\tLPN\tError\tWhen';
    const text = lines.length ? [header, ...lines].join('\n') : header;
    copyTextToClipboard(text, 'errorLog.copy');
  }

  function copyTextToClipboard(text, traceEvent) {
    navigator.clipboard.writeText(text).then(() => trace(traceEvent)).catch(() => prompt('Copy:', text));
  }

  function parseQueueText(text) {
    const lines = text.split(/\r?\n/).map((l) => l.trim()).filter(Boolean);
    const queue = [];
    for (const line of lines) {
      const parts = line.split(/[,\t|]/).map((p) => p.trim());
      if (parts.length < 2) continue;
      queue.push({
        container: parts[0],
        lpn: parts[1],
        sortCode: parts[2] || '',
      });
    }
    return queue;
  }

  function loadQueueFromTextarea() {
    const text = ($('#lsh-queue-input') || {}).value || '';
    state.queue = parseQueueText(text);
    state.queueIndex = 0;
    state.rowStatuses = state.queue.map(() => ROW_STATUS.PENDING);
    state.results = [];
    const marked = applyKnownRowsToQueue();
    persistState();
    renderAll();
    trace('queue.loaded', { count: state.queue.length, alreadyKnown: marked, nextIndex: state.queueIndex });
  }

  let progressTicker = null;

  function getQueueProgress() {
    syncRowStatuses();
    const total = state.queue.length;
    let complete = 0;
    let error = 0;
    let pending = 0;
    let active = 0;
    for (let i = 0; i < total; i++) {
      const s = state.rowStatuses[i];
      if (s === ROW_STATUS.COMPLETE) complete += 1;
      else if (s === ROW_STATUS.ERROR) error += 1;
      else if (s === ROW_STATUS.ACTIVE) active += 1;
      else pending += 1;
    }
    return {
      total,
      complete,
      error,
      pending,
      active,
      remaining: pending + active,
    };
  }

  function formatDuration(ms) {
    if (ms == null || !Number.isFinite(ms) || ms < 0) return '—';
    const sec = Math.max(0, Math.ceil(ms / 1000));
    const h = Math.floor(sec / 3600);
    const m = Math.floor((sec % 3600) / 60);
    const s = sec % 60;
    if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    return `${m}:${String(s).padStart(2, '0')}`;
  }

  function formatClockTime(date) {
    if (!date || !(date instanceof Date) || Number.isNaN(date.getTime())) return '—';
    return date.toLocaleTimeString([], { hour: 'numeric', minute: '2-digit' });
  }

  function getSessionJobsCompleted() {
    const stored = state.sessionJobsCompleted || 0;
    if (stored >= 1) return stored;
    const startedAt = state.sessionStartedAt || 0;
    if (!startedAt) return 0;
    const fromResults = (state.results || []).filter(
      (r) => r.status === 'complete' && r.at && new Date(r.at).getTime() >= startedAt
    ).length;
    return Math.max(stored, fromResults);
  }

  function refreshSessionEtaDeadline(remainingJobs) {
    const startedAt = state.sessionStartedAt || 0;
    const jobsCompleted = getSessionJobsCompleted();
    if (!startedAt || jobsCompleted < 1 || remainingJobs <= 0) {
      state.sessionEtaDeadline = 0;
      return;
    }
    const elapsed = Date.now() - startedAt;
    const avgMs = elapsed / jobsCompleted;
    state.sessionEtaDeadline = Date.now() + avgMs * remainingJobs;
    trace('progress.etaRefresh', {
      jobsCompleted,
      remainingJobs,
      avgMs: Math.round(avgMs),
      deadline: new Date(state.sessionEtaDeadline).toISOString(),
    });
  }

  function resolveSessionEtaDeadline(remainingJobs) {
    if (state.sessionEtaDeadline > Date.now() - 1000) return state.sessionEtaDeadline;
    const startedAt = state.sessionStartedAt || 0;
    const jobsCompleted = getSessionJobsCompleted();
    if (!startedAt || jobsCompleted < 1 || remainingJobs <= 0) return 0;
    const elapsed = Date.now() - startedAt;
    const avgMs = elapsed / jobsCompleted;
    return Date.now() + avgMs * remainingJobs;
  }

  function getProgressTiming(progress) {
    if (!progress.total) {
      return { countdownLabel: 'Remaining: —', etaLabel: 'ETA: —' };
    }
    if (progress.remaining === 0) {
      return {
        countdownLabel: 'Remaining: 0 jobs · 0:00',
        etaLabel: 'ETA: Done',
      };
    }
    const jobsLabel = `${progress.remaining} job${progress.remaining === 1 ? '' : 's'}`;
    const startedAt = state.sessionStartedAt || 0;
    const jobsCompleted = getSessionJobsCompleted();
    if (!state.running || !startedAt) {
      return {
        countdownLabel: `Remaining: ${jobsLabel}`,
        etaLabel: 'ETA: —',
      };
    }
    if (jobsCompleted < 1) {
      return {
        countdownLabel: `Remaining: ${jobsLabel} · estimating…`,
        etaLabel: 'ETA: calculating…',
      };
    }
    const deadline = resolveSessionEtaDeadline(progress.remaining);
    const countdownMs = Math.max(0, deadline - Date.now());
    return {
      countdownLabel: `Remaining: ${jobsLabel} · ${formatDuration(countdownMs)}`,
      etaLabel: `ETA: ${formatClockTime(new Date(deadline))}`,
    };
  }

  function renderProgressStrip() {
    const textEl = $('#lsh-progress-text');
    const countdownEl = $('#lsh-progress-countdown');
    const etaEl = $('#lsh-progress-eta');
    const fillEl = $('#lsh-progress-fill');
    const errorEl = $('#lsh-progress-error');
    if (!textEl || !fillEl) return;

    const progress = getQueueProgress();
    const pctComplete = progress.total ? (progress.complete / progress.total) * 100 : 0;
    const pctError = progress.total ? (progress.error / progress.total) * 100 : 0;
    const timing = getProgressTiming(progress);

    textEl.innerHTML = progress.total
      ? `<strong>${progress.complete}</strong> / ${progress.total} complete` +
        (progress.error ? ` · <span style="color:#e74c3c">${progress.error} error</span>` : '')
      : '<strong>0</strong> / 0 complete';
    if (countdownEl) countdownEl.textContent = timing.countdownLabel;
    if (etaEl) etaEl.textContent = timing.etaLabel;
    fillEl.style.width = `${Math.min(100, pctComplete)}%`;
    if (errorEl) {
      errorEl.style.left = `${Math.min(100, pctComplete)}%`;
      errorEl.style.width = `${Math.min(100 - pctComplete, pctError)}%`;
    }
  }

  function ensureProgressTicker() {
    if (state.running && !progressTicker) startProgressTicker();
  }

  function startProgressTicker() {
    stopProgressTicker();
    renderProgressStrip();
    progressTicker = setInterval(renderProgressStrip, 1000);
  }

  function stopProgressTicker() {
    if (progressTicker) {
      clearInterval(progressTicker);
      progressTicker = null;
    }
    renderProgressStrip();
  }

  }

  function csvEscape(val) {
    const s = String(val ?? '');
    if (/[",\n\r]/.test(s)) return `"${s.replace(/"/g, '""')}"`;
    return s;
  }

  function findResultForQueueRow(row, index) {
    const pairKey = makePairKey(getRowContainer(row), getRowLpn(row));
    return (
      state.results.find((r) => r.index === index) ||
      state.results.find(
        (r) => makePairKey(getRowContainer(r.row), getRowLpn(r.row) || r.lpn) === pairKey
      ) ||
      null
    );
  }

  function findErrorForQueueRow(row) {
    const pairKey = makePairKey(getRowContainer(row), getRowLpn(row));
    return (state.errorLog || []).find((e) => makePairKey(e.container, e.lpn) === pairKey) || null;
  }

  function buildExportCsv() {
    syncRowStatuses();
    const exportedAt = new Date().toISOString();
    const metaLines = [
      `# Liquidation Sort Helper v${SCRIPT_VERSION}`,
      `# exported_at,${csvEscape(exportedAt)}`,
      `# queue_length,${state.queue.length}`,
      `# queue_index,${state.queueIndex}`,
      `# session_jobs_completed,${state.sessionJobsCompleted || 0}`,
      '',
    ];
    const headers = [
      'queue_index',
      'container',
      'lpn',
      'queue_sort_code',
      'status',
      'result_status',
      'tote_id',
      'result_sort_code',
      'completed_at',
      'error_message',
    ];
    const rows = state.queue.map((row, i) => {
      const result = findResultForQueueRow(row, i);
      const err = findErrorForQueueRow(row);
      return [
        i + 1,
        getRowContainer(row),
        getRowLpn(row),
        row.sortCode || '',
        state.rowStatuses[i] || ROW_STATUS.PENDING,
        result?.status || '',
        result?.toteId || '',
        result?.sortCode || '',
        result?.at || '',
        err?.message || result?.message || '',
      ]
        .map(csvEscape)
        .join(',');
    });
    return metaLines.concat([headers.join(',')], rows).join('\r\n');
  }

  function downloadTextFile(filename, content, mimeType) {
    const blob = new Blob([content], { type: mimeType });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.download = filename;
    link.style.display = 'none';
    document.body.appendChild(link);
    link.click();
    link.remove();
    setTimeout(() => URL.revokeObjectURL(url), 1000);
  }

  function exportResultsCsv() {
    if (!state.queue.length) {
      alert('Queue is empty. Load a queue before exporting.');
      return;
    }
    const csv = buildExportCsv();
    const stamp = new Date().toISOString().replace(/[:.]/g, '-').slice(0, 19);
    downloadTextFile(`liquidation-sort-export-${stamp}.csv`, csv, 'text/csv;charset=utf-8');
    trace('export.csv', { rows: state.queue.length });
  }

  function runSelfChecks() {
    const checks = [];

  function init() {
    trace('init', { version: SCRIPT_VERSION, url: location.href });
    buildPanel();

    if (state.sessionArmed && !state.running) {
      state.resumeRequired = true;
      state.running = false;
      trace('init.sessionArmed', { message: 'Deliberate Resume required; no auto-start' });
    }

    if (!state.running && (state.toteMappingPrompt || state.toteFullPrompt)) {
      state.toteMappingPrompt = null;
      state.toteFullPrompt = null;
      persistState();
      trace('init.clearStalePrompts');
    }

    const queueInput = $('#lsh-queue-input');
    if (queueInput && state.queue.length) {
      queueInput.value = state.queue
        .map((r) => `${r.container},${r.lpn}${r.sortCode ? ',' + r.sortCode : ''}`)
        .join('\n');
    }

    runSelfChecks();
    if (state.queue.length) applyKnownRowsToQueue();
    renderAll();
    trace('init.complete');
  }

    document.addEventListener('DOMContentLoaded', () => setTimeout(init, TIMING.initDelayMs));
