    function initializeToggleDragging(toggle) {
        let pointerId = null;
        let startX = 0;
        let startY = 0;
        let originLeft = 0;
        let originTop = 0;
        let moved = false;

        toggle.addEventListener('pointerdown', function(event) {
            pointerId = event.pointerId;
            startX = event.clientX;
            startY = event.clientY;
            const rect = toggle.getBoundingClientRect();
            originLeft = rect.left;
            originTop = rect.top;
            moved = false;
            toggle.dataset.dragging = 'false';
            toggle.style.left = `${originLeft}px`;
            toggle.style.top = `${originTop}px`;
            toggle.style.right = 'auto';
            toggle.style.bottom = 'auto';
            toggle.setPointerCapture(pointerId);
        });

        toggle.addEventListener('pointermove', function(event) {
            if (pointerId !== event.pointerId) {
                return;
            }

            const deltaX = event.clientX - startX;
            const deltaY = event.clientY - startY;
            if (Math.abs(deltaX) > 4 || Math.abs(deltaY) > 4) {
                moved = true;
                toggle.dataset.dragging = 'true';
            }

            if (!moved) {
                return;
            }

            toggle.style.left = `${originLeft + deltaX}px`;
            toggle.style.top = `${originTop + deltaY}px`;
        });

        function stopDragging(event) {
            if (pointerId !== event.pointerId) {
                return;
            }
            pointerId = null;
            toggle.releasePointerCapture(event.pointerId);
            window.setTimeout(function() {
                toggle.dataset.dragging = 'false';
            }, 0);
        }

        toggle.addEventListener('pointerup', stopDragging);
        toggle.addEventListener('pointercancel', stopDragging);
    }

        const month = String(now.getMonth() + 1).padStart(2, '0');

        return `${year}-${month}-${day}`;
    }

    async function runLookup() {
        const warehouseId = DEFAULT_WAREHOUSE_ID;
        const queryDate = getTodayString();
        const logins = parseLogins(document.getElementById(`${SCRIPT_ID}-logins`).value);

        resetDebug(warehouseId, queryDate, logins);

        if (logins.length === 0) {
            updateStatus('Paste at least one login.', 'error');
            addDebugError('validate', 'No logins provided');
            return;
        }

        updateStatus('Fetching attendance report...');
        renderResults([]);

        try {
            const attendanceRows = await fetchAttendanceRows(warehouseId, queryDate);
            const attendanceIndex = buildAttendanceIndex(attendanceRows);
            addDebugStep('attendanceIndexSummary', {
                rows: attendanceRows.length,
                keys: Object.keys(attendanceIndex.byKey).length,
                uniqueNames: Object.keys(attendanceIndex.byName).length
            });

            updateStatus(`Checking ${logins.length} login(s)...`);

            const resolvedUsers = await mapWithConcurrency(logins, 8, async (login) => {
                try {
                    const user = await lookupUser(login, warehouseId, queryDate);
                    return {
                        login,
                        ok: true,
                        user
                    };
                } catch (error) {
                    addDebugError(`lookupUser:${login}`, error);
                    return {
                        login,
                        ok: false,
                        error: formatError(error)
                    };
                }
            });

            addDebugStep('resolvedUsers', resolvedUsers);

            const results = resolvedUsers.map((entry) => {
                if (!entry.ok) {
                    return {
                        login: entry.login,
                        status: 'Lookup Failed',
                        statusClass: 'unknown',
                        lastPunch: '-',
                        details: entry.error,
                        raw: null
                    };
                }

                const clockResult = determineClockStatus(entry.login, entry.user, attendanceRows, attendanceIndex);
                return {
                    login: entry.login,
                    status: clockResult.statusLabel,
                    statusClass: clockResult.statusClass,
                    lastPunch: clockResult.lastPunchTime || '-',
                    details: clockResult.details,
                    raw: {
                        user: entry.user,
                        matchedRow: clockResult.matchedRow
                    }
                };
            });

            lastDebugPayload.results = results;
            renderResults(results);

            const failures = results.filter((item) => item.statusClass === 'unknown').length;
            updateStatus(
                failures > 0
                    ? `Completed with ${failures} item(s) needing review.`
                    : `Completed for ${results.length} login(s).`,
                failures > 0 ? 'error' : 'success'
            );
        } catch (error) {
            addDebugError('runLookup', error);
            updateStatus(`Lookup failed: ${formatError(error)}`, 'error');
        }
    }

    async function lookupUser(login, warehouseId, queryDate) {
        addDebugStep(`lookupUser:start:${login}`, null);
        const pageHtml = await fetchTimeDetailsPage(login, warehouseId, queryDate);
        const parsedPage = parseTimeDetailsPage(pageHtml, login);
        const employeeId = parsedPage.employeeId || inferEmployeeIdFromCurrentPage(login);
        const displayName = parsedPage.displayName || '';
        const nameVariants = buildNameVariants(displayName);
        const identifiers = buildIdentifierCandidates(login, employeeId, parsedPage.identifiers);

        addDebugStep(`lookupUser:summary:${login}`, {
            login,
            employeeId: employeeId || null,
            displayName: displayName || null,
            identifiers: identifiers.slice(0, 10),
            pageStatus: parsedPage.pageStatus || null,
            lastPunchTime: parsedPage.lastPunchTime || null
        });

        if (!parsedPage.pageFound && !employeeId && nameVariants.length === 0) {
            throw new Error(`FCLM time details page was not found for login ${login}.`);
        }

        return {
            login,
            employeeId,
            displayName,
            nameVariants,
            identifiers,
            pageFound: parsedPage.pageFound,
            pageStatus: parsedPage.pageStatus,
            lastPunchTime: parsedPage.lastPunchTime,
            raw: parsedPage
        };
    }

    async function fetchTimeDetailsPage(login, warehouseId, queryDate) {
        const url = buildTimeDetailsUrl(login, warehouseId, queryDate);
        addDebugStep(`lookupUser:url:${login}`, url);
        return gmRequest(url);
    }

    function buildTimeDetailsUrl(login, warehouseId, queryDate) {
        const url = new URL('/employee/timeDetails', window.location.origin);
        const params = url.searchParams;
        params.set('warehouseId', warehouseId);
        params.set('employeeId', login);

        if (queryDate) {
            const range = buildQueryDateRange(queryDate);
            params.set('startTime', range.startTime);
            params.set('endTime', range.endTime);
        }

        return url.toString();
    }

    function inferEmployeeIdFromCurrentPage(login) {
        const params = new URLSearchParams(window.location.search);
        const pageEmployeeId = (params.get('employeeId') || '').trim();
        if (!pageEmployeeId) {
            return '';
        }

        const bodyText = document.body ? document.body.innerText : '';
        const normalizedLogin = normalizeKey(login);
        if (!normalizedLogin || !normalizeKey(bodyText).includes(normalizedLogin)) {
            return '';
        }

        return pageEmployeeId;
    }

    function buildQueryDateRange(queryDate) {
        const start = new Date(`${queryDate}T00:00:00`);
        const end = new Date(start.getTime());
        end.setDate(end.getDate() + 1);

        return {
            startTime: formatPortalDateTime(start),
            endTime: formatPortalDateTime(end)
        };
    }

    function formatPortalDateTime(date) {
        const year = date.getFullYear();
        const month = String(date.getMonth() + 1).padStart(2, '0');
        const day = String(date.getDate()).padStart(2, '0');
        const hours = String(date.getHours()).padStart(2, '0');
        const minutes = String(date.getMinutes()).padStart(2, '0');
        const seconds = String(date.getSeconds()).padStart(2, '0');
        const offsetMinutes = -date.getTimezoneOffset();
        const sign = offsetMinutes >= 0 ? '+' : '-';
        const offsetHours = String(Math.floor(Math.abs(offsetMinutes) / 60)).padStart(2, '0');
        const offsetRemainder = String(Math.abs(offsetMinutes) % 60).padStart(2, '0');
        return `${year}-${month}-${day}T${hours}:${minutes}:${seconds}${sign}${offsetHours}${offsetRemainder}`;
    }

    function parseTimeDetailsPage(html, login) {
        const doc = new DOMParser().parseFromString(html, 'text/html');
        const bodyText = normalizeKey(doc.body ? doc.body.textContent : '');
        const pageFound = !bodyText.includes('http status 404') && !bodyText.includes('not found');
        const pageStatus = extractClockStatusFromDocument(doc);
        const displayName = extractDisplayNameFromDocument(doc);
        const identifiers = extractIdentifiersFromDocument(doc, login);
        const employeeId = identifiers.find((value) => /^\d{6,12}$/.test(value)) || '';

        return {
            pageFound,
            employeeId,
            displayName,
            identifiers,
            pageStatus,
            lastPunchTime: pageStatus.lastPunchTime || '',
            title: doc.title || ''
        };
    }

    function extractDisplayNameFromDocument(doc) {
        const dtNodes = Array.from(doc.querySelectorAll('dt'));
        for (const dtNode of dtNodes) {
            const label = normalizeKey(dtNode.textContent);
            if (!label.includes('name') && !label.includes('associate')) {
                continue;
            }

            const ddNode = dtNode.nextElementSibling;
            const text = ddNode ? ddNode.textContent.trim() : '';
            if (text) {
                return text;
            }
        }

        const selectors = [
            '.employeeInfo h1',
            '.employee-name',
            '.empName',
            '.employeeInfo dd',
            'h1'
        ];

        for (const selector of selectors) {
            const element = doc.querySelector(selector);
            const text = element ? element.textContent.trim() : '';
            if (text) {
                return text;
            }
        }

        return '';
    }

    function extractIdentifiersFromDocument(doc, login) {
        const candidates = new Set();

        const dtNodes = Array.from(doc.querySelectorAll('dt'));
        for (const dtNode of dtNodes) {
            const label = normalizeKey(dtNode.textContent);
            if (!label.includes('employee id') && !label.includes('badge') && !label.includes('login')) {
                continue;
            }

            const ddNode = dtNode.nextElementSibling;
            const text = ddNode ? ddNode.textContent.trim() : '';
            if (text) {
                candidates.add(text);
            }
        }

        const links = Array.from(doc.querySelectorAll('a[href*="employeeId="]'));
        for (const link of links) {
            try {
                const href = new URL(link.href, window.location.origin);
                const employeeId = (href.searchParams.get('employeeId') || '').trim();
                if (employeeId && employeeId !== login) {
                    candidates.add(employeeId);
                }
            } catch (error) {
                // Ignore malformed URLs in the returned HTML.
            }
        }

        const bodyText = doc.body ? doc.body.textContent : '';
        const numericMatches = bodyText.match(/\b\d{6,12}\b/g) || [];
        numericMatches.slice(0, 100).forEach((value) => candidates.add(value.trim()));

        return Array.from(candidates).filter(Boolean);
    }

    function buildIdentifierCandidates(login, employeeId, extractedIdentifiers) {
        const candidates = new Set();
        [login, employeeId].forEach((value) => {
            const normalized = String(value || '').trim();
            if (normalized) {
                candidates.add(normalized);
            }
        });

        if (Array.isArray(extractedIdentifiers)) {
            extractedIdentifiers.forEach((value) => {
                const normalized = String(value || '').trim();
                if (normalized) {
                    candidates.add(normalized);
                }
            });
        }

        return Array.from(candidates);
    }

    function extractClockStatusFromDocument(doc) {
        const rows = Array.from(doc.querySelectorAll('tr.clock-seg, tr[class*="clock"]'));
        const lastRow = rows.length > 0 ? rows[rows.length - 1] : null;
        if (!lastRow) {
            return {
                statusLabel: '',
                statusClass: 'unknown',
                lastPunchTime: '',
                details: 'No clock segments found on time details page.'
            };
        }

        const className = (lastRow.className || '').toLowerCase();
        const text = normalizeKey(lastRow.textContent);
        const rightAlignCells = Array.from(lastRow.querySelectorAll('td.rightAlign'));
        const lastPunchTime = rightAlignCells.length > 0
            ? rightAlignCells[rightAlignCells.length - 1].textContent.trim()
            : '';

        if (className.includes('on-clock') || className.includes('clockpaid') || className.includes('paid')) {
            return {
                statusLabel: 'Clocked In',
                statusClass: 'in',
                lastPunchTime,
                details: 'Resolved from FCLM time details page.'
            };
        }

        if (className.includes('off-clock') || className.includes('un-paid') || className.includes('clockunpaid')) {
            return {
                statusLabel: 'Clocked Out',
                statusClass: 'out',
                lastPunchTime,
                details: 'Resolved from FCLM time details page.'
            };
        }

        if (text.includes('clocked in') || text.includes(' in ')) {
            return {
                statusLabel: 'Clocked In',
                statusClass: 'in',
                lastPunchTime,
                details: 'Resolved from FCLM time details page text.'
            };
        }

        if (text.includes('clocked out') || text.includes(' out ')) {
            return {
                statusLabel: 'Clocked Out',
                statusClass: 'out',
                lastPunchTime,
                details: 'Resolved from FCLM time details page text.'
            };
        }

        return {
            statusLabel: '',
            statusClass: 'unknown',
            lastPunchTime,
            details: 'Clock segments found, but status could not be inferred.'
        };
    }

    function buildNameVariants(displayName) {
        if (!displayName) {
            return [];
        }

        const variants = new Set();
        const base = normalizeName(displayName);
        const swapped = normalizeName(swapNameOrder(displayName));

        if (base) {
            variants.add(base);
        }

        if (swapped) {
            variants.add(swapped);
        }

        return Array.from(variants);
    }

    function normalizeName(value) {
        return String(value || '')
            .toLowerCase()
            .replace(/[^a-z0-9]+/g, ' ')
            .trim()
            .replace(/\s+/g, ' ');
    }

    function swapNameOrder(value) {
        const trimmed = String(value || '').trim();
        if (!trimmed) {
            return '';
        }

        if (trimmed.includes(',')) {
            const parts = trimmed.split(',').map((part) => part.trim()).filter(Boolean);
            if (parts.length >= 2) {
                return `${parts.slice(1).join(' ')} ${parts[0]}`.trim();
            }
        }

        const words = trimmed.split(/\s+/).filter(Boolean);
        if (words.length >= 2) {
            const first = words.shift();
            return `${words.join(' ')} ${first}`.trim();
        }

        return trimmed;
    }

    async function fetchAttendanceRows(warehouseId, queryDate) {
        const datesToFetch = buildAttendanceDates(queryDate);
        const csvPayloads = await Promise.all(datesToFetch.map(async (dateValue) => {
            const csvText = await fetchAttendanceCsv(warehouseId, dateValue);
            addDebugStep(`attendanceCsvPreview:${dateValue}`, csvText.slice(0, 500));
            return csvToArray(csvText, ',');
        }));

        return csvPayloads.flat();
    }

    function buildAttendanceDates(queryDate) {
        const selectedDate = new Date(`${queryDate}T00:00:00`);
        const previousDate = new Date(selectedDate.getTime());
        previousDate.setDate(previousDate.getDate() - 1);
        return [formatDateOnly(previousDate), queryDate];
    }

    function formatDateOnly(date) {
        const year = date.getFullYear();
        const month = String(date.getMonth() + 1).padStart(2, '0');
        const day = String(date.getDate()).padStart(2, '0');
        return `${year}-${month}-${day}`;
    }

    async function fetchAttendanceCsv(warehouseId, queryDate) {
        const url = `https://fclm-portal.amazon.com/reports/employeeAttendance?reportFormat=CSV&warehouseId=${encodeURIComponent(warehouseId)}&spanType=Day&startDateDay=${encodeURIComponent(queryDate)}`;
        addDebugStep(`fetchAttendanceCsv:url:${queryDate}`, url);
        return gmRequest(url);
    }

    function determineClockStatus(login, user, attendanceRows, attendanceIndex) {
        const identifierCandidates = Array.isArray(user.identifiers) && user.identifiers.length > 0
            ? user.identifiers
            : [user.employeeId, login];
        const candidateKeys = identifierCandidates.map(normalizeKey).filter(Boolean);
        const candidateNames = Array.isArray(user.nameVariants) ? user.nameVariants : [];

        let matchedPunch = null;

        candidateKeys.forEach((key) => {
            const indexed = attendanceIndex.byKey[key];
            if (indexed && (!matchedPunch || indexed.timestamp > matchedPunch.timestamp)) {
                matchedPunch = indexed;
            }
        });

        candidateNames.forEach((nameKey) => {
            const indexed = attendanceIndex.byName[nameKey];
            if (indexed && (!matchedPunch || indexed.timestamp > matchedPunch.timestamp)) {
                matchedPunch = indexed;
            }
        });

        if (!matchedPunch) {
            matchedPunch = scanRowsForMatch(attendanceRows, login, user.employeeId, candidateNames);
        }

        if (!matchedPunch) {
            return {
                statusLabel: user.pageFound === false ? 'Lookup Failed' : 'Clocked Out',
                statusClass: user.pageFound === false ? 'unknown' : 'out',
                lastPunchTime: '',
                details: user.pageFound === false
                    ? `Could not load time details for ${login}.`
                    : `No recent attendance punch matched login ${login}${user.employeeId ? ` / employee ${user.employeeId}` : ''}${user.displayName ? ` / name ${user.displayName}` : ''}.`,
                matchedRow: null
            };
        }

        const normalizedPunchType = normalizePunchType(matchedPunch.type);
        const recentPunch = isWithinLastHours(matchedPunch.timestamp, 24);
        const details = recentPunch && normalizedPunchType.className === 'out'
            ? `${matchedPunch.type || 'Unknown punch'} via ${matchedPunch.sourceKey} (recent punch, not actively clocked in)`
            : `${matchedPunch.type || 'Unknown punch'}${matchedPunch.sourceKey ? ` via ${matchedPunch.sourceKey}` : ''}`;

        return {
            statusLabel: normalizedPunchType.label,
            statusClass: normalizedPunchType.className,
            lastPunchTime: matchedPunch.displayTime,
            details,
            matchedRow: matchedPunch.row
        };
    }

    function buildAttendanceIndex(rows) {
        const byKey = {};
        const byName = {};

        rows.forEach((row) => {
            if (!Array.isArray(row) || row.length < 6) {
                return;
            }

            const employeeId = String(row[0] || '').trim();
            const employeeName = String(row[1] || '').trim();
            const punchType = String(row[4] || '').trim();
            const punchTime = String(row[5] || '').trim();

            if (!employeeId || !punchTime) {
                return;
            }

            const timestamp = new Date(punchTime).getTime();
            if (Number.isNaN(timestamp)) {
                return;
            }

            const normalizedEmployeeId = normalizeKey(employeeId);
            const record = {
                type: punchType,
                timestamp,
                displayTime: punchTime,
                row,
                sourceKey: employeeId
            };

            if (!byKey[normalizedEmployeeId] || timestamp > byKey[normalizedEmployeeId].timestamp) {
                byKey[normalizedEmployeeId] = record;
            }

            const normalizedName = normalizeName(employeeName);
            if (normalizedName && (!byName[normalizedName] || timestamp > byName[normalizedName].timestamp)) {
                byName[normalizedName] = {
                    ...record,
                    sourceKey: employeeName
                };
            }

            const swappedName = normalizeName(swapNameOrder(employeeName));
            if (swappedName && (!byName[swappedName] || timestamp > byName[swappedName].timestamp)) {
                byName[swappedName] = {
                    ...record,
                    sourceKey: employeeName
                };
            }

            row.forEach((cell) => {
                const normalizedCell = normalizeKey(cell);
                if (!normalizedCell) {
                    return;
                }

                if (!byKey[normalizedCell] || timestamp > byKey[normalizedCell].timestamp) {
                    byKey[normalizedCell] = {
                        ...record,
                        sourceKey: String(cell).trim()
                    };
                }
            });
        });

        return { byKey, byName };
    }

    function scanRowsForMatch(rows, login, employeeId, candidateNames) {
        const normalizedLogin = normalizeKey(login);
        const normalizedEmployeeId = normalizeKey(employeeId);
        const normalizedNames = Array.isArray(candidateNames) ? candidateNames : [];
        let matchedPunch = null;

        rows.forEach((row) => {
            if (!Array.isArray(row) || row.length < 6) {
                return;
            }

            const normalizedCells = row.map(normalizeKey);
            const normalizedRowName = normalizeName(row[1] || '');
            const isNameMatch = normalizedNames.includes(normalizedRowName);
            const isMatch = normalizedCells.includes(normalizedLogin) || normalizedCells.includes(normalizedEmployeeId) || isNameMatch;
            if (!isMatch) {
                return;
            }

            const punchType = String(row[4] || '').trim();
            const punchTime = String(row[5] || '').trim();
            const timestamp = new Date(punchTime).getTime();

            if (Number.isNaN(timestamp)) {
                return;
            }

            if (!matchedPunch || timestamp > matchedPunch.timestamp) {
                matchedPunch = {
                    type: punchType,
                    timestamp,
                    displayTime: punchTime,
                    row,
                    sourceKey: normalizedCells.includes(normalizedEmployeeId)
                        ? employeeId
                        : (isNameMatch ? row[1] : login)
                };
            }
        });

        return matchedPunch;
    }

    function normalizePunchType(punchType) {
        const raw = String(punchType || '').toLowerCase();

        if (raw.includes('out') || raw.includes('meal start')) {
            return {
                label: 'Clocked Out',
                className: 'out'
            };
        }

        if (raw.includes('in') || raw.includes('meal end') || raw.includes('return')) {
            return {
                label: 'Clocked In',
                className: 'in'
            };
        }

        return {
            label: punchType ? `Unknown (${punchType})` : 'Unknown',
            className: 'unknown'
        };
    }

    function isWithinLastHours(timestamp, hours) {
        const now = Date.now();
        const diffMs = now - timestamp;
        return diffMs >= 0 && diffMs <= hours * 60 * 60 * 1000;
    }

    function normalizeKey(value) {
        return String(value || '').trim().toLowerCase();
    }

    function renderResults(results) {
        const body = document.getElementById(`${SCRIPT_ID}-results-body`);
        if (!body) {
            return;
        }

        if (!results || results.length === 0) {
            body.innerHTML = '<tr><td colspan="3">No results yet.</td></tr>';
            return;
        }

        body.innerHTML = results.map((result) => `
            <tr>
                <td>${escapeHtml(result.login)}</td>
                <td><span class="fclm-pill ${escapeHtml(result.statusClass)}">${escapeHtml(result.status)}</span></td>
                <td>${escapeHtml(result.lastPunch || '-')}</td>
            </tr>
        `).join('');
    }

    async function copyDebugToClipboard() {
        lastDebugPayload.generatedAt = new Date().toISOString();
        const text = JSON.stringify(lastDebugPayload, null, 2);

        try {
            await writeClipboard(text);
            updateStatus('Debug copied to clipboard.', 'success');
        } catch (error) {
            addDebugError('copyDebugToClipboard', error);
            updateStatus(`Copy failed: ${formatError(error)}`, 'error');
        }
    }

    async function copyClockedInLogins() {
        const clockedInLogins = (lastDebugPayload.results || [])
            .filter((result) => result && result.statusClass === 'in')
            .map((result) => result.login)
            .filter(Boolean);

        if (clockedInLogins.length === 0) {
            updateStatus('No Clocked In logins to copy.', 'error');
            return;
        }

        try {
            await writeClipboard(clockedInLogins.join('\n'));
            updateStatus(`Copied ${clockedInLogins.length} Clocked In login(s).`, 'success');
        } catch (error) {
            addDebugError('copyClockedInLogins', error);
            updateStatus(`Copy failed: ${formatError(error)}`, 'error');
        }
    }

    function writeClipboard(text) {
        if (navigator.clipboard && navigator.clipboard.writeText) {
            return navigator.clipboard.writeText(text);
        }

        return new Promise((resolve, reject) => {
            try {
                const textarea = document.createElement('textarea');
                textarea.value = text;
                textarea.setAttribute('readonly', 'readonly');
                textarea.style.position = 'fixed';
                textarea.style.top = '-9999px';
                document.body.appendChild(textarea);
                textarea.select();
                const ok = document.execCommand('copy');
                document.body.removeChild(textarea);

                if (!ok) {
                    reject(new Error('document.execCommand("copy") returned false.'));
                    return;
                }

                resolve();
            } catch (error) {
                reject(error);
            }
        });
    }

    function gmRequest(url) {
        return new Promise((resolve, reject) => {
            GM_xmlhttpRequest({
                method: 'GET',
                url,
                onload: function(response) {
                    if (response.status >= 200 && response.status < 300) {
                        resolve(response.responseText);
                        return;
                    }

                    reject(new Error(`Request failed (${response.status}) for ${url}`));
                },
                onerror: function(error) {
                    reject(new Error(`Network error for ${url}: ${JSON.stringify(error)}`));
                }
            });
        });
    }

    async function mapWithConcurrency(items, concurrency, mapper) {
        const results = new Array(items.length);
        let nextIndex = 0;

        async function worker() {
            while (nextIndex < items.length) {
                const currentIndex = nextIndex;
                nextIndex += 1;
                results[currentIndex] = await mapper(items[currentIndex], currentIndex);
            }
        }

        const workerCount = Math.min(concurrency, items.length);
        await Promise.all(Array.from({ length: workerCount }, worker));
        return results;
    }

    function csvToArray(strData, strDelimiter) {
        const delimiter = strDelimiter || ',';
        const objPattern = new RegExp(
            (
                '(\\' + delimiter + '|\\r?\\n|\\r|^)' +
                '(?:"([^"]*(?:""[^"]*)*)"|' +
                '([^"\\' + delimiter + '\\r\\n]*))'
            ),
            'gi'
        );
        const arrData = [[]];
        let arrMatches;

        while ((arrMatches = objPattern.exec(strData))) {
            const strMatchedDelimiter = arrMatches[1];
            if (strMatchedDelimiter.length && strMatchedDelimiter !== delimiter) {
                arrData.push([]);
            }

            const strMatchedValue = arrMatches[2]
                ? arrMatches[2].replace(/""/g, '"')
                : arrMatches[3];

            arrData[arrData.length - 1].push(strMatchedValue);
        }

        return arrData;
    }

    function escapeHtml(value) {
        return String(value || '')
            .replace(/&/g, '&amp;')
            .replace(/</g, '&lt;')
            .replace(/>/g, '&gt;')
            .replace(/"/g, '&quot;')
            .replace(/'/g, '&#39;');
    }

    function formatError(error) {
        if (error instanceof Error) {
            return error.message;
        }

        if (typeof error === 'string') {
            return error;
        }

        try {
            return JSON.stringify(error);
        } catch (jsonError) {
            return String(error);
        }
    }
