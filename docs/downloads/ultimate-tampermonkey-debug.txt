	async function copyText(text) {
		if (typeof GM_setClipboard === "function") {
			GM_setClipboard(text, "text");
			return;
		}
		if (navigator.clipboard?.writeText) {
			await navigator.clipboard.writeText(text);
			return;
		}
		const textarea = document.createElement("textarea");
		textarea.value = text;
		textarea.style.position = "fixed";
		textarea.style.opacity = "0";
		document.body.appendChild(textarea);
		textarea.select();
		document.execCommand("copy");
		textarea.remove();
	}

	async function waitForAccuracySnapshot(previousSnapshot) {
		const startedAt = Date.now();
		const summaryRoot = getVisualBlock(summaryVisualTitle)?.parentElement ?? document.body;
		let mutationCount = 0;
		let lastKey = "";
		let stableCount = 0;
		const observer = new MutationObserver(() => {
			mutationCount += 1;
		});
		observer.observe(summaryRoot, {
			childList: true,
			subtree: true,
			characterData: true,
			attributes: true
		});
		try {
			await delay(400);
			while (Date.now() - startedAt < 12e3) {
				const snapshot = readAccuracySnapshot();
				if (snapshot) {
					const key = `${snapshot.sellableAccuracy}|${snapshot.unsellableAccuracy}`;
					if (key === lastKey) stableCount += 1;
					else {
						lastKey = key;
						stableCount = 1;
					}
					const previousKey = previousSnapshot ? `${previousSnapshot.sellableAccuracy}|${previousSnapshot.unsellableAccuracy}` : "";
					const changedFromPrevious = !previousSnapshot || key !== previousKey;
					if (stableCount >= 2 && (changedFromPrevious || mutationCount > 0)) return {
						snapshot,
						mutationCount,
						changedFromPrevious,
						timedOut: false
					};
				}
				await delay(150);
			}
			return {
				snapshot: readAccuracySnapshot(),
				mutationCount,
				changedFromPrevious: false,
				timedOut: true
			};
		} finally {
			observer.disconnect();
		}
	}

	function getApplicationTargets(cell) {
		if (!cell) return [];
		const targets = [];
		let cursor = cell;
		while (cursor) {
			if (cursor.getAttribute("role") === "application" && !targets.includes(cursor)) targets.push(cursor);
			cursor = cursor.parentElement;
		}
		return targets;
	}
	async function runKeyboardStrategies(target, previousSnapshot) {
		const cell = getLoginCell(target);
		if (!cell) return {
			strategy: "missing-cell",
			snapshot: null,
			changedFromPrevious: false,
			selectionEvidence: false,
			timedOut: false
		};
		const beforeCellState = getCellState(cell);
		const strategies = [{
			label: "cell-key-enter",
			run: async () => {
				await attemptKeyboardSelection(cell, "Enter");
			}
		}, {
			label: "cell-key-space",
			run: async () => {
				await attemptKeyboardSelection(cell, " ");
			}
		}];
		for (const appTarget of getApplicationTargets(cell)) {
			strategies.push({
				label: "app-key-enter",
				run: async () => {
					await attemptKeyboardSelection(appTarget, "Enter");
				}
			});
			strategies.push({
				label: "app-key-space",
				run: async () => {
					await attemptKeyboardSelection(appTarget, " ");
				}
			});
		}
		let bestSameValueResult = null;
		for (const strategy of strategies) {
			await strategy.run();
			const waitResult = await waitForAccuracySnapshot(previousSnapshot);
			const selectionEvidence = beforeCellState !== getCellState(getLoginCell(target));
			if (waitResult.snapshot && waitResult.changedFromPrevious) return {
				strategy: strategy.label,
				snapshot: waitResult.snapshot,
				changedFromPrevious: true,
				selectionEvidence,
				timedOut: waitResult.timedOut
			};
			if (!bestSameValueResult && waitResult.snapshot && selectionEvidence && !waitResult.timedOut) bestSameValueResult = {
				strategy: strategy.label,
				snapshot: waitResult.snapshot,
				changedFromPrevious: false,
				selectionEvidence: true,
				timedOut: false
			};
		}
		return bestSameValueResult ?? {
			strategy: "no-keyboard-strategy",
			snapshot: readAccuracySnapshot(),
			changedFromPrevious: false,
			selectionEvidence: false,
			timedOut: false
		};
	}
	async function verifySameValueCapture(loginTargets, results, target, candidateSnapshot) {
		const candidateKey = getSnapshotKey(candidateSnapshot);
		const anchorResult = results.find((result) => {
			const snapshot = parseSnapshotText(result.snapshot);
			return snapshot && parsePercentValue(snapshot.sellableAccuracy) !== null && parsePercentValue(snapshot.unsellableAccuracy) !== null && getSnapshotKey(snapshot) !== candidateKey;
		});
		if (!anchorResult) return false;
		const anchorTarget = loginTargets.find((item) => item.login === anchorResult.login);
		if (!anchorTarget) return false;
		const movedAway = await runKeyboardStrategies(anchorTarget, candidateSnapshot);
		if (!movedAway.snapshot || !movedAway.changedFromPrevious) return false;
		const returned = await runKeyboardStrategies(target, movedAway.snapshot);
		return getSnapshotKey(returned.snapshot) === candidateKey && returned.changedFromPrevious;
	}

							mutationCount: 0,
							changedFromPrevious: false,
							timedOut: false,
							snapshot: snapshotToText(readAccuracySnapshot()),
							outcome: `error:${message}`
						});
						break;
					}
				}
				lastRunContext.finalSnapshot = snapshotToText(readAccuracySnapshot());
				pushLog("Automated debug run complete.");
				processingClick = false;
				render();
			})().catch((error) => {
				lastRunContext.fatalError = getErrorMessage(error);
				lastRunContext.finalSnapshot = snapshotToText(readAccuracySnapshot());
				pushLog(`Run failed: ${getErrorMessage(error)}`);
				processingClick = false;
				render();
			});
		});
		copyButton.addEventListener("click", () => {
			copyText(buildDebugReport(logs, lastRunContext)).catch((error) => {
				pushLog(`Copy failed: ${error instanceof Error ? error.message : String(error)}`);
			});
		});
		icon.addEventListener("click", () => {
			if (movedDuringPointer) {
				movedDuringPointer = false;
				return;
			}
			panelOpen = !panelOpen;
			render();
		});
		closeButton.addEventListener("click", () => {
			panelOpen = false;
			render();
		});
		icon.addEventListener("pointerdown", (event) => {
			isDragging = true;
			movedDuringPointer = false;
			pointerOffsetX = event.clientX - position.x;
			pointerOffsetY = event.clientY - position.y;
			icon.setPointerCapture(event.pointerId);
		});
		icon.addEventListener("pointermove", (event) => {
			if (!isDragging) return;
			const nextX = clamp(event.clientX - pointerOffsetX, 12, window.innerWidth - 56);
			const nextY = clamp(event.clientY - pointerOffsetY, 12, window.innerHeight - 56);
			if (Math.abs(nextX - position.x) > 2 || Math.abs(nextY - position.y) > 2) movedDuringPointer = true;
			position = {
				x: nextX,
				y: nextY
			};
			applyIconPosition();
			if (panelOpen) positionPanel();
		});
		icon.addEventListener("pointerup", (event) => {
			if (!isDragging) return;
			isDragging = false;
			icon.releasePointerCapture(event.pointerId);
		});
		icon.addEventListener("pointercancel", () => {
			isDragging = false;
		});
		document.addEventListener("click", (event) => {
			if (!panelOpen) return;
			const targetNode = event.target;
			if (targetNode && (panel.contains(targetNode) || icon.contains(targetNode))) return;
			panelOpen = false;
			render();
		});
		document.addEventListener("keydown", (event) => {
			if (event.key === "Escape") {
				panelOpen = false;
				render();
			}
		});
		window.addEventListener("resize", () => {
			position = {
				x: clamp(position.x, 12, window.innerWidth - 56),
				y: clamp(position.y, 12, window.innerHeight - 56)
			};
			applyIconPosition();
			if (panelOpen) positionPanel();
		});
		render();
		document.body.append(icon, panel);
	}
	onDocumentReady(() => {
		installDebugger();
	});
	//#endregion
})();
