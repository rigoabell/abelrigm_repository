// ==UserScript==
// @name         Apollo Audit Login Helper 28796
// @namespace    https://apollo-audit.corp.amazon.com/
// @version      1.0.1
// @description  Adds a lightweight runner panel to Apollo audit 28796.
// @match        https://apollo-audit.corp.amazon.com/audits/new*
// @grant        none
// @run-at       document-idle
// ==/UserScript==

(function () {
  "use strict";

  const STORAGE_KEY = "apolloAuditSavedLogins28796";
  const COMMENT_STORAGE_KEY = "apolloAuditSavedComments28796";
  const PANEL_ID = "apollo-audit-login-helper-28796";
  const TARGET_AUDIT_TYPE_ID = "28796";
  const LOGIN_INPUT_ID = "apollo-employee-login";
  const INPUT_WATCH_FLAG = "apolloAuditHelperWatcherBound28796";
  const ESCALATION_INPUT_ID = "audit_properties_Associate's Audit Escalation Rate";
  const UNDERSTANDS_IMPORTANCE_SELECT_ID = "audit_properties_Engagement Purpose – Does associate understand the importance of correct Item Match evaluation";
  const ENGAGEMENT_NOTES_INPUT_ID = "audit_properties_Engagement Notes";
  const LPN_INPUT_IDS = [
    "audit_properties_LPN 1",
    "audit_properties_LPN 2",
    "audit_properties_LPN 3"
  ];
  const LPN_PREFIX = "LPNN";
  const LPN_RANDOM_CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
  const POST_FILL_SUBMIT_DELAY_MS = 1;

  function normalizeLogin(value) {
    return String(value || "").trim().toLowerCase();
  }

  function loadLogins() {
    try {
      const parsed = JSON.parse(localStorage.getItem(STORAGE_KEY) || "[]");
      if (!Array.isArray(parsed)) {
        return [];
      }

      return parsed
        .map((entry) => {
          if (typeof entry === "string") {
            const normalized = normalizeLogin(entry);
            return normalized ? { login: normalized, audited: false } : null;
          }

          if (!entry || typeof entry !== "object") {
            return null;
          }

          const normalized = normalizeLogin(entry.login);
          if (!normalized) {
            return null;
          }

          return {
            login: normalized,
            audited: Boolean(entry.audited)
          };
        })
        .filter(Boolean);
    } catch (error) {
      console.warn("Apollo Audit Helper 28796: failed to read saved logins", error);
      return [];
    }
  }

  function saveLogins(logins) {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(logins));
  }

  function loadComments() {
    return String(localStorage.getItem(COMMENT_STORAGE_KEY) || "");
  }

  function saveComments(value) {
    localStorage.setItem(COMMENT_STORAGE_KEY, String(value || ""));
  }

  function getLoginInput() {
    return document.getElementById(LOGIN_INPUT_ID);
  }

  function dispatchFieldEvents(element) {
    ["input", "change", "blur"].forEach((eventName) => {
      element.dispatchEvent(new Event(eventName, { bubbles: true }));
    });
  }

  function chooseRandomValue(values) {
    if (!values.length) {
      return "";
    }

    const randomIndex = Math.floor(Math.random() * values.length);
    return values[randomIndex];
  }

  function chooseRandomSubset(values, minCount, maxCount) {
    if (!values.length) {
      return [];
    }

    const shuffled = [...values];
    for (let index = shuffled.length - 1; index > 0; index -= 1) {
      const swapIndex = Math.floor(Math.random() * (index + 1));
      const current = shuffled[index];
      shuffled[index] = shuffled[swapIndex];
      shuffled[swapIndex] = current;
    }

    const safeMin = Math.max(1, Math.min(minCount, shuffled.length));
    const safeMax = Math.max(safeMin, Math.min(maxCount, shuffled.length));
    const count = safeMin + Math.floor(Math.random() * (safeMax - safeMin + 1));
    return shuffled.slice(0, count);
  }

  function setInputValueById(inputId, value) {
    const input = document.getElementById(inputId);
    if (!input) {
      return;
    }

    input.value = value;
    dispatchFieldEvents(input);
  }

  function setSelectValue(selectId, value) {
    const select = document.getElementById(selectId);
    if (!select) {
      return "";
    }

    select.value = value;
    dispatchFieldEvents(select);
    return select.value;
  }

  function randomFromChars(length) {
    let value = "";

    for (let index = 0; index < length; index += 1) {
      const randomIndex = Math.floor(Math.random() * LPN_RANDOM_CHARS.length);
      value += LPN_RANDOM_CHARS[randomIndex];
    }

    return value;
  }

  function generateRandomLpn() {
    return `${LPN_PREFIX}${randomFromChars(9)}`;
  }

  function generateRandomEscalationRate() {
    const min = 10;
    const max = 90;
    return String(Math.floor(Math.random() * (max - min + 1)) + min);
  }

  function fillRandomLpns() {
    LPN_INPUT_IDS.forEach((inputId) => {
      setInputValueById(inputId, generateRandomLpn());
    });
  }

  function fillLogin(login) {
    const input = getLoginInput();
    if (!input) {
      alert("Apollo Audit Helper 28796: associate login field was not found.");
      return;
    }

    input.value = login;
    dispatchFieldEvents(input);
    input.focus();
    input.select();
  }

  function addLogins(loginsToAdd) {
    const existing = loadLogins();
    const merged = [...existing];
    let addedCount = 0;

    loginsToAdd.forEach((login) => {
      const normalized = normalizeLogin(login);
      if (!normalized || merged.some((entry) => entry.login === normalized)) {
        return;
      }

      merged.unshift({ login: normalized, audited: false });
      addedCount += 1;
    });

    if (addedCount > 0) {
      saveLogins(merged);
    }

    return addedCount;
  }

  function addLogin(login) {
    return addLogins([login]) > 0;
  }

  function clearLogins() {
    saveLogins([]);
  }

  function markLoginAudited(login, audited) {
    const normalized = normalizeLogin(login);
    if (!normalized) {
      return;
    }

    const updated = loadLogins().map((entry) => {
      if (entry.login !== normalized) {
        return entry;
      }

      return {
        login: entry.login,
        audited: Boolean(audited)
      };
    });

    saveLogins(updated);
  }

  function getNextUnauditedLogin() {
    return loadLogins().find((entry) => !entry.audited) || null;
  }

  function parseLoginList(value) {
    return String(value || "")
      .split(/[\s,;]+/)
      .map(normalizeLogin)
      .filter(Boolean);
  }

  function parseCommentList(value) {
    return String(value || "")
      .split(/\r?\n/)
      .map((entry) => entry.trim())
      .filter(Boolean);
  }

  function fillRandomComment() {
    const notes = parseCommentList(loadComments());
    const randomComment = chooseRandomValue(notes);
    if (!randomComment) {
      return;
    }

    setInputValueById(ENGAGEMENT_NOTES_INPUT_ID, randomComment);
  }

  async function copyTextToClipboard(text) {
    if (navigator.clipboard && typeof navigator.clipboard.writeText === "function") {
      await navigator.clipboard.writeText(text);
      return;
    }

    const tempTextArea = document.createElement("textarea");
    tempTextArea.value = text;
    tempTextArea.style.position = "fixed";
    tempTextArea.style.opacity = "0";
    document.body.appendChild(tempTextArea);
    tempTextArea.focus();
    tempTextArea.select();
    document.execCommand("copy");
    document.body.removeChild(tempTextArea);
  }

  function getLoginsByAuditState(audited) {
    return loadLogins()
      .filter((entry) => entry.audited === audited)
      .map((entry) => entry.login);
  }

  async function copyLoginsByAuditState(audited) {
    const logins = getLoginsByAuditState(audited);
    if (logins.length === 0) {
      alert(`Apollo Audit Helper 28796: no ${audited ? "audited" : "unaudited"} logins to copy.`);
      return;
    }

    try {
      await copyTextToClipboard(logins.join("\n"));
    } catch (error) {
      alert("Apollo Audit Helper 28796: failed to copy logins.");
    }
  }

  function escapeForRegex(value) {
    return value.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
  }

  function normalizeLabelText(value) {
    return String(value || "")
      .replace(/\s+/g, " ")
      .replace(/[’]/g, "'")
      .trim();
  }

  function findFormGroupByLabelText(labelText) {
    const normalizedTarget = normalizeLabelText(labelText);
    const labels = Array.from(document.querySelectorAll("#audits-form label"));

    const matchedLabel = labels.find((label) => {
      const text = normalizeLabelText(label.textContent);
      return text === normalizedTarget || text.includes(normalizedTarget);
    });

    return matchedLabel ? matchedLabel.closest(".form-group") : null;
  }

  function getFormGroup(questionText, fallbackSelector) {
    return findFormGroupByLabelText(questionText) || document.querySelector(fallbackSelector);
  }

  function fillEscalationRate() {
    setInputValueById(ESCALATION_INPUT_ID, generateRandomEscalationRate());
  }

  function renderPanel(panel) {
    const list = panel.querySelector(".apollo-helper-list");
    const input = panel.querySelector(".apollo-helper-input");
    const savedLogins = loadLogins();
    const currentLogin = normalizeLogin(getLoginInput()?.value);
    const progress = panel.querySelector(".apollo-helper-progress");
    const auditedCount = savedLogins.filter((entry) => entry.audited).length;
    const totalCount = savedLogins.length;
    const percentAudited = totalCount === 0 ? 0 : Math.round((auditedCount / totalCount) * 100);

    progress.textContent = `${percentAudited}% audited (${auditedCount}/${totalCount})`;
    list.innerHTML = "";

    if (savedLogins.length === 0) {
      const emptyState = document.createElement("div");
      emptyState.className = "apollo-helper-item";
      emptyState.textContent = "No saved logins yet.";
      list.appendChild(emptyState);
    } else {
      savedLogins.forEach((entry) => {
        const item = document.createElement("div");
        item.className = "apollo-helper-item";
        if (entry.login === currentLogin) {
          item.classList.add("is-current");
        }

        const loginLine = document.createElement("div");
        loginLine.className = "apollo-helper-login";
        loginLine.textContent = entry.login;
        loginLine.addEventListener("click", () => fillLogin(entry.login));

        const stateLine = document.createElement("div");
        stateLine.className = "apollo-helper-state";
        stateLine.textContent = entry.audited ? "audited" : "unaudited";

        item.append(loginLine, stateLine);
        list.appendChild(item);
      });
    }

    input.value = "";
  }

  function getFieldLabelText(field) {
    if (!field) {
      return "";
    }

    const id = field.id;
    if (id) {
      const directLabel = document.querySelector(`label[for="${CSS.escape(id)}"]`);
      if (directLabel) {
        return normalizeLabelText(directLabel.textContent);
      }
    }

    const formGroup = field.closest(".form-group");
    if (!formGroup) {
      return field.name || field.id || field.type || "unknown field";
    }

    const groupLabel = formGroup.querySelector("label.form-control-label, label.form-control-label.control-label");
    if (groupLabel) {
      return normalizeLabelText(groupLabel.textContent);
    }

    return field.name || field.id || field.type || "unknown field";
  }

  function getInvalidFieldSummary(form) {
    const fields = Array.from(form.querySelectorAll("input, select, textarea"));
    const invalidFields = fields.filter((field) => {
      if (field.disabled) {
        return false;
      }

      return typeof field.checkValidity === "function" && !field.checkValidity();
    });

    const uniqueSummaries = [];
    invalidFields.forEach((field) => {
      const summary = getFieldLabelText(field);
      if (!uniqueSummaries.includes(summary)) {
        uniqueSummaries.push(summary);
      }
    });

    return uniqueSummaries;
  }

  function hookFormAutosave(panel) {
    const submitButton = document.getElementById("audit-submit");
    if (!submitButton || submitButton.dataset.apolloAuditHelperBound28796 === "true") {
      return;
    }

    submitButton.dataset.apolloAuditHelperBound28796 = "true";
    submitButton.addEventListener("click", () => {
      const loginInput = getLoginInput();
      if (loginInput && normalizeLogin(loginInput.value)) {
        addLogin(loginInput.value);
        markLoginAudited(loginInput.value, true);
        renderPanel(panel);
      }
    });
  }

  function hookLoginSync(panel) {
    const loginInput = getLoginInput();
    if (!loginInput || loginInput.dataset[INPUT_WATCH_FLAG] === "true") {
      return;
    }

    loginInput.dataset[INPUT_WATCH_FLAG] = "true";
    ["input", "change", "blur"].forEach((eventName) => {
      loginInput.addEventListener(eventName, () => renderPanel(panel));
    });
  }

  function submitForm() {
    const submitButton = document.getElementById("audit-submit");
    if (submitButton) {
      if (document.activeElement && typeof document.activeElement.blur === "function") {
        document.activeElement.blur();
      }
      submitButton.click();
      return;
    }

    alert("Apollo Audit Helper 28796: submit action was not found.");
  }

  function runNextAudit(panel) {
    const nextEntry = getNextUnauditedLogin();
    if (!nextEntry) {
      alert("Apollo Audit Helper 28796: no unaudited saved logins are available.");
      return;
    }

    fillLogin(nextEntry.login);
    fillEscalationRate();
    setSelectValue(UNDERSTANDS_IMPORTANCE_SELECT_ID, "Yes");
    fillRandomLpns();
    fillRandomComment();
    markLoginAudited(nextEntry.login, true);
    renderPanel(panel);

    window.setTimeout(() => {
      submitForm();
    }, POST_FILL_SUBMIT_DELAY_MS);
  }

  function createPanel() {
    if (document.getElementById(PANEL_ID)) {
      return;
    }

    buildStyles();

    const panel = document.createElement("aside");
    panel.id = PANEL_ID;
    panel.innerHTML = `
      <div class="apollo-helper-header">
        <h2 class="apollo-helper-title">Saved Logins</h2>
        <p class="apollo-helper-subtitle">Press Enter to save one or many logins.</p>
        <div class="apollo-helper-progress"></div>
      </div>
      <input class="apollo-helper-input" type="text" placeholder="Paste logins, then press Enter" />
      <textarea class="apollo-helper-comments" placeholder="One comment per line. RUN will pick one at random."></textarea>
      <button class="apollo-helper-button apollo-helper-copy apollo-helper-copy-audited" type="button">Copy Audited</button>
      <button class="apollo-helper-button apollo-helper-copy apollo-helper-copy-unaudited" type="button">Copy Unaudited</button>
      <button class="apollo-helper-button apollo-helper-run" type="button">RUN</button>
      <button class="apollo-helper-button apollo-helper-clear" type="button">Wipe Memory</button>
      <div class="apollo-helper-list"></div>
    `;

    const helperInput = panel.querySelector(".apollo-helper-input");
    const commentsInput = panel.querySelector(".apollo-helper-comments");
    const copyAuditedButton = panel.querySelector(".apollo-helper-copy-audited");
    const copyUnauditedButton = panel.querySelector(".apollo-helper-copy-unaudited");
    const runButton = panel.querySelector(".apollo-helper-run");
    const clearButton = panel.querySelector(".apollo-helper-clear");

    commentsInput.value = loadComments();

    helperInput.addEventListener("keydown", (event) => {
      if (event.key === "Enter") {
        event.preventDefault();
        const addedCount = addLogins(parseLoginList(helperInput.value));
        if (addedCount > 0) {
          renderPanel(panel);
        }
      }
    });

    commentsInput.addEventListener("input", () => {
      saveComments(commentsInput.value);
    });

    copyAuditedButton.addEventListener("click", () => {
      void copyLoginsByAuditState(true);
    });

    copyUnauditedButton.addEventListener("click", () => {
      void copyLoginsByAuditState(false);
    });

    runButton.addEventListener("click", () => runNextAudit(panel));

    clearButton.addEventListener("click", () => {
      if (!window.confirm("Wipe all saved login history from this browser?")) {
        return;
      }

      clearLogins();
      renderPanel(panel);
    });

    document.body.appendChild(panel);
    renderPanel(panel);
    hookFormAutosave(panel);
    hookLoginSync(panel);
  }

  function init() {
    const auditTypeId = new URLSearchParams(window.location.search).get("audit_type_id");
    if (window.location.pathname !== "/audits/new" || auditTypeId !== TARGET_AUDIT_TYPE_ID) {
      return;
    }

    createPanel();
  }

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", init, { once: true });
  } else {
    init();
  }
})();
