  function parseLogins(rawValue) {
    return Array.from(
      new Set(
        String(rawValue || '')
          .split(/[\s,;]+/)
          .map((value) => value.trim().toLowerCase())
          .filter(Boolean)
      )
    );
  }

  function pad2(n) {
    return String(n).padStart(2, '0');
  }

  function formatPortalDateTime(date) {
    const offsetMinutes = -date.getTimezoneOffset();
    const sign = offsetMinutes >= 0 ? '+' : '-';
    const offsetHours = String(Math.floor(Math.abs(offsetMinutes) / 60)).padStart(2, '0');
    const offsetRemainder = String(Math.abs(offsetMinutes) % 60).padStart(2, '0');
    return (
      `${date.getFullYear()}-${pad2(date.getMonth() + 1)}-${pad2(date.getDate())}` +
      `T${pad2(date.getHours())}:${pad2(date.getMinutes())}:${pad2(date.getSeconds())}` +
      `${sign}${offsetHours}${offsetRemainder}`
    );
  }

  function formatDatetimeLocal(date) {
    return (
      `${date.getFullYear()}-${pad2(date.getMonth() + 1)}-${pad2(date.getDate())}` +
      `T${pad2(date.getHours())}:${pad2(date.getMinutes())}`
    );
  }

  function formatRangeLabel(start, end) {
    return `${formatDatetimeLocal(start).replace('T', ' ')} → ${formatDatetimeLocal(end).replace('T', ' ')}`;
  }

  function parseDatetimeLocal(value) {
    const text = cleanText(value);
    if (!text) {
      return null;
    }
    const date = new Date(text);
    if (Number.isNaN(date.getTime())) {
      return null;
    }
    return date;
  }

  function getDefaultRange() {
    const start = new Date();
    start.setHours(0, 0, 0, 0);
    return {
      start,
      end: new Date(start.getTime() + CHUNK_MS),
    };
  }

  function clampRange(start, end) {
    if (!(start instanceof Date) || Number.isNaN(start.getTime())) {
      return getDefaultRange();
    }
    if (!(end instanceof Date) || Number.isNaN(end.getTime()) || end.getTime() <= start.getTime()) {
      return {
        start,
        end: new Date(start.getTime() + CHUNK_MS),
      };
    }
    if (end.getTime() - start.getTime() > MAX_RANGE_MS) {
      return {
        start,
        end: new Date(start.getTime() + MAX_RANGE_MS),
      };
    }
    return { start, end };
  }

  function splitRangeIntoChunks(start, end) {
    const chunks = [];
    let cursor = start.getTime();
    const endMs = end.getTime();
    while (cursor < endMs) {
      const next = Math.min(cursor + CHUNK_MS, endMs);
      chunks.push({
        start: new Date(cursor),
        end: new Date(next),
      });
      cursor = next;
    }
    return chunks;
  }

  function formatRangeDuration(start, end) {
    const totalMinutes = Math.max(0, Math.round((end.getTime() - start.getTime()) / 60000));
    const days = Math.floor(totalMinutes / (60 * 24));
    const hours = Math.floor((totalMinutes % (60 * 24)) / 60);
    const minutes = totalMinutes % 60;
    const parts = [];
    if (days > 0) {
      parts.push(`${days}d`);
    }
    if (hours > 0) {
      parts.push(`${hours}h`);
    }
    if (minutes > 0 || parts.length === 0) {
      parts.push(`${minutes}m`);
    }
    return parts.join(' ');
  }

  function validateRange(start, end) {
    if (!start || !end) {
      return 'Set both a start and an end date/time.';
    }
    if (end.getTime() <= start.getTime()) {
      return 'End must be after start.';
    }
    if (end.getTime() - start.getTime() > MAX_RANGE_MS) {
      return `Range cannot be more than ${MAX_RANGE_DAYS} days.`;
    }
    return '';
  }

  function buildQueryDateRange(start, end) {
    return {
      startTime: formatPortalDateTime(start),
      endTime: formatPortalDateTime(end),
      durationMs: end.getTime() - start.getTime(),
      durationLabel: formatRangeDuration(start, end),
    };
  }

  function readRangeFromInputs(clampValues) {
    const startInput = document.getElementById(`${SCRIPT_ID}-start`);
    const endInput = document.getElementById(`${SCRIPT_ID}-end`);
    const parsedStart = parseDatetimeLocal(startInput && startInput.value);

    }

  }

  function cleanText(value) {
    return String(value || '')
      .replace(/\u00a0/g, ' ')
      .replace(/\s+/g, ' ')
      .trim();
  }

  function parseDurationToSeconds(durationStr) {
    const text = cleanText(durationStr);
    if (!text) {
      return 0;
    }
    const parts = text.split(':').map((n) => Number(n));
    if (parts.some(Number.isNaN)) {
      return 0;
    }
    if (parts.length === 2) {
      return parts[0] * 60 + parts[1];
    }
    if (parts.length === 3) {
      return parts[0] * 3600 + parts[1] * 60 + parts[2];
    }
    return 0;
  }

  function formatSeconds(totalSeconds) {
    const total = Math.max(0, Math.round(totalSeconds || 0));
    const hours = Math.floor(total / 3600);
    const minutes = Math.floor((total % 3600) / 60);
    if (hours > 0 && minutes > 0) {
      return `${hours}h ${minutes}m`;
    }
    if (hours > 0) {
      return `${hours}h`;
    }
    return `${minutes}m`;
  }

  function formatPct(numerator, denominator) {
    if (!denominator) {
      return '0.0%';
    }
    return `${((numerator / denominator) * 100).toFixed(1)}%`;
  }

  function truncate(value, maxChars) {
    const text = String(value || '');
    if (text.length <= maxChars) {
      return text;
    }
    return `${text.slice(0, maxChars)}…`;
  }

  function looksLikeAuth(html, title) {
    const haystack = `${title || ''}\n${html || ''}`.toLowerCase();
    return (
      haystack.includes('sign in with midway') ||
      haystack.includes('midway-auth') ||
      /<title>[^<]*midway/i.test(html || '')
    );
  }

  function looksLikeNotFound(html, title) {
    const haystack = `${title || ''}\n${html || ''}`.toLowerCase();
    return haystack.includes('http status 404') || haystack.includes('was not found') || haystack.includes('not found');
  }

  function extractDefinitionValue(doc, labelPattern) {
    for (const term of doc.querySelectorAll('dt')) {
      if (labelPattern.test(cleanText(term.textContent))) {
        return cleanText(term.nextElementSibling && term.nextElementSibling.textContent);
      }
    }
    return '';
  }

  function classifyFunctionRow(row) {
    const className = row.className || '';
    const text = cleanText(row.textContent);
    const titleText = cleanText((row.querySelector('td:first-child') || row).textContent);
    const durationText = cleanText(
      (row.querySelector('td.rightAlign') || row.querySelector('td:last-child') || {}).textContent
    );
    const isIndirect =
      row.classList.contains('indirect') ||
      /\bindirect\b/i.test(className) ||
      /\bindirect\b/i.test(text);
    const isTimeOffTask =
      row.classList.contains('timeOffTask') ||
      /\btimeofftask\b/i.test(className) ||
      /\btime off task\b/i.test(text);
    const hasPathMarker = /[♦◆]|â™¦/.test(titleText);
    let bucket = 'other';
    if (isTimeOffTask) {
      bucket = 'tot';
    } else if (isIndirect) {
      bucket = 'idle';
    } else if (hasPathMarker || row.classList.contains('direct')) {
      bucket = 'active';
    }
    return {
      className,
      titleText: truncate(titleText, 160),
      durationText,
      seconds: parseDurationToSeconds(durationText),
      isIndirect,
      isTimeOffTask,
      hasPathMarker,
      bucket,
    };
  }

  function parseTimeDetailsHtml(html, login, url, httpStatus) {
    const doc = new DOMParser().parseFromString(html || '', 'text/html');
    const title = cleanText(doc.title);
    const bodyText = cleanText(doc.body ? doc.body.textContent : '');
    const auth = looksLikeAuth(html, title);
    const notFound = looksLikeNotFound(html, title);
    const gantt = doc.querySelector('table.ganttChart');
    const functionRows = Array.from(doc.querySelectorAll('tr.function-seg'));
    const clockRows = Array.from(doc.querySelectorAll('tr.clock-seg, tr[class*="clock"]'));

    const classified = functionRows.map(classifyFunctionRow);
    const idleSeconds = classified
      .filter((row) => row.bucket === 'idle')
      .reduce((sum, row) => sum + row.seconds, 0);
    const totSeconds = classified
      .filter((row) => row.bucket === 'tot')
      .reduce((sum, row) => sum + row.seconds, 0);
    const activeSeconds = classified
      .filter((row) => row.bucket === 'active')
      .reduce((sum, row) => sum + row.seconds, 0);

    let onClockSeconds = 0;
    let offClockSeconds = 0;
    const clockSamples = clockRows.slice(0, 12).map((row) => {
      const className = row.className || '';
      const text = cleanText(row.textContent);
      const durationText = cleanText(
        (row.querySelector('td.rightAlign') || row.querySelector('td:last-child') || {}).textContent
      );
      const seconds = parseDurationToSeconds(durationText);
      const isOnClock =
        row.classList.contains('on-clock') ||
        row.classList.contains('paid') ||
        row.classList.contains('ClockPaid') ||
        /\bon-clock\b/i.test(className) ||
        /\bonclock\/paid\b/i.test(text);
      const isOffClock =
        row.classList.contains('off-clock') ||
        row.classList.contains('un-paid') ||
        /\boff-clock\b/i.test(className);
      if (isOnClock) {
        onClockSeconds += seconds;
      } else if (isOffClock) {
        offClockSeconds += seconds;
      }
      return { className, durationText, seconds, isOnClock, isOffClock };
    });

    const titleNode = doc.querySelector('td.title > span.fold-control, .employeeInfo h1, h1');
    const titleText = cleanText(titleNode && titleNode.textContent);
    const loginMatch = titleText.match(/\(([^()]+)\)\s*$/);
    const displayName = loginMatch ? titleText.slice(0, loginMatch.index).trim() : titleText;
    const resolvedLogin = (loginMatch && loginMatch[1]) || login;
    const photo = doc.querySelector('img.photo');
    let employeeId = extractDefinitionValue(doc, /^employee\s*id$/i);
    if (!employeeId && photo && photo.getAttribute('src')) {
      try {
        const photoUrl = new URL(photo.getAttribute('src'), FCLM_ORIGIN);
        employeeId = photoUrl.searchParams.get('employeeid') || photoUrl.searchParams.get('employeeId') || '';
      } catch {
        employeeId = '';
      }
    }

    const classCounts = {};
    classified.forEach((row) => {
      const key = row.className || '(none)';
      classCounts[key] = (classCounts[key] || 0) + 1;
    });

    let note = '';
    if (auth) {
      note = 'FCLM auth required. Open fclm-portal.amazon.com and sign in with Midway, then retry.';
    } else if (notFound) {
      note = `No FCLM time details found for ${login}.`;
    } else if (!gantt && functionRows.length === 0) {
      note = 'Time details loaded but no Gantt/function segments were present for this date.';
    }

    return {
      login,
      resolvedLogin,
      displayName: displayName || '',
      employeeId: employeeId || '',
      idleSeconds,
      totSeconds,
      activeSeconds,
      onClockSeconds,
      offClockSeconds,
      idleTime: formatSeconds(idleSeconds),
      totTime: formatSeconds(totSeconds),
      activeTime: formatSeconds(activeSeconds),
      onClockTime: formatSeconds(onClockSeconds),
      idlePct: formatPct(idleSeconds, onClockSeconds),
      note,
      ok: !auth && !notFound,
      debug: {
        url,
        httpStatus,
        htmlLength: (html || '').length,
        title,
        looksLikeAuth: auth,
        looksLikeNotFound: notFound,
        ganttPresent: Boolean(gantt),
        ganttClass: gantt ? gantt.className : null,
        functionSegCount: functionRows.length,
        clockSegCount: clockRows.length,
        classCounts,
        idleSeconds,
        totSeconds,
        activeSeconds,
        onClockSeconds,
        offClockSeconds,
        functionRows: classified.slice(0, MAX_DEBUG_FUNCTION_ROWS),
        functionRowsOmitted: Math.max(0, classified.length - MAX_DEBUG_FUNCTION_ROWS),
        clockSamples,
        bodyPreview: truncate(bodyText, MAX_DEBUG_HTML_CHARS),
      },
    };
  }

  function gmRequest(url) {
    return new Promise((resolve, reject) => {
      if (runControl.stopRequested) {
        reject(LookupStopped());
        return;
      }
      let settled = false;
      let handle = null;
      function finish(callback) {
        if (settled) {
          return;
        }
        settled = true;
        runControl.requests.delete(handle);
        callback();
      }
      handle = GM_xmlhttpRequest({
        method: 'GET',
        url,
        timeout: REQUEST_TIMEOUT_MS,
        onload(response) {
          finish(function () {
            if (runControl.stopRequested) {
              reject(LookupStopped());
              return;
            }
            resolve({
              url,
              status: response.status,
              responseText: response.responseText || '',
              finalUrl: response.finalUrl || url,
            });
          });
        },
        onerror() {
          finish(function () {
            reject(new Error(`Network error for ${url}`));
          });
        },
        ontimeout() {
          finish(function () {
            reject(new Error(`Timeout for ${url}`));
          });
        },
        onabort() {
          finish(function () {
            reject(LookupStopped());
          });
        },
      });
      if (handle) {
        runControl.requests.add(handle);
      }
    });
  }

  function mergeParsedChunks(login, parsedChunks) {
    const merged = {
      login,
      resolvedLogin: login,
      displayName: '',
      employeeId: '',
      idleSeconds: 0,
      totSeconds: 0,
      activeSeconds: 0,
      onClockSeconds: 0,
      offClockSeconds: 0,
      note: '',
      ok: false,
      debug: {
        chunkCount: parsedChunks.length,
        chunks: [],
      },
    };

    const notes = [];
    for (const parsed of parsedChunks) {
      if (parsed.debug) {
        merged.debug.chunks.push(parsed.debug);
      }
      if (parsed.debug && parsed.debug.looksLikeAuth) {
        merged.note = parsed.note || 'FCLM auth required. Open fclm-portal.amazon.com and sign in with Midway, then retry.';
        merged.ok = false;
        merged.idleTime = '—';
        merged.totTime = '—';
        merged.activeTime = '—';
        merged.onClockTime = '—';
        merged.idlePct = '—';
        return merged;
      }
      if (parsed.ok) {
        merged.ok = true;
        merged.idleSeconds += Number(parsed.idleSeconds) || 0;
        merged.totSeconds += Number(parsed.totSeconds) || 0;
        merged.activeSeconds += Number(parsed.activeSeconds) || 0;
        merged.onClockSeconds += Number(parsed.onClockSeconds) || 0;
        merged.offClockSeconds += Number(parsed.offClockSeconds) || 0;
        if (parsed.displayName) {
          merged.displayName = parsed.displayName;
        }
        if (parsed.employeeId) {
          merged.employeeId = parsed.employeeId;
        }
        if (parsed.resolvedLogin) {
          merged.resolvedLogin = parsed.resolvedLogin;
        }
      } else if (parsed.note) {
        notes.push(parsed.note);
      }
    }

    merged.idleTime = formatSeconds(merged.idleSeconds);
    merged.totTime = formatSeconds(merged.totSeconds);
    merged.activeTime = formatSeconds(merged.activeSeconds);
    merged.onClockTime = formatSeconds(merged.onClockSeconds);
    merged.idlePct = formatPct(merged.idleSeconds, merged.onClockSeconds);
    if (!merged.ok) {
      merged.note = notes[0] || `No FCLM time details found for ${login}.`;
      merged.idleTime = '—';
      merged.totTime = '—';
      merged.activeTime = '—';
      merged.onClockTime = '—';
      merged.idlePct = '—';
    }
    return merged;
  }

  function stoppedPartialResult(login, parsedChunks, expectedWindows) {
    const merged = mergeParsedChunks(login, parsedChunks);
    merged.stopped = true;
    if (merged.ok) {
      merged.note = `Stopped after ${parsedChunks.length} of ${expectedWindows} windows.`;
    } else {
      merged.note = 'Stopped before time details were loaded.';
      merged.idleTime = '—';
      merged.onClockTime = '—';
      merged.idlePct = '—';
      merged.totTime = '—';
    }
    return merged;
  }

  async function lookupIdleForLogin(login, warehouseId, start, end) {
    const windows = splitRangeIntoChunks(start, end);
    runControl.currentLogin = login;
    addDebugStep(`lookup:start:${login}`, {
      windows: windows.length,
      startTime: formatPortalDateTime(start),
      endTime: formatPortalDateTime(end),
    });
    updateProgress();

    const parsedChunks = [];
    for (let index = 0; index < windows.length; index += 1) {
      if (runControl.stopRequested) {
        return stoppedPartialResult(login, parsedChunks, windows.length);
      }
      const windowRange = windows[index];
      const url = buildTimeDetailsUrl(login, warehouseId, windowRange.start, windowRange.end);
      addDebugStep(`lookup:window:${login}`, {
        index,
        url,
      });
      let response;
      try {
        response = await gmRequest(url);
      } catch (error) {
        if (isStoppedError(error)) {
          return stoppedPartialResult(login, parsedChunks, windows.length);
        }
        throw error;
      }
      markRequestDone();
      addDebugStep(`lookup:http:${login}`, {
        index,
        status: response.status,
        htmlLength: (response.responseText || '').length,
        finalUrl: response.finalUrl,
      });
      if (response.status < 200 || response.status >= 400) {
        throw new Error(`FCLM returned HTTP ${response.status} for ${login}`);
      }
      const parsed = parseTimeDetailsHtml(response.responseText, login, url, response.status);
      parsedChunks.push(parsed);
      if (parsed.debug && parsed.debug.looksLikeAuth) {
        runControl.done += Math.max(0, windows.length - parsedChunks.length);
        updateProgress();
        break;
      }
    }

    return mergeParsedChunks(login, parsedChunks);
  }

  async function mapWithConcurrency(items, concurrency, mapper) {
    const results = new Array(items.length);
    let nextIndex = 0;
    async function worker() {
      while (nextIndex < items.length) {
        if (runControl.stopRequested) {
          return;
        }
        const currentIndex = nextIndex;
        nextIndex += 1;
        results[currentIndex] = await mapper(items[currentIndex], currentIndex);
      }
    }
    const workerCount = Math.min(concurrency, items.length);
    await Promise.all(Array.from({ length: workerCount }, worker));
    return results.filter(function (entry) {
      return entry != null;
    });
  }

  function buildDayProgressionRows() {
    const days = (lastChartData && lastChartData.days) || [];
    const seriesMap = new Map(((lastChartData && lastChartData.series) || []).map((row) => [row.login, row]));
    const rows = [];
    (lastLookupResults || []).forEach((result) => {
      const series = seriesMap.get(result.login);

      const byDay = new Map(points.map((point) => [point.start, point]));
      if (!days.length) {
        rows.push({
          login: result.login,
          day: 'All days',
          idleTime: formatSeconds(result.idleSeconds),
          onClockTime: formatSeconds(result.onClockSeconds),
          idlePct: formatPct(result.idleSeconds, result.onClockSeconds),
          idlePctValue: result.onClockSeconds
            ? ((Number(result.idleSeconds) || 0) / result.onClockSeconds) * 100
            : 0,
          start: '',
        });
        return;
      }
      days.forEach((day) => {
        const point = byDay.get(day.start);
        const idleSeconds = point ? Number(point.idleSeconds) || 0 : 0;
        const onClockSeconds = point ? Number(point.onClockSeconds) || 0 : 0;
        rows.push({
          login: result.login,
          day: formatDayLabel(day.start),
          idleTime: formatSeconds(idleSeconds),
          onClockTime: formatSeconds(onClockSeconds),
          idlePct: formatPct(idleSeconds, onClockSeconds),
          idlePctValue: onClockSeconds ? (idleSeconds / onClockSeconds) * 100 : 0,
          start: day.start,
        });
      });
    });
    return rows;
  }

    const match = String(startTime || '').match(/^(\d{4})-(\d{2})-(\d{2})/);

  function buildDayTimeCardUrl(login, startTime) {
    const day = formatPortalDayParam(startTime);

  function buildChartData(results, debugResults) {
    const logins = (results || [])
      .filter((row) => row && row.ok)
      .map((row) => {
        const idleSeconds = Number(row.idleSeconds) || 0;
        const totSeconds = Number(row.totSeconds) || 0;
        const activeSeconds = Number(row.activeSeconds) || 0;
        const onClockSeconds = Number(row.onClockSeconds) || 0;
        return {
          login: row.login || '',
          idleSeconds,
          totSeconds,
          activeSeconds,
          onClockSeconds,
          otherSeconds: Math.max(0, onClockSeconds - idleSeconds - totSeconds - activeSeconds),
          idlePct: onClockSeconds ? (idleSeconds / onClockSeconds) * 100 : 0,
        };
      })
      .sort((a, b) => b.idleSeconds - a.idleSeconds)
      .slice(0, 40);

    const dayMap = new Map();
    const seriesMap = new Map();
    (debugResults || []).forEach((entry) => {
      const login = entry && entry.login ? entry.login : '';
      if (!login) {
        return;
      }
      const chunks = entry.parse && Array.isArray(entry.parse.chunks) ? entry.parse.chunks : [];
      const points = [];
      chunks.forEach((chunk) => {
        if (!chunk || chunk.looksLikeAuth) {
          return;
        }
        const start = chunkStartFromUrl(chunk.url);
        if (!start) {
          return;
        }
        const idleSeconds = Number(chunk.idleSeconds) || 0;
        const totSeconds = Number(chunk.totSeconds) || 0;
        const activeSeconds = Number(chunk.activeSeconds) || 0;
        const onClockSeconds = Number(chunk.onClockSeconds) || 0;
        const point = {
          start,
          idleSeconds,
          totSeconds,
          activeSeconds,
          onClockSeconds,
          idlePct: onClockSeconds ? (idleSeconds / onClockSeconds) * 100 : 0,
        };
        points.push(point);
        const existing = dayMap.get(start) || {
          start,
          idleSeconds: 0,
          totSeconds: 0,
          activeSeconds: 0,
          onClockSeconds: 0,
        };
        existing.idleSeconds += idleSeconds;
        existing.totSeconds += totSeconds;
        existing.activeSeconds += activeSeconds;
        existing.onClockSeconds += onClockSeconds;
        dayMap.set(start, existing);
      });
      if (points.length) {
        seriesMap.set(login, points);
      }
    });

    const days = Array.from(dayMap.values())
      .sort((a, b) => a.start.localeCompare(b.start))
      .map((day) => ({
        ...day,
        idlePct: day.onClockSeconds ? (day.idleSeconds / day.onClockSeconds) * 100 : 0,
      }));

    return {
      logins,
      days,
      series: logins
        .map((row) => ({
          login: row.login,
          points: seriesMap.get(row.login) || [],
        }))
        .filter((row) => row.points.length),
    };
  }

  function niceCeiling(value) {
    if (value <= 0) {
      return 1;
    }
    const exp = Math.pow(10, Math.floor(Math.log10(value)));
    const n = value / exp;
    const nice = n <= 1 ? 1 : n <= 2 ? 2 : n <= 5 ? 5 : 10;
    return nice * exp;
  }

  function yTicks(maxValue, count) {
    const top = niceCeiling(maxValue || 1);
    const step = niceCeiling(top / count);
    const values = [];
    for (let value = 0; value <= top + 1e-9; value += step) {
      values.push(value);
    }
    return { max: Math.max(top, values[values.length - 1] || top), values };
  }

  function polarPoint(cx, cy, radius, angle) {
    const rad = ((angle - 90) * Math.PI) / 180;
    return [cx + radius * Math.cos(rad), cy + radius * Math.sin(rad)];
  }

  function pieSlice(cx, cy, radius, startAngle, endAngle) {
    const [x1, y1] = polarPoint(cx, cy, radius, endAngle);
    const [x2, y2] = polarPoint(cx, cy, radius, startAngle);
    const large = endAngle - startAngle > 180 ? 1 : 0;
    return `M ${cx} ${cy} L ${x2} ${y2} A ${radius} ${radius} 0 ${large} 1 ${x1} ${y1} Z`;
  }

  function idleHeatColor(pct) {
    const n = Number(pct) || 0;
    if (n < 5) {

      return '#fde68a';
    }
    if (n < 25) {
      return '#fbbf24';
    }
    if (n < 30) {
      return '#fb923c';
    }
    return '#ef4444';
  }

  function bindChartTooltips(root) {
    if (!root) {
      return;
    }
    let tip = document.getElementById(`${SCRIPT_ID}-chart-tip`);
    if (!tip) {
      tip = document.createElement('div');
      tip.id = `${SCRIPT_ID}-chart-tip`;
      document.body.appendChild(tip);
    }
    function hideTip() {
      tip.style.display = 'none';
    }
    function showTip(event) {
      const target = event.currentTarget;
      const text = target.getAttribute('data-tip');
      if (!text) {
        return;
      }
      tip.textContent = text;
      tip.style.display = 'block';
      const pad = 12;
      const offsetX = 14;
      const offsetY = 10;
      let left = event.clientX + offsetX;
      let top = event.clientY - tip.offsetHeight - offsetY;
      if (left + tip.offsetWidth > window.innerWidth - pad) {
        left = window.innerWidth - tip.offsetWidth - pad;
      }
      if (top < pad) {
        top = pad;
      }
      tip.style.left = `${Math.max(pad, left)}px`;
      tip.style.top = `${top}px`;
    }
    root.querySelectorAll('[data-tip]').forEach((node) => {
      node.addEventListener('pointerenter', showTip);
      node.addEventListener('pointermove', showTip);
      node.addEventListener('pointerleave', hideTip);
    });
    hideTip();
  }

    const onClock = logins.reduce((sum, row) => sum + row.onClockSeconds, 0);
    if (!total) {
      return '<div class="chart-empty">No on-clock time to graph.</div>';
    }

      slices.length === 1

                `${slice.label} ${formatSeconds(slice.seconds)}`

  function svgMultiLine(series, days) {

    const right = 12;
    const top = 12;
    const bottom = 28;
    const plotW = width - left - right;
    const plotH = height - top - bottom;

      0.25,
      ...lines.flatMap((row) => row.points.map((point) => point.idleSeconds / 3600))
    );
    const ticks = yTicks(maxIdleHours, 4);
    const xFor = (index) => left + (days.length === 1 ? plotW / 2 : (index / (days.length - 1)) * plotW);
    const yFor = (seconds) => top + plotH - (seconds / 3600 / ticks.max) * plotH;
    const grid = ticks.values
      .map((tick) => {

        return `<line x1="${left}" y1="${y.toFixed(1)}" x2="${left + plotW}" y2="${y.toFixed(1)}" stroke="${CHART_COLORS.grid}"></line>

            formatDayLabel(day.start)
          )}</text>`

      .join('');

      .map((row, seriesIndex) => {

          .map((day, index) => {

  function svgStackedDays(days) {
    const width = 460;
    const height = 180;
    const left = 44;
    const right = 12;
    const top = 12;
    const bottom = 28;
    const plotW = width - left - right;
    const plotH = height - top - bottom;
    const maxHours = Math.max(
      1,
      ...days.map((day) => (day.idleSeconds + day.totSeconds + day.activeSeconds + Math.max(0, day.onClockSeconds - day.idleSeconds - day.totSeconds - day.activeSeconds)) / 3600)
    );
    const ticks = yTicks(maxHours, 4);
    const gap = days.length > 10 ? 2 : 6;
    const barWidth = Math.max(8, Math.floor((plotW - gap * (days.length - 1)) / days.length));
    const grid = ticks.values
      .map((tick) => {
        const y = top + plotH - (tick / ticks.max) * plotH;
        return `<line x1="${left}" y1="${y.toFixed(1)}" x2="${left + plotW}" y2="${y.toFixed(1)}" stroke="${CHART_COLORS.grid}"></line>
          <text x="${left - 6}" y="${y + 3}" text-anchor="end" fill="${CHART_COLORS.text}" font-size="9">${tick}h</text>`;
      })
      .join('');
    const bars = days
      .map((day, index) => {
        const x = left + index * (barWidth + gap);
        const parts = [
          { seconds: day.idleSeconds, color: CHART_COLORS.idle, label: 'Idle' },
          { seconds: day.totSeconds, color: CHART_COLORS.tot, label: 'ToT' },
          { seconds: day.activeSeconds, color: CHART_COLORS.active, label: 'Active' },
        ];
        let y = top + plotH;
        const rects = parts
          .map((part) => {
            const h = (part.seconds / 3600 / ticks.max) * plotH;
            y -= h;
            if (h <= 0) {
              return '';
            }
            return `<rect x="${x}" y="${y.toFixed(1)}" width="${barWidth}" height="${Math.max(1, h).toFixed(1)}" fill="${part.color}">
              <title>${escapeHtml(`${formatDayLabel(day.start)} · ${part.label} ${formatSeconds(part.seconds)}`)}</title>
            </rect>`;
          })
          .join('');
        return `${rects}<text x="${x + barWidth / 2}" y="${height - 8}" text-anchor="middle" fill="${CHART_COLORS.text}" font-size="9">${escapeHtml(
          formatDayLabel(day.start)
        )}</text>`;
      })
      .join('');
    return `<svg viewBox="0 0 ${width} ${height}" role="img">${grid}${bars}</svg>${chartLegend([
      { color: CHART_COLORS.idle, label: 'Idle' },
      { color: CHART_COLORS.tot, label: 'Time off task' },
      { color: CHART_COLORS.active, label: 'Active' },
    ])}`;
  }

  function wrapCanvasText(ctx, text, maxWidth) {
    const words = String(text || '')
      .replace(/\s+/g, ' ')
      .trim()
      .split(' ')
      .filter(Boolean);
    if (!words.length) {
      return [];
    }
    const lines = [];
    let current = '';
    words.forEach((word) => {
      const next = current ? `${current} ${word}` : word;
      if (ctx.measureText(next).width <= maxWidth) {
        current = next;
        return;
      }
      if (current) {
        lines.push(current);
      }
      current = word;
    });
    if (current) {
      lines.push(current);
    }
    return lines;
  }

  function downloadChartPng(card) {
    if (!card) {
      return;
    }
    const svg = card.querySelector('svg');
    const title = (card.querySelector('.chart-title') && card.querySelector('.chart-title').textContent) || 'chart';
    if (!svg) {
      updateStatus('This card has no graph to download.', 'error');
      return;
    }
    const clone = svg.cloneNode(true);
    clone.setAttribute('xmlns', 'http://www.w3.org/2000/svg');
    const box = svg.viewBox && svg.viewBox.baseVal;
    const graphWidth = Math.max(1, Math.round(box && box.width ? box.width : svg.clientWidth || 460));
    const graphHeight = Math.max(1, Math.round(box && box.height ? box.height : svg.clientHeight || 200));
    clone.setAttribute('width', String(graphWidth));
    clone.setAttribute('height', String(graphHeight));
    const xml = new XMLSerializer().serializeToString(clone);
    const legend = readChartLegend(card);
    const notes = readChartNotes(card);
    const image = new Image();
    image.onload = function () {
      const scale = 2;
      const pad = 28;
      const gap = 16;
      const contentW = Math.max(graphWidth, 560);
      const measure = document.createElement('canvas').getContext('2d');
      measure.font = '700 18px "Segoe UI", system-ui, sans-serif';
      const titleLines = wrapCanvasText(measure, title.toUpperCase(), contentW);
      measure.font = '600 13px "Segoe UI", system-ui, sans-serif';
      const noteLines = notes.flatMap((note) => wrapCanvasText(measure, note, contentW));
      const swatch = 10;
      const legendGap = 16;
      const legendItems = legend.map((item) => {
        const textW = measure.measureText(item.label).width;
        return { ...item, width: swatch + 6 + textW };
      });
      const legendRows = [];
      let row = [];
      let rowWidth = 0;
      legendItems.forEach((item) => {
        const next = rowWidth + (row.length ? legendGap : 0) + item.width;
        if (row.length && next > contentW) {
          legendRows.push(row);
          row = [item];
          rowWidth = item.width;
          return;
        }
        row.push(item);
        rowWidth = next;
      });
      if (row.length) {
        legendRows.push(row);
      }
      const titleH = titleLines.length * 24;
      const legendH = legendRows.length ? legendRows.length * 22 : 0;
      const notesH = noteLines.length ? noteLines.length * 18 : 0;
      const extra =
        pad +
        titleH +
        (titleH ? gap : 0) +
        (legendH ? gap : 0) +
        legendH +
        (notesH ? gap : 0) +
        notesH +
        pad;
      const canvas = document.createElement('canvas');
      canvas.width = Math.round((contentW + pad * 2) * scale);
      canvas.height = Math.round((graphHeight + extra) * scale);
      const ctx = canvas.getContext('2d');
      ctx.scale(scale, scale);
      ctx.fillStyle = '#ffffff';
      ctx.fillRect(0, 0, canvas.width / scale, canvas.height / scale);
      ctx.fillStyle = '#f8fafc';
      roundRect(ctx, 8, 8, canvas.width / scale - 16, canvas.height / scale - 16, 12);
      ctx.fill();
      let y = pad + 8;
      ctx.fillStyle = '#6b7280';
      ctx.font = '700 14px "Segoe UI", system-ui, sans-serif';
      ctx.textBaseline = 'top';
      titleLines.forEach((line) => {
        ctx.fillText(line, pad, y);
        y += 24;
      });
      if (titleH) {
        y += gap / 2;
      }
      const graphX = pad + Math.max(0, (contentW - graphWidth) / 2);
      ctx.drawImage(image, graphX, y, graphWidth, graphHeight);
      y += graphHeight;
      if (legendRows.length) {
        y += gap;
        ctx.font = '600 12px "Segoe UI", system-ui, sans-serif';
        ctx.textBaseline = 'middle';
        legendRows.forEach((legendRow) => {
          const total = legendRow.reduce((sum, item, index) => sum + item.width + (index ? legendGap : 0), 0);
          let x = pad + Math.max(0, (contentW - total) / 2);
          legendRow.forEach((item) => {
            ctx.fillStyle = item.color;
            roundRect(ctx, x, y + 6, swatch, swatch, 2);
            ctx.fill();
            ctx.fillStyle = '#4b5563';
            ctx.fillText(item.label, x + swatch + 6, y + 11);
            x += item.width + legendGap;
          });
          y += 22;
        });
      }
      if (noteLines.length) {
        y += gap / 2;
        ctx.fillStyle = '#6b7280';
        ctx.font = '500 12px "Segoe UI", system-ui, sans-serif';
        ctx.textBaseline = 'top';
        noteLines.forEach((line) => {
          ctx.fillText(line, pad, y);
          y += 18;
        });
      }
      canvas.toBlob(function (blob) {
        if (!blob) {
          updateStatus('Could not create PNG.', 'error');
          return;
        }
        const stamp = new Date().toISOString().slice(0, 19).replace(/[:T]/g, '-');
        downloadBlob(blob, `fclm-idle-${slugChartName(title)}-${stamp}.png`);
        updateStatus(`Downloaded ${title} as PNG.`, 'success');
      }, 'image/png');
    };
    image.onerror = function () {
      updateStatus('Could not render chart PNG.', 'error');
    };
    image.src = `data:image/svg+xml;charset=utf-8,${encodeURIComponent(xml)}`;
  }

  function roundRect(ctx, x, y, width, height, radius) {
    const r = Math.min(radius, width / 2, height / 2);
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.arcTo(x + width, y, x + width, y + height, r);
    ctx.arcTo(x + width, y + height, x, y + height, r);
    ctx.arcTo(x, y + height, x, y, r);
    ctx.arcTo(x, y, x + width, y, r);
    ctx.closePath();
  }

  async function writeClipboard(text) {
    try {
      if (typeof GM_setClipboard === 'function') {
        GM_setClipboard(text, 'text');
        return;
      }
    } catch {
      // Fall through to browser clipboard.
    }
    if (navigator.clipboard && navigator.clipboard.writeText) {
      await navigator.clipboard.writeText(text);
      return;
    }
    const textarea = document.createElement('textarea');
    textarea.value = text;
    textarea.setAttribute('readonly', 'readonly');
    textarea.style.position = 'fixed';
    textarea.style.top = '-9999px';
    document.body.appendChild(textarea);
    textarea.select();
    const ok = document.execCommand('copy');
    document.body.removeChild(textarea);
    if (!ok) {
      throw new Error('Clipboard copy failed.');
    }
  }

  async function runLookup() {
    const warehouseInput = document.getElementById(`${SCRIPT_ID}-warehouse`);
    const loginInput = document.getElementById(`${SCRIPT_ID}-logins`);
    const warehouseId = cleanText(warehouseInput.value).toUpperCase() || DEFAULT_WAREHOUSE_ID;
    const { start, end } = readRangeFromInputs(true);
    const logins = parseLogins(loginInput.value);
    const rangeError = validateRange(start, end);

    warehouseInput.value = warehouseId;
    storeState({
      warehouseId,
      startLocal: formatDatetimeLocal(start),
      endLocal: formatDatetimeLocal(end),
      logins: loginInput.value,
    });
    resetDebug(warehouseId, start, end, logins);
    updateSummary(warehouseId, start, end, logins.length);

    if (rangeError) {
      updateStatus(rangeError, 'error');
      addDebugError('validate', rangeError);
      return;
    }

    if (logins.length === 0) {
      updateStatus('Paste at least one login.', 'error');
      addDebugError('validate', 'No logins provided');
      return;
    }

    const windows = splitRangeIntoChunks(start, end);
    const totalRequests = logins.length * windows.length;
    startRunControl(totalRequests);
    renderResults([]);
    updateStatus(
      windows.length > 1
        ? `Checking idle time for ${logins.length} login(s) across ${windows.length} 24h windows…`
        : `Checking idle time for ${logins.length} login(s)…`
    );

    try {
      const resolved = await mapWithConcurrency(logins, LOOKUP_CONCURRENCY, async (login) => {
        try {
          const parsed = await lookupIdleForLogin(login, warehouseId, start, end);
          return {
            login,
            ok: parsed.ok,
            ...parsed,
          };
        } catch (error) {
          if (isStoppedError(error)) {
            return stoppedPartialResult(login, [], windows.length);
          }
          addDebugError(`lookup:${login}`, error);
          markRequestDone();
          return {
            login,
            ok: false,
            idleSeconds: 0,
            onClockSeconds: 0,
            idleTime: '—',
            onClockTime: '—',
            idlePct: '—',
            totTime: '—',
            note: formatError(error),
            debug: { error: formatError(error) },
          };
        }
      });

      const results = resolved
        .map((entry) => ({
          login: entry.login,
          resolvedLogin: entry.resolvedLogin || entry.login,
          displayName: entry.displayName || '',
          employeeId: entry.employeeId || '',
          idleSeconds: entry.idleSeconds || 0,
          totSeconds: entry.totSeconds || 0,
          activeSeconds: entry.activeSeconds || 0,
          onClockSeconds: entry.onClockSeconds || 0,
          idleTime: entry.idleTime || '0m',
          totTime: entry.totTime || '0m',
          activeTime: entry.activeTime || '0m',
          onClockTime: entry.onClockTime || '0m',
          idlePct: entry.idlePct || '0.0%',
          note: entry.note || '',
          ok: Boolean(entry.ok),
        }));

      lastDebugPayload.results = resolved.map((entry) => ({
        login: entry.login,
        ok: entry.ok,
        displayName: entry.displayName || '',
        employeeId: entry.employeeId || '',
        idleTime: entry.idleTime,
        idlePct: entry.idlePct,
        onClockTime: entry.onClockTime,
        totTime: entry.totTime,
        note: entry.note || '',
        parse: entry.debug || null,
      }));

      addDebugStep(runControl.stopRequested ? 'stopped' : 'completed', {
        count: results.length,
        failures: results.filter((item) => !item.ok).length,
        withIdle: results.filter((item) => item.idleSeconds > 0).length,
        done: runControl.done,
        total: runControl.total,
      });

      renderResults(results);
      layoutOverlay();
      const failures = results.filter((item) => !item.ok).length;
      const stopped = runControl.stopRequested;
      let statusMessage;
      if (stopped) {
        statusMessage = `Stopped. ${results.length} login(s) kept (${runControl.done}/${runControl.total} requests).`;
      } else if (failures > 0) {
        statusMessage = `Done. ${results.length - failures} ok, ${failures} failed. Copy Debug if idle looks wrong.`;
      } else {
        statusMessage = `Done. Idle time loaded for ${results.length} login(s).`;
      }
      updateStatus(statusMessage, stopped || failures > 0 ? 'error' : 'success');
      persistFormState();
      persistSession(results, statusMessage);
    } catch (error) {
      addDebugError('runLookup', error);
      updateStatus(`Lookup failed: ${formatError(error)}`, 'error');
    } finally {
      resetRunControl();
      setLookupButtons(false);
      updateProgress();
    }
  }

  function getViewportSize() {
    const viewport = window.visualViewport;
    return {
      width: Math.max(1, Math.round(viewport && viewport.width ? viewport.width : window.innerWidth)),
      height: Math.max(1, Math.round(viewport && viewport.height ? viewport.height : window.innerHeight)),
    };
  }

  function clamp(value, min, max) {
    return Math.min(max, Math.max(min, value));
  }

  function placeToggleAtBottomCenter(toggle) {
    if (!toggle || toggle.dataset.moved === 'true') {
      return;
    }
    const size = getViewportSize();
    const left = Math.max(0, (size.width - toggle.offsetWidth) / 2);
    const top = Math.max(0, size.height - toggle.offsetHeight - TOGGLE_EDGE);
    toggle.style.position = 'fixed';
    toggle.style.left = `${left}px`;
    toggle.style.top = `${top}px`;
    toggle.style.right = 'auto';
    toggle.style.bottom = 'auto';
    toggle.style.transform = 'none';
  }

  function keepToggleInViewport(toggle) {
    if (!toggle) {
      return;
    }
    const size = getViewportSize();
    const rect = toggle.getBoundingClientRect();
    const left = clamp(rect.left, 0, Math.max(0, size.width - rect.width));
    const top = clamp(rect.top, 0, Math.max(0, size.height - rect.height));
    toggle.style.position = 'fixed';
    toggle.style.left = `${left}px`;
    toggle.style.top = `${top}px`;
    toggle.style.right = 'auto';
    toggle.style.bottom = 'auto';
    toggle.style.transform = 'none';
  }

  function layoutOverlay() {
    const toggle = document.getElementById(`${SCRIPT_ID}-toggle`);
    const panel = document.getElementById(`${SCRIPT_ID}-panel`);
    if (!toggle) {
      return;
    }
    if (toggle.dataset.moved === 'true') {
      keepToggleInViewport(toggle);
    } else {
      placeToggleAtBottomCenter(toggle);
    }
    if (!panel) {
      return;
    }
    const size = getViewportSize();
    const toggleRect = toggle.getBoundingClientRect();
    const maxWidth = Math.max(160, Math.min(PANEL_WIDTH, size.width - VIEW_PAD * 2));
    panel.style.position = 'fixed';
    panel.style.width = `${maxWidth}px`;
    panel.style.maxWidth = `${Math.max(280, size.width - VIEW_PAD * 2)}px`;
    panel.style.right = 'auto';
    panel.style.transform = 'none';

    const spaceAbove = toggleRect.top - TOGGLE_GAP - VIEW_PAD;
    const spaceBelow = size.height - toggleRect.bottom - TOGGLE_GAP - VIEW_PAD;
    const placeAbove = spaceAbove >= 160 || spaceAbove >= spaceBelow;
    const available = Math.max(120, placeAbove ? spaceAbove : spaceBelow);
    const maxHeight = Math.min(available, size.height - VIEW_PAD * 2);
    panel.style.maxHeight = `${maxHeight}px`;
    panel.style.height = 'auto';

    let left = toggleRect.left + toggleRect.width / 2 - maxWidth / 2;
    left = clamp(left, VIEW_PAD, Math.max(VIEW_PAD, size.width - maxWidth - VIEW_PAD));
    panel.style.left = `${left}px`;

    if (placeAbove) {
      panel.style.top = 'auto';
      panel.style.bottom = `${size.height - toggleRect.top + TOGGLE_GAP}px`;
    } else {
      panel.style.bottom = 'auto';
      panel.style.top = `${toggleRect.bottom + TOGGLE_GAP}px`;
    }
  }

  function initializeToggleDragging(toggle) {
    let pointerId = null;
    let startX = 0;
    let startY = 0;
    let originLeft = 0;
    let originTop = 0;
    let moved = false;

    toggle.addEventListener('pointerdown', function (event) {
      pointerId = event.pointerId;
      startX = event.clientX;
      startY = event.clientY;
      const rect = toggle.getBoundingClientRect();
      originLeft = rect.left;
      originTop = rect.top;
      moved = false;
      toggle.dataset.dragging = 'false';
      toggle.style.left = `${originLeft}px`;
      toggle.style.top = `${originTop}px`;
      toggle.style.right = 'auto';
      toggle.style.bottom = 'auto';
      toggle.style.transform = 'none';
      toggle.setPointerCapture(pointerId);
    });

    toggle.addEventListener('pointermove', function (event) {
      if (pointerId !== event.pointerId) {
        return;
      }
      const deltaX = event.clientX - startX;
      const deltaY = event.clientY - startY;
      if (Math.abs(deltaX) > 4 || Math.abs(deltaY) > 4) {
        moved = true;
        toggle.dataset.dragging = 'true';
        toggle.dataset.moved = 'true';
      }
      if (!moved) {
        return;
      }
      const size = getViewportSize();
      const left = clamp(originLeft + deltaX, 0, Math.max(0, size.width - toggle.offsetWidth));
      const top = clamp(originTop + deltaY, 0, Math.max(0, size.height - toggle.offsetHeight));
      toggle.style.left = `${left}px`;
      toggle.style.top = `${top}px`;
      layoutOverlay();
    });

    function stopDragging(event) {
      if (pointerId !== event.pointerId) {
        return;
      }
      pointerId = null;
      toggle.releasePointerCapture(event.pointerId);
      if (moved) {
        layoutOverlay();
      }
      window.setTimeout(function () {
        toggle.dataset.dragging = 'false';
      }, 0);
    }

    toggle.addEventListener('pointerup', stopDragging);
    toggle.addEventListener('pointercancel', stopDragging);
  }

  function bindViewportLayout() {
    window.addEventListener('resize', layoutOverlay);
    window.addEventListener('orientationchange', layoutOverlay);
    if (window.visualViewport) {
      window.visualViewport.addEventListener('resize', layoutOverlay);
      window.visualViewport.addEventListener('scroll', layoutOverlay);
    }
  }

  function init() {
    if (!document.body) {
      setTimeout(init, 100);
      return;
    }
    injectStyles();
    createToggleButton();
    createPanel();
    bindViewportLayout();
    layoutOverlay();
    window.addEventListener('load', layoutOverlay);
  }
