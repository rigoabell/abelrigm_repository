  function traceWorkflow(event, data = {}) {
    const entry = {

      sortCodeText: readSortCodeText().slice(0, 40) || null,
      ...data,
    };
    workflowTrace.push(entry);
    if (workflowTrace.length > WORKFLOW_TRACE_MAX) {
      workflowTrace.splice(0, workflowTrace.length - WORKFLOW_TRACE_MAX);
    }
    console.log('[LS Trace]', event, data);
  }

  /**

   */
  function traceWatch(decision, detail = {}) {
    const key = decision + '|' + JSON.stringify(detail);
    if (key === lastWatchTraceKey) return;
    lastWatchTraceKey = key;
    traceWorkflow('watch.' + decision, detail);
  }

  function decodeAmazonFieldText(value) {
    const raw = String(value ?? '').trim();
    if (!raw) return '';
    try {
      return decodeURIComponent(raw.replace(/\+/g, ' ')).trim();
    } catch {
      return raw;
    }
  }

  function isAmazonPlaceholderValue(value) {
    const text = decodeAmazonFieldText(value).toLowerCase();
    if (!text) return true;
    for (const step of Object.values(STEPS)) {
      if (text === step.placeholder.toLowerCase()) return true;
    }
    return /^scan\s/.test(text);
  }

  function isValidSourceContainerId(value) {
    const id = decodeAmazonFieldText(value);
    if (!id || isAmazonPlaceholderValue(id)) return false;
    return /^ts[a-z0-9]+$/i.test(id);
  }

  function isValidLpn(value) {
    const lpn = decodeAmazonFieldText(value);
    if (!lpn || isAmazonPlaceholderValue(lpn)) return false;
    return /^lpn[a-z0-9]+$/i.test(lpn);
  }

  function releaseAutomation() {
    automationHoldDepth = Math.max(0, automationHoldDepth - 1);
    if (automationHoldDepth === 0) {
      const waiters = automationUnblockWaiters.splice(0);
      waiters.forEach((w) => w());
    }
  }

  function releaseAllAutomationHolds() {

  /** Reset every Container→LPN that errored back to pending (Not Done). */
  function forgetErrorLpns() {
    const keysForgotten = new Set();
    for (const r of getErrorResults()) {
      keysForgotten.add(rowKey(r.container, r.lpn));
    }

    if (!keysForgotten.size) {
      log('No error LPNs to reset.');
      updateResetErrorStatusesBtn();
      return 0;
    }

    for (const key of keysForgotten) {
      rowStatuses[key] = 'pending';
    }

    sortCodeResults = sortCodeResults.filter((r) => {
      const key = rowKey(r.container, r.lpn);
      if (!keysForgotten.has(key)) return true;
      return !isErrorResult(r);
    });

    rebuildMovedCountByContainer();
    bumpRowStatusVersion();
    scheduleRenderQueueList();
    lastExceptionKey = '';
    setErrorHalt(false);
    scheduleSaveState();
    updateResetErrorStatusesBtn();

    log(
      `Reset error status on ${keysForgotten.size} LPN(s) — marked Not Done. ` +
        'They will be retried when automation reaches them.'
    );
    return keysForgotten.size;
  }

  function copyTextToClipboard(text, successLog) {
    if (!text) return;
    if (typeof GM_setClipboard === 'function') {
      GM_setClipboard(text, { type: 'text', mimetype: 'text/plain' });
      if (successLog) log(successLog);
      return;
    }
    navigator.clipboard
      .writeText(text)
      .then(() => {
        if (successLog) log(successLog);
      })
      .catch(() => log('Copy failed.'));
  }

  /**
   * @param {string} title

   * @param {'error' | 'confirm'} [mode]
   * @param {boolean} [forceManualDismiss] — never auto-close (Amazon exceptions)
   */
  function showBlockerPopup(
    title,
    body,
    errorCtx,
    onDismiss,
    recoverAfterAutoClose = false,
    mode = 'error',
    forceManualDismiss = false
  ) {
    if (blockerAutoCloseTimer) {
      clearTimeout(blockerAutoCloseTimer);
      blockerAutoCloseTimer = null;
    }

    const overlay = document.createElement('div');
    overlay.id = 'ls-blocker-overlay';
    overlay.setAttribute('role', 'alertdialog');
    overlay.setAttribute('aria-modal', 'true');

    const card = document.createElement('div');
    card.className =
      mode === 'confirm' ? 'ls-blocker-card ls-blocker-confirm' : 'ls-blocker-card';

    const h = document.createElement('h2');
    h.className = 'ls-blocker-title';
    h.textContent = title;

    const pairLine = document.createElement('div');
    pairLine.className = 'ls-blocker-pair';
    pairLine.textContent = `Container: ${errorCtx.container || '—'}\nLPN: ${errorCtx.lpn || '—'}`;

    const p = document.createElement('p');
    p.className = 'ls-blocker-body';
    p.textContent = body;

    const actions = document.createElement('div');
    actions.className = 'ls-blocker-actions';

    const copyPairBtn = document.createElement('button');
    copyPairBtn.type = 'button';
    copyPairBtn.className = 'ls-blocker-secondary';
    copyPairBtn.textContent = 'Copy container + LPN';
    copyPairBtn.addEventListener('click', () => {
      copyTextToClipboard(
        `${errorCtx.container || ''}\t${errorCtx.lpn || ''}`,
        'Copied container + LPN.'
      );
    });

    const ok = document.createElement('button');
    ok.type = 'button';
    ok.className = mode === 'confirm' ? 'ls-blocker-ok ls-confirm-ok' : 'ls-blocker-ok';
    ok.textContent = 'OK';

    let dismissed = false;
    const dismiss = () => {
      if (dismissed || !overlay.isConnected) return;
      dismissed = true;
      if (blockerAutoCloseTimer) {
        clearTimeout(blockerAutoCloseTimer);
        blockerAutoCloseTimer = null;
      }
      overlay.remove();
      releaseAutomation();
      onDismiss?.();
      if (mode === 'error' && (autoCloseErrors || recoverAfterAutoClose)) {
        if (sessionArmed) {
          void skipToNextJobAfterAutoCloseError();
        } else if (autoCloseErrors) {
          void dismissAmazonExceptionModal('[Auto-close]');
        }
      }
    };

    ok.addEventListener('click', dismiss);

    const stopProgramBtn = document.createElement('button');
    stopProgramBtn.type = 'button';
    stopProgramBtn.className = 'ls-blocker-stop';
    stopProgramBtn.textContent = 'STOP PROGRAM';
    stopProgramBtn.title = 'Halt automation (same as Stop on the panel)';
    stopProgramBtn.addEventListener('click', () => {
      if (dismissed || !overlay.isConnected) return;
      dismissed = true;
      if (blockerAutoCloseTimer) {
        clearTimeout(blockerAutoCloseTimer);
        blockerAutoCloseTimer = null;
      }
      overlay.remove();
      releaseAutomation();
      onDismiss?.();
      void onStop();
    });

    actions.append(copyPairBtn, stopProgramBtn, ok);

    if (mode === 'error' && autoCloseErrors && !forceManualDismiss) {
      const hint = document.createElement('div');
      hint.className = 'ls-blocker-autoclose-hint';
      hint.textContent = 'Auto-clicking OK — then Continue / Update / next LPN';
      card.append(h, pairLine, p, hint, actions);
      blockerAutoCloseTimer = setTimeout(dismiss, AUTO_CLOSE_ERROR_MS);
    } else {
      card.append(h, pairLine, p, actions);
    }

    overlay.appendChild(card);
    document.body.appendChild(overlay);
    holdAutomation();
    ok.focus();
  }

  function rebuildMovedCountByContainer() {
    const next = new Map();
    for (const r of sortCodeResults) {
      if (jobRowStatus(r.sortCode) === 'Error') continue;
      const key = (r.container || '').trim().toLowerCase();
      if (!key) continue;
      next.set(key, (next.get(key) || 0) + 1);
    }
    movedCountByContainerKey = next;
  }

  function countLpnsMovedIntoContainer(container) {
    const key = (container || '').trim().toLowerCase();
    if (!key) return 0;
    return movedCountByContainerKey.get(key) || 0;
  }

  function invalidateMappingListCache() {
    cachedMappingList = null;
    cachedMappingText = '';
  }

  function jobRowStatus(sortCode) {
    return String(sortCode).startsWith('EXCEPTION') ? 'Error' : 'Moved';
  }

  function parseExceptionReason(sortCode) {
    const s = String(sortCode || '').trim();
    if (!s.startsWith('EXCEPTION')) return '';
    if (s.startsWith('EXCEPTION: ')) return s.slice('EXCEPTION: '.length);
    return s.replace(/^EXCEPTION:?\s*/i, '').trim();
  }

  function isErrorResult(r) {
    if (jobRowStatus(r.sortCode) === 'Error') return true;
    return rowStatuses[rowKey(r.container, r.lpn)] === 'error';
  }

  function getErrorDisplayReason(r) {
    if (jobRowStatus(r.sortCode) === 'Error') {
      return parseExceptionReason(r.sortCode) || 'Exception';
    }
    const parts = [];
    if (r.sortCode) parts.push(String(r.sortCode));
    const units = String(r.toteUnits || '');
    if (units.includes('FULL')) parts.push('Destination tote full');
    else if (r.tote) parts.push(`Tote: ${r.tote}`);
    return parts.join(' · ') || 'Error';
  }

  function getErrorResults() {
    return sortCodeResults.filter((r) => isErrorResult(r));
  }

  function buildErrorsListCopyTsv() {
    const rows = getErrorResults().map(
      (r) => `${r.container || ''}\t${r.lpn || ''}\t${getErrorDisplayReason(r)}`
    );
    return ['container\tlpn\treason', ...rows].join('\n');
  }

  function isValidDashboardSortCode(sortCode) {
    const s = String(sortCode || '').trim();
    if (!s) return false;
    if (s === '(not read)') return false;
    if (isAmazonPlaceholderValue(s)) return false;
    if (s.startsWith('EXCEPTION')) return false;
    return true;
  }

  /**
   * Always 5 routes from the panel, merged with session LPN counts per destination tote.
   * @returns {{
   *   routeIndex: number,
   *   routeLabel: string,
   *   sortCode: string,
   *   sortCodeDisplay: string,
   *   totalLpns: number,
   *   destinations: { toteId: string, lpnCount: number, units: number }[]
   * }[]}
   */
  function buildSortCodeDestinationDashboard() {
    /** @type {Map<string, { displayName: string, totes: Map<string, { displayName: string, lpnCount: number }> }>} */
    const fromResults = new Map();

    for (const r of sortCodeResults) {
      if (jobRowStatus(r.sortCode) === 'Error') continue;
      const sc = String(r.sortCode || '').trim();
      if (!isValidDashboardSortCode(sc)) continue;
      const tote = String(r.tote || '').trim();
      if (!tote || !isValidSourceContainerId(tote)) continue;

      const sk = sc.toLowerCase();
      if (!fromResults.has(sk)) {
        fromResults.set(sk, { displayName: sc, totes: new Map() });
      }
      const g = fromResults.get(sk);
      const tk = tote.toLowerCase();
      if (!g.totes.has(tk)) {
        g.totes.set(tk, { displayName: tote, lpnCount: 0 });
      }
      g.totes.get(tk).lpnCount += 1;
    }

    const slots = getSortSlotsFromInputs();
    const routes = [];

    for (let i = 0; i < SORT_SLOT_COUNT; i++) {
      const slot = slots[i] || { sortCode: '', toteIds: [] };
      const sortCode = String(slot.sortCode || '').trim();
      /** @type {Map<string, { displayName: string, lpnCount: number }>} */
      const toteMap = new Map();

      for (const id of slot.toteIds || []) {
        const tid = String(id).trim();
        if (!tid || !isValidSourceContainerId(tid)) continue;
        toteMap.set(tid.toLowerCase(), { displayName: tid, lpnCount: 0 });
      }

      if (isValidDashboardSortCode(sortCode)) {
        const rg = fromResults.get(sortCode.toLowerCase());
        if (rg) {
          for (const [tk, t] of rg.totes) {
            if (toteMap.has(tk)) {
              toteMap.get(tk).lpnCount = t.lpnCount;
            } else {
              toteMap.set(tk, { displayName: t.displayName, lpnCount: t.lpnCount });
            }
          }
        }
      }

      const destinations = [...toteMap.values()]
        .map((t) => ({
          toteId: t.displayName,
          lpnCount: t.lpnCount,
          units: getToteCount(t.displayName),
        }))
        .sort((a, b) => a.toteId.localeCompare(b.toteId));

      const totalLpns = destinations.reduce((n, d) => n + d.lpnCount, 0);

      routes.push({
        routeIndex: i + 1,
        routeLabel: `Route ${i + 1}`,
        sortCode,
        sortCodeDisplay: sortCode || 'Not set',
        destinations,
        totalLpns,
      });
    }

    return routes;
  }

  function buildDestinationDashboardTsv() {
    const rows = [];
    for (const route of buildSortCodeDestinationDashboard()) {
      const sc = route.sortCode || route.sortCodeDisplay;
      for (const dest of route.destinations) {
        rows.push(
          `${route.routeLabel}\t${sc}\t${dest.toteId}\t${dest.lpnCount}\t${dest.units}/${TOTE_MAX_UNITS}`
        );
      }
    }
    return ['route\tsort_code\tdestination_container\tlpns_moved\tunits', ...rows].join('\n');
  }

  /**
   * After an error: Continue (C) if shown, Update if shown, advance queue, resume page steps.
   * @param {string} [logPrefix]
   */
  async function recoverAndAdvanceToNextLpn(logPrefix = '[Recover]') {
    if (!sessionArmed) return false;

    postBarcodeHandled = false;
    barcodeDone = false;

    if (isAmazonExceptionShown()) {
      log(`${logPrefix} Clicking Amazon Continue (C)…`);
      const cont = await clickExceptionContinue();
      if (!cont) {
        log(`${logPrefix} Continue (#exception-btn) not clicked — fix page manually.`);
        return false;
      }
      await sleep(POST_UPDATE_WAIT_MS);
    }

    if (shouldClickUpdateToFinishJob()) {
      const updated = await finishJobWithUpdate(logPrefix);
      if (!updated) {
        log(`${logPrefix} Update not clicked — fix page manually.`);
        postBarcodeHandled = false;
        barcodeDone = false;
        return false;
      }
    } else {
      log(`${logPrefix} No completed scan to close — advancing queue without Update.`);
    }

    postBarcodeHandled = false;
    barcodeDone = false;
    const hasMore = advanceToNextContainer();
    resetStepFlagsForCurrentItem();
    lastSubmittedLpn = null;

    if (!hasMore) {
      log(`${logPrefix} Queue finished.`);
      return true;
    }

    if (isStaleBarcodeOnPage()) {
      await resetPageWorkflowState(logPrefix, { light: true });
    }

    await waitForPageReadyForNextLpn();

    setRowStatusByIndex(queueIndex, 'active');
    await continueWorkflowAfterLpnFinished();
    return true;
  }

  async function ensurePageMatchesQueueContainer() {
    const expected = getContainerValue();
    const onPage = readDisplayedSourceContainer();
    if (!expected || !onPage) return true;
    if (onPage.toUpperCase() === expected.toUpperCase()) return true;

    log(
      `[Workflow] Page container ${onPage} ≠ queue ${expected} — clicking Update to reset.`
    );
    await clickPageResetUpdates();
    sourceContainerDone = false;
    barcodeDone = false;
    postBarcodeHandled = false;
    lastSubmittedLpn = null;
    await sleep(STEP_SETTLE_MS);
    syncStepFlagsFromPageState();
    return (
      !readDisplayedSourceContainer() ||
      readDisplayedSourceContainer().toUpperCase() === expected.toUpperCase() ||
      isFieldStepActive(SOURCE_CONTAINER_SELECTOR)
    );
  }

  /**
   * @param {{ betweenJobs?: boolean, allowWhileRunning?: boolean, skipSortType?: boolean }} [opts]
   */
  async function resumeWorkflowFromPageState(opts = {}) {
    const allowWhileRunning = Boolean(opts.allowWhileRunning);
    const skipSortType = Boolean(opts.skipSortType);
    traceWorkflow('resume.enter', { ...opts, allowWhileRunning, skipSortType });
    if (!canRunAutomation()) {
      traceWorkflow('resume.abort', { reason: 'cannot-run' });
      return;
    }
    if (pageResetInProgress) {
      const start = Date.now();
      while (pageResetInProgress && Date.now() - start < PAGE_RESET_WAIT_MS) {
        await sleep(STEP_VERIFY_POLL_MS);
      }
      if (pageResetInProgress) return;
    }

    const clean = await ensureCleanPageBeforeWork('[Workflow]', {
      betweenJobs: Boolean(opts.betweenJobs),
    });
    if (!clean) {
      log('[Workflow] Paused — fix the page or press Start again.');
      return;
    }

    if (isStaleBarcodeOnPage()) {
      traceWorkflow('resume.staleBarcode', { onPage: readBarcodeOnPage() });
      log(
        `[Workflow] Page barcode ${readBarcodeOnPage()} ≠ queue ${getCurrentLpn()} — resetting page.`
      );
      await resetPageWorkflowState('[Workflow]', { light: opts.betweenJobs });
    } else if (!opts.betweenJobs && isBarcodeBeforeSourceStep()) {
      traceWorkflow('resume.barcodeBeforeSource', {});
      log(
        '[Workflow] LPN on page before source container — clicking Update to reset steps.'
      );
      await resetPageWorkflowState('[Workflow]');
    }

    syncStepFlagsFromPageState();

    const pageContainer = readDisplayedSourceContainer();
    const mappingContainer = getContainerValue();
    const containerMismatch =
      pageContainer &&
      mappingContainer &&
      pageContainer.toUpperCase() !== mappingContainer.toUpperCase();

    tryMarkSourceContainerDoneFromPage();

    if (containerMismatch && !isAwaitingSourceContainerScan()) {
      if (opts.betweenJobs) {
        traceWorkflow('resume.containerMismatch.betweenJobs', { pageContainer, mappingContainer });
        log(
          `[Job] Next container ${mappingContainer}` +
            (pageContainer ? ` (page label still shows ${pageContainer})` : '') +
            ' — scanning source container, not Update.'
        );
        sourceContainerDone = false;
        barcodeDone = false;
        postBarcodeHandled = false;
        lastSubmittedLpn = null;
      } else {
        const ok = await ensurePageMatchesQueueContainer();
        if (!ok) {
          log(
            `[Workflow] Still on container ${readDisplayedSourceContainer() || '?'} — ` +
              `queue expects ${mappingContainer}. Click Update on the page, then Start.`
          );
          return;
        }
      }
    } else if (containerMismatch && isAwaitingSourceContainerScan()) {
      sourceContainerDone = false;
      barcodeDone = false;
    }

    if (!skipSortType && needsSortTypeEntry()) {
      log('[Workflow] Sort type step — entering WDSorting.');
      const result = await runSortTypeStep();
      if (result.reason === 'exception' || isAmazonExceptionShown()) {
        await processAmazonException();
        return;
      }
      if (!result.advanced) {
        log('[Workflow] Sort type not accepted — click Update on the page if needed, then Start again.');
        return;
      }
      await sleep(STEP_SETTLE_MS);
      syncStepFlagsFromPageState();
    } else if (isSortTypeSatisfiedOnPage()) {
      log('[Workflow] Sort type already set on page — skipping scan.');
    }

    if (!canRunAutomation()) return;

    syncStepFlagsFromPageState();

    if (!sourceContainerDone) {
      const srcPlaceholder = isAmazonPlaceholderValue(readFieldValue(SOURCE_CONTAINER_SELECTOR));
      if (isFieldStepActive(SOURCE_CONTAINER_SELECTOR) || (isFieldVisible(SOURCE_CONTAINER_SELECTOR) && srcPlaceholder)) {
        if (!isFieldStepActive(SOURCE_CONTAINER_SELECTOR) && srcPlaceholder) {
          log('[Source container] Field visible — submitting container.');
        }
        await tryAutoSourceContainerForced(allowWhileRunning);
      }
    }

    if (!canRunAutomation()) return;

    syncStepFlagsFromPageState();

    if (
      isBarcodeEmptyErrorOnPage() &&
      isBarcodeFieldAwaitingLpn() &&
      getCurrentLpn()
    ) {
      log('[Workflow] "Barcode is empty" on page — submitting queue LPN.');
    }

    if (
      !barcodeDone &&
      getCurrentLpn() &&
      (canSubmitBarcodeNow() || isBarcodeFieldAwaitingLpn())
    ) {
      resolvePendingLpn({ forceLog: true });
      await tryAutoBarcode(allowWhileRunning);
    } else if (!barcodeDone && getCurrentLpn() && !sourceContainerDone) {
      log(
        '[Workflow] Waiting for source container — complete that step on the page or press Start again.'
      );
    } else if (!barcodeDone && getCurrentLpn() && isStaleBarcodeOnPage()) {
      log(
        `[Workflow] Stale barcode ${readBarcodeOnPage()} on page — press Start again after Update clears it.`
      );
    }
  }

  /**
   * One job = WDSorting through Update. Call after Update to begin the next queue row.
   * @param {boolean} [allowWhileRunning] — finish path runs inside handlePostBarcode (running=true)
   */
  async function startNextJob(allowWhileRunning = false) {
    if (!sessionArmed) return false;
    if (jobStartupInProgress) return false;

    jobStartupInProgress = true;
    try {
      return await runStartNextJobSteps(allowWhileRunning);
    } finally {
      jobStartupInProgress = false;
    }
  }

  async function runStartNextJobSteps(allowWhileRunning = false) {
    traceWorkflow('job.start', { allowWhileRunning });
    if (needsSortTypeEntry()) {
      log('[Job] Start — scanning WDSorting.');
      const result = await runSortTypeStep();
      if (result.reason === 'exception' || isAmazonExceptionShown()) {
        await processAmazonException();
        return false;
      }
      if (!result.advanced) {
        log('[Job] WDSorting not accepted — click Update on the page if needed.');
        return false;
      }
      await sleep(STEP_SETTLE_MS);
      syncStepFlagsFromPageState();
      log('[Job] WDSorting accepted — container → LPN.');
    } else if (isSortTypeSatisfiedOnPage()) {
      log('[Job] Start — sort type already set; continuing.');
    }

    tryMarkSourceContainerDoneFromPage();
    syncStepFlagsFromPageState();

    if (!canRunAutomation()) return false;
    await resumeWorkflowFromPageState({
      betweenJobs: true,
      allowWhileRunning,
      skipSortType: true,
    });
    return true;
  }

  /** After Update on a finished LPN: advance queue and run startNextJob. */
  async function continueWorkflowAfterLpnFinished() {
    if (!sessionArmed) return;
    await sleep(POST_UPDATE_WAIT_MS);
    if (!sessionArmed) return;
    await startNextJob(true);
  }

  async function skipToNextJobAfterAutoCloseError() {
    if (!sessionArmed) return;
    if (running) {
      log('[Recover] Auto-close skipped — another step is running.');
      return;
    }
    setRunningFlag(true);
    try {
      log('[Recover] Auto-close — same as OK: Continue/Update if needed, then next LPN.');
      await recoverAndAdvanceToNextLpn('[Recover]');
    } catch (err) {
      log('Recover error: ' + (err instanceof Error ? err.message : String(err)));
      console.error('[Liquidation Sort]', err);
    } finally {
      setRunningFlag(false);
    }
  }

  /**
   * @param {string} text
   * @returns {string[]}
   */
  function parseDestinationContainers(text) {
    const ids = [];
    const seen = new Set();
    for (const line of String(text).split(/\r?\n/)) {
      for (const part of line.split(/[\t,;]+/)) {
        const id = part.trim();
        if (!id) continue;
        const key = id.toLowerCase();
        if (seen.has(key)) continue;
        seen.add(key);
        ids.push(id);
      }
    }
    return ids;
  }

  /**
   * @param {unknown} raw
   * @returns {{ sortCode: string, toteIds: string[] }}
   */
  function normalizeSortSlot(raw) {
    const sortCode = String(raw?.sortCode || '').trim();
    let toteIds = [];
    if (Array.isArray(raw?.toteIds)) {
      toteIds = raw.toteIds.map((t) => String(t).trim()).filter(Boolean);
    } else if (raw?.toteId) {
      toteIds = parseDestinationContainers(String(raw.toteId));
    }
    return { sortCode, toteIds };
  }

  /**
   * @param {string} reason
   * @param {string} sortCode
   * @param {string} tote
   * @param {string[] | undefined} toteIds
   */
  function toteBlockerMessage(reason, sortCode, tote, toteIds) {
    switch (reason) {
      case 'no-mapping':
        return (
          `Sort code "${sortCode}" has no destination containers.\n\n` +
          'Add this sort code and one or more destination container IDs (one per line) in a route on the right.'
        );
      case 'tote-full':
        return (
          `Destination "${tote}" is full (${TOTE_MAX_UNITS}/${TOTE_MAX_UNITS} units).\n\n` +
          'Add another destination container on a new line in that route, or clear counts with Clear.'
        );
      case 'all-full': {
        const list = (toteIds && toteIds.length ? toteIds : [tote]).filter(Boolean).join('\n');
        return (
          `Every destination container for sort code "${sortCode}" is full (${TOTE_MAX_UNITS} units each):\n\n` +
          `${list}\n\n` +
          'Add another container ID on a new line in that route, then retry.'
        );
      }
      case 'field-missing':
        return (
          'The scan-tote field did not appear on the page.\n\n' +
          'The LPN may not have been accepted, or the workflow is on an unexpected step.'
        );
      case 'submit-failed':
        return (
          `Could not submit destination tote "${tote}".\n\n` +
          'Check the page for errors and that the tote ID is valid.'
        );
      default:
        return `Could not move LPN into tote (reason: ${reason || 'unknown'}).`;
    }
  }

  function rowKey(container, lpn) {
    return `${String(container).trim().toLowerCase()}|${String(lpn).trim()}`;
  }

  function getRowStatus(container, lpn, index) {
    const key = rowKey(container, lpn);
    const saved = rowStatuses[key];
    if (saved === 'complete' || saved === 'error') return saved;
    if (sessionArmed && index === queueIndex) return 'active';
    return saved || 'pending';
  }

  /** Row already finished successfully (green Done dot). */
  function isRowDone(index) {
    const pair = getMappingList()[index];
    if (!pair) return false;
    return rowStatuses[rowKey(pair.container, pair.lpn)] === 'complete';
  }

  /**
   * Advance queueIndex past any rows already marked Done.
   * @returns {boolean} true if a pending row remains
   */
  function skipDoneRowsFromCurrent() {
    const list = getMappingList();
    const start = queueIndex;
    while (queueIndex < list.length && isRowDone(queueIndex)) {
      queueIndex += 1;
    }
    if (queueIndex > start) {
      log(`Skipped ${queueIndex - start} row(s) already marked Done.`);
      scheduleRenderQueueList();
      scheduleSaveState();
    }
    return queueIndex < list.length;
  }

  /** Skip Done rows; leave queueIndex on the next pending row. */
  function ensureCurrentRowIsPending() {
    if (!isRowDone(queueIndex)) return true;
    if (!skipDoneRowsFromCurrent()) return false;
    resetStepFlagsForCurrentItem();
    setRowStatusByIndex(queueIndex, 'active');
    return true;
  }

  function setRowStatusByIndex(index, status) {
    const pair = getMappingList()[index];
    if (!pair) return;
    rowStatuses[rowKey(pair.container, pair.lpn)] = status;
    bumpRowStatusVersion();
    scheduleRenderQueueList();
    scheduleSaveState();
    if (status === 'error') updateResetErrorStatusesBtn();
  }

  function syncRowStatusesFromMapping() {
    const list = getMappingList();
    const next = {};
    for (const pair of list) {
      const key = rowKey(pair.container, pair.lpn);
      next[key] = rowStatuses[key] || 'pending';
    }
    rowStatuses = next;
    bumpRowStatusVersion();
    scheduleRenderQueueList();
    scheduleSaveState();
  }

  function appendQueueListRow(pair, index, frag) {
    const status = getRowStatus(pair.container, pair.lpn, index);
    const row = document.createElement('div');
    row.className = 'ls-queue-row';
    if (index === queueIndex) row.classList.add('ls-queue-row-current');
    const dot = document.createElement('span');
    dot.className = `ls-dot ls-dot-${status}`;
    dot.title =
      status === 'complete'
        ? 'Complete'
        : status === 'active'
          ? 'In progress'
          : status === 'error'
            ? 'Error'
            : 'Not started';
    const text = document.createElement('span');
    text.className = 'ls-queue-text';
    text.textContent = `${pair.container}\t${pair.lpn}`;
    row.append(dot, text);
    frag.appendChild(row);
  }

  function bumpRowStatusVersion() {
    rowStatusVersion++;
    progressStatsCache = null;
  }

  function getProgressStats() {
    if (progressStatsCache && progressStatsCacheVersion === rowStatusVersion) {
      return progressStatsCache;
    }
    const list = getMappingList();
    const total = list.length;
    let moved = 0;
    let error = 0;
    for (const pair of list) {
      const s = rowStatuses[rowKey(pair.container, pair.lpn)];
      if (s === 'complete') moved++;
      else if (s === 'error') error++;
    }
    const percent = total ? Math.round((moved / total) * 100) : 0;
    progressStatsCache = {
      total,
      moved,
      error,
      remaining: Math.max(0, total - moved),
      percent,
    };
    progressStatsCacheVersion = rowStatusVersion;
    return progressStatsCache;
  }

  function getActiveElapsedMs() {
    if (!sessionStartedAt) return 0;
    let paused = timingPausedMs;
    if (pauseStartedAt) paused += Date.now() - pauseStartedAt;
    return Math.max(0, Date.now() - sessionStartedAt - paused);
  }

  function estimateRemainingMs(stats) {
    const { total, moved } = stats;
    const left = total - moved;
    if (left <= 0) return 0;
    if (!sessionStartedAt || moved === 0) return null;
    const avg = getActiveElapsedMs() / moved;
    return Math.round(avg * left);
  }

  function formatCountdown(ms) {
    if (ms == null || !Number.isFinite(ms)) return '—:—:—';
    const s = Math.max(0, Math.ceil(ms / 1000));
    const h = Math.floor(s / 3600);
    const m = Math.floor((s % 3600) / 60);
    const sec = s % 60;
    if (h > 0) {
      return `${h}:${String(m).padStart(2, '0')}:${String(sec).padStart(2, '0')}`;
    }
    return `${String(m).padStart(2, '0')}:${String(sec).padStart(2, '0')}`;
  }

  function formatEtaClock(deadlineMs) {
    if (!deadlineMs || !Number.isFinite(deadlineMs)) return '—';
    return new Date(deadlineMs).toLocaleTimeString([], {
      hour: 'numeric',
      minute: '2-digit',
    });
  }

  function resetProgressTiming() {
    sessionStartedAt = null;
    timingPausedMs = 0;
    pauseStartedAt = null;
    frozenCountdownMs = null;
  }

  function armProgressTiming() {
    if (!sessionStartedAt) sessionStartedAt = Date.now();
    if (pauseStartedAt) {
      timingPausedMs += Date.now() - pauseStartedAt;
      pauseStartedAt = null;
      frozenCountdownMs = null;
    }
  }

  function pauseProgressTiming() {
    if (!sessionStartedAt || pauseStartedAt) return;
    pauseStartedAt = Date.now();
    const stats = getProgressStats();
    const rem = estimateRemainingMs(stats);
    if (rem != null) frozenCountdownMs = rem;
  }

  function updateProgressUI() {
    if (!progressBarFillEl) return;
    const stats = getProgressStats();
    const done = stats.total > 0 && stats.moved >= stats.total;

    progressBarFillEl.style.width = `${stats.percent}%`;
    if (progressPctEl) progressPctEl.textContent = `${stats.percent}%`;
    if (progressDetailEl) {
      progressDetailEl.textContent =
        stats.total > 0
          ? `${stats.moved} / ${stats.total} moved${stats.error ? ` · ${stats.error} error` : ''}`
          : 'Paste queue to begin';
    }

    if (!etaClockEl || !countdownEl) return;

    if (done) {
      etaClockEl.textContent = 'Done';
      countdownEl.textContent = '00:00';
      return;
    }

    if (pauseStartedAt) {
      etaClockEl.textContent = 'Paused';
      countdownEl.textContent = formatCountdown(frozenCountdownMs);
      return;
    }

    if (!sessionStartedAt || stats.moved === 0) {
      etaClockEl.textContent = 'ETA —';
      countdownEl.textContent = '—:—:—';
      return;
    }

    const remainingMs = estimateRemainingMs(stats);
    if (remainingMs == null) {
      etaClockEl.textContent = 'ETA —';
      countdownEl.textContent = '—:—:—';
      return;
    }

    frozenCountdownMs = remainingMs;
    const deadline = Date.now() + remainingMs;
    etaClockEl.textContent = `ETA ${formatEtaClock(deadline)}`;
    countdownEl.textContent = formatCountdown(remainingMs);
  }

  function startProgressTicker() {
    if (progressTicker) return;
    progressTicker = setInterval(() => {
      if (!pauseStartedAt && sessionStartedAt) {
        const stats = getProgressStats();
        if (stats.moved > 0 && stats.moved < stats.total) {
          const rem = estimateRemainingMs(stats);
          if (rem != null) frozenCountdownMs = rem;
        }
      }
      updateProgressUI();
    }, 1000);
  }

  function renderQueueList() {
    if (!queueListEl) return;
    const list = getMappingList();
    queueListEl.innerHTML = '';
    if (!list.length) {
      queueListEl.innerHTML =
        '<div class="ls-queue-empty">Paste container → LPN lines above</div>';
      updateProgressUI();
      return;
    }

    const frag = document.createDocumentFragment();
    list.forEach((pair, index) => {
      appendQueueListRow(pair, index, frag);
    });
    queueListEl.appendChild(frag);
    updateProgressUI();
  }

  function getSortSlotsFromInputs() {
    const slots = [];
    for (let i = 0; i < SORT_SLOT_COUNT; i++) {
      slots.push({
        sortCode: (slotSortInputs[i]?.value || '').trim(),
        toteIds: parseDestinationContainers(slotToteInputs[i]?.value || ''),
      });
    }
    return slots;
  }

  function findSortCodeSlotIndex(sortCode) {
    const key = (sortCode || '').trim().toLowerCase();
    if (!key) return -1;
    const slots = getSortSlotsFromInputs();
    for (let i = 0; i < slots.length; i++) {
      const slotKey = (slots[i].sortCode || '').trim().toLowerCase();
      if (slotKey && slotKey === key) return i;
    }
    return -1;
  }

  function findFirstEmptySortCodeSlotIndex() {
    const slots = getSortSlotsFromInputs();
    for (let i = 0; i < slots.length; i++) {
      if (!(slots[i].sortCode || '').trim()) return i;
    }
    return -1;
  }

  /**
   * Puts an unknown sort code into the first empty route (1–5) after user confirms.
   * @param {string} sortCode
   * @returns {Promise<{ assigned: boolean, slotIndex: number }>}
   */
  async function ensureSortCodeAssignedToSlot(sortCode) {
    const trimmed = (sortCode || '').trim();
    if (!trimmed) return { assigned: false, slotIndex: -1 };

    const existingIdx = findSortCodeSlotIndex(trimmed);
    if (existingIdx >= 0) {
      return { assigned: false, slotIndex: existingIdx };
    }

    const emptyIdx = findFirstEmptySortCodeSlotIndex();
    if (emptyIdx < 0) {
      log(`[Sort code] All ${SORT_SLOT_COUNT} routes have a sort code — cannot auto-assign "${trimmed}".`);
      return { assigned: false, slotIndex: -1 };
    }

    await alertUserConfirm(
      'New sort code — confirm route',
      `Amazon showed sort code:\n"${trimmed}"\n\n` +
        `Save it to Route ${emptyIdx + 1} on the right?\n\n` +
        'After OK, add destination container ID(s) on that route if needed.',
      getBlockerContext()
    );

    if (slotSortInputs[emptyIdx]) {
      slotSortInputs[emptyIdx].value = trimmed;
    }
    log(`[Sort code] Assigned "${trimmed}" → Route ${emptyIdx + 1}.`);
    void storageSetJson(STORAGE_KEY_SORT_SLOTS, getSortSlotsFromInputs());
    scheduleSaveState();
    renderSlotCounts();
    return { assigned: true, slotIndex: emptyIdx };
  }

  /**
   * @param {string} toteId
   * @param {string} sortCode
   * @param {number} count
   */
  async function confirmDestinationToteFull(toteId, sortCode, count) {
    const key = (toteId || '').trim().toLowerCase();
    if (!key || count < TOTE_MAX_UNITS) return;
    if (acknowledgedFullTotes.has(key)) return;

    await alertUserConfirm(
      `Destination full (${TOTE_MAX_UNITS} units)`,
      `Destination container:\n"${toteId}"\n` +
        `Sort code: ${sortCode || '(unknown)'}\n\n` +
        `This destination now has ${count}/${TOTE_MAX_UNITS} LPNs.\n\n` +
        'Click OK when you are ready to use the next destination on that route ' +
        '(add another ID on a new line on the right if needed).',
      getBlockerContext()
    );
    acknowledgedFullTotes.add(key);
  }

  function renderSlotCounts() {
    for (let i = 0; i < SORT_SLOT_COUNT; i++) {
      const toteIds = parseDestinationContainers(slotToteInputs[i]?.value || '');
      const el = slotCountEls[i];
      if (!el) continue;
      el.classList.remove('ls-count-full');
      if (!toteIds.length) {
        el.textContent = `Add destinations (one per line, max ${TOTE_MAX_UNITS} each)`;

      const parts = toteIds.map((id) => {

        const full = n >= TOTE_MAX_UNITS;
        return `${id}: ${n}/${TOTE_MAX_UNITS}${full ? ' FULL' : ''}`;
      });
      el.textContent = parts.join(' · ');
      el.classList.toggle('ls-count-full', toteIds.every((id) => getToteCount(id) >= TOTE_MAX_UNITS));
    }
  }

  function setNativeInputValue(input, value) {
    const proto =
      input instanceof HTMLTextAreaElement
        ? HTMLTextAreaElement.prototype
        : HTMLInputElement.prototype;
    const desc = Object.getOwnPropertyDescriptor(proto, 'value');
    if (desc && desc.set) {
      desc.set.call(input, value);
    } else {
      input.value = value;
    }
  }

  function keyInitFor(ch, ctrl) {
    let code = '';
    if (/[a-zA-Z]/.test(ch)) code = 'Key' + ch.toUpperCase();
    else if (/[0-9]/.test(ch)) code = 'Digit' + ch;
    return {
      key: ch,
      code,
      bubbles: true,
      cancelable: true,
      composed: true,
      ctrlKey: !!ctrl,
      keyCode: ch.charCodeAt(0),
      which: ch.charCodeAt(0),
    };
  }

  function isFieldStepActive(selector) {
    const field = document.querySelector(selector);
    if (!field) return false;
    const style = getComputedStyle(field);
    if (style.display === 'none' || style.visibility === 'hidden' || Number(style.opacity) === 0) {
      return false;
    }
    const rect = field.getBoundingClientRect();
    if (rect.width <= 0 || rect.height <= 0) return false;
    if (field.offsetParent === null && style.position !== 'fixed') return false;
    return true;
  }

  function isFieldVisible(selector) {
    const field = document.querySelector(selector);
    if (!field) return false;
    const style = getComputedStyle(field);
    if (style.display === 'none' || style.visibility === 'hidden' || Number(style.opacity) === 0) {
      return false;
    }
    const rect = field.getBoundingClientRect();
    return rect.width > 0 && rect.height > 0;
  }

  function readFieldValue(selector) {
    const field = document.querySelector(selector);
    if (!field) return '';
    return decodeAmazonFieldText(field.value);
  }

  function isElementClickable(el) {
    if (!el) return false;
    const style = getComputedStyle(el);
    if (style.display === 'none' || style.visibility === 'hidden' || Number(style.opacity) === 0) {
      return false;
    }
    const rect = el.getBoundingClientRect();
    return rect.width > 0 && rect.height > 0;
  }

  /** Source container shown in the page summary row (not only the hidden scan field). */
  function readDisplayedSourceContainer() {
    const fieldVal = readFieldValue(SOURCE_CONTAINER_SELECTOR);
    if (isValidSourceContainerId(fieldVal)) return fieldVal;

    for (const el of document.querySelectorAll('td, th, span, label, div, a')) {
      const label = (el.textContent || '').replace(/\s+/g, ' ').trim();
      if (!/^source\s*container$/i.test(label)) continue;
      const row = el.closest('tr') || el.parentElement?.parentElement;
      if (!row) continue;
      for (const cell of row.querySelectorAll('td, span, div, a')) {
        const id = decodeAmazonFieldText(cell.textContent || '');
        if (isValidSourceContainerId(id)) return id;
      }
    }
    return '';
  }

  function readBarcodeOnPage() {
    const val = readFieldValue(BARCODE_SELECTOR);
    return isValidLpn(val) ? val : '';
  }

  function isStaleBarcodeOnPage() {
    const onPage = readBarcodeOnPage();
    const expected = getCurrentLpn();
    if (!onPage || !expected) return false;
    return onPage.toUpperCase() !== expected.toUpperCase();
  }

  /** Sort code text still in #sort-code-id (may linger in DOM after Update). */

    const text = readSortCodeText();
    return Boolean(text && text.length > 2);
  }

  /**
   * Both scan fields visible with placeholders — normal at workflow start on Amazon.
   * Only treat as stuck after sort type was already accepted (WDSorting entered).
   */
  function hasConflictingScanSteps() {
    const sortVal = readFieldValue(SORT_TYPE_SELECTOR);
    const sortTypeEntered =
      !isAmazonPlaceholderValue(sortVal) && /wdsorting/i.test(sortVal);
    if (!sortTypeEntered) return false;
    return (
      isAmazonPlaceholderValue(readFieldValue(SOURCE_CONTAINER_SELECTOR)) &&
      isBarcodeFieldAwaitingLpn() &&
      !readDisplayedSourceContainer() &&
      !isValidSourceContainerId(readFieldValue(SOURCE_CONTAINER_SELECTOR))
    );
  }

  function needsSortTypeEntry() {
    if (isSortTypeSatisfiedOnPage()) return false;
    const sortVal = readFieldValue(SORT_TYPE_SELECTOR);
    if (!isAmazonPlaceholderValue(sortVal) && /wdsorting/i.test(sortVal)) return false;
    return isFieldVisible(SORT_TYPE_SELECTOR) || isFieldStepActive(SORT_TYPE_SELECTOR);
  }

  /**
   * Prior LPN left sort code in #sort-code-id while barcode waits — only when mid-item,
   * not after Update when sort-type step is back (old label often stays in DOM).
   */
  function isStaleSortCodeBeforeBarcode() {
    if (!isSortCodeElementPopulated()) return false;
    if (!isBarcodeFieldAwaitingLpn()) return false;
    if (isFieldStepActive(SCAN_TOTE_SELECTOR)) return false;
    const toteVal = readFieldValue(SCAN_TOTE_SELECTOR);
    if (!isAmazonPlaceholderValue(toteVal) && /^ts[a-z0-9]+$/i.test(toteVal)) {
      return false;
    }
    const sortTypeVal = readFieldValue(SORT_TYPE_SELECTOR);
    if (
      isFieldStepActive(SORT_TYPE_SELECTOR) ||
      (isFieldVisible(SORT_TYPE_SELECTOR) && isAmazonPlaceholderValue(sortTypeVal))
    ) {
      return false;
    }
    if (!isAmazonPlaceholderValue(sortTypeVal) && /wdsorting/i.test(sortTypeVal)) {
      return false;
    }
    return true;
  }

  function isSortTypeSatisfiedOnPage() {
    if (isStaleBarcodeOnPage()) return false;
    if (isStaleSortCodeBeforeBarcode()) return false;
    if (isSortCodeReady()) return true;
    const sortVal = readFieldValue(SORT_TYPE_SELECTOR);
    if (!isAmazonPlaceholderValue(sortVal) && /wdsorting/i.test(sortVal)) return true;
    if (readDisplayedSourceContainer()) return true;
    const srcVal = readFieldValue(SOURCE_CONTAINER_SELECTOR);
    if (isValidSourceContainerId(srcVal)) return true;
    if (isFieldStepActive(SCAN_TOTE_SELECTOR)) return true;
    return false;
  }

  async function waitForSortCodeCleared(timeoutMs = PAGE_RESET_WAIT_MS) {
    const start = Date.now();
    while (Date.now() - start < timeoutMs) {
      if (!isSortCodeElementPopulated()) return true;
      if (isFieldStepActive(SORT_TYPE_SELECTOR)) return true;
      const sortTypeVal = readFieldValue(SORT_TYPE_SELECTOR);
      if (isFieldVisible(SORT_TYPE_SELECTOR) && isAmazonPlaceholderValue(sortTypeVal)) {
        return true;
      }
      await sleep(STEP_VERIFY_POLL_MS);
    }
    return !isSortCodeElementPopulated() || isFieldStepActive(SORT_TYPE_SELECTOR);
  }

  async function resetPageWorkflowState(logPrefix = '[Workflow]', opts = {}) {
    const light = Boolean(opts.light);
    const maxAttempts = light ? 1 : 3;
    traceWorkflow('reset.begin', { logPrefix, light, maxAttempts });

    if (pageResetInProgress) {
      log(`${logPrefix} Page reset already in progress — waiting…`);
      const start = Date.now();
      while (pageResetInProgress && Date.now() - start < PAGE_RESET_WAIT_MS) {
        await sleep(STEP_VERIFY_POLL_MS);
      }
      if (pageResetInProgress) return;
    }
    pageResetInProgress = true;
    try {
      log(`${logPrefix} Resetting page (Update)…`);
      for (let attempt = 1; attempt <= maxAttempts; attempt++) {
        await clickPageResetUpdates();
        await sleep(POST_UPDATE_WAIT_MS);
        if (!isSortCodeElementPopulated()) break;
        if (attempt < maxAttempts) {
          log(`${logPrefix} Still stale after Update (${attempt}/${maxAttempts}) — retrying…`);
        }
      }
      await waitForSortCodeCleared();
      sourceContainerDone = false;
      barcodeDone = false;
      postBarcodeHandled = false;
      lastSubmittedLpn = null;
      syncStepFlagsFromPageState();
      if (isStaleSortCodeBeforeBarcode()) {
        log(
          `${logPrefix} Page may still be stuck — click Update on Amazon once, then Start again.`
        );
      } else {
        log(`${logPrefix} Page reset complete.`);
      }
    } finally {
      pageResetInProgress = false;
      traceWorkflow('reset.end', { logPrefix });
    }
  }

  /**
   * @param {string} [logPrefix]
   * @param {{ betweenJobs?: boolean }} [opts] — betweenJobs: do not halt queue if sort-code label lingers in DOM
   */
  async function ensureCleanPageBeforeWork(logPrefix = '[Workflow]', opts = {}) {
    const betweenJobs = Boolean(opts.betweenJobs);

    if (isStaleBarcodeOnPage()) {
      await resetPageWorkflowState(logPrefix);
      if (isStaleBarcodeOnPage()) {
        log(`${logPrefix} Wrong barcode still on page after reset — cannot continue.`);
        return false;
      }
    }

    if (!betweenJobs && isStaleSortCodeBeforeBarcode()) {
      await resetPageWorkflowState(logPrefix);
      if (isStaleSortCodeBeforeBarcode()) {
        log(
          `${logPrefix} Sort code still on screen — click Update on Amazon, then press Start again.`
        );
        return false;
      }
    }

    return true;
  }

  /**
   * Wrong order: a real LPN is on the page before the source container was accepted.
   * Not the normal new-job layout (sort-type + barcode placeholders + lingering sort-code label).
   */
  function isBarcodeBeforeSourceStep() {
    if (sourceContainerDone || isSortCodeReady()) return false;
    if (needsSortTypeEntry()) return false;

    const sortTypeVal = readFieldValue(SORT_TYPE_SELECTOR);
    if (isFieldStepActive(SORT_TYPE_SELECTOR)) return false;
    if (isFieldVisible(SORT_TYPE_SELECTOR) && isAmazonPlaceholderValue(sortTypeVal)) {
      return false;
    }

    const barcodeVal = readFieldValue(BARCODE_SELECTOR);
    if (!isValidLpn(barcodeVal)) return false;

    const srcVal = readFieldValue(SOURCE_CONTAINER_SELECTOR);
    if (isValidSourceContainerId(srcVal) || readDisplayedSourceContainer()) return false;
    return true;
  }

  function isBarcodeEmptyErrorOnPage() {
    const body = (document.body?.innerText || '').replace(/\s+/g, ' ');
    return /barcode\s+is\s+empty/i.test(body);
  }

  /**
   * Infer sourceContainerDone / barcodeDone from Amazon UI so Start does not
   * blindly reset when the page is already mid-workflow.
   */
  function syncStepFlagsFromPageState() {
    const pageContainer = readDisplayedSourceContainer();
    const mappingContainer = getContainerValue();
    const sortOnPage = isSortCodeElementPopulated();
    const sortReady = isSortCodeReady();
    const barcodeVal = readFieldValue(BARCODE_SELECTOR);
    const awaitingBarcode = isBarcodeFieldAwaitingLpn();
    const currentLpn = getCurrentLpn();

    if (
      pageContainer &&
      mappingContainer &&
      pageContainer.toUpperCase() !== mappingContainer.toUpperCase()
    ) {
      sourceContainerDone = false;
      barcodeDone = false;
      return;
    }

    if (isStaleBarcodeOnPage()) {
      sourceContainerDone = false;
      barcodeDone = false;
      return;
    }

    if (sortOnPage && awaitingBarcode && !needsSortTypeEntry()) {
      sourceContainerDone = false;
      barcodeDone = false;
      return;
    }

    if (sortReady) {
      sourceContainerDone = true;
      if (isValidLpn(barcodeVal) && currentLpn) {
        barcodeDone = barcodeVal.toUpperCase() === currentLpn.toUpperCase();
      } else if (!awaitingBarcode && !isFieldStepActive(BARCODE_SELECTOR)) {
        barcodeDone = true;
      } else {
        barcodeDone = false;
      }
      return;
    }

    const srcFieldVal = readFieldValue(SOURCE_CONTAINER_SELECTOR);
    if (isValidSourceContainerId(srcFieldVal)) {
      sourceContainerDone = true;
    } else if (pageContainer && isValidSourceContainerId(pageContainer)) {
      sourceContainerDone = true;
    } else if (sortReady) {
      sourceContainerDone = true;
    }

    if (awaitingBarcode || isFieldStepActive(BARCODE_SELECTOR)) {
      barcodeDone = false;
    }
  }

  function isBarcodeFieldAwaitingLpn() {
    if (!isFieldVisible(BARCODE_SELECTOR)) return false;
    return isAmazonPlaceholderValue(readFieldValue(BARCODE_SELECTOR));
  }

  function canSubmitBarcodeNow() {
    return isFieldStepActive(BARCODE_SELECTOR) || isBarcodeFieldAwaitingLpn();
  }

  /**
   * @param {string} selector
   * @param {string} placeholder
   * @param {string} typedValue
   */
  async function waitForStepAdvanced(selector, placeholder, typedValue, label) {
    const start = Date.now();
    while (Date.now() - start < STEP_VERIFY_TIMEOUT_MS) {
      if (isAmazonExceptionShown()) {
        return { advanced: false, reason: 'exception' };
      }
      if (fieldStepAdvanced(selector, placeholder, typedValue)) {
        log(`[${label}] Verified step advanced (${Date.now() - start}ms).`);
        return { advanced: true };
      }
      await sleep(STEP_VERIFY_POLL_MS);
    }
    log(`[${label}] Step verify timed out after ${STEP_VERIFY_TIMEOUT_MS / 1000}s.`);
    return {
      advanced: fieldStepAdvanced(selector, placeholder, typedValue),
      reason: 'verify-timeout',
    };
  }

  async function verifySourceContainerAccepted(container) {
    const expected = decodeAmazonFieldText(container).toUpperCase();
    const start = Date.now();
    while (Date.now() - start < STEP_VERIFY_TIMEOUT_MS) {
      if (isAmazonExceptionShown()) return { ok: false, reason: 'exception' };
      const srcVal = readFieldValue(SOURCE_CONTAINER_SELECTOR);
      const onPage = readDisplayedSourceContainer();
      const srcOk =
        isValidSourceContainerId(srcVal) && srcVal.toUpperCase() === expected;
      const pageOk = onPage && onPage.toUpperCase() === expected;
      if ((srcOk || pageOk) && (isBarcodeFieldAwaitingLpn() || isFieldStepActive(BARCODE_SELECTOR))) {
        log(`[Source container] Verified ${container} accepted (${Date.now() - start}ms).`);
        return { ok: true };
      }
      if (
        !isFieldStepActive(SOURCE_CONTAINER_SELECTOR) &&
        (isBarcodeFieldAwaitingLpn() || isFieldStepActive(BARCODE_SELECTOR))
      ) {
        log(`[Source container] Barcode step open (${Date.now() - start}ms).`);
        return { ok: true };
      }
      await sleep(STEP_VERIFY_POLL_MS);
    }
    return { ok: false, reason: 'verify-timeout' };
  }

  async function verifyBarcodeScanAccepted(lpn) {
    const expected = decodeAmazonFieldText(lpn).toUpperCase();
    const start = Date.now();
    while (Date.now() - start < STEP_VERIFY_TIMEOUT_MS) {
      if (isAmazonExceptionShown()) return { ok: false, reason: 'exception' };
      if (isStaleSortCodeBeforeBarcode()) {
        return { ok: false, reason: 'stale-sort-before-barcode' };
      }
      if (isSortCodeReady()) {
        const fieldVal = readFieldValue(BARCODE_SELECTOR);
        if (!isValidLpn(fieldVal) || fieldVal.toUpperCase() === expected) {
          log(`[Barcode] Sort code ready (${Date.now() - start}ms): ${readSortCodeText()}`);
          return { ok: true, sortCode: readSortCodeText() };
        }
      }
      const fieldVal = readFieldValue(BARCODE_SELECTOR);
      if (isValidLpn(fieldVal) && fieldVal.toUpperCase() === expected && isSortCodeReady()) {
        return { ok: true, sortCode: readSortCodeText() };
      }
      await sleep(STEP_VERIFY_POLL_MS);
    }
    return { ok: false, reason: 'verify-timeout' };
  }

  async function verifyToteMoveOutcome() {
    await sleep(STEP_SETTLE_MS);
    const start = Date.now();
    while (Date.now() - start < MOVE_VERIFY_TIMEOUT_MS) {
      if (isAmazonExceptionShown()) {
        await sleep(150);
        if (isAmazonExceptionShown()) {
          return { ok: false, reason: 'move-exception' };
        }
      }
      const toteVal = readFieldValue(SCAN_TOTE_SELECTOR);
      const toteStillActive =
        isFieldStepActive(SCAN_TOTE_SELECTOR) && !isAmazonPlaceholderValue(toteVal);
      if (!toteStillActive) {
        log(`[Tote] Move settled (${Date.now() - start}ms).`);
        return { ok: true };
      }
      await sleep(STEP_VERIFY_POLL_MS);
    }
    if (isAmazonExceptionShown()) return { ok: false, reason: 'move-exception' };
    log('[Tote] Move verify timed out — tote field still active.');
    return { ok: false, reason: 'move-timeout' };
  }

  async function waitForPageReadyForNextLpn() {
    const start = Date.now();
    while (Date.now() - start < PAGE_RESET_WAIT_MS) {
      if (isAmazonExceptionShown()) return false;
      if (isStaleSortCodeBeforeBarcode() || isStaleBarcodeOnPage()) {
        await sleep(STEP_VERIFY_POLL_MS);
        continue;
      }
      if (
        isFieldStepActive(SORT_TYPE_SELECTOR) ||
        isFieldStepActive(SOURCE_CONTAINER_SELECTOR) ||
        (isBarcodeFieldAwaitingLpn() && !isSortCodeElementPopulated())
      ) {
        return true;
      }
      await sleep(STEP_VERIFY_POLL_MS);
    }
    return !isStaleSortCodeBeforeBarcode() && !isStaleBarcodeOnPage();
  }

  function fieldStepAdvanced(selector, placeholder, typedValue) {
    const field = document.querySelector(selector);
    const fieldVal = field ? decodeAmazonFieldText(field.value) : '';
    const stillActive = isFieldStepActive(selector);
    const typed = decodeAmazonFieldText(typedValue);

    if (selector === BARCODE_SELECTOR) {
      if (!isValidLpn(typed)) return false;
      if (isAmazonPlaceholderValue(fieldVal)) {
        return isSortCodeReady();
      }
      if (fieldVal.toUpperCase() === typed.toUpperCase()) return true;
      if (!stillActive && isSortCodeReady() && !isAmazonPlaceholderValue(fieldVal)) return true;
      return false;
    }

    return (
      !stillActive ||
      (fieldVal &&
        fieldVal.toLowerCase() !== placeholder.toLowerCase() &&
        !isAmazonPlaceholderValue(fieldVal))
    );
  }

  /**
   * Ordered container → LPN pairs (top line = first to process on Start).
   * @param {string} [text]
   * @returns {{ container: string, lpn: string }[]}
   */
  function parseContainerLpnList(text) {
    const src = text !== undefined ? text : getMappingText();
    const list = [];
    for (const line of src.split(/\r?\n/)) {
      const trimmed = line.trim();
      if (!trimmed || trimmed.startsWith('#')) continue;
      const parts = trimmed.split(/[\t\s]+/).filter(Boolean);
      if (parts.length >= 2) {
        const container = decodeAmazonFieldText(parts[0]);
        const lpn = decodeAmazonFieldText(parts[1]);
        if (!isValidSourceContainerId(container) || !isValidLpn(lpn)) {
          continue;
        }
        list.push({ container, lpn });
      }
    }
    return list;
  }

  function getMappingList() {
    const text = getMappingText();
    if (cachedMappingList && text === cachedMappingText) {
      return cachedMappingList;
    }
    cachedMappingText = text;
    cachedMappingList = parseContainerLpnList(text);
    return cachedMappingList;
  }

    const { quiet = false, forceLog = false } = options;
    const pair = getCurrentPair();

        if (sessionArmed) {
          const ctx = getBlockerContext();
          void alertLpnBlocked(
            `no-lpn|${queueIndex}|${ctx.container}`,

            `Queue row ${queueIndex + 1} has container "${ctx.container}" but no LPN.\n\n` +
              'Add a tab-separated LPN on that line in the mapping list.',
            ctx
          );
        }
      } else if (forceLog) {
        log('[Barcode] No container at current queue index — check mapping list.');
        if (sessionArmed) {
          const ctx = getBlockerContext();
          void alertLpnBlocked(
            `no-queue|${queueIndex}`,
            'Cannot scan — queue row missing',
            `There is no container/LPN pair at queue index ${queueIndex + 1}.\n\n` +
              'Check your mapping list or press Clear and paste again.',
            ctx
          );

  /**
   * @param {string} sortCode
   * @returns {{ tote: string | null, toteIds: string[], reason?: string }}
   */
  function pickDestinationToteForSortCode(sortCode) {
    if (!sortCode) return { tote: null, toteIds: [] };
    const key = sortCode.trim().toLowerCase();
    for (const slot of getSortSlotsFromInputs()) {
      if (!slot.sortCode || slot.sortCode.trim().toLowerCase() !== key) continue;
      const toteIds = slot.toteIds || [];
      if (!toteIds.length) return { tote: null, toteIds: [], reason: 'no-mapping' };
      for (const id of toteIds) {
        if (getToteCount(id) < TOTE_MAX_UNITS) {
          return { tote: id, toteIds };
        }
      }
      return { tote: null, toteIds, reason: 'all-full' };
    }
    return { tote: null, toteIds: [], reason: 'no-mapping' };
  }

  function getToteCount(toteId) {
    if (!toteId) return 0;

    if (!toteId) return 0;

    return toteCounts[toteId];

  function buildResultsTsv() {
    const header = 'container\tlpn\tsort_code\ttote\ttote_units';
    const rows = sortCodeResults.map(
      (r) => `${r.container}\t${r.lpn}\t${r.sortCode}\t${r.tote}\t${r.toteUnits}`

  }

  /**
   * @param {{ container: string, lpn: string }} pair
   * @param {string} sortCode
   * @param {{ tote?: string, toteUnits?: string | number, status?: 'complete' | 'error' }} [extra]
   */
  function recordSortCodeResult(pair, sortCode, extra = {}) {
    const code = sortCode || '';
    const last = sortCodeResults[sortCodeResults.length - 1];
    const sameAsLast =
      last &&
      last.container === pair.container &&
      last.lpn === pair.lpn &&
      last.sortCode === code &&
      jobRowStatus(code) !== 'Error' &&
      jobRowStatus(last.sortCode) !== 'Error';

    if (sameAsLast) {
      last.tote = extra.tote ?? last.tote;
      last.toteUnits = String(extra.toteUnits ?? last.toteUnits ?? '');
    } else {
      sortCodeResults.push({
        container: pair.container,
        lpn: pair.lpn,
        sortCode: code,
        tote: extra.tote ?? '',
        toteUnits: String(extra.toteUnits ?? ''),
      });
    }
    const st = extra.status || (String(sortCode).startsWith('EXCEPTION') ? 'error' : 'complete');
    rowStatuses[rowKey(pair.container, pair.lpn)] = st;
    rebuildMovedCountByContainer();
    bumpRowStatusVersion();
    scheduleRenderQueueList();
    scheduleSaveState();
    updateResetErrorStatusesBtn();
  }

  /**
   * @param {string} sortCode
   */
  async function submitToteForSortCode(sortCode) {
    const pick = pickDestinationToteForSortCode(sortCode);
    if (!pick.tote) {
      if (pick.reason === 'all-full') {
        log(
          `[Tote] All destinations full for "${sortCode}" — ` +
            pick.toteIds.map((id) => `${id}=${getToteCount(id)}`).join(', ')
        );
        await alertUserConfirm(
          'All destinations full',
          `Every destination listed for sort code "${sortCode}" is at ${TOTE_MAX_UNITS} units:\n\n` +
            pick.toteIds.join('\n') +
            '\n\nAdd a new destination container on a new line in that route on the right, then OK.',
          getBlockerContext()
        );
        return {
          ok: false,
          tote: pick.toteIds[0] || '',
          reason: 'all-full',
          count: TOTE_MAX_UNITS,
          toteIds: pick.toteIds,
        };
      }
      log(`[Tote] No mapping for sort code "${sortCode}" — fill a sort-code slot on the right.`);
      return { ok: false, tote: '', reason: 'no-mapping', toteIds: pick.toteIds };
    }

    const tote = pick.tote;
    const count = getToteCount(tote);
    log(
      `[Tote] Sort code "${sortCode}" → "${tote}" (${count}/${TOTE_MAX_UNITS}; ` +
        `${pick.toteIds.length} destination(s) in route)`
    );

    const ready = await waitForFieldActive(SCAN_TOTE_SELECTOR, 10000);
    if (!ready) {
      log('[Tote] #scan-tote-textfield not visible yet.');
      return { ok: false, tote, reason: 'field-missing', count };
    }

    const s = STEPS.scanTote;
    const result = await runFieldStep({
      selector: s.selector,
      enterParentIds: s.enterParentIds,
      value: tote,
      label: 'Scan tote',
      placeholder: s.placeholder,
    });

    if (!result.advanced) {
      if (result.reason === 'exception' || isAmazonExceptionShown()) {
        return { ok: false, tote, reason: 'move-exception', count };
      }
      return { ok: false, tote, reason: 'submit-failed', count };
    }

    const move = await verifyToteMoveOutcome();
    if (!move.ok) {
      log(
        move.reason === 'move-exception'
          ? '[Tote] Amazon move error (e.g. "Item could not be moved").'
          : '[Tote] Move did not complete before timeout.'
      );
      return { ok: false, tote, reason: move.reason || 'move-failed', count };
    }

    const newCount = incrementToteCount(tote);
    log(`[Tote] Submitted "${tote}" — now ${newCount}/${TOTE_MAX_UNITS} units.`);
    if (newCount >= TOTE_MAX_UNITS) {
      await confirmDestinationToteFull(tote, sortCode, newCount);
    }
    return { ok: true, tote, count: newCount };
  }

  /**
   * @param {HTMLElement} el
   * @param {string} logPrefix
   */
  async function clickElement(el, logPrefix) {
    /** @param {string} label */
    const tryClick = (label, fn) => {
      try {
        fn();
        log(`${logPrefix} clicked (${label}).`);
        return true;
      } catch (err) {
        log(`${logPrefix} ${label} failed: ${err instanceof Error ? err.message : String(err)}`);
        return false;
      }
    };

    if (tryClick('native .click', () => el.click())) {
      await sleep(POST_UPDATE_WAIT_MS);
      return true;
    }

    if (
      tryClick('MouseEvent', () => {
        el.dispatchEvent(
          new MouseEvent('click', { bubbles: true, cancelable: true, composed: true })
        );
      })
    ) {
      await sleep(POST_UPDATE_WAIT_MS);
      return true;
    }

    if (
      tryClick('createEvent', () => {
        const ev = document.createEvent('MouseEvents');
        ev.initEvent('click', true, true);
        el.dispatchEvent(ev);
      })
    ) {
      await sleep(POST_UPDATE_WAIT_MS);
      return true;
    }

    log(`${logPrefix} all click methods failed.`);
    return false;
  }

  }

  async function processAmazonException() {
    if (!canAutoHandleExceptions() || exceptionFlowActive) return;
    if (!isAmazonExceptionShown()) return;

    const ctx = getActiveLpnContext();
    const detail = readExceptionDetail();
    const key = `${ctx.container}|${ctx.lpn}|${detail}`;
    const isRepeat = lastExceptionKey === key;

    if (isRepeat && !autoCloseErrors) return;
    if (isRepeat && autoCloseErrors && !isExceptionBtnVisible()) return;

    exceptionFlowActive = true;
    setRunningFlag(true);
    try {
      postBarcodeHandled = false;

      if (!isRepeat) {
        log(`[Exception] ${(detail || 'Amazon exception').slice(0, 160)}`);
        recordExceptionResult(ctx.container, ctx.lpn, detail);
        setRowStatusByIndex(queueIndex, 'error');
        lastExceptionKey = key;
        updateResetErrorStatusesBtn();
      } else {
        log('[Exception] Same error still on page — retrying auto-close.');
      }

      if (autoCloseErrors) {
        await dismissAmazonExceptionModal('[Exception]');
        if (sessionArmed) {
          await recoverAndAdvanceToNextLpn('[Exception]');
        } else {
          barcodeDone = false;
          sourceContainerDone = false;
          lastSubmittedLpn = null;
          syncStepFlagsFromPageState();
          log(
            '[Exception] Continue clicked. Press Start to resume the queue (or Stop was active).'
          );
        }
        return;
      }

      const autoHint =
        '\n\nPress OK — the script will press Continue (C), Update if shown, then start the next LPN.';

      await alertLpnBlocked(
        `exception|${key}`,
        'Amazon exception — cannot sort this LPN',
        `Queue: ${formatQueueLabel()}\n\n${detail || 'Amazon reported an exception on this scan.'}${autoHint}`,
        ctx,
        false,
        false
      );

      if (canRunAutomation()) {
        await recoverAndAdvanceToNextLpn('[Exception]');
      }
    } catch (err) {
      log('Error: ' + (err instanceof Error ? err.message : String(err)));
      console.error('[Liquidation Sort]', err);
    } finally {
      exceptionFlowActive = false;
      setRunningFlag(false);
    }
  }

  async function tryAutoException() {
    await processAmazonException();
  }

  async function handlePostBarcode() {
    if (!canRunAutomation() || postBarcodeHandled) {
      traceWorkflow('postBarcode.skip', {
        canRun: canRunAutomation(),
        postBarcodeHandled,
      });
      return;
    }
    postBarcodeHandled = true;
    traceWorkflow('postBarcode.begin', {});

    try {
      const pair = getCurrentPair();
      if (!pair) {
        log('[Sort code] No active queue item.');
        const ctx = getBlockerContext();
        await alertLpnBlocked(
          `no-pair-post|${queueIndex}`,
          'Cannot finish LPN',
          'No active queue row while processing the sort code after barcode scan.',
          ctx,
          true
        );
        if (shouldStopOnError()) postBarcodeHandled = false;
        return;
      }

      const ctx = getBlockerContext();
      log('[Sort code] Waiting for #sort-code-id…');
      const sortCode = await waitForSortCode();
      if (sortCode === SORT_CODE_EXCEPTION) {
        log('[Sort code] Amazon exception detected — not waiting full timeout.');
        await processAmazonException();
        return;
      }
      const recorded = sortCode || '(not read)';

      if (sortCode) {
        const slotAssign = await ensureSortCodeAssignedToSlot(sortCode);
        if (slotAssign.assigned) {
          log(`[Sort code] Saved in Route ${slotAssign.slotIndex + 1}.`);
        }
      }

      if (!sortCode) {
        if (isAmazonExceptionShown()) {
          await processAmazonException();
          return;
        }
        recordSortCodeResult(pair, recorded, { status: 'error' });
        await alertLpnBlocked(
          `sort-code-timeout|${ctx.lpn}|${queueIndex}`,
          'LPN stuck — no sort code',
          `Queue: ${ctx.queueLabel}\n\n` +
            `Amazon did not show a sort code within ${SORT_CODE_WAIT_MS / 1000} seconds after the LPN scan.\n\n` +
            'The LPN may be invalid, the container may be wrong, or the page may need manual action.',
          ctx,
          true
        );
        if (shouldStopOnError()) {
          postBarcodeHandled = false;
          barcodeDone = false;
        }
        return;
      }

      let toteExtra = { tote: '', toteUnits: '' };
      if (sortCode && sessionArmed) {
        const toteResult = await submitToteForSortCode(sortCode);
        toteExtra = {
          tote: toteResult.tote || '',
          toteUnits:
            toteResult.count !== undefined
              ? String(toteResult.count)
              : toteResult.reason === 'tote-full' || toteResult.reason === 'all-full'
                ? `${TOTE_MAX_UNITS}+FULL`
                : '',
        };
        if (!toteResult.ok) {
          if (toteResult.reason === 'move-exception' || isAmazonExceptionShown()) {
            postBarcodeHandled = false;
            await processAmazonException();
            return;
          }
          recordSortCodeResult(pair, recorded, { ...toteExtra, status: 'error' });
          await alertLpnBlocked(
            `tote|${ctx.lpn}|${sortCode}|${toteResult.reason}`,
            'LPN cannot move into destination tote',
            `Queue: ${ctx.queueLabel}\nSort code: ${sortCode}\n\n` +
              toteBlockerMessage(
                toteResult.reason || 'unknown',
                sortCode,
                toteResult.tote || '',
                toteResult.toteIds
              ),
            ctx,
            true
          );
          if (shouldStopOnError()) {
            postBarcodeHandled = false;
            barcodeDone = false;
          }
          return;
        }
      }

      recordSortCodeResult(pair, recorded, { ...toteExtra, status: 'complete' });

      const clicked = await finishJobWithUpdate('[Job]');
      if (!clicked) {
        await alertLpnBlocked(
          `update-failed|${ctx.lpn}|${queueIndex}`,
          'LPN cannot move on — Update failed',
          `Queue: ${ctx.queueLabel}\nSort code: ${recorded}\n\n` +
            'The script could not click Update (#change-sort-type) after the tote move.\n\n' +
            'Click Update on the page manually or check for a blocking error.',
          ctx,
          true
        );
        if (shouldStopOnError()) {
          postBarcodeHandled = false;
          barcodeDone = false;
        }
        return;
      }

      if (!sessionArmed) {
        log('[Finish] Update clicked. Press Start to continue the queue.');
        return;
      }

      if (!canRunAutomation()) return;

      const hasMore = advanceToNextContainer();
      resetStepFlagsForCurrentItem();

      if (!hasMore) {
        log('Queue finished — no pending rows left.');
        return;
      }

      log(`[Queue] Next: ${formatQueueLabel()}`);
      setRowStatusByIndex(queueIndex, 'active');
      traceWorkflow('postBarcode.advance', { queueIndex });
      await continueWorkflowAfterLpnFinished();
      traceWorkflow('postBarcode.chainDone', {});
    } catch (err) {
      traceWorkflow('postBarcode.error', { err: err instanceof Error ? err.message : String(err) });
      log('Error: ' + (err instanceof Error ? err.message : String(err)));
      console.error('[Liquidation Sort]', err);
      const ctx = getBlockerContext();
      await alertLpnBlocked(
        `post-barcode-error|${ctx.lpn}|${queueIndex}`,
        'Error after LPN scan',
        `Queue: ${ctx.queueLabel}\n\n` + (err instanceof Error ? err.message : String(err)),
        ctx,
        true
      );
      if (shouldStopOnError()) {
        postBarcodeHandled = false;
        barcodeDone = false;
      }
    }
  }

  function advanceToNextContainer() {
    const list = getMappingList();
    queueIndex += 1;
    if (!skipDoneRowsFromCurrent()) {
      log(`Batch complete — no pending rows left (${list.length} in mapping).`);
      return false;
    }
    resetStepFlagsForCurrentItem();
    log(`Next in queue: ${formatQueueLabel()}`);
    return true;
  }

  async function storageGet(key, fallback) {
    if (typeof GM_getValue === 'function') {
      return GM_getValue(key, fallback);
    }
    try {
      const v = localStorage.getItem(key);
      return v !== null ? v : fallback;
    } catch {
      return fallback;
    }
  }

  async function storageSet(key, value) {
    if (typeof GM_setValue === 'function') {
      GM_setValue(key, value);
      return;
    }
    try {
      localStorage.setItem(key, value);
    } catch {
      /* ignore */
    }
  }

  async function storageGetJson(key, fallback) {
    const raw = await storageGet(key, null);
    if (raw === null || raw === undefined) return fallback;
    if (typeof raw === 'object') return raw;
    try {
      return JSON.parse(String(raw));
    } catch {
      return fallback;
    }
  }

  async function storageSetJson(key, value) {
    await storageSet(key, JSON.stringify(value));
  }

  function applySortSlotsToInputs(slots) {
    const list = Array.isArray(slots) ? slots : DEFAULT_SORT_SLOTS;
    for (let i = 0; i < SORT_SLOT_COUNT; i++) {
      const s = normalizeSortSlot(list[i] || {});
      if (slotSortInputs[i]) slotSortInputs[i].value = s.sortCode || '';
      if (slotToteInputs[i]) slotToteInputs[i].value = s.toteIds.join('\n');
    }
    renderSlotCounts();
  }

  async function savePersistedState() {
    await storageSet(STORAGE_KEY_MAPPING, getMappingText());
    await storageSetJson(STORAGE_KEY_SORT_SLOTS, getSortSlotsFromInputs());
    await storageSetJson(STORAGE_KEY_STATE, {
      v: 7,
      queueIndex,
      sortCodeResults,
      toteCounts,
      rowStatuses,
      sessionArmed,
      sessionStartedAt,
      timingPausedMs,
      pauseStartedAt,
      frozenCountdownMs,
      autoCloseErrors,
      acknowledgedFullTotes: [...acknowledgedFullTotes],
    });
  }

  async function loadPersistedState() {
    const state = await storageGetJson(STORAGE_KEY_STATE, null);
    if (!state || typeof state !== 'object') return false;

    if (typeof state.queueIndex === 'number' && state.queueIndex >= 0) {
      queueIndex = state.queueIndex;
    }
    if (Array.isArray(state.sortCodeResults)) {
      sortCodeResults = state.sortCodeResults.map((r) => ({
        container: r.container || '',
        lpn: r.lpn || '',
        sortCode: r.sortCode || '',
        tote: r.tote || '',
        toteUnits: r.toteUnits ?? '',
      }));
    }
    if (state.toteCounts && typeof state.toteCounts === 'object') {
      toteCounts = state.toteCounts;
    }
    if (state.rowStatuses && typeof state.rowStatuses === 'object') {
      rowStatuses = state.rowStatuses;
    }
    for (const r of sortCodeResults) {
      const key = rowKey(r.container, r.lpn);
      if (!rowStatuses[key]) {
        rowStatuses[key] = String(r.sortCode).startsWith('EXCEPTION') ? 'error' : 'complete';
      }
    }
    if (typeof state.sessionStartedAt === 'number') {
      sessionStartedAt = state.sessionStartedAt;
    }
    if (typeof state.timingPausedMs === 'number') {
      timingPausedMs = state.timingPausedMs;
    }
    if (typeof state.pauseStartedAt === 'number') {
      pauseStartedAt = state.pauseStartedAt;
    }
    if (typeof state.frozenCountdownMs === 'number') {
      frozenCountdownMs = state.frozenCountdownMs;
    }
    if (typeof state.autoCloseErrors === 'boolean') {
      autoCloseErrors = state.autoCloseErrors;
    }
    if (typeof state.sessionArmed === 'boolean') {
      sessionArmed = state.sessionArmed;
    }
    if (Array.isArray(state.acknowledgedFullTotes)) {
      acknowledgedFullTotes = new Set(state.acknowledgedFullTotes.map((t) => String(t).toLowerCase()));
    }
    updateAutoCloseToggleBtn();
    rebuildMovedCountByContainer();
    bumpRowStatusVersion();
    renderQueueList();
    renderSlotCounts();
    updateProgressUI();
    updateResetErrorStatusesBtn();
    if (sessionArmed) {
      armProgressTiming();
      log('Restored armed session — continuing queue.');
    }
    return true;
  }

  async function performType(target, value, enterParentIds, label) {
    target.focus();

    dispatchKey(target, 'keydown', {
      key: 'Control',
      code: 'ControlLeft',
      bubbles: true,
      cancelable: true,
      composed: true,
      ctrlKey: true,
      location: 1,
    });

    for (const ch of value) {
      const init = keyInitFor(ch, true);
      dispatchKey(target, 'keydown', init);
      dispatchKey(target, 'keypress', { ...init, charCode: ch.charCodeAt(0) });
      dispatchKey(target, 'keyup', init);
      if (KEY_DELAY_MS) await sleep(KEY_DELAY_MS);
    }

    dispatchKey(target, 'keyup', {
      key: 'Control',
      code: 'ControlLeft',
      bubbles: true,
      cancelable: true,
      composed: true,
      ctrlKey: false,
      location: 1,
    });

    setNativeInputValue(target, value);
    target.dispatchEvent(
      new InputEvent('beforeinput', {
        bubbles: true,
        cancelable: true,
        data: value,
        inputType: 'insertText',
      })
    );
    target.dispatchEvent(
      new InputEvent('input', {
        bubbles: true,
        cancelable: true,
        data: value,
        inputType: 'insertText',
      })
    );
    target.dispatchEvent(new Event('change', { bubbles: true }));

    await sleep(PRE_ENTER_MS);

    const enterInit = {
      key: 'Enter',
      code: 'Enter',
      bubbles: true,
      cancelable: true,
      composed: true,
      keyCode: 13,
      which: 13,
    };

    const enterTargets = [
      target,
      ...enterParentIds.map((id) => document.getElementById(id)),
      document,
    ].filter(Boolean);

    for (const el of enterTargets) {
      dispatchKey(el, 'keydown', enterInit);
      dispatchKey(el, 'keypress', enterInit);
      dispatchKey(el, 'keyup', enterInit);
    }

    const form = target.form;
    if (form) {
      form.dispatchEvent(new Event('submit', { bubbles: true, cancelable: true }));
    }

    log(`[${label}] Typed "${value}" and dispatched Enter.`);
  }

  async function runFieldStep(config) {
    const { selector, enterParentIds, value, label, placeholder } = config;
    const field = document.querySelector(selector);
    if (!field) {
      log(`[${label}] Field not found (${selector}).`);
      return { ok: false, reason: 'missing-field', advanced: false, label };
    }

    if (!value) {
      log(`[${label}] No value to type.`);
      return { ok: false, reason: 'empty-value', advanced: false, label };
    }

    if (isAmazonPlaceholderValue(value)) {
      log(`[${label}] Refusing to type Amazon placeholder text "${value}".`);
      return { ok: false, reason: 'placeholder-value', advanced: false, label };
    }

    log(`[${label}] Typing into #${field.id} (was: "${field.value}")`);
    field.focus();
    field.click();
    setNativeInputValue(field, '');
    field.dispatchEvent(new Event('input', { bubbles: true }));
    await sleep(10);

    await performType(field, value, enterParentIds, label);

    await sleep(STEP_SETTLE_MS);

    const verify = await waitForStepAdvanced(selector, placeholder, value, label);
    const advanced = verify.advanced;

    if (!advanced) {
      log(`[${label}] Step did not advance — ${verify.reason || 'check page or Copy Debug'}.`);
      if (selector === BARCODE_SELECTOR && isAmazonPlaceholderValue(field.value)) {
        log(`[${label}] Barcode field still shows placeholder — LPN was not accepted.`);
      }
    }

    return {
      ok: true,
      advanced,
      reason: advanced ? undefined : verify.reason || 'step-stuck',
      label,
    };
  }

  async function tryAutoSourceContainerForced(allowWhileRunning = false) {
    if (!canRunAutomation() || sourceContainerDone) return;
    if (running && !allowWhileRunning) return;
    if (!ensureCurrentRowIsPending()) return;
    const field = document.querySelector(SOURCE_CONTAINER_SELECTOR);
    if (!field || !getContainerValue()) return;

    const acquiredLock = !running;
    if (acquiredLock) setRunningFlag(true);
    try {
      setRowStatusByIndex(queueIndex, 'active');
      log(`[Source container] Submitting ${formatQueueLabel()}`);
      const result = await runSourceContainerStep();
      if (result.reason === 'exception' || isAmazonExceptionShown()) {
        await processAmazonException();
      } else if (result.advanced) {
        const verified = await verifySourceContainerAccepted(getContainerValue());
        if (verified.ok) {
          sourceContainerDone = true;
          clearBarcodeBlock();
          resolvePendingLpn({ forceLog: true });
          if (canRunAutomation()) await tryAutoBarcode(allowWhileRunning);
        } else if (verified.reason === 'exception' || isAmazonExceptionShown()) {
          await processAmazonException();
        } else {
          log('[Source container] Container typed but barcode step never opened.');
        }
      }
    } finally {
      if (acquiredLock) setRunningFlag(false);
    }
  }

  async function tryAutoSourceContainer() {
    if (!canRunAutomation() || sourceContainerDone || running) return;
    if (!ensureCurrentRowIsPending()) return;

    const srcPlaceholder = isAmazonPlaceholderValue(readFieldValue(SOURCE_CONTAINER_SELECTOR));
    if (
      !isFieldStepActive(SOURCE_CONTAINER_SELECTOR) &&
      isFieldVisible(SOURCE_CONTAINER_SELECTOR) &&
      srcPlaceholder &&
      getContainerValue()
    ) {
      await tryAutoSourceContainerForced();
      return;
    }
    if (!isFieldStepActive(SOURCE_CONTAINER_SELECTOR)) return;

    const container = getContainerValue();
    if (!container) {
      log('[Source container] Step detected — add rows to the mapping list.');
      const ctx = getBlockerContext();
        await alertLpnBlocked(
          `no-container|${queueIndex}`,
          'Cannot load container',
          `Queue row ${queueIndex + 1} has no container ID in the mapping list.\n\n` +
            'Add container and LPN (tab-separated) before pressing Start.',
        ctx,
        true
      );
      return;
    }

    setRunningFlag(true);
    try {
      setRowStatusByIndex(queueIndex, 'active');
      log(`[Source container] Submitting ${formatQueueLabel()}`);
      const result = await runSourceContainerStep();
      if (result.reason === 'exception' || isAmazonExceptionShown()) {
        await processAmazonException();
      } else if (result.advanced) {
        const verified = await verifySourceContainerAccepted(container);
        if (verified.ok) {
          sourceContainerDone = true;
          clearBarcodeBlock();
          resolvePendingLpn({ forceLog: true });
          if (canRunAutomation()) await tryAutoBarcode(false);
        } else if (verified.reason === 'exception' || isAmazonExceptionShown()) {
          await processAmazonException();
        } else {
          const ctx = getBlockerContext();
          setRowStatusByIndex(queueIndex, 'error');
          await alertLpnBlocked(
            `source-verify|${container}|${queueIndex}`,
            'Container not verified',
            `Queue: ${ctx.queueLabel}\n\n` +
              `Amazon did not open the barcode step after container "${container}".\n\n` +
              'Press Update on the page and Start again.',
            ctx,
            true
          );
        }
      } else if (result.reason === 'no-container') {
        const ctx = getBlockerContext();
        await alertLpnBlocked(
          `source-empty|${queueIndex}`,
          'Cannot load container',
          'No container value for the current queue row.',
          ctx,
          true
        );
      } else if (!result.advanced) {
        const ctx = getBlockerContext();
        setRowStatusByIndex(queueIndex, 'error');
        await alertLpnBlocked(
          `source-stuck|${container}|${queueIndex}`,
          'Container not accepted',
          `Queue: ${ctx.queueLabel}\n\n` +
            `Amazon did not accept source container "${container}".\n\n` +
            'The LPN cannot be scanned until the container step clears. Check the page for errors.',
          ctx,
          true
        );
      }
    } catch (err) {
      log('Error: ' + (err instanceof Error ? err.message : String(err)));
      console.error('[Liquidation Sort]', err);
    } finally {
      setRunningFlag(false);
    }
  }

  async function tryAutoBarcode(allowWhileRunning = false) {
    if (!canRunAutomation() || barcodeDone) return;
    if (running && !allowWhileRunning) return;
    if (isStaleSortCodeBeforeBarcode()) {
      if (!sourceContainerDone && (isAwaitingSourceContainerScan() || needsSortTypeEntry())) {
        traceWorkflow('barcode.deferStaleSort', {
          sortCode: readSortCodeText(),
          reason: 'need-source-first',
        });
        await tryAutoSourceContainerForced(allowWhileRunning);
        return;
      }
      traceWorkflow('barcode.staleSortReset', { sortCode: readSortCodeText() });
      log(
        `[Barcode] Sort code "${readSortCodeText()}" still on page — resetting before scanning ${getCurrentLpn()}.`
      );
      await resetPageWorkflowState('[Barcode]', { light: true });
      if (canRunAutomation()) {
        await resumeWorkflowFromPageState({ betweenJobs: true, allowWhileRunning, skipSortType: true });
      }
      return;
    }
    if (isStaleBarcodeOnPage()) {
      log(
        `[Barcode] Page shows ${readBarcodeOnPage()} but queue expects ${getCurrentLpn()} — resetting.`
      );
      await resetPageWorkflowState('[Barcode]');
      if (canRunAutomation()) await resumeWorkflowFromPageState();
      return;
    }
    if (!sourceContainerDone) {
      syncStepFlagsFromPageState();
      if (!sourceContainerDone) {
        return;
      }
    }
    if (!ensureCurrentRowIsPending()) return;
    if (!canSubmitBarcodeNow()) return;
    if (barcodeBlocked) {
      resolvePendingLpn({ forceLog: true });
      return;
    }

    const lpn = decodeAmazonFieldText(pendingLpn || getCurrentLpn() || '');
    if (!isValidLpn(lpn)) {
      resolvePendingLpn({ forceLog: true });
      return;
    }
    pendingLpn = lpn;
    lastSubmittedLpn = lpn;

    const acquiredLock = !running;
    if (acquiredLock) setRunningFlag(true);
    try {
      if (isBarcodeFieldAwaitingLpn() && isSortCodeReady()) {
        log(`[Barcode] Sort code visible but barcode still placeholder — submitting LPN ${lpn}`);
      } else {
        log(`[Barcode] Submitting LPN ${lpn} — ${formatQueueLabel()}`);
      }
      const result = await runBarcodeStep(lpn);
      if (result.reason === 'exception' || isAmazonExceptionShown()) {
        await processAmazonException();
      } else if (result.advanced) {
        const verified = await verifyBarcodeScanAccepted(lpn);
        if (!verified.ok) {
          if (verified.reason === 'exception' || isAmazonExceptionShown()) {
            await processAmazonException();
          } else if (verified.reason === 'stale-sort-before-barcode') {
            log('[Barcode] Stale sort code before scan finished — resetting page.');
            await resetPageWorkflowState('[Barcode]');
            if (canRunAutomation()) await resumeWorkflowFromPageState();
          } else {
            const ctx = getBlockerContext();
            setRowStatusByIndex(queueIndex, 'error');
            await alertLpnBlocked(
              `barcode-verify|${lpn}|${queueIndex}`,
              'LPN not verified',
              `Queue: ${ctx.queueLabel}\n\n` +
                'Barcode was submitted but Amazon did not show a sort code in time.\n\n' +
                'Check the page for errors or press Update and Start again.',
              ctx,
              true
            );
          }
          return;
        }
        barcodeDone = true;
        log(`[Barcode] Verified — ${formatQueueLabel()} → ${verified.sortCode || readSortCodeText()}`);
        await handlePostBarcode();
      } else if (isAmazonExceptionShown()) {
        await processAmazonException();
      } else {
        const ctx = getBlockerContext();
        setRowStatusByIndex(queueIndex, 'error');
        await alertLpnBlocked(
          `barcode-stuck|${lpn}|${queueIndex}`,
          'LPN not accepted',
          `Queue: ${ctx.queueLabel}\n\n` +
            'Amazon kept the barcode scan step open after submitting this LPN.\n\n' +
            'Common causes: wrong container, LPN not in this container, duplicate scan, or a page error above the scan field.',
          ctx,
          true
        );
      }
    } catch (err) {
      log('Error: ' + (err instanceof Error ? err.message : String(err)));
      console.error('[Liquidation Sort]', err);
      const ctx = getBlockerContext();
      setRowStatusByIndex(queueIndex, 'error');
      await alertLpnBlocked(
        `barcode-error|${lpn}|${queueIndex}`,
        'LPN scan error',
        `An error occurred while submitting this LPN:\n\n` +
          (err instanceof Error ? err.message : String(err)),
        ctx,
        true
      );
    } finally {
      if (acquiredLock) setRunningFlag(false);
    }
  }

  function watchWorkflowSteps() {
    const check = () => {
      clearStuckRunningIfNeeded();

      if (
        canAutoHandleExceptions() &&
        isAmazonExceptionShown() &&
        !exceptionFlowActive &&
        !running
      ) {
        traceWatch('exception', {});
        void processAmazonException();
        return;
      }

      const staleSort = isStaleSortCodeBeforeBarcode();
      const staleBc = isStaleBarcodeOnPage();
      if (
        staleSort &&
        !sourceContainerDone &&
        (needsSortTypeEntry() || isAwaitingSourceContainerScan())
      ) {
        traceWatch('staleSortNeedSource', {
          sortCode: readSortCodeText(),
          needsSortType: needsSortTypeEntry(),
        });
        if (canRunAutomation() && !running && !jobStartupInProgress) {
          void tryAutoSourceContainer();
        }
      } else if (
        canRunAutomation() &&
        !running &&
        !exceptionFlowActive &&
        !pageResetInProgress &&
        !jobStartupInProgress &&
        (staleSort || staleBc)
      ) {
        const now = Date.now();
        if (now - lastStaleRecoverAt < STALE_RECOVER_COOLDOWN_MS) {
          traceWatch('staleRecoverCooldown', {});
          return;
        }
        traceWatch('staleRecover', { staleSort, staleBc });
        lastStaleRecoverAt = now;
        setRunningFlag(true);
        void resumeWorkflowFromPageState({
          betweenJobs: true,
          allowWhileRunning: true,
          skipSortType: true,
        }).finally(() => {
          setRunningFlag(false);
        });
        return;
      }

      if (!canRunAutomation()) {
        traceWatch('idleDisarmed', {});
        return;
      }
      if (jobStartupInProgress) {
        traceWatch('idleJobStartup', {});
        return;
      }
      if (running) {
        traceWatch('idleRunning', {});
        return;
      }

      traceWatch('poll', {
        srcDone: sourceContainerDone,
        bcDone: barcodeDone,
        needsSort: needsSortTypeEntry(),
        awaitSrc: isAwaitingSourceContainerScan(),
      });
      tryAutoException();
      tryAutoSourceContainer();
      if (sourceContainerDone) {
        tryAutoBarcode();
      } else {
        traceWatch('skipBarcodeNeedSource', {});
      }
      if (canRunAutomation() && !postBarcodeHandled && !running) {
        if (
          !barcodeDone &&
          getCurrentLpn() &&
          (canSubmitBarcodeNow() || isBarcodeFieldAwaitingLpn())
        ) {
          setRunningFlag(true);
          tryAutoBarcode()
            .catch((err) => console.error('[Liquidation Sort]', err))
            .finally(() => {
              setRunningFlag(false);
            });
          return;
        }
        if (barcodeDone) {
          setRunningFlag(true);
          handlePostBarcode()
            .catch((err) => console.error('[Liquidation Sort]', err))
            .finally(() => {
              setRunningFlag(false);
            });
        }
      }
    };

    const observer = new MutationObserver(check);
    observer.observe(document.documentElement, { childList: true, subtree: true });

    setInterval(check, WORKFLOW_POLL_MS);
  }

  function wasMidJobWhenStopping(wasArmed) {
    if (!wasArmed) return false;
    const pair = getCurrentPair();
    if (!pair) return false;
    const key = rowKey(pair.container, pair.lpn);
    if (rowStatuses[key] === 'active') return true;
    return (
      sourceContainerDone ||
      barcodeDone ||
      postBarcodeHandled ||
      running ||
      exceptionFlowActive
    );
  }

  async function onStop() {
    const wasArmed = sessionArmed;
    const midJob = wasMidJobWhenStopping(wasArmed);

    sessionArmed = false;
    onStartActive = false;
    setRunningFlag(false);
    exceptionFlowActive = false;
    setErrorHalt(false);
    forceCloseBlockerOverlay();
    releaseAllAutomationHolds();
    pauseProgressTiming();
    scheduleSaveState();

    const pair = getCurrentPair();
    if (pair && midJob) {
      rowStatuses[rowKey(pair.container, pair.lpn)] = 'pending';
      bumpRowStatusVersion();
    }

    resetStepFlagsForCurrentItem();
    lastSubmittedLpn = null;

    if (midJob) {
      const clicked = await clickPageResetUpdates();
      log(
        clicked
          ? '[Stop] Current LPN marked Not Done; clicked Update to reset page.'
          : '[Stop] Current LPN marked Not Done; Update link not found on page.'
      );
    } else {
      log('--- Stop --- Automation halted. Press Start to continue.');
    }

    renderQueueList();
    updateProgressUI();
    savePersistedState();
  }

  async function onStartFromTop(startBtn) {

  async function onStart(startBtn) {
    if (onStartActive) {
      log('Start already in progress…');
      return;
    }

    resetStuckAutomationFlags();

    if (document.getElementById('ls-dashboard-overlay')) {
      log('Close the Jobs dashboard before starting.');
      return;
    }

    onStartActive = true;
    startBtn.disabled = true;

    sessionArmed = true;
    workflowTrace = [];
    lastWatchTraceKey = '';
    traceWorkflow('session.start', { queueIndex, label: formatQueueLabel() });
    armProgressTiming();
    lastExceptionKey = '';
    lastSubmittedLpn = null;
    resetStepFlagsForCurrentItem();
    updateProgressUI();
    scheduleSaveState();

    try {
      log('--- Start ---');
      const list = getMappingList();
      if (!list.length) {
        log('Mapping is empty — paste container + LPN rows first.');
        sessionArmed = false;
        return;
      }

      if (!skipDoneRowsFromCurrent()) {
        log('Every row in the mapping is already marked Done.');
        sessionArmed = false;
        return;
      }

      log(`Queue: ${list.length} row(s). Active: ${formatQueueLabel()}`);
      setRowStatusByIndex(queueIndex, 'active');

      await resumeWorkflowFromPageState();
    } catch (err) {
      log('Error: ' + (err instanceof Error ? err.message : String(err)));
      console.error('[Liquidation Sort]', err);
    } finally {
      onStartActive = false;
      startBtn.disabled = false;
      scheduleSaveState();
    }
  }

  function copyDebug() {
    const payload = gatherDebugPayload();
    const text = JSON.stringify(payload, null, 2);

    const done = (ok) => {
      log(ok ? 'Debug copied to clipboard.' : 'Copy failed — see browser console (F12).');
      if (!ok) console.log('[Liquidation Sort] Debug payload:', payload);
    };

    if (typeof GM_setClipboard === 'function') {
      GM_setClipboard(text, { type: 'text', mimetype: 'text/plain' });
      done(true);
      return;
    }

    navigator.clipboard
      .writeText(text)
      .then(() => done(true))
      .catch(() => done(false));
  }

  async function loadAllPersisted() {
    const [savedMapping, savedSlots] = await Promise.all([
      storageGet(STORAGE_KEY_MAPPING, DEFAULT_CONTAINER_LPN_TEXT),
      storageGetJson(STORAGE_KEY_SORT_SLOTS, DEFAULT_SORT_SLOTS),
    ]);
    if (mappingInput && savedMapping) mappingInput.value = savedMapping;
    applySortSlotsToInputs(savedSlots);
    const restored = await loadPersistedState();
    syncRowStatusesFromMapping();
    return restored;
  }
