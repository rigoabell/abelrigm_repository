  function loadLogins() {
    try {
      const parsed = JSON.parse(localStorage.getItem(STORAGE_KEY) || "[]");
      if (!Array.isArray(parsed)) {
        return [];
      }

      return parsed
        .map((entry) => {
          if (typeof entry === "string") {
            const normalized = normalizeLogin(entry);
            return normalized ? { login: normalized, audited: false, attributes: [] } : null;
          }

          if (!entry || typeof entry !== "object") {
            return null;
          }

          const normalized = normalizeLogin(entry.login);
          if (!normalized) {
            return null;
          }

          return {
            login: normalized,
            audited: Boolean(entry.audited),
            attributes: normalizeStoredAttributes(entry.attributes),
            lpns: normalizeStoredLpns(entry.lpns),
          };
        })
        .filter(Boolean);
    } catch (error) {
      console.warn("Apollo Audit Helper 29836: failed to read saved logins", error);
      return [];
    }
  }

  function normalizeStoredLpns(value) {

      return [];

      .map((item) => {

          return lpn ? { lpn, date: "" } : null;

        const lpn = String(item?.lpn || "").trim().toUpperCase();
        if (!lpn) {
          return null;
        }
        return { lpn, date: String(item?.date || "").trim() };
      })
      .filter(Boolean);
  }

  function headerLabel(value) {
    return String(value || "")
      .replace(/<[^>]+>/g, " ")
      .replace(/\s+/g, " ")
      .trim()
      .replace(/\s*:\s*activate to sort[\s\S]*$/i, "");

  function isLpnHeader(value) {
    const text = headerLabel(value).toLowerCase();
    return text === "lpn" || /^lpn\b/.test(text);
  }

  function lpnColumn(headers) {
    const index = (headers || []).findIndex((header) => isLpnHeader(header));
    return index >= 0 ? index : 6;

  function looksLikeLpn(value) {
    return /^LPN[A-Z0-9]+$/.test(String(value || "").replace(/\s+/g, "").toUpperCase());
  }

  function uniqueRecentLpns(headers, rows, limit) {
    const column = lpnColumn(headers);
    const max = Number.isFinite(limit) ? limit : Infinity;
    const seen = new Set();
    const lpns = [];
    for (const row of rows || []) {
      const lpn = String(row[column] || "").replace(/\s+/g, " ").trim().toUpperCase();
      if (!looksLikeLpn(lpn) || seen.has(lpn)) {
        continue;
      }
      seen.add(lpn);
      lpns.push(lpn);
      if (lpns.length >= max) {
        break;
      }
    }
    return lpns;
  }

  function recentLpnDetails(headers, rows, limit) {
    const column = lpnColumn(headers);
    let dateColumn = (headers || []).findIndex((header) => /^date\b/i.test(headerLabel(header)));
    if (dateColumn < 0) {
      const dated = (rows || []).some((cells) => /^\d{4}-\d{2}-\d{2}\b/.test(String(cells && cells[0] || "").trim()));
      dateColumn = dated ? 0 : -1;
    }
    return uniqueRecentLpns(headers, rows, limit).map((lpn) => {
      const row = (rows || []).find((cells) => String(cells[column] || "").replace(/\s+/g, " ").trim().toUpperCase() === lpn);
      const date = dateColumn >= 0 && row ? String(row[dateColumn] || "").replace(/\s+/g, " ").trim() : "";
      return { lpn, date };
    });
  }

  function chooseRandomSubset(items, minCount, maxCount) {
    if (!items.length) {
      return [];
    }

    const shuffled = [...items];
    for (let index = shuffled.length - 1; index > 0; index -= 1) {
      const swapIndex = Math.floor(Math.random() * (index + 1));
      const current = shuffled[index];
      shuffled[index] = shuffled[swapIndex];
      shuffled[swapIndex] = current;
    }

    const safeMin = Math.max(1, Math.min(minCount, shuffled.length));
    const safeMax = Math.max(safeMin, Math.min(maxCount, shuffled.length));
    const count = safeMin + Math.floor(Math.random() * (safeMax - safeMin + 1));
    return shuffled.slice(0, count);
  }

  function mapAttributeToOpportunity(attribute) {
    const normalized = String(attribute || "")
      .toLowerCase()
      .replace(/[^a-z0-9]+/g, " ")
      .trim();
    if (!normalized) {
      return null;
    }
    if (normalized.includes("item") && normalized.includes("damage")) {
      return "item_damage";
    }
    if (normalized.includes("package") && normalized.includes("damage")) {
      return "package_damage";
    }
    if (normalized.includes("seal")) {
      return "seal_type_and_condition";
    }
    if (normalized.includes("missing")) {
      return "missing_parts";
    }
    return null;
  }

      const key = mapAttributeToOpportunity(row.attribute);
      if (!key || selectedKeys.has(key)) {
        return;
      }
      selectedKeys.add(key);
      mapped.push({
        key,
        attribute: row.attribute,
        accuracy: row.accuracy,
      });
    });

    if (!selectedKeys.size) {
      return { source: "random", selected: fillRandomOpportunities() };
    }

    OPPORTUNITIES.forEach((opportunity) => {
      const checked = selectedKeys.has(opportunity.key);
      setCheckboxChecked(opportunity.checkboxId, checked);
      fillRandomSubDefect(opportunity, checked);
    });

    return {
      source: "quicksight",
      selected: mapped.map((row) => {
        const opportunity = OPPORTUNITIES.find((entry) => entry.key === row.key);
        return `${opportunity?.label || row.attribute}${row.accuracy ? ` ${row.accuracy}` : ""}`;
      }),
    };
  }

  function ingestPastedQueue(rows) {
    const existing = loadLogins();
    const grouped = new Map();

    (rows || []).forEach((row) => {
      const login = normalizeLogin(row?.login);
      if (!login || login === "login") {
        return;
      }
      if (!grouped.has(login)) {
        grouped.set(login, []);
      }
      const attribute = String(row?.attribute || "").trim();
      if (!attribute || /^attribute$/i.test(attribute)) {
        return;
      }
      const accuracy = String(row?.accuracy || "").trim();
      const list = grouped.get(login);
      if (!list.some((item) => item.attribute === attribute && item.accuracy === accuracy)) {
        list.push({ attribute, accuracy: /^accuracy$/i.test(accuracy) ? "" : accuracy });
      }
    });

    const next = existing.map((entry) => ({ ...entry }));
    let added = 0;
    let updated = 0;
    const newEntries = [];

    grouped.forEach((attributes, login) => {
      const found = next.find((entry) => entry.login === login);
      if (found) {
        if (attributes.length) {
          found.attributes = attributes;
          updated += 1;
        }
        return;
      }
      newEntries.push({
        login,
        audited: false,
        attributes,
        lpns: [],
      });
      added += 1;
    });

    if (added || updated) {
      saveLogins([...newEntries, ...next]);
    }

    STATE.lastPaste = {
      at: new Date().toISOString(),
      rows: (rows || []).length,
      uniqueLogins: grouped.size,
      added,
      updated,
    };
    diag("paste queue", STATE.lastPaste);
    return { added, updated, uniqueLogins: grouped.size };
  }

  function looksLikeAuditTable(text) {
    const sample = String(text || "").trim();
    if (!sample) {
      return false;
    }
    if (/login\s+attribute\s+accuracy/i.test(sample)) {
      return true;
    }
    return sample.includes("\t") && /\d+(?:\.\d+)?%/.test(sample);
  }

  function expandCollapsedTable(value) {
    let text = String(value || "").replace(/^\uFEFF/, "");
    if (!/[\r\n\u2028\u2029]/.test(text)) {
      text = text
        .replace(/(\bAccuracy\b)\s+/i, "$1\n")
        .replace(/(\d+(?:\.\d+)?%)\s+(?=\S)/g, "$1\n");
    }
    return text;
  }

  function parseTableLine(line) {
    const trimmed = String(line || "").trim();
    if (!trimmed) {
      return null;
    }

    let cols;
    if (trimmed.includes("\t")) {
      cols = trimmed.split("\t").map((cell) => cell.trim());
    } else if (trimmed.includes(",")) {
      cols = trimmed.split(",").map((cell) => cell.trim());
    } else {
      const match = trimmed.match(/^(\S+)\s+(.+?)\s+(\d+(?:\.\d+)?%?)$/);
      cols = match ? [match[1], match[2], match[3]] : [trimmed];
    }

    const login = normalizeLogin(cols[0]);
    if (!login || login === "login") {
      return null;
    }

    const attribute = cols[1] && !/^attribute$/i.test(cols[1]) ? cols[1] : "";
    const accuracy = cols[2] && !/^accuracy$/i.test(cols[2]) ? cols[2] : "";
    return { login, attribute, accuracy };
  }

  function parsePastedQueue(value) {
    const lines = expandCollapsedTable(value)
      .split(/\r\n|\n|\r|\u2028|\u2029/)
      .map((line) => line.trim())
      .filter(Boolean);
    if (!lines.length) {
      return [];
    }

    const parsed = lines.map(parseTableLine).filter(Boolean);
    if (parsed.some((row) => row.attribute) || looksLikeAuditTable(value)) {
      return parsed;
    }

    return parseLoginList(value).map((login) => ({
      login,
      attribute: "",
      accuracy: "",
    }));
  }

  function consumePendingText(panel) {
    const helperInput = panel.querySelector(".apollo-helper-input");
    const loginInput = getLoginInput();
    const sources = [
      { source: "helper", text: helperInput?.value || "", always: true },
      { source: "login-field", text: loginInput?.value || "", always: false },
    ];
    let changed = false;

    sources.forEach((item) => {
      if (!item.text.trim()) {
        return;
      }
      if (!item.always && !looksLikeAuditTable(item.text)) {
        return;
      }
      const rows = parsePastedQueue(item.text);
      diag("parse paste", {
        source: item.source,
        chars: item.text.length,
        preview: item.text.slice(0, 180),
        rowCount: rows.length,
      });
      if (!rows.length) {
        return;
      }
      ingestPastedQueue(rows);
      changed = true;
      if (item.source === "helper" && helperInput) {
        helperInput.value = "";
      }
      if (item.source === "login-field" && loginInput) {
        loginInput.value = "";
        dispatchFieldEvents(loginInput);
      }
    });

    if (changed) {
      renderPanel(panel);
    }
    return changed;
  }

  function ingestClipboardText(panel, text, source) {
    const rows = parsePastedQueue(text);
    diag("parse paste", {
      source,
      chars: String(text || "").length,
      preview: String(text || "").slice(0, 180),
      rowCount: rows.length,
    });
    if (!rows.length) {
      return false;
    }
    ingestPastedQueue(rows);
    const helperInput = panel.querySelector(".apollo-helper-input");
    if (helperInput) {

  async function copyTextToClipboard(text) {
    if (navigator.clipboard && typeof navigator.clipboard.writeText === "function") {
      await navigator.clipboard.writeText(text);
      return;
    }

    const tempTextArea = document.createElement("textarea");
    tempTextArea.value = text;
    tempTextArea.style.position = "fixed";
    tempTextArea.style.opacity = "0";
    document.body.appendChild(tempTextArea);
    tempTextArea.focus();
    tempTextArea.select();
    document.execCommand("copy");
    document.body.removeChild(tempTextArea);
  }

  function snapshotHistoryTable(doc) {
    const table = doc.getElementById("table-inventory-history");
    const info = doc.getElementById("table-inventory-history_info");
    const infoText = headerLabel(info ? info.textContent : "");
    if (!table) {
      return { tableFound: false, headers: [], rows: [], rowCount: 0, infoText, emptyMessage: "", processing: false };
    }
    const processingEl = doc.querySelector("#table-inventory-history_processing");
    let processing = false;
    if (processingEl && doc.defaultView && doc.defaultView.getComputedStyle) {
      const style = doc.defaultView.getComputedStyle(processingEl);
      processing = style.display !== "none" && style.visibility !== "hidden";
    }
    const headerRows = [...table.querySelectorAll("thead tr")].map((tr) =>
      [...tr.children].map((cell) => domHeaderLabel(cell))
    );
    const headers = headerRows.find((row) => row.some((cell) => isLpnHeader(cell))) || headerRows[headerRows.length - 1] || [];
    const rows = [];
    let emptyMessage = "";
    for (const tr of table.querySelectorAll("tbody tr")) {
      const cells = [...tr.children].filter((el) => el.tagName === "TD").map((td) => headerLabel(td.textContent));
      const joined = cells.join(" ").toLowerCase();
      if (tr.querySelector("td.dataTables_empty, .dataTables_empty") || /no matching records|no data available/.test(joined)) {
        emptyMessage = cells.join(" ").trim();
        continue;
      }
      if (cells.filter(Boolean).length < 2) {
        continue;
      }
      rows.push(cells);
    }
    if (!emptyMessage && (/of 0 entries/i.test(infoText) || /showing 0 to 0/i.test(infoText))) {
      emptyMessage = infoText;
    }
    return { tableFound: true, headers, rows, rowCount: rows.length, infoText, emptyMessage, processing };
  }

  function scrollHistoryTable(doc, win) {
    try {
      const table = doc.getElementById("table-inventory-history");
      const section = doc.querySelector('[data-section-type="inventory-history"]') || table;
      if (section && win) {
        const top = section.getBoundingClientRect().top + (win.scrollY || win.pageYOffset || 0);
        win.scrollTo(0, Math.max(0, top - 40));
      }
    } catch (error) {
      // Scrolling only nudges a lazy section into view.
    }
  }

  async function waitForHistoryTable(doc, win, timeoutMs) {
    const started = Date.now();
    let lastKey = "";
    let stableSince = 0;
    let latest = snapshotHistoryTable(doc);
    while (Date.now() - started < timeoutMs) {
      scrollHistoryTable(doc, win);
      latest = snapshotHistoryTable(doc);
      const key = [latest.tableFound, latest.processing, latest.rowCount, latest.emptyMessage, latest.infoText].join("|");
      const ready = latest.tableFound && !latest.processing && (latest.rowCount > 0 || latest.emptyMessage);
      if (ready && key === lastKey && Date.now() - stableSince >= 700) {
        latest.stable = true;
        return latest;
      }
      if (key !== lastKey) {
        stableSince = Date.now();
      }
      lastKey = key;
      await sleep(200);
    }
    latest.stable = false;
    return latest;
  }

  function historyResultFromSnapshot(snap) {
    const details = recentLpnDetails(snap.headers, snap.rows, 3);
    const lpns = details.map((item) => item.lpn);
    const empty = Boolean(snap.stable && snap.tableFound && snap.rowCount === 0 && snap.emptyMessage);
    return {
      lpns,
      details,
      empty,
      settled: lpns.length > 0 || empty,
      error: lpns.length || empty ? "" : "Inventory history did not finish loading.",
      report: {
        tableFound: snap.tableFound,
        stable: Boolean(snap.stable),
        headers: snap.headers,
        rowCount: snap.rowCount,
        infoText: snap.infoText,
        emptyMessage: snap.emptyMessage,
        sampleRows: (snap.rows || []).slice(0, 4),
      },
    };
  }

  function pageMatchesHistoryJob(job) {
    try {
      const wanted = new URL(job.requestUrl);
      if (location.origin !== wanted.origin) {
        return false;
      }
      const herePath = location.pathname.replace(/\/+$/, "").toUpperCase();
      const wantedPath = wanted.pathname.replace(/\/+$/, "").toUpperCase();
      if (herePath !== wantedPath) {
        return false;
      }
      const hereLogin = (new URLSearchParams(location.search).get("s") || "").toLowerCase();
      const wantedLogin = (wanted.searchParams.get("s") || "").toLowerCase();
      return hereLogin === wantedLogin;
    } catch (error) {
      return false;
    }
  }

  async function widenHistoryRange() {
    const section = document.querySelector('[data-section-type="inventory-history"]');
    const start = document.getElementById("searchStart");
    const end = document.getElementById("searchEnd");
    if (!section || !start || !end) {
      return false;
    }
    const button = [...section.querySelectorAll("button, input[type='submit'], input[type='button'], a")].find((node) => {
      return headerLabel(node.value || node.textContent).toLowerCase() === "search";
    });
    if (!button) {
      return false;
    }
    const startOn = new Date();
    startOn.setDate(startOn.getDate() - 180);
    const pad = (value) => String(value).padStart(2, "0");
    const format = (date) => `${pad(date.getMonth() + 1)}/${pad(date.getDate())}/${date.getFullYear()}`;
    start.value = format(startOn);
    end.value = format(new Date());
    start.dispatchEvent(new Event("change", { bubbles: true }));
    end.dispatchEvent(new Event("change", { bubbles: true }));
    button.click();
    await sleep(400);
    return true;
  }

  function rememberedHistoryJob() {
    let stored = "";
    try {
      stored = sessionStorage.getItem(HISTORY_MARK) || "";
    } catch (error) {
      stored = "";
    }
    const id = earlyHistoryJobId || stored || readHistoryJobId();
    if (!id) {
      return null;
    }
    const job = gmGetValue(HISTORY_JOB_KEY, null);
    if (!job || job.id !== id || Date.now() - job.createdAt > 120000) {
      clearHistoryMark();
      return null;
    }
    if (job.status !== "pending" && job.status !== "running") {
      clearHistoryMark();
      return null;
    }
    if (!pageMatchesHistoryJob(job)) {
      return null;
    }
    return job;
  }

  async function fulfillHistoryJob(job) {
    gmSetValue(HISTORY_JOB_KEY, { ...job, status: "running", updatedAt: Date.now() });
    let result;
    try {
      const snap = await waitForHistoryTable(document, window, 40000);
      result = historyResultFromSnapshot(snap);
      if (result.empty) {
        const widened = await widenHistoryRange();
        if (widened) {
          const next = await waitForHistoryTable(document, window, 20000);
          const widenedResult = historyResultFromSnapshot(next);
          widenedResult.report = { ...(widenedResult.report || {}), widened: true };
          result = widenedResult;
        }
      }
    } catch (error) {
      result = { lpns: [], details: [], empty: false, settled: false, error: String(error?.message || error), report: null };
    }
    gmSetValue(HISTORY_JOB_KEY, {
      ...job,
      status: result.lpns.length || result.settled ? "done" : "error",
      result,
      updatedAt: Date.now(),
    });
    clearHistoryMark();
    window.setTimeout(() => {
      try {
        window.close();
      } catch (error) {
        // The opener closes the tab if this one stays open.
      }
    }, 250);
  }

  function fetchAssociateHistory(login) {
    const requestUrl = historyRequestUrl(login);
    if (typeof GM_openInTab !== "function" || typeof GM_getValue !== "function") {
      return Promise.resolve({
        lpns: [],
        details: [],
        error: "Tampermonkey storage is required to read inventory history. Reinstall the script so the grants apply.",
        report: null,
      });
    }
    return new Promise((resolve) => {
      const id = `${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 8)}`;
      const job = { id, status: "pending", requestUrl, createdAt: Date.now(), result: null };
      gmSetValue(HISTORY_JOB_KEY, job);
      const url = new URL(requestUrl);
      url.hash = `apollo29836-lpn=${encodeURIComponent(id)}`;
      let tab = null;
      try {
        tab = GM_openInTab(url.toString(), { active: false, insert: true, setParent: true });
      } catch (error) {
        resolve({ lpns: [], details: [], error: String(error?.message || error), report: null });
        return;
      }
      let settled = false;
      let listenerId = null;
      const timer = window.setTimeout(() => finish({
        lpns: [],
        details: [],
        error: "Timed out waiting for inventory history.",
        report: null,
      }), 45000);

      function finish(result) {
        if (settled) {
          return;
        }
        settled = true;
        window.clearTimeout(timer);
        window.clearInterval(poll);
        if (listenerId != null && typeof GM_removeValueChangeListener === "function") {
          try {
            GM_removeValueChangeListener(listenerId);
          } catch (error) {
            // The listener goes away with the page.
          }
        }
        if (tab && typeof tab.close === "function") {
          try {
            tab.close();
          } catch (error) {
            // The worker tab closes itself.
          }
        }
        resolve(result);
      }

      function consider(value) {
        if (!value || value.id !== id) {
          return;
        }
        if (value.status === "done" || value.status === "error") {
          finish(value.result || { lpns: [], details: [], error: "Inventory history returned no result.", report: null });
        }
      }

      if (typeof GM_addValueChangeListener === "function") {
        try {
          listenerId = GM_addValueChangeListener(HISTORY_JOB_KEY, (_name, _oldValue, value) => consider(value));
        } catch (error) {
          listenerId = null;
        }
      }
      const poll = window.setInterval(() => consider(gmGetValue(HISTORY_JOB_KEY, null)), 400);
    });
  }

  function readFilledLpns() {
    return LPN_INPUT_IDS.map((id) => String(document.getElementById(id)?.value || "").trim().toUpperCase());
  }

  function historyAlreadyApplied(login) {
    if (!STATE.applied || STATE.applied.login !== normalizeLogin(login)) {
      return false;
    }
    const filled = readFilledLpns();
    return STATE.applied.lpns.length > 0 && STATE.applied.lpns.every((lpn, index) => filled[index] === lpn);
  }

  function setStatus(panel, message, kind) {
    const status = panel.querySelector(".apollo-helper-status");
    if (!status) {
      return;
    }
    status.textContent = message;
    status.classList.toggle("is-ok", kind === "ok");
    status.classList.toggle("is-error", kind === "error");
    status.classList.toggle("is-busy", kind === "busy");
  }

  function hookFormAutosave(panel) {
    const submitButton = document.getElementById("audit-submit");
    if (!submitButton || submitButton.dataset[SUBMIT_HOOK_FLAG] === "true") {
      return;
    }

    submitButton.dataset[SUBMIT_HOOK_FLAG] = "true";
    submitButton.addEventListener(
      "click",
      (event) => {
        const login = normalizeLogin(getLoginInput()?.value);
        if (!login) {
          return;
        }
        if (STATE.allowSubmit) {
          STATE.allowSubmit = false;
          addLogin(login);
          markLoginAudited(login, true);
          diag("marked audited", { login, lpns: STATE.applied?.lpns || readFilledLpns() });
          renderPanel(panel);
          return;
        }
        event.preventDefault();
        event.stopImmediatePropagation();
        if (STATE.running) {
          setStatus(panel, "Still reading inventory history…", "busy");
          return;
        }
        if (historyAlreadyApplied(login)) {
          window.clearTimeout(STATE.submitTimer);
          STATE.submitTimer = 0;
          STATE.allowSubmit = true;
          submitButton.click();
          return;
        }
        void applyHistoryAndSubmit(panel, login, submitButton);
      },
      true
    );
  }

  async function applyHistoryAndSubmit(panel, login, submitButton) {
    STATE.running = true;
    setStatus(panel, `Reading inventory history for ${login}…`, "busy");
    try {
      const history = await fetchAssociateHistory(login);
      STATE.lastHistory = { at: new Date().toISOString(), login, ...history };
      diag("submit history", STATE.lastHistory);
      if (!history.lpns.length) {
        setStatus(panel, history.error || `No inventory-history LPNs for ${login}. Not submitted.`, "error");
        return;
      }
      fillHistoryLpns(history.details);
      STATE.applied = { login, lpns: history.lpns, details: history.details };
      rememberLpns(login, history.details);
      renderPanel(panel);
      setStatus(panel, `Using ${history.lpns.length} history LPNs for ${login}.`, "ok");
      STATE.allowSubmit = true;
      submitButton.click();
    } catch (error) {
      const message = String(error?.message || error);
      STATE.lastError = message;
      diag("submit history failed", message);
      setStatus(panel, message, "error");
    } finally {
      STATE.running = false;
    }
  }

  function hookLoginSync(panel) {
    const loginInput = getLoginInput();
    if (!loginInput || loginInput.dataset[INPUT_WATCH_FLAG] === "true") {
      return;
    }

    loginInput.dataset[INPUT_WATCH_FLAG] = "true";
    ["input", "change", "blur"].forEach((eventName) => {
      loginInput.addEventListener(eventName, () => renderPanel(panel));
    });
  }

  function submitForm() {
    const form = document.getElementById("audits-form");
    if (form && typeof form.reportValidity === "function" && !form.reportValidity()) {
      STATE.allowSubmit = false;
      alert("Apollo Audit Helper 29836: form validation failed before submit.");
      return;
    }

    const submitButton = document.getElementById("audit-submit");
    if (submitButton) {
      if (document.activeElement && typeof document.activeElement.blur === "function") {
        document.activeElement.blur();
      }
      STATE.allowSubmit = true;
      submitButton.click();
      return;
    }

    STATE.allowSubmit = false;
    alert("Apollo Audit Helper 29836: submit button was not found.");
  }

  async function runNextAudit(panel) {
    if (STATE.running) {
      return;
    }
    consumePendingText(panel);
    const nextEntry = getNextUnauditedLogin();
    if (!nextEntry) {
      setStatus(panel, "No unaudited associates left in the queue.", "ok");
      return;
    }

    const runButton = panel.querySelector(".apollo-helper-run");
    STATE.running = true;
    if (runButton) {
      runButton.disabled = true;
    }
    setStatus(panel, `Reading inventory history for ${nextEntry.login}…`, "busy");

    try {
      fillLogin(nextEntry.login);
      const opportunities = fillRandomAuditFields(nextEntry);
      const history = await fetchAssociateHistory(nextEntry.login);
      STATE.lastHistory = { at: new Date().toISOString(), login: nextEntry.login, ...history };
      if (!history.lpns.length) {
        STATE.lastFill = {
          at: new Date().toISOString(),
          login: nextEntry.login,
          attributes: normalizeStoredAttributes(nextEntry.attributes),
          opportunities,
          lpns: [],
          lpnSource: "inventory-history",
          error: history.error || "No LPNs",
          autoSubmit: false,
        };
        STATE.lastError = STATE.lastFill.error;
        diag("history miss", STATE.lastFill);
        setStatus(panel, history.error || `No inventory-history LPNs for ${nextEntry.login}. Not submitted.`, "error");
        renderPanel(panel);
        return;
      }

      fillHistoryLpns(history.details);
      STATE.applied = { login: nextEntry.login, lpns: history.lpns, details: history.details };
      rememberLpns(nextEntry.login, history.details);
      STATE.lastFill = {
        at: new Date().toISOString(),
        login: nextEntry.login,
        attributes: normalizeStoredAttributes(nextEntry.attributes),
        opportunities,
        lpns: history.lpns,
        lpnSource: "inventory-history",
        autoSubmit: AUTO_SUBMIT,
      };
      STATE.lastError = null;
      diag("run fill", STATE.lastFill);
      renderPanel(panel);
      setStatus(panel, `${history.lpns.length} history LPNs ready for ${nextEntry.login}.`, "ok");
    } catch (error) {
      STATE.lastError = String(error?.message || error);
      diag("run fill failed", STATE.lastError);
      setStatus(panel, STATE.lastError, "error");
      return;
    } finally {
      STATE.running = false;
      if (runButton) {
        runButton.disabled = false;
      }
    }

    if (!AUTO_SUBMIT) {
      return;
    }

    window.clearTimeout(STATE.submitTimer);
    STATE.submitTimer = window.setTimeout(() => {
      STATE.submitTimer = 0;
      submitForm();
    }, POST_FILL_SUBMIT_DELAY_MS);
  }

  function watchAuditForm(panel) {
    hookFormAutosave(panel);
    hookLoginSync(panel);
    const started = Date.now();
    const timer = window.setInterval(() => {
      hookFormAutosave(panel);
      hookLoginSync(panel);
      if (document.getElementById("audit-submit") || Date.now() - started > 20000) {
        window.clearInterval(timer);
      }
    }, 400);
  }

  function isApolloAuditPage() {
    const auditTypeId = new URLSearchParams(window.location.search).get("audit_type_id");
    return window.location.pathname === "/audits/new" && auditTypeId === TARGET_AUDIT_TYPE_ID;
  }

  function boot() {
    const job = rememberedHistoryJob();
    if (job) {
      void fulfillHistoryJob(job);
      return;
    }
    if (!isApolloAuditPage()) {
      return;
    }
    createPanel();
    diag("loaded", { href: location.href });
  }

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", boot, { once: true });
  } else {
    boot();
  }
