  function setNativeValue(input, value) {
    const descriptor =
      Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, "value") ||
      Object.getOwnPropertyDescriptor(input, "value");

    if (descriptor && descriptor.set) {
      descriptor.set.call(input, value);
    } else {
      input.value = value;
    }

    input.dispatchEvent(new Event("input", { bubbles: true }));
    input.dispatchEvent(new Event("change", { bubbles: true }));
  }

  function sleep(ms) {
    return new Promise((resolve) => window.setTimeout(resolve, ms));
  }

  async function waitFor(predicate, timeoutMs) {
    const started = Date.now();
    while (Date.now() - started < timeoutMs) {
      const value = predicate();
      if (value) {
        return value;
      }
      await sleep(150);
    }
    throw new Error("Timed out waiting for page state");
  }

  function normalizeText(text) {
    return (text || "").replace(/\s+/g, " ").trim();
  }

  function formatHumanDateTime(value) {
    if (!value) {
      return "";
    }

    const date = new Date(value);
    if (Number.isNaN(date.getTime())) {
      return String(value);
    }

    return date.toLocaleString(undefined, {
      year: "numeric",
      month: "numeric",
      day: "numeric",
      hour: "numeric",
      minute: "2-digit",
      second: "2-digit",
    });
  }

  function getWorkbookSheetName(date = new Date()) {
    const pad = (value) => String(value).padStart(2, "0");
    const year = String(date.getFullYear()).slice(-2);
    const safeName = `scan-history ${pad(date.getMonth() + 1)}-${pad(date.getDate())}-${year} ${pad(
      date.getHours()
    )}-${pad(date.getMinutes())}-${pad(date.getSeconds())}`;

    // Excel sheet names cannot contain : \ / ? * [ ] and cannot exceed 31 chars.
    return safeName.replace(/[:\\/?*\[\]]/g, "-").slice(0, 31);
  }

  function escapeHtml(value) {
    return String(value)
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;")
      .replace(/'/g, "&#39;");
  }

  function escapeAttribute(value) {
    return escapeHtml(value).replace(/`/g, "&#96;");
  }

  function clamp(value, min, max) {
    return Math.min(max, Math.max(min, value));
  }

  function parseAgeToMinutes(value) {
    const text = normalizeText(value);
    if (!text) {
      return null;
    }

    const days = Number((text.match(/(\d+)d/i) || [])[1] || 0);
    const hours = Number((text.match(/(\d+)h/i) || [])[1] || 0);
    const minutes = Number((text.match(/(\d+)m/i) || [])[1] || 0);
    return days * 24 * 60 + hours * 60 + minutes;
  }

  function interpolateColor(start, end, progress) {
    return start.map((channel, index) => Math.round(channel + (end[index] - channel) * progress));
  }

  function rgbToCss(rgb) {
    return `rgb(${rgb.join(", ")})`;
  }

  function mixWithWhite(rgb, amount) {
    return rgb.map((channel) => Math.round(channel + (255 - channel) * amount));
  }

  function getAgeVisuals(ageText) {
    const ageMinutes = parseAgeToMinutes(ageText);
    const green = [67, 190, 92];
    const yellow = [235, 194, 59];
    const red = [222, 76, 63];
    const yellowThreshold = 3 * 24 * 60;
    const redThreshold = 5 * 24 * 60;

    let baseColor = green;
    if (ageMinutes != null) {
      if (ageMinutes <= yellowThreshold) {
        const progress = clamp(ageMinutes / yellowThreshold, 0, 1);
        baseColor = interpolateColor(green, yellow, progress);
      } else if (ageMinutes < redThreshold) {
        const progress = clamp((ageMinutes - yellowThreshold) / (redThreshold - yellowThreshold), 0, 1);
        baseColor = interpolateColor(yellow, red, progress);
      } else {
        baseColor = red;
      }
    }

    const start = mixWithWhite(baseColor, 0.86);
    const end = mixWithWhite(baseColor, 0.78);
    const border = mixWithWhite(baseColor, 0.56);

    return {
      textColor: rgbToCss(baseColor),
      borderColor: rgbToCss(border),
      background: `linear-gradient(135deg, ${rgbToCss(start)} 0%, ${rgbToCss(end)} 100%)`,
    };
  }

  function copyText(text) {
    if (typeof GM_setClipboard === "function") {
      GM_setClipboard(text);
      return Promise.resolve();
    }

    if (navigator.clipboard?.writeText) {
      return navigator.clipboard.writeText(text);
    }

    throw new Error("Clipboard API unavailable");
  }

  function ensureXLSXLoaded() {
    const library =
      window.XLSX ||
      globalThis.XLSX ||
      (typeof XLSX !== "undefined" ? XLSX : null);

    if (!library) {
      return Promise.reject(
        new Error("XLSX library unavailable. Re-save the Tampermonkey script so the @require dependency loads.")
      );
    }

    return Promise.resolve(library);
  }

  function focusPanelInput() {
    if (Date.now() < state.suppressAutoFocusUntil) {
      return;
    }

    const scanInput = getPanelScanInput();
    if (scanInput instanceof HTMLInputElement && !state.panelHidden) {
      window.requestAnimationFrame(() => {
        scanInput.focus();
        scanInput.select();
      });
    }
  }

  function getScaleMode() {
    const pixelRatio = window.devicePixelRatio || 1;
    return pixelRatio >= 1.4 ? "high" : "normal";
  }

  function applyPanelScaleMode() {
    const panel = getPanel();
    const launcher = getLauncher();
    if (panel) {
      panel.setAttribute("data-scale-mode", getScaleMode());
    }
    if (launcher) {
      launcher.setAttribute("data-scale-mode", getScaleMode());
    }
  }

  function clearScheduledFocusRestore() {
    state.focusRestoreTimerIds.forEach((timerId) => window.clearTimeout(timerId));
    state.focusRestoreTimerIds = [];
  }

  function schedulePanelFocusRestore(delays) {
    clearScheduledFocusRestore();
    state.focusRestoreTimerIds = delays.map((delay) =>
      window.setTimeout(() => {
        focusPanelInput();
      }, delay)
    );
  }

  function suppressAutoFocus(ms) {
    state.suppressAutoFocusUntil = Date.now() + ms;
    clearScheduledFocusRestore();
  }

  function isEditableTarget(target) {
    if (!(target instanceof Element)) {
      return false;
    }

    return Boolean(target.closest("input, textarea, select, [contenteditable=''], [contenteditable='true']"));
  }

  function isVisibleElement(element) {
    return Boolean(
      element &&
        element instanceof HTMLElement &&
        element.isConnected &&
        element.getClientRects().length > 0 &&
        window.getComputedStyle(element).visibility !== "hidden" &&
        window.getComputedStyle(element).display !== "none"
    );
  }

  function queryPreferredElement(selector) {
    const matches = Array.from(document.querySelectorAll(selector));
    return matches.reverse().find(isVisibleElement) || matches[0] || null;
  }

  function queryVisibleElement(selector) {
    return Array.from(document.querySelectorAll(selector)).reverse().find(isVisibleElement) || null;
  }

      tableNext: queryPreferredElement("#result-inventory-details_next"),

  function isDefaultAction(action) {
    return DEFAULT_ACTIONS.includes(normalizeText(action));
  }

  function normalizeActionValue(action) {
    const normalized = normalizeText(action);
    if (!normalized || normalized === "Idle") {
      return "Nothing";
    }

    return normalized;
  }

  function getAvailableActions() {
    return [...DEFAULT_ACTIONS, ...state.customActions.filter((action) => !isDefaultAction(action))];
  }

  function toHistoryEntry(entry) {
    return {
      historyEntryId: normalizeText(entry?.historyEntryId || "") || `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
      scannedValue: normalizeText(entry?.scannedValue),
      containerDisplay: normalizeText(entry?.containerDisplay || entry?.scannedValue),
      capturedAt: entry?.capturedAt || new Date().toISOString(),
      details: {
        "Total units": normalizeText(entry?.details?.["Total units"] || ""),
        "Oldest Units Age": normalizeText(entry?.details?.["Oldest Units Age"] || ""),
        "Parent Container": normalizeText(entry?.details?.["Parent Container"] || ""),
        "Grandparent Container": normalizeText(entry?.details?.["Grandparent Container"] || ""),
      },
      totalRecords: Number(entry?.totalRecords || 0),
      reasonCodes: Array.isArray(entry?.reasonCodes) ? entry.reasonCodes.map(normalizeText).filter(Boolean) : [],
      notFound: Boolean(entry?.notFound),
    };
  }

  function toActionRecord(entry, existingAction) {
    return {
      scannedValue: normalizeText(entry?.scannedValue),
      containerDisplay: normalizeText(entry?.containerDisplay || entry?.scannedValue),
      capturedAt: entry?.capturedAt || new Date().toISOString(),
      totalUnits: normalizeText(entry?.details?.["Total units"] || entry?.totalUnits || ""),
      oldestUnitsAge: normalizeText(entry?.details?.["Oldest Units Age"] || entry?.oldestUnitsAge || ""),
      parentContainer: normalizeText(entry?.details?.["Parent Container"] || entry?.parentContainer || ""),
      grandparentContainer: normalizeText(entry?.details?.["Grandparent Container"] || entry?.grandparentContainer || ""),
      reasonCodes: Array.isArray(entry?.reasonCodes) ? entry.reasonCodes.map(normalizeText).filter(Boolean) : [],
      notFound: Boolean(entry?.notFound),
      action: normalizeActionValue(existingAction || entry?.action || "Nothing"),
    };
  }

  function saveHistory() {
    try {
      localStorage.setItem(HISTORY_STORAGE_KEY, JSON.stringify(state.history.map(toHistoryEntry)));
    } catch (_error) {
      // Ignore storage failures and keep runtime history working.
    }
  }

  function saveActionRecords() {
    try {
      localStorage.setItem(ACTION_RECORDS_STORAGE_KEY, JSON.stringify(state.actionRecords));
    } catch (_error) {
      // Ignore storage failures and keep runtime behavior working.
    }
  }

  function loadActionRecords() {
    try {
      const raw = localStorage.getItem(ACTION_RECORDS_STORAGE_KEY);
      if (!raw) {
        return {};
      }

      const parsed = JSON.parse(raw);
      if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) {
        return {};
      }

      return Object.fromEntries(
        Object.entries(parsed)
          .map(([key, value]) => [key, toActionRecord(value)])
          .filter(([, value]) => value.scannedValue)
      );
    } catch (_error) {
      return {};
    }
  }

  function saveCustomActions() {
    try {
      localStorage.setItem(CUSTOM_ACTIONS_STORAGE_KEY, JSON.stringify(state.customActions));
    } catch (_error) {
      // Ignore storage failures and keep runtime behavior working.
    }
  }

  function loadCustomActions() {
    try {
      const raw = localStorage.getItem(CUSTOM_ACTIONS_STORAGE_KEY);
      if (!raw) {
        return [];
      }

      const parsed = JSON.parse(raw);
      if (!Array.isArray(parsed)) {
        return [];
      }

      return parsed
        .map(normalizeText)
        .filter(Boolean)
        .filter((action) => !isDefaultAction(action));
    } catch (_error) {
      return [];
    }
  }

  function loadHistory() {
    try {
      const raw = localStorage.getItem(HISTORY_STORAGE_KEY);
      if (!raw) {
        return [];
      }

      const parsed = JSON.parse(raw);
      if (!Array.isArray(parsed)) {
        return [];
      }

      return parsed
        .map(toHistoryEntry)
        .filter((entry) => entry.scannedValue)
        .slice(0, HISTORY_LIMIT);
    } catch (_error) {
      return [];
    }
  }

  function extractNodeData(node) {
    if (!node) {
      return {
        text: "",
        href: "",
        html: "",
        links: [],
      };
    }

    const text = normalizeText(node.textContent);
    const html = node.innerHTML || "";
    const links = Array.from(node.querySelectorAll("a"))
      .map((anchor) => ({
        text: normalizeText(anchor.textContent),
        href: anchor.href || anchor.getAttribute("href") || "",
      }))
      .filter((link) => link.text || link.href);

    return {
      text,
      href: links[0]?.href || "",
      html,
      links,
    };
  }

  function renderLinkedValue(valueData, fallbackText) {
    const text = normalizeText(valueData?.text || fallbackText || "-") || "-";
    const href = valueData?.href || "";
    if (!href) {
      return escapeHtml(text);
    }

    return `<a class="tm-link" href="${escapeAttribute(href)}" target="_blank" rel="noopener noreferrer">${escapeHtml(text)}</a>`;
  }

  function parseDetailsList(container) {
    if (!container) {
      return {
        pairs: {},
        meta: {},
      };
    }

    const pairs = {};
    const meta = {};
    const nodes = Array.from(container.children);
    let currentKey = "";

    for (const node of nodes) {
      const tag = node.tagName?.toLowerCase();
      const text = normalizeText(node.textContent);
      if (!text) {
        continue;
      }

      if (tag === "dt") {
        currentKey = text;
        if (!(currentKey in pairs)) {
          pairs[currentKey] = "";
        }
      } else if (tag === "dd" && currentKey) {
        const data = extractNodeData(node);
        pairs[currentKey] = data.text;
        meta[currentKey] = data;
      }
    }

    return { pairs, meta };
  }

  function parseTable(table) {
    if (!table) {
      return { headers: [], rows: [] };
    }

    const headers = Array.from(table.querySelectorAll("thead th")).map((th, index) => {
      const clone = th.cloneNode(true);
      clone.querySelectorAll("script, style").forEach((node) => node.remove());
      return normalizeText(clone.textContent) || `Column ${index + 1}`;
    });

    const rows = Array.from(table.querySelectorAll("tbody tr")).map((tr) => {
      const cellsData = Array.from(tr.querySelectorAll("td")).map((td) => extractNodeData(td));
      const cells = cellsData.map((cell) => cell.text);
      const mapped = {};
      const mappedMeta = {};

      cells.forEach((value, index) => {
        mapped[headers[index] || `Column ${index + 1}`] = value;
        mappedMeta[headers[index] || `Column ${index + 1}`] = cellsData[index];
      });

      return { values: cells, mapped, cellsData, mappedMeta };
    });

    return { headers, rows };
  }

  function parseTotalRecords(rawText, fallback) {
    const match = normalizeText(rawText).match(/Showing\s+\d+\s+to\s+\d+\s+of\s+(\d+)\s+inventory records/i);
    return match ? Number(match[1]) : fallback;
  }

  function buildPaginationState(previousButton, nextButton) {
    function getButtonState(button) {
      if (!(button instanceof HTMLElement)) {
        return {
          visible: false,
          enabled: false,
          text: "",
          className: "",
        };
      }

      const className = button.className || "";
      const disabled = /paginate_disabled/i.test(className) || button.getAttribute("aria-disabled") === "true";
      return {
        visible: true,
        enabled: !disabled,
        text: normalizeText(button.textContent),
        className,
      };
    }

    return {
      previous: getButtonState(previousButton),
      next: getButtonState(nextButton),
    };
  }

  function extractResult(scannedValue) {
    const { popoverHeader, popoverContent, detailsList, resultsTable, tablePrevious, tableNext } = getSiteElements();
    const detailInfo = parseDetailsList(detailsList);
    const table = parseTable(resultsTable);
    const title = normalizeText(popoverHeader?.textContent) || "Container details";
    const rawText = normalizeText(popoverContent?.innerText || "");
    const containerDisplay = normalizeText((rawText.match(/^(.+?)\s+Total units\b/i) || [])[1] || scannedValue);
    const contentLinks = extractNodeData(popoverContent).links;
    const reasonCodes = Array.from(
      new Set(
        table.rows
          .map((row) => row.mapped.Disposition || row.mapped["Reason Code"] || row.mapped["Inventory Type"] || row.mapped.Owner || "")
          .map(normalizeText)
          .filter(Boolean)
      )
    );

    return {
      scannedValue,
      title,
      capturedAt: new Date().toISOString(),
      containerDisplay,
      details: detailInfo.pairs,
      detailMeta: detailInfo.meta,
      table,
      totalRecords: parseTotalRecords(rawText, table.rows.length),
      reasonCodes,
      rawText,
      contentLinks,
      pagination: buildPaginationState(tablePrevious, tableNext),
      debugHtml: {
        popoverHeaderHtml: popoverHeader?.outerHTML || "",
        popoverContentHtml: popoverContent?.innerHTML || "",
        detailsListHtml: detailsList?.innerHTML || "",
        resultsTableHtml: resultsTable?.outerHTML || "",
      },
    };
  }

  function isPlaceholderResult(result) {
    const detailsValues = Object.values(result.details || {}).map(normalizeText);
    const hasOnlyNoneDetails = detailsValues.length > 0 && detailsValues.every((value) => !value || value === "None");
    const hasNoRows = !result.table?.rows?.length;
    const rawText = normalizeText(result.rawText);

    return hasOnlyNoneDetails && hasNoRows && rawText.includes("Total units None") && rawText.includes("Oldest Units Age None");
  }

  function resultContainsScan(result, scannedValue) {
    const normalizedScan = normalizeText(scannedValue).toLowerCase();
    if (!normalizedScan) {
      return false;
    }

    const normalizedDisplay = normalizeText(result.containerDisplay).toLowerCase();
    const normalizedRawText = normalizeText(result.rawText).toLowerCase();
    return normalizedDisplay.includes(normalizedScan) || normalizedRawText.includes(normalizedScan);
  }

  function isResolvedResult(result, scannedValue) {
    if (!result) {
      return false;
    }

    if (isPlaceholderResult(result)) {
      return false;
    }

    return resultContainsScan(result, scannedValue);
  }

      details: {

      },
      table: { headers: [], rows: [] },
      totalRecords: 0,
      reasonCodes: [],
      rawText: "",
      notFound: true,
    };
  }

    return {
      scannedValue,
      title: "Container details",
      capturedAt: new Date().toISOString(),
      containerDisplay: scannedValue,
      details: {
        "Total units": "No record found",
        "Oldest Units Age": "--",
        "Parent Container": "No record found",
        "Grandparent Container": "No record found",
      },
      table: { headers: [], rows: [] },
      totalRecords: 0,
      reasonCodes: [],
      rawText: "No record found",
      notFound: true,
    };
  }

  function upsertHistory(entry) {
    const historyEntry = toHistoryEntry(entry);
    const existingIndex = state.history.findIndex((item) => item.scannedValue === historyEntry.scannedValue);
    if (existingIndex >= 0) {
      state.history = [
        historyEntry,
        ...state.history.filter((item) => item.scannedValue !== historyEntry.scannedValue),
      ].slice(0, HISTORY_LIMIT);
    } else {
      state.history = [historyEntry, ...state.history].slice(0, HISTORY_LIMIT);
    }
    saveHistory();
  }

  function setRecentHistoryScan(scannedValue) {
    const normalizedScan = normalizeText(scannedValue);
    if (!normalizedScan) {
      return;
    }

    upsertHistory({
      scannedValue: normalizedScan,
      containerDisplay: normalizedScan,
      capturedAt: new Date().toISOString(),
      details: {
        "Total units": "",
        "Oldest Units Age": "",
        "Parent Container": "",
        "Grandparent Container": "",
      },
      totalRecords: 0,
      reasonCodes: [],
      notFound: false,
    });
  }

  function upsertActionRecord(entry) {
    const existing = state.actionRecords[entry.scannedValue];
    const nextRecord = toActionRecord(entry, existing?.action);
    state.actionRecords[nextRecord.scannedValue] = nextRecord;
    saveActionRecords();
    return nextRecord;
  }

  function getCurrentAction() {
    if (!state.lastResult?.scannedValue) {
      return "Nothing";
    }

    return state.actionRecords[state.lastResult.scannedValue]?.action || "Nothing";
  }

  function setActionForContainer(scannedValue, action) {
    const normalizedScan = normalizeText(scannedValue);
    const normalizedAction = normalizeActionValue(action);
    if (!normalizedScan) {
      return;
    }

    const existing = state.actionRecords[normalizedScan];
    const source = existing || state.lastResult || { scannedValue: normalizedScan };
    state.actionRecords[normalizedScan] = {
      ...toActionRecord(source, normalizedAction),
      scannedValue: normalizedScan,
      action: normalizedAction,
    };
    saveActionRecords();
  }

  function addCustomAction(actionName) {
    const normalizedAction = normalizeText(actionName);
    if (!normalizedAction || isDefaultAction(normalizedAction)) {
      return null;
    }

    if (!state.customActions.includes(normalizedAction)) {
      state.customActions = [...state.customActions, normalizedAction];
      saveCustomActions();
    }

    return normalizedAction;
  }

  function deleteCustomAction(actionName) {
    const normalizedAction = normalizeText(actionName);
    if (!normalizedAction || isDefaultAction(normalizedAction)) {
      return false;
    }

    state.customActions = state.customActions.filter((action) => action !== normalizedAction);
    Object.keys(state.actionRecords).forEach((scan) => {
      if (state.actionRecords[scan]?.action === normalizedAction) {
        state.actionRecords[scan].action = "Nothing";
      }
    });
    saveCustomActions();
    saveActionRecords();
    return true;
  }

  function renderHistory(historyElement) {
    const searchQuery = normalizeText(state.historySearchQuery).toLowerCase();
    const filteredHistory = state.history.filter((entry) => {
      if (!searchQuery) {
        return true;
      }

      return (
        normalizeText(entry.scannedValue).toLowerCase().includes(searchQuery) ||
        normalizeText(entry.containerDisplay).toLowerCase().includes(searchQuery)
      );
    });

    historyElement.innerHTML = filteredHistory.length
      ? filteredHistory
          .map(
            (entry) => `
              <button class="tm-history-item" data-scan="${escapeHtml(entry.scannedValue)}" type="button">
                <span>${escapeHtml(entry.scannedValue)}</span>
              </button>
            `
          )
          .join("")
      : `<div class='tm-empty'>${state.history.length ? "No matching scans." : "No scans yet."}</div>`;
  }

  function buildHistoryExportPayload() {
    const records = Object.values(state.actionRecords).sort((left, right) =>
      String(right.capturedAt || "").localeCompare(String(left.capturedAt || ""))
    );
    const summaryCounts = records.reduce((accumulator, entry) => {
      const action = normalizeActionValue(entry.action);
      accumulator[action] = (accumulator[action] || 0) + 1;
      return accumulator;
    }, {});
    const summary = Object.entries(summaryCounts)
      .sort((left, right) => right[1] - left[1] || left[0].localeCompare(right[0]))
      .map(([action, count]) => ({
        action,
        count,
        percent: records.length ? `${((count / records.length) * 100).toFixed(1)}%` : "0.0%",
      }));

    return {
      exportedAt: new Date().toISOString(),
      count: records.length,
      summary,
      scans: records.map((entry) => ({
        scannedValue: entry.scannedValue,
        containerDisplay: entry.containerDisplay || entry.scannedValue,
        capturedAt: formatHumanDateTime(entry.capturedAt || ""),
        totalUnits: entry.totalUnits || "",
        oldestUnitsAge: entry.oldestUnitsAge || "",
        parentContainer: entry.parentContainer || "",
        grandparentContainer: entry.grandparentContainer || "",
        reasonCodes: entry.reasonCodes?.join(", ") || "",
        notFound: entry.notFound ? "Yes" : "",
        action: entry.action || "Nothing",
      })),
    };
  }

  function clearHistoryWithConfirmation() {
    if (!state.history.length) {
      return false;
    }

    const confirmed = window.confirm("Clear all saved scan history?");
    if (!confirmed) {
      return false;
    }

    state.history = [];
    saveHistory();
    render();
    return true;
  }

  function getWorkbookType(fileName) {
    const normalized = String(fileName || "").toLowerCase();
    if (normalized.endsWith(".xlsm")) {
      return "xlsm";
    }
    if (normalized.endsWith(".xlsb")) {
      return "xlsb";
    }
    if (normalized.endsWith(".xls")) {
      return "xls";
    }
    return "xlsx";
  }

  async function downloadHistoryWorkbook() {
    const XLSX = await ensureXLSXLoaded();
    const workbook = XLSX.utils.book_new();
    const worksheet = buildHistoryWorksheet(XLSX);
    XLSX.utils.book_append_sheet(workbook, worksheet, getWorkbookSheetName());
    const buffer = XLSX.write(workbook, {
      bookType: "xlsm",
      type: "array",
      bookVBA: true,
    });
    const blob = new Blob([buffer], { type: "application/vnd.ms-excel.sheet.macroEnabled.12" });
    const url = URL.createObjectURL(blob);
    const anchor = document.createElement("a");
    const timestamp = new Date().toISOString().replace(/[:.]/g, "-");
    anchor.href = url;
    anchor.download = `scan-history-${timestamp}.xlsm`;
    document.body.appendChild(anchor);
    anchor.click();
    anchor.remove();
    window.setTimeout(() => URL.revokeObjectURL(url), 1000);
    return anchor.download;
  }

  async function navigateResultPage(direction) {
    if (!state.lastResult || state.busy) {
      return;
    }

    const { tablePrevious, tableNext } = getSiteElements();
    const button = direction === "previous" ? tablePrevious : tableNext;
    if (!(button instanceof HTMLElement)) {
      return;
    }

    const className = button.className || "";
    if (/paginate_disabled/i.test(className) || button.getAttribute("aria-disabled") === "true") {
      return;
    }

    const currentRawText = normalizeText(state.lastResult.rawText || "");
    state.busy = true;
    render();

    try {
      button.click();
      const result = await waitFor(() => {
        const candidate = extractResult(state.lastResult.scannedValue);
        const candidateRawText = normalizeText(candidate.rawText || "");
        if (!candidateRawText || candidateRawText === currentRawText) {
          return false;
        }

        return resultContainsScan(candidate, state.lastResult.scannedValue) ? candidate : false;
      }, 10000);

      state.lastResult = result;
      upsertHistory(result);
    } finally {
      state.busy = false;
      render();
      schedulePanelFocusRestore([50]);
    }
  }

  function render() {
    const panel = getPanel();
    const launcher = getLauncher();
    const backdrop = document.getElementById(`${PANEL_ID}-backdrop`);
    if (!panel || !launcher || !backdrop) {
      return;
    }

    applyPanelScaleMode();

    panel.style.display = state.panelHidden ? "none" : "block";
    launcher.style.display = state.panelHidden ? "inline-flex" : "none";
    backdrop.style.display = state.panelHidden ? "none" : "block";

    const status = panel.querySelector("[data-role='status']");
    const history = panel.querySelector("[data-role='history']");
    const results = panel.querySelector("[data-role='results']");
    const oldestAgeCard = panel.querySelector("[data-role='oldest-age']");
    const oldestAgeValue = panel.querySelector("[data-role='oldest-age-value']");

    if (status) {
      status.textContent = state.busy
        ? `Searching ${state.lastScan || "container"}...`
        : state.lastResult
          ? `Showing results for ${state.lastResult.scannedValue}.`
          : "Ready to scan.";
    }

    const oldestAgeText = state.lastResult?.details?.["Oldest Units Age"] || "--";
    const ageVisuals = getAgeVisuals(oldestAgeText);

    if (oldestAgeCard instanceof HTMLElement) {
      oldestAgeCard.style.background = ageVisuals.background;
      oldestAgeCard.style.borderColor = ageVisuals.borderColor;
    }

    if (oldestAgeValue instanceof HTMLElement) {
      oldestAgeValue.textContent = oldestAgeText;
      oldestAgeValue.style.color = ageVisuals.textColor;
    }

    if (history) {
      renderHistory(history);
    }

    if (results) {
      renderResults(results);
    }
  }

  function handleGlobalScannerKeydown(event) {
    if (!state.globalCapture || state.busy || state.panelHidden) {
      return;
    }

    if (event.defaultPrevented || event.ctrlKey || event.altKey || event.metaKey) {
      return;
    }

    const panel = getPanel();
    if (!panel) {
      return;
    }

    const target = event.target;
    if (target instanceof Element && panel.contains(target)) {
      return;
    }

    if (isEditableTarget(target)) {
      return;
    }

    const now = Date.now();
    if (now - state.lastKeystrokeAt > SCAN_RESET_MS) {
      state.scanBuffer = "";
    }
    state.lastKeystrokeAt = now;

    if (event.key === "Enter") {
      if (!state.scanBuffer) {
        return;
      }
      event.preventDefault();
      event.stopPropagation();
      flushGlobalScanBuffer();
      return;
    }

    if (event.key.length !== 1) {
      return;
    }

    state.scanBuffer += event.key;
    event.preventDefault();
    event.stopPropagation();

    const scanInput = getPanelScanInput();
    if (scanInput instanceof HTMLInputElement) {
      scanInput.value = state.scanBuffer;
    }
  }

  async function performLookup(scannedValue) {
    if (!scannedValue || state.busy) {
      return;
    }

    state.busy = true;
    state.lastScan = scannedValue;
      setRecentHistoryScan(scannedValue);
    render();

    try {
      const { searchInput, searchButton } = getSiteElements();
      if (!searchInput || !searchButton) {
        throw new Error("Could not find the site search controls.");
      }

      const previousRawText = normalizeText(state.lastResult?.rawText || "");

      setNativeValue(searchInput, scannedValue);
      searchButton.click();

      const result = await waitFor(() => {
        const { popoverHeader, popoverContent, noRecordAlert } = getSiteElements();
        if (normalizeText(noRecordAlert?.textContent).includes("No record found")) {
          return buildNoRecordFoundResult(scannedValue);
        }

        if (
          !normalizeText(popoverHeader?.textContent).includes("Container details") ||
          normalizeText(popoverContent?.innerText).length === 0
        ) {
          return false;
        }

        const candidate = extractResult(scannedValue);
        if (normalizeText(candidate.rawText || "") === previousRawText) {
          return false;
        }

        return isResolvedResult(candidate, scannedValue) ? candidate : false;
      }, 15000);

      state.lastResult = result;
      upsertHistory(result);
      upsertActionRecord(result);
    } catch (error) {
      const { noRecordAlert } = getSiteElements();
      if (normalizeText(noRecordAlert?.textContent).includes("No record found")) {
        state.lastResult = buildNoRecordFoundResult(scannedValue);
        upsertHistory(state.lastResult);
        upsertActionRecord(state.lastResult);
        return;
      }

      const latestResult = extractResult(scannedValue);
      const previousRawText = normalizeText(state.lastResult?.rawText || "");

      if (
        isPlaceholderResult(latestResult) &&
        normalizeText(latestResult.rawText || "") !== previousRawText
      ) {
        state.lastResult = buildNotFoundResult(scannedValue);
      } else {
        state.lastResult = {
          scannedValue,
          title: "Lookup failed",
          capturedAt: new Date().toISOString(),
          containerDisplay: scannedValue,
          details: {
            "Total units": "",
            Error: error instanceof Error ? error.message : String(error),
            "Oldest Units Age": "--",
            "Parent Container": "",
            "Grandparent Container": "",
          },
          table: { headers: [], rows: [] },
          totalRecords: 0,
          reasonCodes: [],
          rawText: "",
        };
      }
      upsertHistory(state.lastResult);
      upsertActionRecord(state.lastResult);
    } finally {
      state.busy = false;
      state.scanBuffer = "";
      state.lastKeystrokeAt = 0;

      const scanInput = getPanelScanInput();
      if (scanInput instanceof HTMLInputElement) {
        scanInput.value = "";
      }

      render();
      schedulePanelFocusRestore([50, 250, 750]);
    }
  }

          status.textContent = "Debug copied to clipboard.";

        const status = panel.querySelector("[data-role='status']");
        if (status) {
          status.textContent = `Copy failed: ${error instanceof Error ? error.message : String(error)}`;
        }
      }
    });

      try {
        const fileName = await downloadHistoryWorkbook();
        if (status) {
          status.textContent = `History exported as ${fileName}.`;
        }

        if (status) {
          status.textContent = `History export failed: ${error instanceof Error ? error.message : String(error)}`;
        }

    });

      const cleared = clearHistoryWithConfirmation();
      if (status && cleared) {
        status.textContent = "Scan history cleared.";

    });

      const target = event.target instanceof Element ? event.target.closest("[data-scan]") : null;
      const value = target?.getAttribute("data-scan");
      if (value) {
        performLookup(value);

    });

    results?.addEventListener("click", (event) => {
      const target = event.target instanceof Element
        ? event.target.closest(
            "[data-role='prev-page'], [data-role='next-page'], [data-role='delete-custom-action'], [data-role='action-button'], [data-role='custom-action-button']"
          )
        : null;
      if (!target) {
        return;
      }

      if (target.getAttribute("data-role") === "prev-page") {
        navigateResultPage("previous");
      } else if (target.getAttribute("data-role") === "next-page") {
        navigateResultPage("next");
      } else if (target.getAttribute("data-role") === "delete-custom-action") {
        const currentAction = getCurrentAction();
        if (!currentAction || isDefaultAction(currentAction)) {
          return;
        }

        const confirmed = window.confirm(`Delete custom action "${currentAction}"? Containers using it will be set to Nothing.`);
        if (confirmed) {
          deleteCustomAction(currentAction);
          render();
        }
      } else if (target.getAttribute("data-role") === "action-button" && state.lastResult) {
        const action = target.getAttribute("data-action") || "Nothing";
        setActionForContainer(state.lastResult.scannedValue, action);
        render();
      } else if (target.getAttribute("data-role") === "custom-action-button" && state.lastResult) {
        const actionName = window.prompt("Enter custom action name:");
        const addedAction = addCustomAction(actionName || "");
        if (addedAction) {
          setActionForContainer(state.lastResult.scannedValue, addedAction);
        }
        render();
      }
    });

      const target = event.target;
      if (!(target instanceof Element)) {
        return;
      }

      if (
        target.closest(
          "[data-role='action-button'], [data-role='custom-action-button'], [data-role='delete-custom-action'], [data-role='history-search']"
        )
      ) {
        suppressAutoFocus(4000);
      }
    });

      state.panelHidden = true;
      render();

      state.panelHidden = false;
      render();

    });

      render();
    });

    document.addEventListener("keydown", handleGlobalScannerKeydown, true);
    document.addEventListener("click", (event) => {
      const target = event.target;
      if (!(target instanceof Element) || !panel.contains(target)) {
        state.suppressAutoFocusUntil = 0;
        schedulePanelFocusRestore([0]);
      }
    });
    window.addEventListener("focus", () => {
      state.suppressAutoFocusUntil = 0;
      schedulePanelFocusRestore([0]);
    });

    render();
  }

  function boot() {
    if (!document.body || !document.head) {
      window.setTimeout(boot, 250);
      return;
    }

    state.history = loadHistory();
    state.actionRecords = loadActionRecords();
    state.customActions = loadCustomActions();
    installPanel();
  }
