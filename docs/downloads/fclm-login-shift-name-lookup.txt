// @name         FCLM Login Shift + Name Lookup

  function normalizeLogins(value) {
    return Array.from(
      new Set(
        (value ?? '')
          .split(/[\s,;]+/)
          .map((item) => item.trim())
          .filter(Boolean)
      )
    );
  }

    return new URL(window.location.href).searchParams.get('warehouseId') ?? '';
  }

  function getDefinitionValue(doc, labels) {
    const terms = Array.from(doc.querySelectorAll('dt'));
    for (const term of terms) {
      const termText = term.textContent?.replace(/\s+/g, ' ').trim() ?? '';
      if (!labels.some((label) => termText.toLowerCase() === label.toLowerCase())) {
        continue;
      }

      let sibling = term.nextElementSibling;
      while (sibling && sibling.tagName !== 'DD') {
        sibling = sibling.nextElementSibling;
      }

      const value = sibling?.textContent?.replace(/\s+/g, ' ').trim() ?? '';
      if (value) {
        return value;
      }
    }

    return '';
  }

  function parseFullName(doc) {
    const headerText = doc.querySelector('.fold-control')?.textContent?.replace(/\s+/g, ' ').trim() ?? '';
    if (!headerText) {
      return '';
    }

    const match = headerText.match(/^(.*?)\s*\(/);
    return (match?.[1] ?? headerText).trim();
  }

  function parseShift(doc) {
    const directMatch = getDefinitionValue(doc, shiftLabels);
    if (directMatch) {
      return directMatch;
    }

    const combinedText = doc.body?.textContent?.replace(/\s+/g, ' ').trim() ?? '';
    const fallbackMatch = combinedText.match(
      /\b(?:Shift|Shift Pattern|Schedule|Shift Code)\b\s*[:\-]?\s*([A-Z0-9][A-Z0-9 /\-]{0,40})/i
    );
    return fallbackMatch?.[1]?.trim() ?? '';
  }

  async function copyText(text) {
    if (typeof GM_setClipboard === 'function') {
      GM_setClipboard(text, 'text');
      return;
    }

    if (navigator.clipboard?.writeText) {
      await navigator.clipboard.writeText(text);
      return;
    }

    const textarea = document.createElement('textarea');
    textarea.value = text;
    textarea.style.position = 'fixed';
    textarea.style.opacity = '0';
    document.body.appendChild(textarea);
    textarea.select();
    document.execCommand('copy');
    textarea.remove();
  }

  async function fetchEmployeeDetails(login, warehouseId) {
    const url = new URL('/employee/timeDetails', window.location.origin);
    url.searchParams.set('warehouseId', warehouseId);
    url.searchParams.set('employeeId', login);

    const response = await fetch(url.toString(), {
      credentials: 'include'
    });

    if (!response.ok) {
      throw new Error(`HTTP ${response.status}`);
    }

    const html = await response.text();
    const doc = new DOMParser().parseFromString(html, 'text/html');
    const fullName = parseFullName(doc);
    const shift = parseShift(doc);

    return {
      login,
      fullName: fullName || 'Not found',
      shift: shift || 'Not found'
    };
  }

  function install() {
    if (window[installationKey]) {
      return;
    }
    window[installationKey] = true;

    document.documentElement.appendChild(createStyles());

    const launcher = document.createElement('button');
    launcher.type = 'button';
    launcher.className = 'tm-fclm-launcher';
    launcher.textContent = 'Shift Lookup';

    let overlay = null;

    function closeOverlay() {
      overlay?.remove();
      overlay = null;
    }

    function openOverlay() {
      if (overlay?.isConnected) {
        return;
      }

      overlay = document.createElement('div');
      overlay.className = 'tm-fclm-overlay';

      const modal = document.createElement('div');
      modal.className = 'tm-fclm-modal';

      const header = document.createElement('div');
      header.className = 'tm-fclm-header';

      const title = document.createElement('h2');
      title.className = 'tm-fclm-title';
      title.textContent = 'FCLM Login Shift + Name Lookup';

      const closeButton = document.createElement('button');
      closeButton.type = 'button';
      closeButton.className = 'tm-fclm-close';
      closeButton.textContent = '×';
      closeButton.addEventListener('click', closeOverlay);

      header.appendChild(title);
      header.appendChild(closeButton);

      const content = document.createElement('div');
      content.className = 'tm-fclm-content';

      const help = document.createElement('p');
      help.className = 'tm-fclm-help';
      help.textContent = 'Enter a warehouse and paste one or more logins separated by commas, spaces, or new lines.';

      const warehouseRow = document.createElement('div');
      warehouseRow.className = 'tm-fclm-row';

      const warehouseLabel = document.createElement('label');
      warehouseLabel.className = 'tm-fclm-label';
      warehouseLabel.textContent = 'Warehouse ID';

      const warehouseInput = document.createElement('input');
      warehouseInput.className = 'tm-fclm-input';
      warehouseInput.value = getCurrentWarehouseId();
      warehouseInput.placeholder = 'MEM3';

      warehouseRow.appendChild(warehouseLabel);
      warehouseRow.appendChild(warehouseInput);

      const loginRow = document.createElement('div');
      loginRow.className = 'tm-fclm-row';

      const loginLabel = document.createElement('label');
      loginLabel.className = 'tm-fclm-label';
      loginLabel.textContent = 'Logins';

      const loginInput = document.createElement('textarea');
      loginInput.className = 'tm-fclm-textarea';
      loginInput.placeholder = 'abelrigm\njdoe\nasmith';

      loginRow.appendChild(loginLabel);
      loginRow.appendChild(loginInput);

      const actions = document.createElement('div');
      actions.className = 'tm-fclm-actions';

      const lookupButton = document.createElement('button');
      lookupButton.type = 'button';
      lookupButton.className = 'tm-fclm-button';
      lookupButton.textContent = 'Lookup';

      const copyDbButton = document.createElement('button');
      copyDbButton.type = 'button';
      copyDbButton.className = 'tm-fclm-button tm-fclm-secondary';
      copyDbButton.textContent = 'Copy DB Logins';

      const clearButton = document.createElement('button');
      clearButton.type = 'button';
      clearButton.className = 'tm-fclm-button tm-fclm-secondary';
      clearButton.textContent = 'Clear';

      actions.appendChild(lookupButton);
      actions.appendChild(copyDbButton);
      actions.appendChild(clearButton);

      const status = document.createElement('div');
      status.className = 'tm-fclm-status';
      status.textContent = 'Enter logins to look up.';

      const tableWrap = document.createElement('div');
      tableWrap.className = 'tm-fclm-table-wrap';

      const table = document.createElement('table');
      table.className = 'tm-fclm-table';

      const thead = document.createElement('thead');
      thead.innerHTML = '<tr><th>Login</th><th>Full Name</th><th>Shift</th><th>Status</th></tr>';

      const tbody = document.createElement('tbody');
      table.appendChild(thead);
      table.appendChild(tbody);
      tableWrap.appendChild(table);
      let lookupResults = [];

      async function runLookup() {
        const warehouseId = warehouseInput.value.trim();
        const logins = normalizeLogins(loginInput.value);

        tbody.replaceChildren();
        lookupResults = [];

        if (!warehouseId) {
          status.textContent = 'Warehouse ID is required.';
          return;
        }

        if (logins.length === 0) {
          status.textContent = 'No logins entered.';
          return;
        }

        lookupButton.disabled = true;
        status.textContent = `Looking up ${logins.length} login${logins.length === 1 ? '' : 's'}...`;

        try {
          for (const login of logins) {
            const row = document.createElement('tr');
            row.innerHTML = `<td>${login}</td><td>Loading...</td><td>Loading...</td><td>Working</td>`;
            tbody.appendChild(row);

            try {
              const result = await fetchEmployeeDetails(login, warehouseId);
              row.children[1].textContent = result.fullName;
              row.children[2].textContent = result.shift;
              row.children[3].textContent = 'OK';
              lookupResults.push(result);
            } catch (error) {
              row.children[1].textContent = 'Not found';
              row.children[2].textContent = 'Not found';
              row.children[3].textContent = error instanceof Error ? error.message : String(error);
              row.children[3].className = 'tm-fclm-error';
            }
          }

          status.textContent = `Finished ${logins.length} lookup${logins.length === 1 ? '' : 's'}.`;
        } finally {
          lookupButton.disabled = false;
        }
      }

      lookupButton.addEventListener('click', runLookup);
      copyDbButton.addEventListener('click', async () => {
        const dbLogins = lookupResults
          .filter((result) => /^db/i.test(result.shift))
          .map((result) => result.login);

        if (dbLogins.length === 0) {
          status.textContent = 'No DB-shift logins found in the current results.';
          return;
        }

        copyDbButton.disabled = true;
        try {
          await copyText(dbLogins.join('\n'));
          status.textContent = `Copied ${dbLogins.length} DB login${dbLogins.length === 1 ? '' : 's'}.`;
        } catch (error) {
          status.textContent = `Copy failed: ${error instanceof Error ? error.message : String(error)}`;
        } finally {
          copyDbButton.disabled = false;
        }
      });
      clearButton.addEventListener('click', () => {
        loginInput.value = '';
        tbody.replaceChildren();
        lookupResults = [];
        status.textContent = 'Enter logins to look up.';
        loginInput.focus();
      });

      loginInput.addEventListener('keydown', (event) => {
        if ((event.ctrlKey || event.metaKey) && event.key === 'Enter') {
          runLookup();
        }
      });

      overlay.addEventListener('click', (event) => {
        if (event.target === overlay) {
          closeOverlay();
        }
      });

      content.appendChild(help);
      content.appendChild(warehouseRow);
      content.appendChild(loginRow);
      content.appendChild(actions);
      content.appendChild(status);
      content.appendChild(tableWrap);

      modal.appendChild(header);
      modal.appendChild(content);
      overlay.appendChild(modal);
      document.body.appendChild(overlay);
      loginInput.focus();
    }

    launcher.addEventListener('click', openOverlay);

    document.addEventListener('keydown', (event) => {
      if (event.key === 'Escape' && overlay?.isConnected) {
        closeOverlay();
      }
    });

    document.body.appendChild(launcher);
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', install, { once: true });
  } else {
    install();
  }
})();
