  function requestFclmPage(login) {
    const url = new URL(fclmBaseUrl);
    url.searchParams.set('warehouseId', defaultWarehouseId);
    url.searchParams.set('employeeId', login);

    return new Promise((resolve, reject) => {
      GM_xmlhttpRequest({
        method: 'GET',
        url: url.href,
        timeout: 30000,
        onload(response) {
          if (response.status < 200 || response.status >= 400) {
            reject(new Error(`FCLM returned HTTP ${response.status}`));
            return;
          }
          resolve(response.responseText);
        },
        onerror() {
          reject(new Error('Could not connect to FCLM'));
        },
        ontimeout() {
          reject(new Error('FCLM request timed out'));
        },
      });
    });
  }

  function findDefinitionValue(documentNode, selector, labelPattern) {
    const suppliedValue = normalizeText(
      documentNode.querySelector(selector)?.textContent
    );
    if (suppliedValue) {
      return suppliedValue;
    }

    for (const term of documentNode.querySelectorAll('dt')) {
      if (labelPattern.test(normalizeText(term.textContent))) {
        return normalizeText(term.nextElementSibling?.textContent);
      }
    }

    return '';
  }

  function findShift(documentNode) {
    return findDefinitionValue(
      documentNode,
      'dl.list-side-by-side:nth-of-type(2) > dd:nth-of-type(3)',
      /^shift\b/i
    );
  }

  function findBadgeNumber(documentNode) {
    return findDefinitionValue(
      documentNode,
      'dl.list-side-by-side:nth-of-type(1) > dd:nth-of-type(3)',
      /^badge(\s*(id|number|#))?$/i
    );
  }

  function parseEmployeePage(html, requestedLogin) {
    const documentNode = new DOMParser().parseFromString(html, 'text/html');
    const photo = documentNode.querySelector('img.photo');
    const title = documentNode.querySelector('td.title > span.fold-control');

    if (!photo || !title) {
      const isMidwayPage =
        /midway/i.test(documentNode.title) ||
        /sign in with midway/i.test(documentNode.body?.textContent ?? '');
      throw new Error(
        isMidwayPage
          ? 'FCLM authentication is required; open FCLM and sign in first'
          : 'Employee was not found in FCLM'
      );
    }

    const titleText = normalizeText(title.textContent);
    const loginMatch = titleText.match(/\(([^()]+)\)\s*$/);
    const photoUrl = new URL(photo.getAttribute('src'), fclmBaseUrl).href;
    const employeeId =
      new URL(photoUrl).searchParams.get('employeeid') ||
      new URL(photoUrl).searchParams.get('employeeId') ||
      '';

    return {
      login: loginMatch?.[1] || requestedLogin,
      name: loginMatch ? titleText.slice(0, loginMatch.index).trim() : titleText,
      shift: findShift(documentNode) || 'Shift unavailable',
      badgeNumber: findBadgeNumber(documentNode),
      photoUrl,
      employeeId,
    };
  }

  async function lookupEmployee(login) {
    const html = await requestFclmPage(login);
    return parseEmployeePage(html, login);
  }

  function formatDateOnly(date) {
    const year = date.getFullYear();
    const month = String(date.getMonth() + 1).padStart(2, '0');
    const day = String(date.getDate()).padStart(2, '0');
    return `${year}-${month}-${day}`;
  }

  function requestAttendanceCsv(queryDate) {
    const url = new URL('https://fclm-portal.amazon.com/reports/employeeAttendance');
    url.searchParams.set('reportFormat', 'CSV');
    url.searchParams.set('warehouseId', defaultWarehouseId);
    url.searchParams.set('spanType', 'Day');
    url.searchParams.set('startDateDay', queryDate);

    return new Promise((resolve, reject) => {
      GM_xmlhttpRequest({
        method: 'GET',
        url: url.href,
        timeout: 30000,
        onload(response) {
          if (response.status < 200 || response.status >= 300) {
            reject(new Error(`Attendance report returned HTTP ${response.status}`));
            return;
          }
          resolve(response.responseText);
        },
        onerror() {
          reject(new Error('Could not load the attendance report'));
        },
        ontimeout() {
          reject(new Error('Attendance report request timed out'));
        },
      });
    });
  }

  function parseCsv(csv, delimiter = ',') {
    const pattern = new RegExp(
      `(\\${delimiter}|\\r?\\n|\\r|^)(?:"([^"]*(?:""[^"]*)*)"|([^"\\${delimiter}\\r\\n]*))`,
      'gi'
    );
    const rows = [[]];
    let match;

    while ((match = pattern.exec(csv))) {
      if (match[1].length && match[1] !== delimiter) {
        rows.push([]);
      }
      rows[rows.length - 1].push(match[2] ? match[2].replace(/""/g, '"') : match[3]);
    }

    return rows;
  }

  async function fetchAttendanceRows() {
    const today = new Date();
    const yesterday = new Date(today);
    yesterday.setDate(yesterday.getDate() - 1);
    const reports = await Promise.all(
      [yesterday, today].map(async (date) => parseCsv(await requestAttendanceCsv(formatDateOnly(date))))
    );
    return reports.flat();
  }

  function normalizeName(value) {
    return String(value ?? '')
      .toLowerCase()
      .replace(/[^a-z0-9]+/g, ' ')
      .trim()
      .replace(/\s+/g, ' ');
  }

  function swapNameOrder(value) {
    const parts = String(value ?? '')
      .split(',')
      .map((part) => part.trim())
      .filter(Boolean);
    return parts.length >= 2 ? `${parts.slice(1).join(' ')} ${parts[0]}` : value;
  }

  function normalizePunchType(value) {
    const punchType = String(value ?? '').toLowerCase();

    if (punchType.includes('out') || punchType.includes('meal start')) {
      return { label: 'Clocked Out', className: 'out' };
    }
    if (
      punchType.includes('in') ||
      punchType.includes('meal end') ||
      punchType.includes('return')
    ) {
      return { label: 'Clocked In', className: 'in' };
    }
    return {
      label: value ? `Unknown (${value})` : 'Unknown',
      className: 'unknown',
    };
  }

  function determineClockStatus(employee, attendanceRows) {
    const employeeKeys = new Set(
      [employee.employeeId, employee.badgeNumber, employee.login]
        .map((value) => String(value ?? '').trim().toLowerCase())
        .filter(Boolean)
    );
    const employeeNames = new Set(
      [employee.name, swapNameOrder(employee.name)].map(normalizeName).filter(Boolean)
    );
    let latestPunch = null;

    for (const row of attendanceRows) {
      if (!Array.isArray(row) || row.length < 6) {
        continue;
      }

      const normalizedCells = row.map((cell) => String(cell ?? '').trim().toLowerCase());
      const rowName = normalizeName(row[1]);
      const matches =
        normalizedCells.some((cell) => employeeKeys.has(cell)) || employeeNames.has(rowName);
      if (!matches) {
        continue;
      }

      const timestamp = new Date(String(row[5] ?? '').trim()).getTime();
      if (Number.isNaN(timestamp)) {
        continue;
      }

      if (!latestPunch || timestamp > latestPunch.timestamp) {
        latestPunch = {
          timestamp,
          type: String(row[4] ?? '').trim(),
          displayTime: String(row[5] ?? '').trim(),
        };
      }
    }

    if (!latestPunch) {
      return {
        label: 'Clocked Out',
        className: 'out',
        lastPunch: 'No recent punch found',
      };
    }

  function writeClipboard(text) {
    if (navigator.clipboard?.writeText) {
      return navigator.clipboard.writeText(text);
    }

    const textarea = document.createElement('textarea');
    textarea.value = text;
    textarea.setAttribute('readonly', 'readonly');
    textarea.style.position = 'fixed';
    textarea.style.top = '-9999px';
    document.body.appendChild(textarea);
    textarea.select();
    const copied = document.execCommand('copy');
    textarea.remove();
    return copied ? Promise.resolve() : Promise.reject(new Error('Clipboard copy failed'));
  }

  function writeClipboardTable(html, text) {
    if (navigator.clipboard?.write && typeof ClipboardItem === 'function') {
      return navigator.clipboard.write([
        new ClipboardItem({
          'text/html': new Blob([html], { type: 'text/html' }),
          'text/plain': new Blob([text], { type: 'text/plain' }),
        }),
      ]);
    }

    return writeClipboard(text);
  }

  function escapeXml(value) {
    return String(value ?? '')
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;')
      .replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, '');
  }

  function escapeHtml(value) {
    return String(value ?? '')
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;');
  }

  function columnLetter(index) {
    let position = index + 1;
    let letter = '';
    while (position > 0) {
      const remainder = (position - 1) % 26;
      letter = String.fromCharCode(65 + remainder) + letter;
      position = Math.floor((position - 1) / 26);
    }
    return letter;
  }

  function concatBytes(parts) {
    const total = parts.reduce((sum, part) => sum + part.length, 0);
    const output = new Uint8Array(total);
    let offset = 0;
    for (const part of parts) {
      output.set(part, offset);
      offset += part.length;
    }
    return output;
  }

  function u16le(value) {
    const bytes = new Uint8Array(2);
    new DataView(bytes.buffer).setUint16(0, value, true);
    return bytes;
  }

  function u32le(value) {
    const bytes = new Uint8Array(4);
    new DataView(bytes.buffer).setUint32(0, value, true);
    return bytes;
  }

  function crc32(bytes) {
    let crc = 0xffffffff;
    for (let index = 0; index < bytes.length; index += 1) {
      crc = crc32Table[(crc ^ bytes[index]) & 0xff] ^ (crc >>> 8);
    }
    return (crc ^ 0xffffffff) >>> 0;
  }

  function createZip(files) {
    const encoder = new TextEncoder();
    const locals = [];
    const centrals = [];
    let offset = 0;

    for (const file of files) {
      const nameBytes = encoder.encode(file.name);
      const data = typeof file.data === 'string' ? encoder.encode(file.data) : file.data;
      const checksum = crc32(data);
      const local = concatBytes([
        u32le(0x04034b50),
        u16le(20),
        u16le(0),
        u16le(0),
        u16le(0),
        u16le(0),
        u32le(checksum),
        u32le(data.length),
        u32le(data.length),
        u16le(nameBytes.length),
        u16le(0),
        nameBytes,
        data,
      ]);
      const central = concatBytes([
        u32le(0x02014b50),
        u16le(20),
        u16le(20),
        u16le(0),
        u16le(0),
        u16le(0),
        u16le(0),
        u32le(checksum),
        u32le(data.length),
        u32le(data.length),
        u16le(nameBytes.length),
        u16le(0),
        u16le(0),
        u16le(0),
        u16le(0),
        u32le(0),
        u32le(offset),
        nameBytes,
      ]);
      locals.push(local);
      centrals.push(central);
      offset += local.length;
    }

    const centralDirectory = concatBytes(centrals);
    const endRecord = concatBytes([
      u32le(0x06054b50),
      u16le(0),
      u16le(0),
      u16le(files.length),
      u16le(files.length),
      u32le(centralDirectory.length),
      u32le(offset),
      u16le(0),
    ]);

    return concatBytes([...locals, centralDirectory, endRecord]);
  }

      function getSelectedShifts() {
        return Array.from(shiftOptions.querySelectorAll('input:checked')).map(
          (checkbox) => checkbox.value
        );
      }

      function updateShiftSummary() {
        const selectedShifts = getSelectedShifts();
        shiftFilterSummary.textContent =
          selectedShifts.length === 0
            ? 'All identified shifts'
            : `${selectedShifts.length} shift${selectedShifts.length === 1 ? '' : 's'} selected`;
      }

      function getSelectedWorkSchedules() {
        return Array.from(workDaysOptions.querySelectorAll('input:checked')).map(
          (checkbox) => checkbox.value
        );
      }

      function updateWorkDaysSummary() {
        const selectedSchedules = getSelectedWorkSchedules();
        workDaysSummary.textContent =
          selectedSchedules.length === 0
            ? 'All work schedules'
            : `${selectedSchedules.length} schedule${selectedSchedules.length === 1 ? '' : 's'} selected`;
      }

      function setFilterMode(mode) {
        filterMode = mode === 'days' ? 'days' : 'shift';
        shiftModeButton.classList.toggle('is-active', filterMode === 'shift');
        daysModeButton.classList.toggle('is-active', filterMode === 'days');
        shiftFilter.classList.toggle('tm-photo-filter-hidden', filterMode !== 'shift');
        workDaysFilter.classList.toggle('tm-photo-filter-hidden', filterMode !== 'days');
        applyFilters();
      }

      function setViewMode(mode) {
        viewMode = mode === 'photos' ? 'photos' : 'table';
        photosViewButton.classList.toggle('is-active', viewMode === 'photos');
        tableViewButton.classList.toggle('is-active', viewMode === 'table');
        grid.classList.toggle('tm-photo-filter-hidden', viewMode !== 'photos');
        tableWrap.classList.toggle('tm-photo-filter-hidden', viewMode !== 'table');
        sizeControl.classList.toggle('tm-photo-filter-hidden', viewMode !== 'photos');
        grid.style.display = viewMode === 'photos' ? '' : 'none';
        tableWrap.style.display = viewMode === 'table' ? '' : 'none';
        modal.classList.toggle('is-table', viewMode === 'table');
      }

      function matchesFilters(dataset) {
        const selectedShifts = new Set(getSelectedShifts());
        const selectedSchedules = new Set(getSelectedWorkSchedules());
        const clockMatches =
          !clockFilter.value || dataset.clockStatus === clockFilter.value;
        const scheduleMatches =
          filterMode === 'shift'
            ? selectedShifts.size === 0 || selectedShifts.has(dataset.shift)
            : selectedSchedules.size === 0 ||
              selectedSchedules.has(dataset.workSchedule);
        return clockMatches && scheduleMatches;
      }

      function applyFilters() {
        for (const card of grid.querySelectorAll('.tm-photo-card')) {
          card.style.display = matchesFilters(card.dataset) ? '' : 'none';
        }
        for (const row of tableBody.querySelectorAll('tr')) {
          row.style.display = matchesFilters(row.dataset) ? '' : 'none';
        }
      }

      function createResultTableRow(result, rowNumber) {
        const record = getResultTableRecord(result);
        const row = document.createElement('tr');
        row.dataset.login = record.login;
        row.dataset.clockStatus = record.clockStatusClass;
        row.dataset.shift = record.shiftCode;
        row.dataset.workSchedule = record.workScheduleKey;
        if (!result.ok) {
          row.classList.add('is-error');
        }
        const rowNumberCell = document.createElement('th');
        rowNumberCell.className = 'tm-excel-row-num';
        rowNumberCell.textContent = String(rowNumber);
        row.appendChild(rowNumberCell);
        for (const [key] of resultTableColumns) {
          const cell = document.createElement('td');
          cell.textContent = record[key];
          if (key === 'clockStatus') {
            cell.classList.add(`tm-excel-status-${record.clockStatusClass}`);
          }
          row.appendChild(cell);
        }
        return row;
      }

      function syncResultsTable() {
        tableBody.replaceChildren();
        (lastDebugPayload.results || []).forEach((result, index) => {
          tableBody.appendChild(createResultTableRow(result, index + 2));
        });
        applyFilters();
      }

      function getVisibleTableRecords() {
        return (lastDebugPayload.results || [])
          .map(getResultTableRecord)
          .filter((record) =>
            matchesFilters({
              clockStatus: record.clockStatusClass,
              shift: record.shiftCode,
              workSchedule: record.workScheduleKey,
            })
          );
      }

      function refreshShiftFilter() {
        const selectedShifts = new Set(getSelectedShifts());
        const shifts = Array.from(grid.querySelectorAll('.tm-photo-card'))
          .map((card) => card.dataset.shift)
          .filter((shift) => shift && shift !== 'Shift unavailable');
        const uniqueShifts = Array.from(new Set(shifts)).sort();
        shiftOptions.replaceChildren();
        for (const shift of uniqueShifts) {
          const label = document.createElement('label');
          label.className = 'tm-photo-shift-option';
          const checkbox = document.createElement('input');
          checkbox.type = 'checkbox';
          checkbox.value = shift;
          checkbox.checked = selectedShifts.has(shift);
          const text = document.createElement('span');
          text.textContent = shift;
          label.append(checkbox, text);
          shiftOptions.appendChild(label);
        }
        if (uniqueShifts.length === 0) {
          const empty = document.createElement('div');
          empty.className = 'tm-photo-shift-option';
          empty.textContent = 'No shifts identified yet';
          shiftOptions.appendChild(empty);
        }
        updateShiftSummary();
      }

      function refreshWorkDaysFilter() {
        const selectedSchedules = new Set(getSelectedWorkSchedules());
        const schedulesByKey = new Map();
        for (const result of lastDebugPayload.results || []) {
          if (!result.ok) {
            continue;
          }
          const schedule = result.workSchedule || decodeShiftCode(result.shift);
          if (schedule.known) {
            schedulesByKey.set(schedule.key, schedule.label);
          }
        }

        workDaysOptions.replaceChildren();
        for (const [key, labelText] of Array.from(schedulesByKey.entries()).sort((a, b) =>
          a[1].localeCompare(b[1])
        )) {
          const label = document.createElement('label');
          label.className = 'tm-photo-shift-option';
          const checkbox = document.createElement('input');
          checkbox.type = 'checkbox';
          checkbox.value = key;
          checkbox.checked = selectedSchedules.has(key);
          const text = document.createElement('span');
          text.textContent = labelText;
          label.append(checkbox, text);
          workDaysOptions.appendChild(label);
        }
        if (schedulesByKey.size === 0) {
          const empty = document.createElement('div');
          empty.className = 'tm-photo-shift-option';
          empty.textContent = 'No work schedules identified yet';
          workDaysOptions.appendChild(empty);
        }
        updateWorkDaysSummary();
      }

      function applyPhotoSize(value) {
        const size = Math.min(300, Math.max(120, Number(value) || 180));
        sizeSlider.value = String(size);
        sizeLabel.textContent = `Photo size: ${size}px`;
        grid.style.setProperty('--tm-photo-card-min', `${size}px`);
      }

      function updateProgress(
        completed,
        total,
        startedAt,
        isComplete = false,
        refreshEstimate = false
      ) {
        const percent = total === 0 ? 0 : Math.round((completed / total) * 100);
        progressBar.style.width = `${percent}%`;
        const elapsed = Date.now() - startedAt;

        if (isComplete) {
          progressText.textContent = `${completed}/${total} jobs complete in ${formatCountdown(elapsed)}.`;
          return;
        }
        if (completed === 0) {
          progressText.textContent = `0/${total} jobs complete · Calculating ETA...`;
          return;
        }

        if (refreshEstimate || expectedCompletionAt === null) {
          expectedCompletionAt = Date.now() + (elapsed / completed) * (total - completed);
        }
        const remainingMs = Math.max(0, expectedCompletionAt - Date.now());
        const expectedAt = new Date(expectedCompletionAt);
        progressText.textContent =
          `${completed}/${total} jobs complete · ETA ${expectedAt.toLocaleTimeString()} ` +
          `(${formatCountdown(remainingMs)} remaining)`;
      }

      function saveCurrentState() {
        storePanelState({
          inputValue: input.value,
          clockFilter: clockFilter.value,
          filterMode,
          viewMode,
          selectedShifts: getSelectedShifts(),
          selectedWorkSchedules: getSelectedWorkSchedules(),
          photoSize: sizeSlider.value,
          statusText: status.textContent,
          progressText: progressText.textContent,
          progressWidth: progressBar.style.width,
          debug: lastDebugPayload,
        });
      }

      function restoreSavedState() {
        const savedState = loadPanelState();
        if (!savedState) {
          return;
        }

        input.value = savedState.inputValue || '';
        if (savedState.debug) {
          lastDebugPayload = savedState.debug;
        }
        for (const result of savedState.debug?.results || []) {
          grid.appendChild(createStoredPhotoCard(result));
        }
        syncResultsTable();
        refreshShiftFilter();
        refreshWorkDaysFilter();
        clockFilter.value = savedState.clockFilter || '';
        const savedShifts = new Set(
          Array.isArray(savedState.selectedShifts)
            ? savedState.selectedShifts
            : [savedState.shiftFilter].filter(Boolean)
        );
        for (const checkbox of shiftOptions.querySelectorAll('input')) {
          checkbox.checked = savedShifts.has(checkbox.value);
        }
        updateShiftSummary();
        const savedSchedules = new Set(savedState.selectedWorkSchedules || []);
        for (const checkbox of workDaysOptions.querySelectorAll('input')) {
          checkbox.checked = savedSchedules.has(checkbox.value);
        }
        updateWorkDaysSummary();
        setFilterMode(savedState.filterMode || 'shift');
        applyPhotoSize(savedState.photoSize || 180);
        applyFilters();
        status.textContent = savedState.statusText || 'Saved results restored.';
        progressText.textContent =
          savedState.progressText ||
          `${savedState.debug?.results?.length || 0} saved jobs restored.`;
        progressBar.style.width =
          savedState.progressWidth || (savedState.debug?.results?.length ? '100%' : '0%');
      }

      function renderEmployees() {
        const logins = normalizeLoginInput(input.value);
        grid.replaceChildren();
        tableBody.replaceChildren();
        activeRunId += 1;
        const runId = activeRunId;
        window.clearInterval(progressTimer);
        progressTimer = null;
        shiftOptions.replaceChildren();
        workDaysOptions.replaceChildren();
        clockFilter.value = '';
        updateShiftSummary();
        updateWorkDaysSummary();
        setFilterMode('shift');
        expectedCompletionAt = null;

        if (logins.length === 0) {
          status.textContent = 'No logins entered.';
          progressBar.style.width = '0%';
          progressText.textContent = '0 jobs queued.';
          return;
        }

        const startedAt = Date.now();
        let completed = 0;
        renderButton.disabled = true;
        lastDebugPayload = {
          version: '0.9.3',
          generatedAt: new Date().toISOString(),
          warehouseId: defaultWarehouseId,
          logins: [...logins],
          attendanceRows: 0,
          startedAt: new Date(startedAt).toISOString(),
          completedAt: null,
          durationMs: null,
          results: [],
          errors: [],
        };
        updateProgress(0, logins.length, startedAt);

        const attendancePromise = fetchAttendanceRows().then((rows) => {
          if (runId === activeRunId) {
            lastDebugPayload.attendanceRows = rows.length;
          }
          return rows;
        });
        const completions = [];
        for (const login of logins) {
          const job = createPhotoCard(login, attendancePromise);
          grid.appendChild(job.card);
          completions.push(
            job.completion.then((result) => {
              if (runId !== activeRunId) {
                return result;
              }
              completed += 1;
              lastDebugPayload.results.push(result);
              if (!result.ok) {
                lastDebugPayload.errors.push({ login, error: result.error });
              }
              refreshShiftFilter();
              refreshWorkDaysFilter();
              syncResultsTable();
              applyFilters();
              updateProgress(completed, logins.length, startedAt, false, true);
              saveCurrentState();
              return result;
            })
          );
        }

        progressTimer = window.setInterval(() => {
          if (runId === activeRunId) {
            updateProgress(completed, logins.length, startedAt);
          }
        }, 1000);

        Promise.all(completions).then(() => {
          if (runId !== activeRunId) {
            return;
          }
          window.clearInterval(progressTimer);
          progressTimer = null;
          renderButton.disabled = false;
          const completedAt = new Date();
          lastDebugPayload.generatedAt = completedAt.toISOString();
          lastDebugPayload.completedAt = completedAt.toISOString();
          lastDebugPayload.durationMs = Date.now() - startedAt;
          updateProgress(completed, logins.length, startedAt, true);
          const failures = lastDebugPayload.errors.length;
          status.textContent = failures
            ? `Completed with ${failures} failed lookup${failures === 1 ? '' : 's'}.`
            : `Completed ${logins.length} employee lookup${logins.length === 1 ? '' : 's'}.`;
          saveCurrentState();
        });

        status.textContent = `Looking up ${logins.length} employee${logins.length === 1 ? '' : 's'} in FCLM.`;
        saveCurrentState();
      }

      setFilterMode('shift');
      restoreSavedState();
      setViewMode('table');
      renderButton.addEventListener('click', renderEmployees);
      clearButton.addEventListener('click', () => {
        activeRunId += 1;
        window.clearInterval(progressTimer);
        progressTimer = null;
        renderButton.disabled = false;
        input.value = '';
        grid.replaceChildren();
        tableBody.replaceChildren();
        clockFilter.value = '';
        shiftOptions.replaceChildren();
        workDaysOptions.replaceChildren();
        updateShiftSummary();
        updateWorkDaysSummary();
        setFilterMode('shift');
        setViewMode('table');
        applyPhotoSize(180);
        progressBar.style.width = '0%';
        progressText.textContent = '0 jobs queued.';
        status.textContent = 'Enter one or more logins.';
        lastDebugPayload = {
          version: '0.9.3',
          generatedAt: null,
          warehouseId: defaultWarehouseId,
          logins: [],
          attendanceRows: 0,
          startedAt: null,
          completedAt: null,
          durationMs: null,
          results: [],
          errors: [],
        };
        clearPanelState();
        input.focus();
      });
      copyDebugButton.addEventListener('click', async () => {
        lastDebugPayload.generatedAt = new Date().toISOString();
        try {
          await writeClipboard(JSON.stringify(lastDebugPayload, null, 2));
          status.textContent = 'Debug copied to clipboard.';
        } catch (error) {
          status.textContent = `Debug copy failed: ${formatError(error)}`;
        }
      });
      copyLoginsButton.addEventListener('click', async () => {
        const visibleLogins = Array.from(grid.querySelectorAll('.tm-photo-card'))
          .filter((card) => card.style.display !== 'none')
          .map((card) => card.dataset.login)
          .filter(Boolean);

        if (visibleLogins.length === 0) {
          status.textContent = 'No visible logins to copy.';
          return;
        }

        try {
          await writeClipboard(visibleLogins.join('\n'));
          status.textContent = `Copied ${visibleLogins.length} visible login${visibleLogins.length === 1 ? '' : 's'}.`;
        } catch (error) {
          status.textContent = `Login copy failed: ${formatError(error)}`;
        }
      });
      copyTableButton.addEventListener('click', async () => {
        const records = getVisibleTableRecords();
        if (records.length === 0) {
          status.textContent = 'No visible table rows to copy.';
          return;
        }

        try {
          const payload = buildTableCopyPayload(records);
          await writeClipboardTable(payload.html, payload.text);
          status.textContent = `Copied ${records.length} table row${records.length === 1 ? '' : 's'}.`;
        } catch (error) {
          status.textContent = `Table copy failed: ${formatError(error)}`;
        }
      });
      exportExcelButton.addEventListener('click', () => {
        const records = getVisibleTableRecords();
        if (records.length === 0) {
          status.textContent = 'No visible table rows to export.';
          return;
        }

        const filename = `fclm-photo-lookup-${defaultWarehouseId}-${formatDateOnly(new Date())}.xlsx`;
        downloadBlob(buildExcelWorkbook(records), filename);
        status.textContent = `Exported ${records.length} row${records.length === 1 ? '' : 's'} to ${filename}.`;
      });

        applyFilters();
        saveCurrentState();
      });
      shiftOptions.addEventListener('change', () => {
        updateShiftSummary();
        applyFilters();
        saveCurrentState();
      });
      workDaysOptions.addEventListener('change', () => {
        updateWorkDaysSummary();
        applyFilters();
        saveCurrentState();
      });
      shiftModeButton.addEventListener('click', () => {
        setFilterMode('shift');
        saveCurrentState();
      });
      daysModeButton.addEventListener('click', () => {
        setFilterMode('days');
        saveCurrentState();
      });
      photosViewButton.addEventListener('click', () => {
        setViewMode('photos');
        saveCurrentState();
      });
      tableViewButton.addEventListener('click', () => {
        setViewMode('table');
        saveCurrentState();
      });
      sizeSlider.addEventListener('input', () => {
        applyPhotoSize(sizeSlider.value);
        saveCurrentState();
      });
      input.addEventListener('input', saveCurrentState);

      input.addEventListener('keydown', (event) => {
        if ((event.ctrlKey || event.metaKey) && event.key === 'Enter') {
          renderEmployees();
        }
      });

      overlay.addEventListener('click', (event) => {
        if (event.target === overlay) {
          closeOverlay();
        }
      });

      content.appendChild(help);
      content.appendChild(versionNote);
      content.appendChild(input);
      content.appendChild(actions);
      content.appendChild(status);
      content.appendChild(progress);
      content.appendChild(progressText);
      content.appendChild(filters);
      content.appendChild(grid);
      content.appendChild(tableWrap);

      modal.appendChild(header);
      modal.appendChild(content);
      overlay.appendChild(modal);
      document.body.appendChild(overlay);
      input.focus();
    }

    launcher.addEventListener('click', openOverlay);

    document.addEventListener('keydown', (event) => {
      if (event.key === 'Escape' && overlay?.isConnected) {
        closeOverlay();
      }
    });

    document.body.appendChild(launcher);

    const removeLegacyUi = () => {
      document.querySelectorAll('.tm-photo-launcher').forEach((element) => {
        if (element !== launcher) {
          element.remove();
        }
      });
      document.querySelectorAll('.tm-photo-modal').forEach((element) => {
        if (element.id !== 'tm-photo-modal-v060') {
          element.closest('.tm-photo-overlay')?.remove();
        }
      });
    };
    removeLegacyUi();
    new MutationObserver(removeLegacyUi).observe(document.body, {
      childList: true,
      subtree: true,
    });
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', install, { once: true });
  } else {
    install();
  }
})();
