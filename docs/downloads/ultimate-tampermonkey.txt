	async function copyTextToClipboard(text) {
		if (typeof GM_setClipboard === "function") {
			GM_setClipboard(text, "text");
			return;
		}
		if (navigator.clipboard?.writeText) {
			await navigator.clipboard.writeText(text);
			return;
		}
		const textarea = document.createElement("textarea");
		textarea.value = text;
		textarea.setAttribute("readonly", "true");
		textarea.style.position = "fixed";
		textarea.style.opacity = "0";
		textarea.style.pointerEvents = "none";
		document.body.appendChild(textarea);
		textarea.select();
		const copied = document.execCommand("copy");
		textarea.remove();
		if (!copied) throw new Error("Clipboard copy failed.");
	}

	function getEmployeeLinks(table) {

	function parseDurationToSeconds(durationText) {
		const normalized = cleanText(durationText);
		if (!normalized) return 0;
		const parts = normalized.split(":").map((part) => Number(part));
		if (parts.some((part) => Number.isNaN(part))) return 0;
		if (parts.length === 2) return parts[0] * 60 + parts[1];
		if (parts.length === 3) return parts[0] * 3600 + parts[1] * 60 + parts[2];
		return 0;
	}
	function formatSeconds(totalSeconds) {
		const roundedSeconds = Math.max(0, Math.round(totalSeconds || 0));
		const hours = Math.floor(roundedSeconds / 3600);
		const minutes = Math.floor(roundedSeconds % 3600 / 60);
		const seconds = roundedSeconds % 60;
		if (hours > 0) return `${String(hours).padStart(2, "0")}:${String(minutes).padStart(2, "0")}:${String(seconds).padStart(2, "0")}`;
		return `${String(minutes).padStart(2, "0")}:${String(seconds).padStart(2, "0")}`;
	}
	function formatPercentage(numerator, denominator) {
		if (!denominator) return "0.0%";
		return `${(numerator / denominator * 100).toFixed(1)}%`;
	}
	async function fetchEmployeeDocument(employeeLink) {
		const existingRequest = employeeDocumentCache.get(employeeLink.href);
		if (existingRequest) return existingRequest;
		const request = (async () => {
			const controller = new AbortController();
			const timeoutId = window.setTimeout(() => controller.abort(), requestTimeoutMs);
			try {
				const response = await fetch(employeeLink.href, {
					credentials: "include",
					signal: controller.signal
				});
				if (!response.ok) throw new Error(`Failed to load employee details: ${response.status}`);
				const html = await response.text();
				return new DOMParser().parseFromString(html, "text/html");
			} finally {
				window.clearTimeout(timeoutId);
			}
		})();
		employeeDocumentCache.set(employeeLink.href, request);
		return request;
	}

	function extractIdleMetricsFromDocument(doc) {
		let idleSeconds = 0;
		const functionSegments = Array.from(doc.querySelectorAll("tr.function-seg"));
		for (const row of functionSegments) {
			const className = row.className || "";
			const text = cleanText(row.textContent);
			const isIndirect = row.classList.contains("indirect") || /\bindirect\b/i.test(className) || /\bindirect\b/i.test(text);
			const isTimeOffTask = row.classList.contains("timeOffTask") || /\btimeofftask\b/i.test(className) || /\btime off task\b/i.test(text);
			if (!isIndirect || isTimeOffTask) continue;
			const durationCell = row.querySelector("td.rightAlign") ?? row.querySelector("td:last-child");
			idleSeconds += parseDurationToSeconds(cleanText(durationCell?.textContent));
		}
		let clockedInSeconds = 0;
		const clockRows = Array.from(doc.querySelectorAll("tr.clock-seg"));
		for (const row of clockRows) {
			const className = row.className || "";
			const text = cleanText(row.textContent);
			if (!(row.classList.contains("on-clock") || /\bon-clock\b/i.test(className) || /\bonclock\b/i.test(text) || /\bonclock\/paid\b/i.test(text) || /\bonclock\/unpaid\b/i.test(text))) continue;
			const durationCell = row.querySelector("td.rightAlign") ?? row.querySelector("td:last-child");
			clockedInSeconds += parseDurationToSeconds(cleanText(durationCell?.textContent));
		}
		return {
			idleSeconds,
			clockedInSeconds,
			idleTime: formatSeconds(idleSeconds),
			clockedInTime: formatSeconds(clockedInSeconds),
			idlePct: formatPercentage(idleSeconds, clockedInSeconds)
		};
	}

	function escapeIdentifier(value) {
		if (typeof CSS !== "undefined" && typeof CSS.escape === "function") return CSS.escape(value);
		return value.replace(/[^a-zA-Z0-9_-]/g, (char) => `\\${char}`);
	}
	function isUniqueSelector(selector) {
		try {
			return document.querySelectorAll(selector).length === 1;
		} catch {
			return false;
		}
	}
	function buildSegment(element) {
		if (element.id) return `#${escapeIdentifier(element.id)}`;
		const tagName = element.tagName.toLowerCase();
		const baseSegment = `${tagName}${Array.from(element.classList).filter(Boolean).slice(0, 2).map((className) => `.${escapeIdentifier(className)}`).join("")}`;
		const parent = element.parentElement;
		if (!parent) return baseSegment || tagName;
		const siblingsOfSameType = Array.from(parent.children).filter((sibling) => sibling.tagName === element.tagName);
		if (siblingsOfSameType.length <= 1) return baseSegment || tagName;
		const position = siblingsOfSameType.indexOf(element) + 1;
		return `${baseSegment || tagName}:nth-of-type(${position})`;
	}
	function buildAbsoluteCssSegment(element) {
		const tagName = element.tagName.toLowerCase();
		if (element.id) return `${tagName}#${escapeIdentifier(element.id)}`;
		const parent = element.parentElement;
		if (!parent) return tagName;
		const siblingsOfSameType = Array.from(parent.children).filter((sibling) => sibling.tagName === element.tagName);
		if (siblingsOfSameType.length <= 1) return tagName;
		return `${tagName}:nth-of-type(${siblingsOfSameType.indexOf(element) + 1})`;
	}
	function escapeXPathLiteral(value) {
		if (!value.includes("\"")) return `"${value}"`;
		if (!value.includes("'")) return `'${value}'`;
		return `concat(${value.split("\"").map((part) => `"${part}"`).join(", '\"', ")})`;
	}
	function getCssSelector(element) {
		const segments = [];
		let current = element;
		while (current && current !== document.documentElement) {
			const segment = buildSegment(current);
			segments.unshift(segment);
			const candidateSelector = segments.join(" > ");
			if (isUniqueSelector(candidateSelector)) return candidateSelector;
			current = current.parentElement;
		}
		segments.unshift("html");
		return segments.join(" > ");
	}
	function getCssPath(element) {
		const segments = [];
		let current = element;
		while (current) {
			segments.unshift(buildAbsoluteCssSegment(current));
			current = current.parentElement;
		}
		return segments.join(" > ");
	}
	function getXPath(element) {
		if (element.id) return `//*[@id=${escapeXPathLiteral(element.id)}]`;
		const segments = [];
		let current = element;
		while (current && current.nodeType === Node.ELEMENT_NODE) {
			const tagName = current.tagName.toLowerCase();
			const currentTagName = current.tagName;
			const parent = current.parentElement;
			if (!parent) {
				segments.unshift(`/${tagName}[1]`);
				break;
			}
			const index = Array.from(parent.children).filter((sibling) => sibling.tagName === currentTagName).indexOf(current) + 1;
			segments.unshift(`/${tagName}[${index}]`);
			current = parent;
		}
		return segments.join("");
	}

	function createDomHelpers(logger) {
		return {
			query(selector, root = document) {
				return root.querySelector(selector);
			},
			queryAll(selector, root = document) {
				return Array.from(root.querySelectorAll(selector));
			},
			waitFor(selector, options) {
				const root = options?.root ?? document;
				const timeoutMs = options?.timeoutMs ?? 1e4;
				const existingElement = root.querySelector(selector);
				if (existingElement) return Promise.resolve(existingElement);
				return new Promise((resolve, reject) => {
					const timeoutId = window.setTimeout(() => {
						observer.disconnect();
						reject(/* @__PURE__ */ new Error(`Timed out waiting for selector: ${selector}`));
					}, timeoutMs);
					const observer = new MutationObserver(() => {
						const element = root.querySelector(selector);
						if (!element) return;
						window.clearTimeout(timeoutId);
						observer.disconnect();
						resolve(element);
					});
					const observationTarget = root instanceof Document ? root.documentElement : root;
					if (!observationTarget) {
						window.clearTimeout(timeoutId);
						reject(/* @__PURE__ */ new Error(`Cannot observe selector target for: ${selector}`));
						return;
					}
					observer.observe(observationTarget, {
						childList: true,
						subtree: true
					});
					logger.debug(`Waiting for selector "${selector}"`, { timeoutMs });
				});
			}
		};
	}

	function createStorage(namespace) {
		function getKey(key) {
			return `${namespace}:${key}`;
		}
		return {
			async get(key, fallbackValue) {
				const storageKey = getKey(key);
				if (typeof GM_getValue === "function") return await Promise.resolve(GM_getValue(storageKey, fallbackValue)) ?? fallbackValue;
				const rawValue = window.localStorage.getItem(storageKey);
				if (rawValue === null) return fallbackValue;
				return JSON.parse(rawValue);
			},
			async set(key, value) {
				const storageKey = getKey(key);
				if (typeof GM_setValue === "function") {
					await Promise.resolve(GM_setValue(storageKey, value));
					return;
				}
				window.localStorage.setItem(storageKey, JSON.stringify(value));
			},
			async remove(key) {
				const storageKey = getKey(key);
				if (typeof GM_deleteValue === "function") {
					await Promise.resolve(GM_deleteValue(storageKey));
					return;
				}
				window.localStorage.removeItem(storageKey);
			}
		};
	}
