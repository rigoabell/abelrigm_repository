(function () {
  'use strict';

  const SCRIPT_VERSION = '1.1.0';
  const SETTINGS_KEY = 'abelrigm.c2d.v2';
  const PREVIOUS_SETTINGS_KEY = 'abelrigm.c2d.v1';
  const ABSENT_MS = 1200;
  const HISTORY_LIMIT = 25;
  const MAX_EDGE = 1400;
  const CENTER_REGION = { name: 'center', x: 0.14, y: 0.16, w: 0.72, h: 0.68, maxEdge: 1600 };
  const CAPTURE_REGIONS = [
    CENTER_REGION,
    { name: 'tight', x: 0.22, y: 0.3, w: 0.56, h: 0.4, maxEdge: 1600 },
    { name: 'full', x: 0, y: 0, w: 1, h: 1, maxEdge: 1400 },
  ];
  const NATIVE_2D = ['qr_code', 'data_matrix', 'pdf417', 'aztec'];
  const NATIVE_1D = ['code_128', 'code_39', 'code_93', 'codabar', 'ean_13', 'ean_8', 'itf', 'upc_a', 'upc_e'];
  const ZXING_2D = ['QR_CODE', 'DATA_MATRIX', 'PDF_417', 'AZTEC'];
  const ZXING_1D = ['CODE_128', 'CODE_39', 'CODE_93', 'CODABAR', 'EAN_13', 'EAN_8', 'ITF', 'UPC_A', 'UPC_E'];
  const ONE_D = new Set(NATIVE_1D);
  const FORMAT_LABELS = {
    qr_code: 'QR Code',
    data_matrix: 'Data Matrix',
    pdf417: 'PDF417',
    aztec: 'Aztec',
    code_128: 'Code 128',
    code_39: 'Code 39',
    code_93: 'Code 93',
    codabar: 'Codabar',
    ean_13: 'EAN-13',
    ean_8: 'EAN-8',
    itf: 'ITF',
    upc_a: 'UPC-A',
    upc_e: 'UPC-E',
  };
  const FORMAT_ALIASES = {
    QR_CODE: 'qr_code',
    DATA_MATRIX: 'data_matrix',
    PDF_417: 'pdf417',
    AZTEC: 'aztec',
    CODE_128: 'code_128',
    CODE_39: 'code_39',
    CODE_93: 'code_93',
    CODABAR: 'codabar',
    EAN_13: 'ean_13',
    EAN_8: 'ean_8',
    ITF: 'itf',
    UPC_A: 'upc_a',
    UPC_E: 'upc_e',
  };

  function canonicalFormat(format) {
    const raw = String(format == null ? '' : format);
    if (FORMAT_LABELS[raw]) return raw;
    if (FORMAT_ALIASES[raw]) return FORMAT_ALIASES[raw];
    const lower = raw.toLowerCase();
    if (FORMAT_LABELS[lower]) return lower;
    return lower || 'unknown';
  }

  function labelFor(format) {
    const id = canonicalFormat(format);
    return FORMAT_LABELS[id] || id;
  }

  function nativeFormats(include1d) {
    return include1d ? NATIVE_2D.concat(NATIVE_1D) : NATIVE_2D.slice();
  }

  function zxingFormats(lib, include1d) {
    const names = include1d ? ZXING_2D.concat(ZXING_1D) : ZXING_2D.slice();
    const table = lib && lib.BarcodeFormat || {};
    return names.map(function (name) { return table[name]; }).filter(function (value) { return value != null; });
  }

  function allowsFormat(format, include1d) {
    const id = canonicalFormat(format);
    if (include1d) return id !== 'unknown';
    return !ONE_D.has(id) && id !== 'unknown';
  }

  function displayText(text) {
    return String(text == null ? '' : text).replace(/[\u0000-\u001f]/g, function (ch) {
      if (ch === '\n') return '⏎';
      if (ch === '\r') return '';
      if (ch === '\t') return '⇥';
      if (ch === '\u001d') return '␝';
      return '·';
    });
  }

  function boolSetting(value, fallback) {
    return typeof value === 'boolean' ? value : fallback;
  }

  function sanitizeHistory(value) {
    if (!Array.isArray(value)) return [];
    const rows = [];
    for (let i = 0; i < value.length && rows.length < HISTORY_LIMIT; i++) {
      const row = value[i];
      if (!row || typeof row.text !== 'string' || !row.text) continue;
      rows.push({ text: row.text, format: canonicalFormat(row.format), at: Number(row.at) || 0 });
    }
    return rows;
  }

  function readStored(storage, key) {
    try {
      const raw = storage && storage.getItem(key);
      if (!raw) return null;
      const parsed = JSON.parse(raw);
      if (!parsed || typeof parsed !== 'object' || Array.isArray(parsed)) return null;
      return parsed;
    } catch (error) {
      return null;
    }
  }

  function loadSettings(storage) {
    const current = readStored(storage, SETTINGS_KEY);
    const previous = current ? null : readStored(storage, PREVIOUS_SETTINGS_KEY);
    const parsed = current || previous || {};
    return {
      beep: boolSetting(parsed.beep, true),
      typeIntoField: boolSetting(parsed.typeIntoField, true),
      pressEnter: boolSetting(parsed.pressEnter, true),
      include1d: previous ? true : boolSetting(parsed.include1d, true),
      deviceId: typeof parsed.deviceId === 'string' ? parsed.deviceId : '',
      left: Number.isFinite(parsed.left) ? parsed.left : null,
      top: Number.isFinite(parsed.top) ? parsed.top : null,
      history: sanitizeHistory(parsed.history),
    };
  }

  function saveSettings(storage, settings) {
    try {
      storage.setItem(SETTINGS_KEY, JSON.stringify({
        beep: !!settings.beep,
        typeIntoField: !!settings.typeIntoField,
        pressEnter: !!settings.pressEnter,
        include1d: !!settings.include1d,
        deviceId: settings.deviceId || '',
        left: Number.isFinite(settings.left) ? settings.left : null,
        top: Number.isFinite(settings.top) ? settings.top : null,
        history: sanitizeHistory(settings.history),
      }));
    } catch (error) {}
  }

  function nextHistory(history, scan, limit) {
    const cap = limit || HISTORY_LIMIT;
    const row = { text: scan.text, format: canonicalFormat(scan.format), at: scan.at };
    const rest = (history || []).filter(function (item) {
      return !(item && item.text === row.text && canonicalFormat(item.format) === row.format);
    });
    return [row].concat(rest).slice(0, cap);
  }

  function reduceScan(state, event, now, absentMs) {
    const current = state && state.hold !== undefined ? state : { hold: null, clearAt: 0 };
    const wait = absentMs == null ? ABSENT_MS : absentMs;
    if (!event || event.type === 'miss') {
      if (!current.hold) return { state: current, emit: null };
      if (!current.clearAt) return { state: { hold: current.hold, clearAt: now }, emit: null };
      if (now - current.clearAt >= wait) return { state: { hold: null, clearAt: 0 }, emit: null };
      return { state: current, emit: null };
    }
    const text = event.text;
    const format = canonicalFormat(event.format);
    const same = current.hold && current.hold.text === text && current.hold.format === format;
    if (same) return { state: { hold: current.hold, clearAt: 0 }, emit: null };
    const hold = { text: text, format: format, at: now };
    return { state: { hold: hold, clearAt: 0 }, emit: { text: text, format: format, at: now } };
  }

  function cameraUnavailableReason(env) {
    const secure = env ? env.isSecureContext : (typeof window !== 'undefined' ? window.isSecureContext : true);
    const media = env ? env.mediaDevices : (typeof navigator !== 'undefined' ? navigator.mediaDevices : null);
    if (secure === false) return 'The camera needs a secure page. Open this site with https://.';
    if (!media || typeof media.getUserMedia !== 'function') return 'This browser cannot open a camera.';
    return '';
  }

  function cameraErrorMessage(error) {
    const name = error && error.name;
    if (name === 'NotAllowedError' || name === 'PermissionDeniedError') {
      return 'Camera blocked. Allow the camera for this site in the address bar, then press Start camera.';
    }
    if (name === 'NotFoundError' || name === 'DevicesNotFoundError') return 'No camera was found on this computer.';
    if (name === 'NotReadableError' || name === 'TrackStartError') return 'The camera is busy in another app. Close that app and press Start camera.';
    if (name === 'OverconstrainedError' || name === 'ConstraintNotSatisfiedError') return 'That camera is not available. Choose another one.';
    if (name === 'SecurityError') return 'This page blocked the camera. Open the site with https://.';
    return 'The camera did not start.';
  }

  function canWrite(el) {
    if (!el || (el.nodeType != null && el.nodeType !== 1)) return false;
    if (el.isContentEditable) return true;
    const tag = el.tagName;
    if (tag !== 'INPUT' && tag !== 'TEXTAREA') return false;
    if (el.disabled || el.readOnly) return false;
    if (tag === 'TEXTAREA') return true;
    const type = String(el.type || 'text').toLowerCase();
    return type === 'text' || type === 'search' || type === 'tel' || type === 'url' || type === 'email' || type === 'number';
  }

  function fieldStillThere(el) {
    if (!canWrite(el)) return false;
    if (el.isConnected === false) return false;
    return true;
  }

  function setNativeValue(el, text) {
    const view = (el.ownerDocument && el.ownerDocument.defaultView) || globalThis;
    const Ctor = el.tagName === 'TEXTAREA' ? view.HTMLTextAreaElement : view.HTMLInputElement;
    const desc = Ctor && Object.getOwnPropertyDescriptor(Ctor.prototype, 'value');
    if (desc && typeof desc.set === 'function') {
      try {
        desc.set.call(el, text);
        return;
      } catch (error) {}
    }
    el.value = text;
  }

  function pressEnter(el) {
    if (typeof KeyboardEvent !== 'function') return;
    ['keydown', 'keypress', 'keyup'].forEach(function (type) {
      let event;
      try {
        event = new KeyboardEvent(type, { key: 'Enter', code: 'Enter', bubbles: true, cancelable: true });
      } catch (error) {
        return;
      }
      try {
        Object.defineProperty(event, 'keyCode', { get: function () { return 13; } });
        Object.defineProperty(event, 'which', { get: function () { return 13; } });
      } catch (error) {}
      el.dispatchEvent(event);
    });
  }

  function writeScan(el, text, pressEnterAfter) {
    if (!canWrite(el)) return false;
    if (typeof el.focus === 'function') el.focus();
    if (el.isContentEditable) {
      const doc = el.ownerDocument || document;
      const selection = doc.defaultView && doc.defaultView.getSelection ? doc.defaultView.getSelection() : null;
      if (selection && typeof doc.createRange === 'function') {
        const range = doc.createRange();
        range.selectNodeContents(el);
        selection.removeAllRanges();
        selection.addRange(range);
      }
      let inserted = false;
      try {
        inserted = !!(doc.execCommand && doc.execCommand('insertText', false, text));
      } catch (error) {
        inserted = false;
      }
      if (!inserted) el.textContent = text;
      if (typeof el.dispatchEvent === 'function') el.dispatchEvent(new Event('input', { bubbles: true }));
    } else {
      setNativeValue(el, text);
      if (typeof el.dispatchEvent === 'function') {
        el.dispatchEvent(new Event('input', { bubbles: true }));
        el.dispatchEvent(new Event('change', { bubbles: true }));
      }
    }
    if (pressEnterAfter) pressEnter(el);
    return true;
  }

  function isMiss(error, lib) {
    if (!error) return false;
    if (lib) {
      if (lib.NotFoundException && error instanceof lib.NotFoundException) return true;
      if (lib.ChecksumException && error instanceof lib.ChecksumException) return true;
      if (lib.FormatException && error instanceof lib.FormatException) return true;
    }
    const name = error.name || '';
    return name === 'NotFoundException' || name === 'ChecksumException' || name === 'FormatException';
  }

  function resultFromZxing(lib, result) {
    if (!result || typeof result.getText !== 'function') return null;
    const text = result.getText();
    if (!text) return null;
    const formatValue = typeof result.getBarcodeFormat === 'function' ? result.getBarcodeFormat() : '';
    const formatName = lib && lib.BarcodeFormat ? (lib.BarcodeFormat[formatValue] || formatValue) : formatValue;
    return { text: text, format: canonicalFormat(formatName) };
  }

  function resultFromNative(code) {
    if (!code || !code.rawValue) return null;
    return { text: code.rawValue, format: canonicalFormat(code.format) };
  }

  function makeHints(lib, include1d) {
    const hints = new Map();
    hints.set(lib.DecodeHintType.POSSIBLE_FORMATS, zxingFormats(lib, include1d));
    hints.set(lib.DecodeHintType.TRY_HARDER, true);
    return hints;
  }

  function readFromSource(lib, reader, hints, source, thorough) {
    const binarizers = [lib.HybridBinarizer];
    if (thorough && lib.GlobalHistogramBinarizer) binarizers.push(lib.GlobalHistogramBinarizer);
    for (let b = 0; b < binarizers.length; b++) {
      if (!binarizers[b]) continue;
      const sources = [source];
      if (source && typeof source.invert === 'function') sources.push(source.invert());
      else if (lib.InvertedLuminanceSource) sources.push(new lib.InvertedLuminanceSource(source));
      for (let i = 0; i < sources.length; i++) {
        try {
          const result = reader.decode(new lib.BinaryBitmap(new binarizers[b](sources[i])), hints);
          const hit = resultFromZxing(lib, result);
          if (hit) return hit;
        } catch (error) {
          if (!isMiss(error, lib)) return null;
        } finally {
          reader.reset();
        }
      }
    }
    return null;
  }

  function regionRect(videoWidth, videoHeight, region) {
    const box = region || { x: 0, y: 0, w: 1, h: 1, maxEdge: MAX_EDGE };
    const sx = Math.max(0, Math.round(videoWidth * box.x));
    const sy = Math.max(0, Math.round(videoHeight * box.y));
    const sw = Math.max(1, Math.min(videoWidth - sx, Math.round(videoWidth * box.w)));
    const sh = Math.max(1, Math.min(videoHeight - sy, Math.round(videoHeight * box.h)));
    const maxEdge = box.maxEdge || MAX_EDGE;
    const scale = Math.min(1, maxEdge / Math.max(sw, sh));
    return {
      name: box.name || 'full',
      sx: sx,
      sy: sy,
      sw: sw,
      sh: sh,
      width: Math.max(1, Math.round(sw * scale)),
      height: Math.max(1, Math.round(sh * scale)),
    };
  }

  function debugReport(info) {
    const attempts = (info && info.attempts) || [];
    const lines = [
      'Camera 2D Barcode Scanner — Copy Debug v' + ((info && info.version) || SCRIPT_VERSION),
      'Paste everything below into the chat.',
      'time: ' + ((info && info.time) || ''),
      'page: ' + ((info && info.page) || ''),
      'engine: ' + ((info && info.engine) || ''),
      'camera: ' + ((info && info.camera) || ''),
      'video: ' + ((info && info.video) || ''),
      'linear barcodes: ' + (info && info.linear ? 'on' : 'off'),
      'preview mirrored: no',
      'zxing: ' + (info && info.zxing ? 'yes' : 'no'),
      'built-in detector: ' + (info && info.detector ? 'yes' : 'no'),
      'last status: ' + ((info && info.status) || ''),
      'last value: ' + ((info && info.value) || ''),
      'auto misses: ' + ((info && info.misses) || 0),
      'last capture: ' + ((info && info.capture) || ''),
      'recent attempts:',
    ];
    if (!attempts.length) lines.push('(none)');
    attempts.forEach(function (row) {
      lines.push('- ' + [row.mode, row.detail, row.result].filter(Boolean).join(' | '));
    });
    return lines.join('\n');
  }

  function decodeCanvas(canvas, include1d) {
    const lib = globalThis.ZXing;
    if (!lib || !lib.MultiFormatReader || !canvas) return null;
    const reader = new lib.MultiFormatReader();
    const hints = makeHints(lib, !!include1d);
    const source = new lib.HTMLCanvasElementLuminanceSource(canvas, false);
    return readFromSource(lib, reader, hints, source);
  }

  function drawVideoRegion(video, canvas, ctx, region) {
    const rect = regionRect(video.videoWidth, video.videoHeight, region);
    if (canvas.width !== rect.width || canvas.height !== rect.height) {
      canvas.width = rect.width;
      canvas.height = rect.height;
    }
    ctx.drawImage(video, rect.sx, rect.sy, rect.sw, rect.sh, 0, 0, rect.width, rect.height);
    return rect;
  }

  function drawVideoFrame(video, canvas, ctx, maxEdge) {
    return drawVideoRegion(video, canvas, ctx, { name: 'full', x: 0, y: 0, w: 1, h: 1, maxEdge: maxEdge || MAX_EDGE });
  }

  async function chooseEngine(include1d) {
    const wanted = nativeFormats(include1d);
    const twoD = nativeFormats(false);
    if (typeof BarcodeDetector === 'function') {
      try {
        let supported = wanted.slice();
        if (typeof BarcodeDetector.getSupportedFormats === 'function') {
          supported = await BarcodeDetector.getSupportedFormats();
        }
        const have2d = twoD.every(function (format) { return supported.indexOf(format) !== -1; });
        if (have2d) {
          const missing1d = !!include1d && NATIVE_1D.some(function (format) { return supported.indexOf(format) === -1; });
          if (!missing1d || !(globalThis.ZXing && globalThis.ZXing.MultiFormatReader)) {
            return {
              kind: 'native',
              label: 'built-in',
              formats: wanted.filter(function (format) { return supported.indexOf(format) !== -1; }),
            };
          }
        }
      } catch (error) {}
    }
    if (globalThis.ZXing && globalThis.ZXing.MultiFormatReader) {
      return { kind: 'zxing', label: 'ZXing', formats: zxingFormats(globalThis.ZXing, include1d) };
    }
    return { kind: 'none', label: '', formats: [] };
  }

  function boxArea(code) {
    const box = code && code.boundingBox;
    if (!box) return 0;
    return Math.max(0, box.width) * Math.max(0, box.height);
  }

  function videoConstraints(deviceId) {
    const video = { width: { ideal: 1280 }, height: { ideal: 720 } };

    return { audio: false, video: video };
  }

    function beep() {
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      if (!AudioCtx) return;
      if (!audioCtx) audioCtx = new AudioCtx();
      if (audioCtx.state === 'suspended') audioCtx.resume();
      const now = audioCtx.currentTime;
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.frequency.value = 880;
      gain.gain.setValueAtTime(0.0001, now);
      gain.gain.exponentialRampToValueAtTime(0.06, now + 0.01);
      gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.12);
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.start(now);
      osc.stop(now + 0.13);
    }

    function bindZoom(track) {
      const caps = track && track.getCapabilities ? track.getCapabilities() : {};
      if (!caps.zoom) {
        zoomRow.hidden = true;
        zoom.oninput = null;
        return;
      }
      zoomRow.hidden = false;
      zoom.min = caps.zoom.min;
      zoom.max = caps.zoom.max;
      zoom.step = caps.zoom.step || 0.1;
      const current = track.getSettings ? track.getSettings().zoom : caps.zoom.min;
      zoom.value = current == null ? caps.zoom.min : current;
      zoom.oninput = function () {
        track.applyConstraints({ advanced: [{ zoom: Number(zoom.value) }] }).catch(function () {});
      };
    }

    async function refreshDevices() {
      if (!navigator.mediaDevices || !navigator.mediaDevices.enumerateDevices) return;
      const devices = await navigator.mediaDevices.enumerateDevices();
      const cams = devices.filter(function (device) { return device.kind === 'videoinput'; });
      const previous = cameraSelect.value || settings.deviceId;
      cameraSelect.replaceChildren();
      cams.forEach(function (cam, index) {
        const option = document.createElement('option');
        option.value = cam.deviceId;
        option.textContent = cam.label || ('Camera ' + (index + 1));
        cameraSelect.appendChild(option);
      });
      const match = previous && Array.prototype.some.call(cameraSelect.options, function (option) { return option.value === previous; });
      if (match) cameraSelect.value = previous;
      if (cameraSelect.value) settings.deviceId = cameraSelect.value;
    }

    function stopTracks() {
      if (session.timer) clearTimeout(session.timer);
      session.timer = 0;
      if (session.stream) {
        session.stream.getTracks().forEach(function (track) { track.stop(); });
        session.stream = null;
      }
      if (video.srcObject) video.srcObject = null;
      stage.classList.remove('is-holding');
    }

    function stopCamera(options) {
      session.generation += 1;
      stopTracks();
      setPower(false);
      zoomRow.hidden = true;
      updateEngineLabel();
      if (!options || !options.silent) setStatus('Camera off.');
    }

    function prepareReader() {
      session.detector = null;
      session.reader = null;
      session.hints = null;
      session.scan = { hold: null, clearAt: 0 };
      if (!session.engine) return;
      if (session.engine.kind === 'native') {
        session.detector = new BarcodeDetector({ formats: session.engine.formats });
      } else if (session.engine.kind === 'zxing' && globalThis.ZXing) {
        session.reader = new globalThis.ZXing.MultiFormatReader();
        session.hints = makeHints(globalThis.ZXing, settings.include1d);
      }
    }

    async function rebuildEngine() {
      session.engine = await chooseEngine(settings.include1d);
      prepareReader();
      updateEngineLabel();
    }

    function recordAttempt(row) {
      session.attempts.push(row);
      if (session.attempts.length > 12) session.attempts.shift();
    }

    function deliver(hits, force) {
      const allowedHits = (hits || []).filter(function (hit) {
        return hit && hit.text && allowsFormat(hit.format, settings.include1d);
      });
      if (force && allowedHits.length) {
        const hit = allowedHits[0];
        const scan = { text: hit.text, format: canonicalFormat(hit.format), at: Date.now() };
        session.scan = { hold: scan, clearAt: 0 };
        stage.classList.add('is-holding');
        onEmit(scan);
        return;
      }
      if (!allowedHits.length) {
        session.misses += 1;
        const reduced = reduceScan(session.scan, { type: 'miss' }, Date.now(), ABSENT_MS);
        session.scan = reduced.state;
        stage.classList.toggle('is-holding', !!session.scan.hold);
        return;
      }
      allowedHits.sort(function (a, b) { return (b.area || 0) - (a.area || 0); });
      allowedHits.forEach(function (hit) {
        const reduced = reduceScan(session.scan, { type: 'hit', text: hit.text, format: hit.format }, Date.now(), ABSENT_MS);
        session.scan = reduced.state;
        if (reduced.emit) onEmit(reduced.emit);
      });
      stage.classList.toggle('is-holding', !!session.scan.hold);
    }

    function onEmit(scan) {
      settings.history = nextHistory(settings.history, scan, HISTORY_LIMIT);
      saveSettings(localStorage, settings);
      renderResult(scan);
      renderHistory();
      if (settings.beep) {
        try { beep(); } catch (error) {}
      }
      if (settings.typeIntoField) {
        if (fieldStillThere(lastField)) {
          const wrote = writeScan(lastField, scan.text, settings.pressEnter);
          setStatus(wrote ? ('Typed ' + labelFor(scan.format) + ' into the page.') : (labelFor(scan.format) + ' read.'));
        } else {
          setStatus(labelFor(scan.format) + ' read. Click a text field to type the next one.');
        }
      } else {
        setStatus(labelFor(scan.format) + ' read.');
      }
    }

    function fallbackToZxing() {
      if (!(globalThis.ZXing && globalThis.ZXing.MultiFormatReader)) return false;
      session.engine = { kind: 'zxing', label: 'ZXing', formats: [] };
      prepareReader();
      updateEngineLabel();
      return true;
    }

    function zxingHit(region, thorough) {
      const lib = globalThis.ZXing;
      if (!lib || !session.reader || !scratchCtx) return null;
      const rect = drawVideoRegion(video, scratch, scratchCtx, region);
      session.lastFrame = rect.name + ' ' + rect.width + 'x' + rect.height;
      const source = new lib.HTMLCanvasElementLuminanceSource(scratch, false);
      return readFromSource(lib, session.reader, session.hints, source, thorough);
    }

    async function scanOnce(gen) {
      if (gen !== session.generation || session.capturing) return;
      if (video.readyState < 2 || !video.videoWidth) return;
      if (session.engine && session.engine.kind === 'native' && session.detector) {
        try {
          const codes = await session.detector.detect(video);
          if (gen !== session.generation || session.capturing) return;
          const hits = (codes || []).map(function (code) {
            const hit = resultFromNative(code);
            if (hit) hit.area = boxArea(code);
            return hit;
          }).filter(Boolean);
          if (hits.length) {
            deliver(hits);
            return;
          }
        } catch (error) {
          if (gen !== session.generation) return;
          if (!fallbackToZxing()) {
            setStatus('The built-in decoder stopped. Reload the page.');
            return;
          }
        }
      }
      if (globalThis.ZXing && globalThis.ZXing.MultiFormatReader && !session.reader) {
        session.reader = new globalThis.ZXing.MultiFormatReader();
        session.hints = makeHints(globalThis.ZXing, settings.include1d);
      }
      if (!globalThis.ZXing || !session.reader || !scratchCtx) return;
      try {
        const hit = zxingHit(CENTER_REGION, false);
        if (gen !== session.generation || session.capturing) return;
        deliver(hit ? [hit] : []);
      } catch (error) {
        if (gen !== session.generation || session.capturing) return;
        deliver([]);
      }
    }

    async function forceCapture() {
      if (!session.stream || video.readyState < 2 || !video.videoWidth) {
        setStatus('Start the camera, then press Capture.');
        return;
      }
      const gen = session.generation;
      session.capturing = true;
      captureBtn.disabled = true;
      setStatus('Capturing this frame…');
      const notes = [];
      let hit = null;
      if (session.detector) {
        try {
          const codes = await session.detector.detect(video);
          notes.push('built-in ' + ((codes && codes.length) || 0));
          if (codes && codes.length) hit = resultFromNative(codes[0]);
        } catch (error) {
          notes.push('built-in error ' + ((error && error.name) || 'failed'));
        }
      }
      if (!hit && globalThis.ZXing) {
        if (!session.reader) {
          session.reader = new globalThis.ZXing.MultiFormatReader();
          session.hints = makeHints(globalThis.ZXing, settings.include1d);
        }
        for (let i = 0; i < CAPTURE_REGIONS.length && !hit; i++) {
          try {
            hit = zxingHit(CAPTURE_REGIONS[i], true);
            notes.push(session.lastFrame + (hit ? ' hit' : ' miss'));
          } catch (error) {
            notes.push(CAPTURE_REGIONS[i].name + ' error ' + ((error && error.name) || 'failed'));
          }
        }
      }
      session.lastCapture = notes.join('; ') || 'no decoder';
      recordAttempt({ mode: 'capture', detail: session.lastCapture, result: hit ? (hit.format + ' ' + hit.text) : 'miss' });
      session.capturing = false;
      captureBtn.disabled = false;
      if (gen !== session.generation) return;
      if (hit && allowsFormat(hit.format, settings.include1d)) deliver([hit], true);
      else setStatus('No code in this frame. Fill the box with the barcode and press Capture again.');
    }

    function tick(gen) {
      scanOnce(gen).then(function () {
        if (gen !== session.generation) return;
        session.timer = setTimeout(function () { tick(gen); }, 90);
      });
    }

    async function startCamera() {
      const blocked = cameraUnavailableReason();
      if (blocked) {
        setStatus(blocked);
        setPower(false);
        return;
      }
      const gen = ++session.generation;
      stopTracks();
      setPower(true);
      setStatus('Asking for the camera…');
      updateEngineLabel();
      let stream = null;
      try {
        stream = await navigator.mediaDevices.getUserMedia(videoConstraints(settings.deviceId));
      } catch (error) {
        if (error && (error.name === 'OverconstrainedError' || error.name === 'ConstraintNotSatisfiedError' || error.name === 'NotFoundError')) {
          try {
            const loose = settings.deviceId ? { deviceId: settings.deviceId } : true;
            stream = await navigator.mediaDevices.getUserMedia({ audio: false, video: loose });
          } catch (again) {
            if (gen !== session.generation) return;
            setStatus(cameraErrorMessage(again));
            setPower(false);
            updateEngineLabel();
            return;
          }
        } else {
          if (gen !== session.generation) return;
          setStatus(cameraErrorMessage(error));
          setPower(false);
          updateEngineLabel();
          return;
        }
      }
      if (gen !== session.generation) {
        stream.getTracks().forEach(function (track) { track.stop(); });
        return;
      }
      session.stream = stream;
      video.srcObject = stream;
      try { await video.play(); } catch (error) {}
      const track = stream.getVideoTracks()[0];
      bindZoom(track);
      try { await refreshDevices(); } catch (error) {}
      session.engine = await chooseEngine(settings.include1d);
      if (gen !== session.generation) return;
      prepareReader();
      updateEngineLabel();
      if (!session.engine || session.engine.kind === 'none') {
        setStatus('This browser cannot read barcodes. Reinstall the script so the decoder downloads, or use Chrome or Edge.');
        return;
      }
      setStatus(settings.include1d
        ? 'Reading. Hold a barcode in the box. Capture forces this frame.'
        : 'Reading 2D codes. Turn on linear barcodes for a normal barcode, or press Capture.');
      tick(gen);
    }

    async function copyText(text) {
      try {
        if (navigator.clipboard && navigator.clipboard.writeText) {
          await navigator.clipboard.writeText(text);
          return true;
        }
      } catch (error) {}
      const area = document.createElement('textarea');
      area.value = text;
      area.setAttribute('readonly', '');
      area.style.cssText = 'position:fixed;left:-9999px;top:0;';
      document.body.appendChild(area);
      area.select();
      let ok = false;
      try { ok = document.execCommand('copy'); } catch (error) { ok = false; }
      area.remove();
      return ok;
    }
  }
