// ==UserScript==
// @name         MEM3 Control Tower - PS Board (abelrigm)
// @namespace    https://mem3.amazon.local/
// @version      1.2.0
// @description  Button on CRET Problem Solve that opens a sheet, places each associate, and scans a typed login in or out on Control Tower.
// @author       abelrigm
// @match        https://beta.control-tower.meta.amazon.dev/*
// @match        https://beta.control-tower.meta.amazon.dev/MEM3/*
// @match        https://beta.control-tower.meta.amazon.dev/MEM3/assignments*
// @match        https://control-tower.meta.amazon.dev/*
// @match        https://control-tower.meta.amazon.dev/MEM3/*
// @match        https://control-tower.meta.amazon.dev/MEM3/assignments*
// @run-at       document-idle
// ==/UserScript==

(function () {
    'use strict';

    // Sits beside the productivity panel. It only reads the CRET Problem Solve
    // table and draws its own sheet. The button stays outside the heading so
    // the panel and line-pairs scripts still see the group name alone.
    //
    // Lines 1 and 3 split Even into A (front) and B (stacked behind).
    // Lines 2 and 4 split Odd the same way. The other side is a single station.

    const BTN_ID = 'mem3-ps-board-btn';
    const POPUP_ID = 'mem3-ps-board';
    const PANEL_JUNK = '.mem3-uph-badge, .mem3-line-avg-badge, .mem3-popup-uph, .mem3-popup-clock, .mem3-popup-idle, .mem3-yield';

    const LINES = [
        [
            { id: '1-even-b', label: '1 Even B', col: 2, row: 1 },
            { id: '1-odd', label: '1 Odd', col: 1, row: 2 },
            { id: '1-even-a', label: '1 Even A', col: 2, row: 2 }
        ],
        [
            { id: '2-odd-b', label: '2 Odd B', col: 1, row: 1 },
            { id: '2-odd-a', label: '2 Odd A', col: 1, row: 2 },
            { id: '2-even', label: '2 Even', col: 2, row: 2 }
        ],
        [
            { id: '3-even-b', label: '3 Even B', col: 2, row: 1 },
            { id: '3-odd', label: '3 Odd', col: 1, row: 2 },
            { id: '3-even-a', label: '3 Even A', col: 2, row: 2 }
        ],
        [
            { id: '4-odd-b', label: '4 Odd B', col: 1, row: 1 },
            { id: '4-odd-a', label: '4 Odd A', col: 1, row: 2 },
            { id: '4-even', label: '4 Even', col: 2, row: 2 }
        ]
    ];

    const SUPPORT = [
        { id: 'audit', label: 'Audit PS' },
        { id: 'grey', label: 'Grey Totes' },
        { id: 'gate', label: 'Gatekeeper' },
        { id: 'blank-1', label: '', blank: true },
        { id: 'bigs', label: 'Bigs PS' },
        { id: 'addins', label: 'Add-Ins' },
        { id: 'exceptions', label: 'Exceptions' },
        { id: 'blank-2', label: '', blank: true }
    ];

    const SLOT_SET = Object.create(null);
    for (let line = 0; line < LINES.length; line++) {
        for (let s = 0; s < LINES[line].length; s++) SLOT_SET[LINES[line][s].id] = true;
    }
    for (let s = 0; s < SUPPORT.length; s++) {
        if (!SUPPORT[s].blank) SLOT_SET[SUPPORT[s].id] = true;
    }

    const NOT_LOGIN = {
        assignment: true, employee: true, cret: true, crets: true, problem: true, solve: true,
        odd: true, even: true, see: true, ps: true, pa: true, the: true, and: true,
        idle: true, uph: true, avg: true, bigs: true, audit: true, audits: true,
        line: true, priority: true, assign: true, unassign: true, add: true,
        cancel: true, close: true, scan: true, search: true, none: true
    };

    function sheetDate(date) {
        const month = date.getMonth() + 1;
        const day = date.getDate();
        function pad(n) { return n < 10 ? '0' + n : String(n); }
        return pad(month) + '/' + pad(day) + '/' + date.getFullYear();
    }

    function sheetShift(date) {
        const hour = date.getHours();
        if (hour >= 6 && hour < 18) return 'Days';
        return 'Nights';
    }

    function lineSlotId(line, side, letter) {
        const splitSide = (line === '1' || line === '3') ? 'even' : 'odd';
        if (side !== splitSide) return line + '-' + side;
        return line + '-' + side + '-' + (letter === 'b' ? 'b' : 'a');
    }

    function slotIdFromAssignment(text) {
        // The process name and the station are often separate lines. textContent
        // glues them, so "Solve" and "Audit" can arrive with no space between.
        const main = String(text || '')
            .replace(/\s+/g, ' ')
            .replace(/crets?\s*problem\s*solve/ig, ' ')
            .split('|')[0];
        const cret = main.match(/cret\s*([1-4])\s*(odd|even)\s*([ab])?\b/i)
            || main.match(/\b([1-4])\s*(odd|even)\s*([ab])?\b/i);
        if (cret) return lineSlotId(cret[1], cret[2].toLowerCase(), (cret[3] || '').toLowerCase());
        const t = main.toLowerCase();
        if (/\baudits?\b/.test(t)) return 'audit';
        if (/\bgr[ae]y\s*totes?\b/.test(t)) return 'grey';
        if (/\bgate\s*keepers?\b/.test(t)) return 'gate';
        if (/\bbigs\b/.test(t)) return 'bigs';
        if (/\badd[\s-]*ins?\b/.test(t)) return 'addins';
        if (/\bexceptions?\b/.test(t)) return 'exceptions';
        return '';
    }

    function shortAssignment(text) {
        return String(text || '')
            .replace(/\s+/g, ' ')
            .replace(/crets?\s*problem\s*solve/ig, ' ')
            .replace(/\s*\|\s*/g, ' · ')
            .replace(/\s+/g, ' ')
            .trim();
    }

    function textWithBreaks(el) {
        const clone = cleanClone(el);
        const doc = clone.ownerDocument || (typeof document !== 'undefined' ? document : null);
        if (clone.querySelectorAll && doc && doc.createTextNode) {
            const breaks = clone.querySelectorAll('br, div, p, li');
            for (let i = 0; i < breaks.length; i++) {
                const node = breaks[i];
                if (!node.parentNode) continue;
                node.parentNode.insertBefore(doc.createTextNode(' '), node);
            }
        }
        return (clone.textContent || '').replace(/\s+/g, ' ').trim();
    }

    function placePeople(people) {
        const slots = Object.create(null);
        const extra = [];
        const seen = Object.create(null);
        for (let i = 0; i < people.length; i++) {
            const person = people[i] || {};
            const login = String(person.login || '').trim();
            if (!login) continue;
            const id = slotIdFromAssignment(person.assignment);
            const key = login.toLowerCase() + '@' + id;
            if (seen[key]) continue;
            seen[key] = true;
            if (!id || !SLOT_SET[id]) {
                extra.push({ login: login, assignment: shortAssignment(person.assignment) });
                continue;
            }
            if (!slots[id]) slots[id] = [];
            slots[id].push(login);
        }
        return { slots: slots, extra: extra };
    }

    function cleanClone(el) {
        const clone = el.cloneNode(true);
        if (clone.querySelectorAll) {
            const junk = clone.querySelectorAll(PANEL_JUNK);
            for (let i = 0; i < junk.length; i++) junk[i].remove();
        }
        return clone;
    }

    function looksLikeLogin(text) {
        return /^[a-z][a-z0-9._-]{1,31}$/i.test(text) && text.length >= 3;
    }

    function loginFromText(text) {
        const trimmed = String(text || '').replace(/\s+/g, ' ').trim();
        if (looksLikeLogin(trimmed) && !NOT_LOGIN[trimmed.toLowerCase()]) return trimmed;
        const tokens = trimmed.toLowerCase().match(/[a-z][a-z0-9._-]{2,31}/g) || [];
        for (let i = 0; i < tokens.length; i++) {
            if (NOT_LOGIN[tokens[i]]) continue;
            if (looksLikeLogin(tokens[i])) return tokens[i];
        }
        return '';
    }

    function headingText(el) {
        const clone = cleanClone(el);
        return (clone.textContent || '').replace(/\s+/g, ' ').trim();
    }

    function isProblemSolveHeading(el) {
        return /\bcret\s+problem\s+solve\b/i.test(headingText(el));
    }

    function findTable(heading) {
        let el = heading;
        for (let i = 0; i < 8 && el; i++) {
            if (el !== heading && el.querySelector) {
                const table = el.querySelector('table');
                if (table) return table;
            }
            el = el.parentElement;
        }
        let sib = heading.nextElementSibling;
        for (let i = 0; i < 4 && sib; i++) {
            if (sib.tagName === 'TABLE') return sib;
            if (sib.querySelector) {
                const table = sib.querySelector('table');
                if (table) return table;
            }
            sib = sib.nextElementSibling;
        }
        return null;
    }

    function findGroup() {
        if (typeof document === 'undefined' || !document.querySelectorAll) return null;
        const all = document.querySelectorAll('h1, h2, h3, h4, h5, h6, [role="heading"]');
        let fallback = null;
        for (let i = 0; i < all.length; i++) {
            if (!isProblemSolveHeading(all[i])) continue;
            if (!fallback) fallback = all[i];
            if (findTable(all[i])) return all[i];
        }
        return fallback;
    }

    function cellText(el) {
        return textWithBreaks(el);
    }

    function rowLogin(tr) {
        const cells = tr.querySelectorAll('td');
        const cell = cells.length ? cells[cells.length - 1] : tr;
        const clone = cleanClone(cell);
        const anchors = clone.querySelectorAll ? clone.querySelectorAll('a') : [];
        for (let i = 0; i < anchors.length; i++) {
            const fromText = loginFromText(anchors[i].textContent);
            if (fromText) return fromText;
            if (anchors[i].getAttribute) {
                const fromTitle = loginFromText(anchors[i].getAttribute('title') || anchors[i].getAttribute('aria-label'));
                if (fromTitle) return fromTitle;
            }
        }
        return loginFromText(clone.textContent);
    }

    function readPeople(table) {
        const people = [];
        if (!table || !table.querySelectorAll) return people;
        const rows = table.querySelectorAll('tbody tr');
        for (let i = 0; i < rows.length; i++) {
            const tr = rows[i];
            const cells = tr.querySelectorAll('td');
            if (!cells.length) continue;
            const assignment = cellText(cells[0]);
            if (!assignment || /^assignment$/i.test(assignment)) continue;
            people.push({
                login: rowLogin(tr),
                assignment: assignment
            });
        }
        return people;
    }

    function readGroup() {
        const heading = findGroup();
        const table = heading ? findTable(heading) : null;
        const people = readPeople(table);
        return {
            heading: heading,
            table: table,
            people: people,
            placed: placePeople(people),
            staffed: people.filter(function (person) { return String(person.login || '').trim(); }).length
        };
    }

    function normalizeLogin(value) {
        return String(value || '').trim().toLowerCase();
    }

    function validLogin(login) {
        return /^[a-z][a-z0-9._-]{2,31}$/.test(login);
    }

    function stationLabel(slotId) {
        let line, slot;
        for (line = 0; line < LINES.length; line++) {
            for (slot = 0; slot < LINES[line].length; slot++) {
                if (LINES[line][slot].id === slotId) return LINES[line][slot].label;
            }
        }
        for (slot = 0; slot < SUPPORT.length; slot++) {
            if (SUPPORT[slot].id === slotId) return SUPPORT[slot].label;
        }
        return '';
    }

    // Scan in only fills an empty station. Scan out removes the typed login,
    // or the only person on the station when the field is empty.
    function scanPlan(slotId, names, rawLogin, action) {
        const login = normalizeLogin(rawLogin);
        const list = [];
        const source = names || [];
        for (let i = 0; i < source.length; i++) {
            const name = normalizeLogin(source[i]);
            if (name) list.push(name);
        }
        if (!slotId || !SLOT_SET[slotId]) {
            return { ok: false, code: 'station', message: 'Click a station on the sheet first.' };
        }
        const label = stationLabel(slotId);
        if (action === 'out') {
            const who = login || (list.length === 1 ? list[0] : '');
            if (!who) {
                return { ok: false, code: 'login', message: 'Type the login to scan out. For example, abelrigm.' };
            }
            if (!validLogin(who)) {
                return { ok: false, code: 'login', message: 'That does not look like a login. For example, abelrigm.' };
            }
            if (list.indexOf(who) === -1) {
                return { ok: false, code: 'missing', message: who + ' is not on ' + label + '.' };
            }
            return { ok: true, action: 'out', login: who, label: label };
        }
        if (!validLogin(login)) {
            return { ok: false, code: 'login', message: 'Type a login. For example, abelrigm.' };
        }
        if (list.indexOf(login) !== -1) {
            return {
                ok: true,
                action: 'already',
                login: login,
                label: label,
                message: login + ' is already on ' + label + '.'
            };
        }
        if (list.length) {
            return {
                ok: false,
                code: 'occupied',
                message: label + ' already has ' + list.join(', ') + '. Scan them out first.'
            };
        }
        return { ok: true, action: 'in', login: login, label: label };
    }

    function stationOf(placed, login) {
        const want = normalizeLogin(login);
        const slots = placed && placed.slots ? placed.slots : {};
        const ids = Object.keys(slots);
        for (let i = 0; i < ids.length; i++) {
            const names = slots[ids[i]] || [];
            for (let n = 0; n < names.length; n++) {
                if (normalizeLogin(names[n]) === want) return ids[i];
            }
        }
        return '';
    }

    function rowsForSlot(table, slotId) {
        const rows = [];
        if (!table || !table.querySelectorAll) return rows;
        const body = table.querySelectorAll('tbody tr');
        for (let i = 0; i < body.length; i++) {
            const cells = body[i].querySelectorAll('td');
            if (!cells.length) continue;
            if (slotIdFromAssignment(cellText(cells[0])) === slotId) rows.push(body[i]);
        }
        return rows;
    }

    function rowHasLogin(row, login) {
        return normalizeLogin(rowLogin(row)) === normalizeLogin(login);
    }

    function sleep(ms) {
        return new Promise(function (resolve) { setTimeout(resolve, ms); });
    }

    function waitFor(read, ms) {
        const start = Date.now();
        return new Promise(function (resolve) {
            (function poll() {
                let hit = null;
                try { hit = read(); } catch (e) { hit = null; }
                if (hit) return resolve(hit);
                if (Date.now() - start > ms) return resolve(null);
                setTimeout(poll, 80);
            })();
        });
    }

    function pageWindow() {
        return (typeof unsafeWindow !== 'undefined' && unsafeWindow) ? unsafeWindow : window;
    }

    function pageClick(el) {
        if (!el) return;
        const win = pageWindow();
        const rect = el.getBoundingClientRect ? el.getBoundingClientRect() : { left: 0, top: 0, width: 2, height: 2 };
        const x = Math.round(rect.left + Math.max(1, rect.width) / 2);
        const y = Math.round(rect.top + Math.max(1, rect.height) / 2);
        const base = {
            bubbles: true, cancelable: true, composed: true, view: win, button: 0,
            clientX: x, clientY: y, screenX: x, screenY: y,
            pointerId: 1, pointerType: 'mouse', isPrimary: true, width: 1, height: 1
        };
        const down = { buttons: 1 };
        const up = { buttons: 0 };
        const keys = Object.keys(base);
        for (let i = 0; i < keys.length; i++) {
            down[keys[i]] = base[keys[i]];
            up[keys[i]] = base[keys[i]];
        }
        try { if (el.focus) el.focus(); } catch (e0) { /* ignore */ }
        try {
            el.dispatchEvent(new win.PointerEvent('pointerdown', down));
            el.dispatchEvent(new win.MouseEvent('mousedown', down));
            el.dispatchEvent(new win.PointerEvent('pointerup', up));
            el.dispatchEvent(new win.MouseEvent('mouseup', up));
            el.dispatchEvent(new win.MouseEvent('click', up));
        } catch (e) { /* fall through */ }
        try { el.click(); } catch (e2) { /* ignore */ }
    }

    function reactFiber(el) {
        if (!el) return null;
        let keys = [];
        try { keys = Object.keys(el); } catch (e) { return null; }
        for (let i = 0; i < keys.length; i++) {
            if (keys[i].indexOf('__reactFiber') === 0 || keys[i].indexOf('__reactInternalInstance') === 0) return el[keys[i]];
        }
        return null;
    }

    function isMdnClickWrapper(fn) {
        if (typeof fn !== 'function') return true;
        const src = Function.prototype.toString.call(fn);
        return src.indexOf('defaultPrevented') !== -1 && src.indexOf('noopener') !== -1;
    }

    function reactClick(el) {
        const fns = [];
        let fiber = reactFiber(el);
        let hops = 0;
        while (fiber && hops < 20) {
            const props = fiber.memoizedProps || fiber.pendingProps;
            if (props && typeof props.onClick === 'function' && !isMdnClickWrapper(props.onClick)) fns.push(props.onClick);
            fiber = fiber.return;
            hops++;
        }
        if (!fns.length) return false;
        const event = {
            target: el,
            currentTarget: el,
            type: 'click',
            button: 0,
            bubbles: true,
            cancelable: true,
            defaultPrevented: false,
            preventDefault: function () { this.defaultPrevented = true; },
            stopPropagation: function () {},
            persist: function () {},
            nativeEvent: { target: el, preventDefault: function () {}, stopPropagation: function () {} },
            isDefaultPrevented: function () { return !!this.defaultPrevented; },
            isPropagationStopped: function () { return false; }
        };
        for (let i = fns.length - 1; i >= 0; i--) {
            try { fns[i](event); } catch (e) { /* keep going */ }
        }
        return true;
    }

    function clickControl(el) {
        if (!el) return;
        try { el.scrollIntoView({ block: 'center', inline: 'nearest' }); } catch (e) { /* ignore */ }
        reactClick(el);
        pageClick(el);
    }

    function visibleButton(pattern) {
        const buttons = document.querySelectorAll('button');
        for (let i = 0; i < buttons.length; i++) {
            const btn = buttons[i];
            if (!pattern.test((btn.textContent || '').replace(/\s+/g, ' ').trim())) continue;
            if (btn.closest('#mem3-run-panel, #mem3-unassign-panel, #mem3-floor-summary')) continue;
            const rect = btn.getBoundingClientRect();
            if (rect.width < 1 || rect.height < 1) continue;
            return btn;
        }
        return null;
    }

    function tooltipIds() {
        const ids = [];
        const tips = document.querySelectorAll('[id^="tooltip-"]');
        for (let i = 0; i < tips.length; i++) {
            const rect = tips[i].getBoundingClientRect();
            if (rect.width >= 1 && rect.height >= 1 && tips[i].id) ids.push(tips[i].id);
        }
        return ids;
    }

    function newUnassignButton(known) {
        const buttons = document.querySelectorAll('button');
        for (let i = 0; i < buttons.length; i++) {
            const btn = buttons[i];
            if (!/^unassign$/i.test((btn.textContent || '').replace(/\s+/g, ' ').trim())) continue;
            const tip = btn.closest('[id^="tooltip-"]');
            if (tip && known[tip.id]) continue;
            const rect = btn.getBoundingClientRect();
            if (rect.width < 1 || rect.height < 1) continue;
            return btn;
        }
        return null;
    }

    function loginAnchor(row, login) {
        const want = normalizeLogin(login);
        const anchors = row.querySelectorAll('a[role="button"], a[mdn-link], a, button');
        for (let i = 0; i < anchors.length; i++) {
            if (normalizeLogin(anchors[i].textContent) === want) return anchors[i];
        }
        return null;
    }

    function isToolbarSearch(input) {
        if (input.closest && input.closest('[id^="tooltip-"]')) return false;
        const hint = ((input.getAttribute('placeholder') || '') + ' ' + (input.id || '')).toLowerCase();
        return hint.indexOf('search for login') !== -1 || hint.indexOf('search-field-') === 0;
    }

    function isAssignSearch(input) {
        if (!input.closest || !input.closest('[id^="tooltip-"]')) return false;
        const hint = ((input.getAttribute('placeholder') || '') + ' ' + (input.id || '')).toLowerCase();
        return hint.indexOf('search for login') !== -1 || hint.indexOf('search-field-') === 0;
    }

    function noneAnchor(row) {
        const anchors = row.querySelectorAll('a[role="button"], a[mdn-link]');
        for (let i = 0; i < anchors.length; i++) {
            if (/^none$/i.test((anchors[i].textContent || '').replace(/\s+/g, ' ').trim())) return anchors[i];
        }
        return null;
    }

    function writeLogin(input, value) {
        const win = pageWindow();
        const proto = input.tagName === 'TEXTAREA'
            ? (win.HTMLTextAreaElement && win.HTMLTextAreaElement.prototype)
            : (win.HTMLInputElement && win.HTMLInputElement.prototype);
        const desc = proto && Object.getOwnPropertyDescriptor(proto, 'value');
        try { input.focus(); } catch (e) { /* ignore */ }
        if (input._valueTracker) {
            try { input._valueTracker.setValue(''); } catch (e2) { /* ignore */ }
        }
        if (desc && desc.set) desc.set.call(input, value);
        else input.value = value;
        try {
            input.dispatchEvent(new win.InputEvent('input', {
                bubbles: true, cancelable: true, data: value, inputType: 'insertText'
            }));
        } catch (e3) {
            input.dispatchEvent(new win.Event('input', { bubbles: true }));
        }
        input.dispatchEvent(new win.Event('change', { bubbles: true }));
    }

    function hasLoginToken(el, login) {
        if (!el) return false;
        const nodes = el.querySelectorAll('td, a, span, div, button');
        const pool = nodes.length ? nodes : [el];
        for (let i = 0; i < pool.length; i++) {
            if (pool[i].children && pool[i].children.length) continue;
            const text = (pool[i].textContent || '').toLowerCase().replace(/\s+/g, ' ').trim();
            if (!text) continue;
            if (text === login) return true;
            const tokens = text.match(/[a-z][a-z0-9._-]{2,31}/g) || [];
            if (tokens.indexOf(login) !== -1) return true;
        }
        return false;
    }

    function assignSearch(known) {
        const inputs = document.querySelectorAll('input');
        for (let i = 0; i < inputs.length; i++) {
            const input = inputs[i];
            if (isToolbarSearch(input) || !isAssignSearch(input)) continue;
            const tip = input.closest('[id^="tooltip-"]');
            if (tip && known[tip.id]) continue;
            const rect = input.getBoundingClientRect();
            if (rect.width < 1 || rect.height < 1) continue;
            return input;
        }
        return null;
    }

    function assignButtonFor(login) {
        const buttons = document.querySelectorAll('button');
        for (let i = 0; i < buttons.length; i++) {
            const btn = buttons[i];
            if (!/^assign$/i.test((btn.textContent || '').replace(/\s+/g, ' ').trim())) continue;
            if (!btn.closest || !btn.closest('[id^="tooltip-"]')) continue;
            const rect = btn.getBoundingClientRect();
            if (rect.width < 1 || rect.height < 1) continue;
            const row = btn.closest('tr') || btn.closest('[id^="tooltip-"]');
            if (row && hasLoginToken(row, login)) return btn;
        }
        return null;
    }

    async function scanOutOnPage(row, login) {
        const anchor = loginAnchor(row, login);
        if (!anchor) return { ok: false, message: login + ' is not on that Control Tower row.' };
        const known = Object.create(null);
        const existing = tooltipIds();
        for (let i = 0; i < existing.length; i++) known[existing[i]] = true;
        clickControl(anchor);
        let button = await waitFor(function () { return newUnassignButton(known); }, 1400);
        if (!button) {
            clickControl(anchor);
            button = await waitFor(function () { return newUnassignButton(known) || visibleButton(/^unassign$/i); }, 1600);
        }
        if (!button) return { ok: false, message: 'Control Tower did not open Unassign for ' + login + '.' };
        await sleep(200);
        clickControl(button);
        const gone = await waitFor(function () { return rowHasLogin(row, login) ? null : true; }, 4000);
        if (!gone) return { ok: false, message: 'Unassign did not remove ' + login + ' from Control Tower.' };
        return { ok: true, message: 'Scanned ' + login + ' out.' };
    }

    async function typeLogin(input, login) {
        writeLogin(input, '');
        let typed = '';
        for (let i = 0; i < login.length; i++) {
            typed += login.charAt(i);
            writeLogin(input, typed);
            await sleep(25);
        }
    }

    async function scanInOnPage(row, login) {
        const opener = noneAnchor(row);
        if (!opener) return { ok: false, message: 'That station is not open. Control Tower does not show None.' };
        const known = Object.create(null);
        const existing = tooltipIds();
        for (let i = 0; i < existing.length; i++) known[existing[i]] = true;
        clickControl(opener);
        const input = await waitFor(function () { return assignSearch(known); }, 1600);
        if (!input) return { ok: false, message: 'Control Tower did not open the login search.' };
        await typeLogin(input, login);
        let button = await waitFor(function () { return assignButtonFor(login); }, 1800);
        if (!button) {
            writeLogin(input, login);
            button = await waitFor(function () { return assignButtonFor(login); }, 1500);
        }
        if (!button) return { ok: false, message: 'Control Tower did not show Assign for ' + login + '.' };
        clickControl(button);
        const arrived = await waitFor(function () { return rowHasLogin(row, login) ? true : null; }, 4000);
        if (!arrived) return { ok: false, message: 'Control Tower did not assign ' + login + '.' };
        return { ok: true, message: 'Scanned ' + login + ' in.' };
    }

    let selectedSlot = '';
    let scanBusy = false;

        const host = document.getElementById(POPUP_ID);

    function setScanStatus(text, kind) {
        const host = document.getElementById(POPUP_ID);
        const node = host && host.shadowRoot ? host.shadowRoot.querySelector('[data-scan-status]') : null;
        if (!node) return;

    function markSelected() {
        const host = document.getElementById(POPUP_ID);
        if (!host || !host.shadowRoot) return;
        const boxes = host.shadowRoot.querySelectorAll('[data-slot]');
        for (let i = 0; i < boxes.length; i++) {
            const on = boxes[i].getAttribute('data-slot') === selectedSlot;
            if (on) boxes[i].classList.add('is-selected');
            else boxes[i].classList.remove('is-selected');
        }
    }

    function chooseSlot(slotId, login) {
        if (!SLOT_SET[slotId] || scanBusy) return;
        selectedSlot = slotId;
        markSelected();
        const input = scanInput();
        if (input && login) input.value = login;
        if (input && input.focus) input.focus();
        const label = stationLabel(slotId);
        setScanStatus(login ? (login + ' on ' + label + '. Scan out removes them from Control Tower.') : ('Selected ' + label + '.'), '');
    }

    async function runScan(action) {
        if (scanBusy) return;
        const input = scanInput();
        const typed = input ? input.value : '';
        const group = readGroup();
        const names = (group.placed.slots && group.placed.slots[selectedSlot]) || [];
        const plan = scanPlan(selectedSlot, names, typed, action);
        if (!plan.ok) {
            setScanStatus(plan.message, 'bad');
            return;
        }
        if (plan.action === 'already') {
            setScanStatus(plan.message, 'ok');
            return;
        }
        if (plan.action === 'in') {
            const elsewhere = stationOf(group.placed, plan.login);
            if (elsewhere && elsewhere !== selectedSlot) {
                setScanStatus(plan.login + ' is already on ' + stationLabel(elsewhere) + '. Scan them out there first.', 'bad');
                return;
            }
        }
        const rows = rowsForSlot(group.table, selectedSlot);
        if (!rows.length) {
            setScanStatus(plan.label + ' is not a row on CRET Problem Solve.', 'bad');
            return;
        }
        let row = rows[0];
        if (plan.action === 'out') {
            for (let i = 0; i < rows.length; i++) {
                if (rowHasLogin(rows[i], plan.login)) row = rows[i];
            }
        } else {
            for (let i = 0; i < rows.length; i++) {
                if (!rowLogin(rows[i])) row = rows[i];
            }
        }
        scanBusy = true;
        setScanStatus((plan.action === 'out' ? 'Scanning ' + plan.login + ' out of ' : 'Scanning ' + plan.login + ' in to ') + plan.label + '…', '');
        let result;
        try {
            result = plan.action === 'out' ? await scanOutOnPage(row, plan.login) : await scanInOnPage(row, plan.login);
        } catch (e) {
            result = { ok: false, message: (e && e.message) ? e.message : 'Control Tower did not take that scan.' };
        }
        scanBusy = false;
        setScanStatus(result.message, result.ok ? 'ok' : 'bad');
        if (result.ok && input) input.value = '';
        tick();
    }

    function slotBox(slot) {
        const box = document.createElement('div');

        if (slot.row) box.style.gridRow = String(slot.row);
        const head = document.createElement('div');

    }

    function fillBody(body, names) {
        const sig = names.join('\n');
        if (body.getAttribute('data-sig') === sig) return;
        body.setAttribute('data-sig', sig);
        body.textContent = '';
        for (let i = 0; i < names.length; i++) {
            const name = document.createElement('button');
            name.type = 'button';
            name.className = 'name';
            name.textContent = names[i];
            name.title = 'Put ' + names[i] + ' in the login field';
            name.addEventListener('click', function (event) {
                event.preventDefault();
                event.stopPropagation();
                const box = body.closest('[data-slot]');
                if (box) chooseSlot(box.getAttribute('data-slot'), name.textContent);
            });
            body.appendChild(name);
        }
    }

    function paint(group) {
        const host = document.getElementById(POPUP_ID);
        if (!host || !host.shadowRoot) return;
        const shadow = host.shadowRoot;
        const now = new Date();
        const dateEl = shadow.querySelector('[data-date]');
        const shiftEl = shadow.querySelector('[data-shift]');
        if (dateEl) dateEl.textContent = sheetDate(now);
        if (shiftEl) shiftEl.textContent = sheetShift(now);
        const boxes = shadow.querySelectorAll('[data-slot]');
        for (let i = 0; i < boxes.length; i++) {
            const id = boxes[i].getAttribute('data-slot');
            const body = boxes[i].querySelector('[data-body]');
            if (!body) continue;
            fillBody(body, group.placed.slots[id] || []);
        }
        markSelected();
        const extra = shadow.querySelector('[data-extra]');
        if (extra) {
            const rows = group.placed.extra;
            const sig = rows.map(function (row) { return row.login + '|' + row.assignment; }).join('\n');
            if (extra.getAttribute('data-sig') !== sig) {
                extra.setAttribute('data-sig', sig);
                extra.textContent = '';
                if (rows.length) {
                    const title = document.createElement('div');
                    title.className = 'extra-title';
                    title.textContent = 'Not on the sheet';
                    extra.appendChild(title);
                    for (let i = 0; i < rows.length; i++) {
                        const row = document.createElement('div');
                        row.className = 'extra-row';
                        row.textContent = rows[i].assignment
                            ? (rows[i].login + ' \u2014 ' + rows[i].assignment)
                            : rows[i].login;
                        extra.appendChild(row);
                    }
                }
            }
            extra.hidden = rows.length === 0;
        }
        const note = shadow.querySelector('[data-note]');
        if (note) {
            const missing = !group.heading;
            note.hidden = !missing;
            note.textContent = missing ? 'CRET Problem Solve is not on this page right now.' : '';
        }
    }

    let pending = false;
    function schedule() {
        if (pending) return;
        pending = true;
        setTimeout(function () {
            pending = false;
            tick();
        }, 200);
    }

    function tick() {
        const group = readGroup();
        if (group.heading) ensureButton(group.heading, group.staffed);
        else {
            const btn = document.getElementById(BTN_ID);
            if (btn) btn.remove();
        }
        if (document.getElementById(POPUP_ID)) paint(group);
    }

    function onKey(event) {
        if (event.key !== 'Escape' || !event.isTrusted) return;
        const path = event.composedPath ? event.composedPath() : [];
        for (let i = 0; i < path.length; i++) {
            if (path[i] && path[i].getAttribute && path[i].hasAttribute && path[i].hasAttribute('data-scan')) return;
        }
        closeBoard();
    }

    function boot() {
        if (typeof document === 'undefined' || !document || typeof document.querySelectorAll !== 'function') return;
        if (!document.body) {
            document.addEventListener('DOMContentLoaded', boot);
            return;
        }
        document.addEventListener('keydown', onKey);
        tick();
        setInterval(tick, 1500);
        if (typeof MutationObserver === 'function') {
            const observer = new MutationObserver(function (records) {
                for (let i = 0; i < records.length; i++) {
                    const node = records[i].target;
                    const el = node && node.nodeType === 1 ? node : (node && node.parentElement);
                    if (!el || !el.closest) {
                        schedule();
                        return;
                    }
                    if (el.closest('#' + POPUP_ID) || el.id === BTN_ID || el.closest('#' + BTN_ID)) continue;
                    schedule();
                    return;
                }
            });
            observer.observe(document.body, { childList: true, subtree: true, characterData: true });
        }
    }

    if (!root.__PS_BOARD_EXPORT) boot();
