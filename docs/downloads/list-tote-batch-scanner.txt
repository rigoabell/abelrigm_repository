// ==UserScript==
// @name         List Tote - Batch Scanner
// @namespace    https://wd-repair-portal-na.aka.amazon.com/
// @version      0.7.1
// @description  Paste tote IDs, run a batch scan, capture each #destination and #units-count, export TSV. Auto-resume across page refreshes.
// @author       you
// @match        https://wd-repair-portal-na.aka.amazon.com/resources/ReCommerceFCWebToolsUI/html/listTote.html*
// @run-at       document-idle
// @grant        none
// ==/UserScript==

(function () {

  // ---------- Config ----------
  const TOTE_RESULT_SELECTOR = '#tote-scannable-id-text';
  const TOTE_INPUT_SELECTOR  = '#tote-id-input-textfield';
  const DEST_SELECTOR        = '#destination';
  const UNITS_SELECTOR       = '#units-count';
  const KEY_DELAY_MS         = 0;
  const PRE_ENTER_MS         = 30;
  const RESULT_WAIT_MS       = 2500;
  const POST_MATCH_SETTLE_MS = 350;
  const POST_SUBMIT_POLL_MS  = 50;
  const BETWEEN_SCANS_MS     = 120;
  const LOG_PREFIX           = '[ListToteAuto]';
  const MAX_DEBUG_ENTRIES    = 1000;
  const STORAGE_KEY          = 'listToteAuto_v1';

  // ---------- State ----------
  const state = {
    debug: [],
    batch: {
      running: false,
      abort: false,
      total: 0,
      processed: 0,

    },

  // ---------- Storage ----------
  function loadStorage() {
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      return raw ? JSON.parse(raw) : {};
    } catch (_) { return {}; }
  }
  function saveStorage(partial) {
    try {
      const current = loadStorage();
      const merged = { ...current, ...partial };
      localStorage.setItem(STORAGE_KEY, JSON.stringify(merged));
    } catch (e) { warn('saveStorage failed', e && e.message); }
  }
  let saveInputTimer = null;
  function saveInputDebounced(value) {
    clearTimeout(saveInputTimer);
    saveInputTimer = setTimeout(() => saveStorage({ batchInput: value }), 300);
  }

  // ---------- Debug log ----------
  function pushDebug(level, args) {
    const msg = args.map(a => {
      if (a instanceof Element) return describeEl(a);
      if (a && typeof a === 'object') {
        try { return JSON.stringify(a); } catch (_) { return String(a); }
      }
      return String(a);
    }).join(' ');
    state.debug.push({ ts: nowStr(), level, msg });
    if (state.debug.length > MAX_DEBUG_ENTRIES) state.debug.shift();
  }
  const log  = (...a) => { pushDebug('LOG',  a); console.log(LOG_PREFIX, ...a); };
  const warn = (...a) => { pushDebug('WARN', a); console.warn(LOG_PREFIX, ...a); };
  const err  = (...a) => { pushDebug('ERR',  a); console.error(LOG_PREFIX, ...a); };

  const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

  function nowStr() {
    const d = new Date();
    return d.toLocaleTimeString([], { hour12: false }) + '.' + String(d.getMilliseconds()).padStart(3, '0');
  }

  function describeEl(el) {
    if (!el) return 'null';
    if (el === document) return 'document';
    if (el === document.body) return 'body';
    const id = el.id ? '#' + el.id : '';
    const cls = el.className && typeof el.className === 'string'
      ? '.' + el.className.split(/\s+/).filter(Boolean).join('.') : '';
    const tag = (el.tagName || '').toLowerCase();
    const type = el.type ? `[type=${el.type}]` : '';
    return `${tag}${id}${cls}${type}`;
  }

  function escapeHtml(s) {
    return String(s).replace(/[&<>"']/g, (c) => ({ '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;' }[c]));
  }

  function fmtDuration(ms) {
    if (!isFinite(ms) || ms < 0) return '—';
    const s = Math.round(ms / 1000);
    if (s < 60) return s + 's';
    const m = Math.floor(s / 60);
    const r = s % 60;
    if (m < 60) return m + 'm ' + r + 's';
    const h = Math.floor(m / 60);
    return h + 'h ' + (m % 60) + 'm';
  }

  function findUnitsEl()  { return document.querySelector(UNITS_SELECTOR); }

    const el = findResultEl();

    return el ? ((el.textContent || '').trim() || null) : null;
  }
  function readUnits() {
    const el = findUnitsEl();
    return el ? ((el.textContent || '').trim() || null) : null;
  }

  // ---------- Typing ----------
  function keyInitFor(ch, ctrl) {
    let code = '';
    if (/[a-zA-Z]/.test(ch))   code = 'Key'   + ch.toUpperCase();
    else if (/[0-9]/.test(ch)) code = 'Digit' + ch;
    return {
      key: ch, code,
      bubbles: true, cancelable: true, composed: true,
      ctrlKey: !!ctrl,
      keyCode: ch.charCodeAt(0),
      which:   ch.charCodeAt(0),
    };
  }
  function dispatchKey(target, type, init) {
    target.dispatchEvent(new KeyboardEvent(type, init));
  }
  function setNativeInputValue(input, value) {
    const proto = input instanceof HTMLTextAreaElement
      ? HTMLTextAreaElement.prototype : HTMLInputElement.prototype;
    const desc = Object.getOwnPropertyDescriptor(proto, 'value');
    if (desc && desc.set) desc.set.call(input, value);
    else input.value = value;
  }
  function resolveTypeTarget() {
    const input = findInputEl();
    if (input) {
      try { input.focus({ preventScroll: true }); } catch (_) { input.focus(); }
      return input;
    }
    return document.activeElement && document.activeElement !== document.body
      ? document.activeElement : document.body;
  }

  // "both" strategy: send Ctrl-held key events AND set value via native setter + input/change/Enter + form submit.
  async function performType(value, target) {
    if (target instanceof HTMLInputElement || target instanceof HTMLTextAreaElement) {
      target.focus();

      dispatchKey(target, 'keydown', {
        key: 'Control', code: 'ControlLeft',
        bubbles: true, cancelable: true, composed: true,
        ctrlKey: true, location: 1,
      });
      for (const ch of value) {
        const init = keyInitFor(ch, true);
        dispatchKey(target, 'keydown',  init);
        dispatchKey(target, 'keypress', { ...init, charCode: ch.charCodeAt(0) });
        dispatchKey(target, 'keyup',    init);
        if (KEY_DELAY_MS) await sleep(KEY_DELAY_MS);
      }
      dispatchKey(target, 'keyup', {
        key: 'Control', code: 'ControlLeft',
        bubbles: true, cancelable: true, composed: true,
        ctrlKey: false, location: 1,
      });

      setNativeInputValue(target, value);
      target.dispatchEvent(new InputEvent('beforeinput', { bubbles: true, cancelable: true, data: value, inputType: 'insertText' }));
      target.dispatchEvent(new InputEvent('input',       { bubbles: true, cancelable: true, data: value, inputType: 'insertText' }));
      target.dispatchEvent(new Event('change',           { bubbles: true }));

      await sleep(PRE_ENTER_MS);

      const enterInit = {
        key: 'Enter', code: 'Enter',
        bubbles: true, cancelable: true, composed: true,
        keyCode: 13, which: 13,
      };
      dispatchKey(target, 'keydown',  enterInit);
      dispatchKey(target, 'keypress', enterInit);
      dispatchKey(target, 'keyup',    enterInit);

      const form = target.form;
      if (form) form.dispatchEvent(new Event('submit', { bubbles: true, cancelable: true }));
    } else {
      warn('No input target found for typing');
    }
  }

  async function waitForResultMatch(expected, timeoutMs) {
    const start = Date.now();
    while (Date.now() - start < timeoutMs) {
      if (readToteResult() === expected) return true;
      await sleep(POST_SUBMIT_POLL_MS);
    }
    return false;
  }

  async function doType(value, source) {
    if (!value) return { ok: false, destination: null, units: null, matched: false };
    const target = resolveTypeTarget();
    log('doType', source, value, 'target=' + describeEl(target));
    try {
      await performType(value, target);
    } catch (e) {
      err('type error', e && e.message);
      return { ok: false, destination: null, units: null, matched: false };
    }
    const matched = await waitForResultMatch(value, RESULT_WAIT_MS);
    let dest = null, units = null;
    if (matched) {
      await sleep(POST_MATCH_SETTLE_MS);
      dest = readDestination();
      units = readUnits();
    } else {
      log('no-match for', value, 'current=', readToteResult());
    }
    return { ok: true, destination: dest, units, matched };
  }

  function applyPersistedUI() {
    const stored = loadStorage();
    const wrap = document.getElementById('lta-panel');
    if (!wrap) return;
    if (stored.panelPos && Number.isFinite(stored.panelPos.left) && Number.isFinite(stored.panelPos.top)) {
      wrap.style.left = stored.panelPos.left + 'px';
      wrap.style.top  = stored.panelPos.top  + 'px';
      wrap.style.right = 'auto';
      state.ui.pos = stored.panelPos;
    }
    if (stored.panelMin) {
      state.ui.minimized = true;
      wrap.classList.add('min');
    }
    if (typeof stored.batchInput === 'string') {
      const ta = wrap.querySelector('#lta-batch-input');
      if (ta) ta.value = stored.batchInput;
    }
    if (Array.isArray(stored.batchResults)) {
      state.batch.results = stored.batchResults;
    }
  }

  function renderProgress() {
    const b = state.batch;
    const fill = document.querySelector('#lta-progress-fill');
    const text = document.querySelector('#lta-progress-text');
    const stats = document.querySelector('#lta-progress-stats');
    const total = b.total || (b.results.length || 0);
    const done = b.running ? b.processed : (b.results.length || b.processed);
    const pct = total > 0 ? Math.round((done / total) * 100) : 0;
    if (fill) fill.style.width = pct + '%';
    if (text) text.textContent = `${done}/${total}${b.running && b.current ? ' · ' + b.current : ''}`;
    if (stats) {
      if (b.startedAt && b.scanned > 0) {
        const elapsed = Date.now() - b.startedAt;
        const avg = elapsed / b.scanned;
        const remaining = Math.max(0, total - b.processed);
        const eta = remaining > 0 && b.running ? avg * remaining : 0;
        const parts = [
          `avg ${(avg / 1000).toFixed(2)}s/scan`,
          `scanned ${b.scanned}`,
        ];
        if (b.skipped) parts.push(`skipped ${b.skipped}`);
        parts.push(`elapsed ${fmtDuration(elapsed)}`);
        if (b.running) parts.push(`ETA ${fmtDuration(eta)}`);
        stats.textContent = parts.join(' · ');
      } else if (b.results.length) {
        stats.textContent = `${b.results.length} saved result${b.results.length === 1 ? '' : 's'}`;
      } else {
        stats.textContent = '—';
      }
    }
  }

  function renderResults() {
    const tbody = document.querySelector('#lta-results tbody');
    const empty = document.querySelector('#lta-results-empty');
    if (!tbody) return;
    const rows = state.batch.results;
    if (empty) empty.style.display = rows.length === 0 ? 'block' : 'none';
    tbody.innerHTML = rows.map((r, i) => {
      let cls = 'r-ok';
      if (r.status === 'type-fail' || r.status === 'no-match') cls = 'r-fail';
      else if (r.status === 'no-dest') cls = 'r-warn';
      else if (r.status === 'skipped') cls = 'r-skip';
      return `<tr class="${cls}">
        <td class="idx">${i + 1}</td>
        <td class="tote">${escapeHtml(r.tote)}</td>
        <td class="dest">${escapeHtml(r.destination ?? '')}</td>
        <td class="units">${escapeHtml(r.units ?? '')}</td>
        <td class="status">${escapeHtml(r.status)}</td>
      </tr>`;
    }).join('');
  }

  function flashButton(sel, text) {
    const btn = document.querySelector(sel);
    if (!btn) return;
    const prev = btn.textContent;
    btn.textContent = text;
    setTimeout(() => { btn.textContent = prev; }, 1200);
  }

  // ---------- Batch ----------
  function parseBatchInput(text) {
    return text
      .split(/[\r\n,;\s]+/)
      .map(s => s.trim())
      .filter(Boolean);
  }

  function upsertResult(entry) {
    const i = state.batch.results.findIndex(r => r.tote === entry.tote);
    if (i >= 0) state.batch.results[i] = entry;
    else state.batch.results.push(entry);
  }

  function persistResults() {
    saveStorage({ batchResults: state.batch.results });
  }

  async function onBatchRun() {
    if (state.batch.running) return;
    const ta = document.querySelector('#lta-batch-input');
    const totes = parseBatchInput(ta?.value || '');
    if (!totes.length) { warn('No tote IDs in input.'); return; }

    const doneOk = new Set(state.batch.results.filter(r => r.status === 'ok').map(r => r.tote));

    state.batch.running = true;
    state.batch.abort = false;
    state.batch.total = totes.length;
    state.batch.processed = 0;
    state.batch.scanned = 0;
    state.batch.skipped = 0;
    state.batch.current = null;
    state.batch.startedAt = Date.now();
    log(`Batch started, totes=${totes.length}, already-done=${doneOk.size}`);
    setControlsRunning(true);
    renderProgress();

    for (let i = 0; i < totes.length; i++) {
      if (state.batch.abort) { log('Batch aborted at index ' + i); break; }
      const tote = totes[i];
      state.batch.processed = i;
      state.batch.current = tote;
      renderProgress();

      if (doneOk.has(tote)) {
        state.batch.skipped++;
        state.batch.processed = i + 1;
        renderProgress();
        continue;
      }

      let result;
      try {
        result = await doType(tote, `batch[${i + 1}/${totes.length}]`);
      } catch (e) {
        err('batch item error', e && e.message);
        result = { ok: false, destination: null, matched: false };
      }

      const status = !result.ok ? 'type-fail'
        : !result.matched ? 'no-match'
        : (result.destination == null ? 'no-dest' : 'ok');

      upsertResult({
        tote,
        destination: result.destination,
        units: result.units,
        ts: nowStr(),
        status,
      });
      state.batch.scanned++;
      state.batch.processed = i + 1;
      renderResults();
      renderProgress();
      persistResults();

      if (i < totes.length - 1) await sleep(BETWEEN_SCANS_MS);
    }

    state.batch.running = false;
    state.batch.current = null;
    setControlsRunning(false);
    renderProgress();
    log(`Batch finished, scanned=${state.batch.scanned}, skipped=${state.batch.skipped}, total-results=${state.batch.results.length}`);
  }

  function onBatchStop() {
    if (!state.batch.running) return;
    state.batch.abort = true;
    log('Batch stop requested');
  }

  async function copyText(text) {
    try {
      await navigator.clipboard.writeText(text);
      return true;
    } catch (_) {
      const ta = document.createElement('textarea');
      ta.value = text;
      ta.style.position = 'fixed'; ta.style.opacity = '0';
      document.body.appendChild(ta);
      ta.select();
      let ok = false;
      try { ok = document.execCommand('copy'); } catch (_) {}
      document.body.removeChild(ta);
      return ok;
    }
  }

  async function onBatchCopy() {
    const rows = state.batch.results;
    const header = 'tote\tdestination\tunits\tstatus\ttimestamp';
    const body = rows.map(r => `${r.tote}\t${r.destination ?? ''}\t${r.units ?? ''}\t${r.status}\t${r.ts}`).join('\n');
    const text = header + '\n' + body;
    const ok = await copyText(text);
    flashButton('#lta-batch-copy', ok ? 'Copied!' : 'Copy failed');
  }

  function onBatchClear() {
    if (state.batch.running) return;
    state.batch.results = [];
    state.batch.total = 0;
    state.batch.processed = 0;
    state.batch.scanned = 0;
    state.batch.skipped = 0;
    state.batch.startedAt = null;
    persistResults();
    renderResults();
    renderProgress();
  }

  async function onCopyDebug() {
    const text = buildDebugDump();
    const ok = await copyText(text);
    flashButton('#lta-copy-debug', ok ? 'Copied!' : 'Copy failed');
  }

  function buildDebugDump() {
    return [
      '=== List Tote Auto debug dump ===',
      'time: ' + new Date().toISOString(),
      'url: ' + location.href,
      'ua: ' + navigator.userAgent,
      'inputPresent: ' + !!findInputEl(),
      'resultPresent: ' + !!findResultEl(),
      'destPresent: ' + !!findDestEl(),
      'unitsPresent: ' + !!findUnitsEl(),
      'currentTote: ' + readToteResult(),
      'currentDestination: ' + readDestination(),
      'currentUnits: ' + readUnits(),
      '',
      '--- Batch state ---',
      JSON.stringify({
        running: state.batch.running,
        total: state.batch.total,
        processed: state.batch.processed,
        scanned: state.batch.scanned,
        skipped: state.batch.skipped,
        results: state.batch.results.length,
      }),
      '',
      '--- Batch results (' + state.batch.results.length + ') ---',
      ...state.batch.results.map((r, i) => `${i + 1}\t${r.ts}\t${r.status}\t${r.tote}\t${r.destination ?? ''}\t${r.units ?? ''}`),
      '',
      '--- Debug log (' + state.debug.length + ') ---',
      ...state.debug.map(d => `${d.ts}  [${d.level}]  ${d.msg}`),
    ].join('\n');
  }

  // ---------- Boot ----------
  function boot() {
    buildPanel();
    applyPersistedUI();
    renderResults();
    renderProgress();
    log('booted on', location.href, 'results-loaded=' + state.batch.results.length);
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', boot);
  } else {
    boot();
  }
})();
