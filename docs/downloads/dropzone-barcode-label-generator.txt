  function rainbowBorderColor(index, total) {
    if (total <= 1) return rainbowHueToHex(RAINBOW_HUE_START);
    const t = index / (total - 1);
    const hue = RAINBOW_HUE_START + t * (RAINBOW_HUE_END - RAINBOW_HUE_START);
    return rainbowHueToHex(hue);
  }

  function rainbowHueToHex(hue) {
    if (!rainbowCtx) {
      rainbowCtx = document.createElement('canvas').getContext('2d');
    }
    rainbowCtx.fillStyle = `hsl(${hue}, 92%, 48%)`;
    const computed = rainbowCtx.fillStyle;
    if (typeof computed === 'string') {
      if (computed.startsWith('#')) {
        return computed.length >= 7 ? computed.slice(0, 7) : computed;
      }
      const m = computed.match(/rgba?\(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)/);
      if (m) {
        const hex = (n) => Number(m[n]).toString(16).padStart(2, '0');
        return `#${hex(1)}${hex(2)}${hex(3)}`;
      }
    }
    return DEFAULTS.borderColor;
  }

  function findDuplicateData(entries) {
    const seen = new Map();
    const dups = [];
    entries.forEach(({ line, entry }) => {
      const key = entry.data.toLowerCase();
      if (!seen.has(key)) {
        seen.set(key, [line]);
      } else {
        seen.get(key).push(line);
      }
    });
    seen.forEach((lines, key) => {
      if (lines.length > 1) dups.push({ data: entries.find((e) => e.entry.data.toLowerCase() === key).entry.data, count: lines.length, lines });
    });
    return dups;
  }

  function dedupeEntries(entries) {
    const seen = new Set();
    return entries.filter(({ entry }) => {
      const key = entry.data.toLowerCase();
      if (seen.has(key)) return false;
      seen.add(key);
      return true;
    });
  }

    };
  }

  function layoutBarcodeAndText(innerX, innerY, innerW, innerH, bcCanvas, textContentH, hasSubtextGap) {
    const padX = 10;
    const minGap = 8;
    const gapCount = hasSubtextGap ? 4 : 3;
    const scaleByW = (innerW - padX * 2) / bcCanvas.width;
    let gap = minGap;
    let bcScale = scaleByW;
    let bcH = bcCanvas.height * bcScale;
    let remaining = innerH - bcH - textContentH;

    if (remaining >= minGap * gapCount) {
      gap = remaining / gapCount;
    } else {
      bcScale = Math.min(scaleByW, (innerH - minGap * gapCount - textContentH) / bcCanvas.height);
      bcH = bcCanvas.height * bcScale;
      remaining = innerH - bcH - textContentH;
      gap = Math.max(minGap, remaining / gapCount);
    }

    const bcW = bcCanvas.width * bcScale;
    return {
      bcX: innerX + (innerW - bcW) / 2,
      bcY: innerY + gap,
      bcW,
      bcH,
      textTop: innerY + gap + bcH + gap,
      gap,
    };
  }

  function normalizeHexColor(raw, fallback) {
    if (!raw) return fallback;
    let v = String(raw).trim().toLowerCase();
    if (/^#[0-9a-f]{8}$/.test(v)) v = v.slice(0, 7);
    if (/^#[0-9a-f]{3}$/.test(v)) {
      v = `#${v[1]}${v[1]}${v[2]}${v[2]}${v[3]}${v[3]}`;
    }
    return /^#[0-9a-f]{6}$/.test(v) ? v : fallback;
  }

  function measureFittedFontSize(ctx, text, maxSize, minSize, maxWidth, weight, family) {
    if (!text) return maxSize;
    let size = maxSize;
    while (size > minSize) {
      ctx.font = `${weight} ${size}px ${family}`;
      if (ctx.measureText(text).width <= maxWidth) return size;
      size -= 1;
    }
    ctx.font = `${weight} ${minSize}px ${family}`;
    return minSize;
  }

  function drawDropzoneLabel(ctx, w, h, bcCanvas, entry, settings) {
    const bw = settings.borderWidth;
    const borderColor = settings.borderColor;
    const bgColor = settings.bgColor;
    const textColor = settings.textColor;
    const subtextColor = settings.subtextColor;
    const innerX = bw;
    const innerY = bw;
    const innerW = w - bw * 2;
    const innerH = h - bw * 2;

    ctx.fillStyle = bgColor;
    ctx.fillRect(0, 0, w, h);
    if (bw > 0) {
      ctx.fillStyle = borderColor;
      ctx.fillRect(0, 0, w, h);
      ctx.fillStyle = bgColor;
      ctx.fillRect(innerX, innerY, innerW, innerH);
    }

    const titleSize = settings.titleSize || DEFAULTS.titleSize;
    const idSize = settings.idSize || DEFAULTS.idSize;
    const subSize = settings.subtextSize || DEFAULTS.subtextSize;
    const lineGap = entry.title ? 8 : 0;
    const hasSub = settings.subtextEnabled && settings.subText;
    const minGap = 8;
    const maxTextW = innerW - 20;
    const idText = entry.subtitle || entry.data;
    const fittedTitleSize = entry.title
      ? measureFittedFontSize(ctx, entry.title, titleSize, 8, maxTextW, '600', 'system-ui, sans-serif')
      : 0;
    const fittedIdSize = measureFittedFontSize(ctx, idText, idSize, 8, maxTextW, '700', 'system-ui, sans-serif');
    let textContentH = entry.title ? fittedTitleSize + lineGap + fittedIdSize : fittedIdSize;
    if (hasSub) textContentH += minGap + subSize;

    const layout = layoutBarcodeAndText(innerX, innerY, innerW, innerH, bcCanvas, textContentH, hasSub);
    ctx.drawImage(bcCanvas, layout.bcX, layout.bcY, layout.bcW, layout.bcH);

    const cx = innerX + innerW / 2;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'top';
    let y = layout.textTop;

    if (entry.title) {
      ctx.fillStyle = textColor;
      ctx.font = `600 ${fittedTitleSize}px system-ui, sans-serif`;
      ctx.fillText(entry.title, cx, y);
      y += fittedTitleSize + lineGap;
    }
    ctx.fillStyle = textColor;
    ctx.font = `700 ${fittedIdSize}px system-ui, sans-serif`;
    ctx.fillText(idText, cx, y);
    y += fittedIdSize;

    if (hasSub) {
      y += layout.gap;
      ctx.fillStyle = subtextColor;
      ctx.font = `500 ${subSize}px system-ui, sans-serif`;
      ctx.fillText(settings.subText, cx, y);
    }
  }

  function drawTagLabel(ctx, w, h, bcCanvas, entry, settings) {
    const bw = settings.borderWidth;
    const borderColor = settings.borderColor;
    const bgColor = settings.bgColor;
    const textColor = settings.textColor;
    const subtextColor = settings.subtextColor;
    const innerX = bw;
    const innerY = bw;
    const innerW = w - bw * 2;
    const innerH = h - bw * 2;

    ctx.fillStyle = bgColor;
    ctx.fillRect(0, 0, w, h);
    if (bw > 0) {
      ctx.fillStyle = borderColor;
      ctx.fillRect(0, 0, w, h);
      ctx.fillStyle = bgColor;
      ctx.fillRect(innerX, innerY, innerW, innerH);
    }

    const mainSize = settings.idSize || DEFAULTS.idSize;
    const subSize = settings.subtextSize || DEFAULTS.subtextSize;
    const hasSub = settings.subtextEnabled && settings.subText;
    const minGap = 8;
    const mainText = entry.title || entry.subtitle || entry.data;
    const maxTextW = innerW - 20;
    const fittedMainSize = measureFittedFontSize(ctx, mainText, mainSize, 8, maxTextW, '700', 'Georgia, "Times New Roman", serif');
    let textContentH = fittedMainSize;
    if (hasSub) textContentH += minGap + subSize;

    const layout = layoutBarcodeAndText(innerX, innerY, innerW, innerH, bcCanvas, textContentH, hasSub);
    ctx.drawImage(bcCanvas, layout.bcX, layout.bcY, layout.bcW, layout.bcH);

    const cx = innerX + innerW / 2;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'top';
    ctx.fillStyle = textColor;
    ctx.font = `700 ${fittedMainSize}px Georgia, "Times New Roman", serif`;
    ctx.fillText(mainText, cx, layout.textTop);

    if (hasSub) {
      ctx.fillStyle = subtextColor;
      ctx.font = `500 ${subSize}px Georgia, "Times New Roman", serif`;
      ctx.fillText(settings.subText, cx, layout.textTop + fittedMainSize + layout.gap);
    }

    return { bcX: layout.bcX, bcY: layout.bcY, bcW: layout.bcW, bcH: layout.bcH, labelText: mainText, fontSize: fittedMainSize };
  }

  function composeLabel(entry, settings) {
    const w = settings.labelWidth;
    const h = settings.labelHeight;
    const out = document.createElement('canvas');
    out.width = w;
    out.height = h;
    const ctx = out.getContext('2d');

    const bcCanvas = renderBarcodeCanvas(entry.data, settings);
    const layout = settings.layout;
    let layoutMeta = {};

    if (layout === 'dropzone') {
      drawDropzoneLabel(ctx, w, h, bcCanvas, entry, settings);
    } else {
      layoutMeta = drawTagLabel(ctx, w, h, bcCanvas, entry, settings) || {};
    }

    return {
      canvas: out,
      meta: {
        labelWidth: w,
        labelHeight: h,
        symbology: settings.symbology,
        barcodeNativeWidth: bcCanvas.width,
        barcodeNativeHeight: bcCanvas.height,
        layout,
        ...layoutMeta,
      },
    };
  }

  function maxScaleForGrid(cols, rows, labelW, labelH, pageW, pageH) {
    let lo = 0.05;
    let hi = 1;
    let best = null;
    for (let i = 0; i < 40; i += 1) {
      const scale = (lo + hi) / 2;
      const w = labelW * scale;
      const h = labelH * scale;
      const gapX = (pageW - cols * w) / (cols + 1);
      const gapY = (pageH - rows * h) / (rows + 1);
      if (gapX >= PAGE_MIN_GAP && gapY >= PAGE_MIN_GAP) {
        best = { cols, rows, gapX, gapY, scale, labelW: w, labelH: h };
        lo = scale;
      } else {
        hi = scale;
      }
    }
    return best;
  }

  function pickPageGrid(perPage, labelW, labelH, pageW, pageH) {
    let best = null;
    for (let cols = 1; cols <= perPage; cols += 1) {
      const rows = Math.ceil(perPage / cols);
      const candidate = maxScaleForGrid(cols, rows, labelW, labelH, pageW, pageH);
      if (!candidate) continue;
      const balance = Math.min(candidate.gapX, candidate.gapY) / Math.max(candidate.gapX, candidate.gapY);
      const score = candidate.scale * 1000 + balance;
      if (!best || score > best.score) {
        best = { ...candidate, score };
      }
    }
    if (best) return best;
    const cols = 1;
    const rows = perPage;
    const w = labelW * 0.1;
    const h = labelH * 0.1;
    return {
      cols,
      rows,
      gapX: Math.max(PAGE_MIN_GAP, (pageW - cols * w) / (cols + 1)),
      gapY: Math.max(PAGE_MIN_GAP, (pageH - rows * h) / (rows + 1)),
      scale: 0.1,
      labelW: w,
      labelH: h,
    };
  }

  function labelPositionOnPage(index, count, grid, pageW, pageH) {
    const { cols, gapX, gapY, labelW, labelH } = grid;
    const row = Math.floor(index / cols);
    const col = index % cols;
    const usedRows = Math.ceil(count / cols);
    const itemsInRow = row === usedRows - 1 ? count - (usedRows - 1) * cols : cols;

    if (itemsInRow < cols) {
      const rowW = itemsInRow * labelW + (itemsInRow + 1) * gapX;
      const rowOffX = (pageW - rowW) / 2;
      const colInRow = index - (usedRows - 1) * cols;
      return {
        x: rowOffX + gapX + colInRow * (labelW + gapX),
        y: gapY + row * (labelH + gapY),
      };
    }

    return {
      x: gapX + col * (labelW + gapX),
      y: gapY + row * (labelH + gapY),
    };
  }

  function loadImage(dataUrl) {
    return new Promise((resolve, reject) => {
      const img = new Image();
      img.onload = () => resolve(img);
      img.onerror = () => reject(new Error('Failed to load label image'));
      img.src = dataUrl;
    });
  }

  function fillPageCopies(images, perPage) {
    if (!images.length || images.length >= perPage) return images;
    const filled = [];
    while (filled.length < perPage) {
      filled.push(images[filled.length % images.length]);
    }
    return filled;
  }

  async function buildPagesFromResults(results) {
    if (!results.length) return { pages: [], grid: null, pageDim: null };
    const pageSettings = getPageSettings();
    const pageDim = getPageDimensions(pageSettings.pageSize);
    const nativeW = results[0].meta.labelWidth;
    const nativeH = results[0].meta.labelHeight;
    const grid = pickPageGrid(pageSettings.labelsPerPage, nativeW, nativeH, pageDim.w, pageDim.h);
    const images = fillPageCopies(
      await Promise.all(results.map((r) => loadImage(r.dataUrl))),
      pageSettings.labelsPerPage,
    );
    const pages = [];

    for (let i = 0; i < images.length; i += pageSettings.labelsPerPage) {
      const chunk = images.slice(i, i + pageSettings.labelsPerPage);
      const canvas = document.createElement('canvas');
      canvas.width = pageDim.w;
      canvas.height = pageDim.h;
      const ctx = canvas.getContext('2d');
      ctx.fillStyle = '#ffffff';
      ctx.fillRect(0, 0, pageDim.w, pageDim.h);

      chunk.forEach((img, idx) => {
        const { x, y } = labelPositionOnPage(idx, chunk.length, grid, pageDim.w, pageDim.h);
        ctx.drawImage(img, x, y, grid.labelW, grid.labelH);
      });

      pages.push({
        name: `page_${pages.length + 1}.png`,
        dataUrl: canvas.toDataURL('image/png'),
        pageSize: pageSettings.pageSize,
      });
    }

    return { pages, grid, pageDim, pageSettings };
  }

  function updatePageLayoutHint(grid, pageCount) {
    const el = $('#dz-page-layout-hint');
    if (!el) return;
    if (!grid || !pageCount) {
      el.textContent = 'Fewer labels than the page count are repeated to fill the sheet. Labels are placed with equal gaps to each other and the page edge.';
      return;
    }
    const scaleNote = `labels at ${Math.round(grid.scale * 100)}%`;
    const gapNote = Math.abs(grid.gapX - grid.gapY) < 2
      ? `${Math.round(grid.gapX)}px gaps`
      : `${Math.round(grid.gapX)}×${Math.round(grid.gapY)}px gaps`;
    el.textContent = `${grid.cols} × ${grid.rows} grid, ${gapNote}, ${scaleNote} → ${pageCount} page(s)`;
  }

  function bindLabelColorPicker(el, index) {
    if (!el) return;
    const apply = () => { applyLabelBorderColor(index, el.value); };
    el.addEventListener('input', apply);
    el.addEventListener('change', apply);
  }

  function syncLabelColorPickers(index, borderColor) {
    root.querySelectorAll(`[data-label-color-idx="${index}"]`).forEach((el) => {
      if (el.value !== borderColor) el.value = borderColor;
    });
  }

  function applyLabelBorderColor(index, rawColor) {
    const result = lastResults[index];
    if (!result?.labelSettings) return;

    const borderColor = normalizeHexColor(rawColor, result.borderColor || DEFAULTS.borderColor);
    if (result.borderColor === borderColor) return;

    result.borderColor = borderColor;
    labelBorderColorOverrides[result.entry.data] = borderColor;
    const settings = { ...result.labelSettings, borderColor };
    result.labelSettings = settings;

    const { canvas, meta } = composeLabel(result.entry, settings);
    result.dataUrl = canvas.toDataURL('image/png');
    result.meta = { ...meta, borderColor };

    syncLabelColorPickers(index, borderColor);
    const img = $(`#dz-label-card-${index} img`);
    if (img) img.src = result.dataUrl;

    clearTimeout(pageRebuildTimer);
    pageRebuildTimer = setTimeout(() => {
      rebuildPages().then(() => scheduleSave());
    }, 120);
  }

  async function rebuildPages() {
    if (!lastResults.length) {
      lastPages = [];
      lastPageGrid = null;
      updatePageLayoutHint(null);
      updateOutputButtons();
      if (previewMode === 'pages') renderPreviewPages();
      return;
    }
    try {
      const { pages, grid } = await buildPagesFromResults(lastResults);
      lastPages = pages;
      lastPageGrid = grid;
      updatePageLayoutHint(grid, pages.length);
      updateOutputButtons();
      if (previewMode === 'pages') renderPreviewPages();
    } catch (err) {
      $('#dz-bg-status').textContent = `Page layout failed: ${err?.message || String(err)}`;
    }
  }

  function runGenerate(entries) {
    const settings = getSettings();
    $('#dz-bg-preview').innerHTML = '';
    lastResults = [];
    lastPages = [];
    lastGenerateErrors = [];

    let errors = 0;
    const errorList = [];
    entries.forEach(({ line, entry }, labelIndex) => {
      const labelSettings = settingsForLabel(settings, labelIndex, entries.length);
      if (labelBorderColorOverrides[entry.data]) {
        labelSettings.borderColor = labelBorderColorOverrides[entry.data];
      }

      try {
        const { canvas, meta } = composeLabel(entry, labelSettings);
        const dataUrl = canvas.toDataURL('image/png');
        const safeName = entry.data.replace(/[^a-zA-Z0-9._-]+/g, '_').slice(0, 80);
        lastResults.push({
          name: `${safeName || 'label'}_${labelIndex + 1}.png`,
          dataUrl,
          entry,
          borderColor: labelSettings.borderColor,
          labelSettings: { ...labelSettings },
          meta: { ...meta, borderColor: labelSettings.borderColor },
        });
      } catch (err) {
        errors += 1;
        const errMsg = String(err);
        errorList.push({ line, data: entry.data, error: errMsg });
        lastGenerateErrors.push({ data: entry.data, error: errMsg });
      }
    });

    const finishGenerate = async () => {
      await rebuildPages();
      const perPage = getPageSettings().labelsPerPage;
      const labelsOnPages = lastResults.length && lastResults.length < perPage
        ? perPage
        : lastResults.length;
      lastDebug = {
        scriptVersion: '1.6.1',
        timestamp: new Date().toISOString(),
        pageUrl: location.href,
        userAgent: navigator.userAgent,
        settings: { ...settings, ...getPageSettings() },
        pageLayout: lastPageGrid,
        pages: lastPages.length,
        labelsOnPages,
        inputLines: entries.map((e) => e.line),
        results: lastResults.map((r) => ({
          data: r.entry.data,
          title: r.entry.title,
          subtitle: r.entry.subtitle,
          subText: settings.subText,
          fileName: r.name,
          meta: r.meta,
        })),
        errors: errorList,
        ...getColorDebugInfo(),
      };
      setPreviewMode(previewMode);
      $('#dz-bg-status').textContent = errors
        ? `Generated ${lastResults.length} unique label(s), ${errors} failed. ${labelsOnPages} on ${lastPages.length} page(s).`
        : `Generated ${lastResults.length} unique label(s), ${labelsOnPages} on ${lastPages.length} page(s).`;
      saveState();
    };

    finishGenerate();
  }

  function generate() {
    flushFormInputs();
    const lines = $('#dz-data').value
      .split(/\r?\n/)
      .map((l) => l.trim())
      .filter(Boolean);

    if (!lines.length) {
      $('#dz-bg-status').textContent = 'Enter at least one barcode value.';
      return;
    }

    const entries = lines
      .map((line) => ({ line, entry: parseLine(line) }))
      .filter(({ entry }) => entry.data);

    const dups = findDuplicateData(entries);
    if (dups.length) {
      showDuplicateDialog(dups, entries, (finalEntries) => {
        if (finalEntries) runGenerate(finalEntries);
      });
      return;
    }

    runGenerate(entries);
  }

  function downloadFolderName() {
    const d = new Date();
    const pad = (n) => String(n).padStart(2, '0');
    return `barcode-labels-${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}_${pad(d.getHours())}${pad(d.getMinutes())}${pad(d.getSeconds())}`;
  }

  const CRC32_TABLE = (() => {
    const table = new Uint32Array(256);
    for (let i = 0; i < 256; i += 1) {
      let c = i;
      for (let j = 0; j < 8; j += 1) c = (c & 1) ? (0xedb88320 ^ (c >>> 1)) : (c >>> 1);
      table[i] = c >>> 0;
    }
    return table;
  })();

  function crc32(bytes) {
    let crc = 0xffffffff;
    for (let i = 0; i < bytes.length; i += 1) {
      crc = CRC32_TABLE[(crc ^ bytes[i]) & 0xff] ^ (crc >>> 8);
    }
    return (crc ^ 0xffffffff) >>> 0;
  }

  function dataUrlToBytes(dataUrl) {
    const base64 = dataUrl.split(',')[1];
    const binary = atob(base64);
    const bytes = new Uint8Array(binary.length);
    for (let i = 0; i < binary.length; i += 1) bytes[i] = binary.charCodeAt(i);
    return bytes;
  }

  /** Fast ZIP (store only — PNGs are already compressed). */
  function buildStoreZip(fileEntries) {
    const encoder = new TextEncoder();
    const localParts = [];
    const central = [];
    let offset = 0;

    fileEntries.forEach(({ path, bytes }) => {
      const name = encoder.encode(path.replace(/\\/g, '/'));
      const checksum = crc32(bytes);
      const size = bytes.length;

      const local = new Uint8Array(30 + name.length);
      const lv = new DataView(local.buffer, local.byteOffset, local.byteLength);
      lv.setUint32(0, 0x04034b50, true);
      lv.setUint16(4, 20, true);
      lv.setUint16(6, 0, true);
      lv.setUint16(8, 0, true);
      lv.setUint32(14, checksum, true);
      lv.setUint32(18, size, true);
      lv.setUint32(22, size, true);
      lv.setUint16(26, name.length, true);
      lv.setUint16(28, 0, true);
      local.set(name, 30);

      const centralRec = new Uint8Array(46 + name.length);
      const cv = new DataView(centralRec.buffer, centralRec.byteOffset, centralRec.byteLength);
      cv.setUint32(0, 0x02014b50, true);
      cv.setUint16(4, 20, true);
      cv.setUint16(6, 20, true);
      cv.setUint16(8, 0, true);
      cv.setUint16(10, 0, true);
      cv.setUint32(16, checksum, true);
      cv.setUint32(20, size, true);
      cv.setUint32(24, size, true);
      cv.setUint16(28, name.length, true);
      cv.setUint16(30, 0, true);
      cv.setUint16(32, 0, true);
      cv.setUint16(34, 0, true);
      cv.setUint16(36, 0, true);
      cv.setUint32(38, 0, true);
      cv.setUint32(42, offset, true);
      centralRec.set(name, 46);

      localParts.push(local, bytes);
      central.push(centralRec);
      offset += local.length + bytes.length;
    });

    const centralSize = central.reduce((sum, rec) => sum + rec.length, 0);
    const end = new Uint8Array(22);
    const ev = new DataView(end.buffer, end.byteOffset, end.byteLength);
    ev.setUint32(0, 0x06054b50, true);
    ev.setUint16(4, 0, true);
    ev.setUint16(6, 0, true);
    ev.setUint16(8, fileEntries.length, true);
    ev.setUint16(10, fileEntries.length, true);
    ev.setUint32(12, centralSize, true);
    ev.setUint32(16, offset, true);
    ev.setUint16(20, 0, true);

    const localSize = localParts.reduce((sum, part) => sum + part.length, 0);
    const total = localSize + centralSize + end.length;
    const out = new Uint8Array(total);
    let pos = 0;
    localParts.forEach((part) => {
      out.set(part, pos);
      pos += part.length;
    });
    central.forEach((rec) => {
      out.set(rec, pos);
      pos += rec.length;
    });
    out.set(end, pos);
    return new Blob([out], { type: 'application/zip' });
  }

  function triggerBlobDownload(blob, filename) {
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = filename;
    a.style.display = 'none';
    document.body.appendChild(a);
    a.click();
    a.remove();
    setTimeout(() => URL.revokeObjectURL(url), 5000);
  }

  function triggerDataUrlDownload(dataUrl, filename) {
    const a = document.createElement('a');
    a.href = dataUrl;
    a.download = filename;
    a.style.display = 'none';
    document.body.appendChild(a);
    a.click();
    a.remove();
  }

  async function downloadAll() {
    if (!lastResults.length) return;

    const status = $('#dz-bg-status');
    const dlBtn = $('#dz-bg-btn-download');

    if (lastResults.length === 1) {
      const { name, dataUrl } = lastResults[0];
      triggerDataUrlDownload(dataUrl, name);
      status.textContent = `Downloaded ${name}`;
      return;
    }

    const folderName = downloadFolderName();
    const zipName = `${folderName}.zip`;
    const total = lastResults.length;

    dlBtn.disabled = true;
    status.textContent = `Preparing ${total} labels…`;

    try {
      await new Promise((resolve) => requestAnimationFrame(resolve));

      const files = [];
      for (let i = 0; i < lastResults.length; i += 1) {
        const { name, dataUrl } = lastResults[i];
        if (i % 4 === 0) {
          status.textContent = `Preparing ${i + 1}/${total}…`;
          await new Promise((resolve) => requestAnimationFrame(resolve));
        }
        files.push({
          path: `${folderName}/${name}`,
          bytes: dataUrlToBytes(dataUrl),
        });
      }

      status.textContent = `Building ${zipName}…`;
      await new Promise((resolve) => requestAnimationFrame(resolve));

      const blob = buildStoreZip(files);
      triggerBlobDownload(blob, zipName);
      status.textContent = `Downloaded ${zipName} (${total} PNGs in ${folderName}/).`;
    } catch (err) {
      status.textContent = `ZIP failed: ${err?.message || String(err)}`;
    } finally {
      dlBtn.disabled = lastResults.length === 0;
    }
  }

  async function downloadPages() {
    if (!lastPages.length) return;

    const status = $('#dz-bg-status');
    const dlBtn = $('#dz-bg-btn-download-pages');
    const folderName = downloadFolderName();
    const zipName = `${folderName}-pages.zip`;
    const total = lastPages.length;

    if (total === 1) {
      triggerDataUrlDownload(lastPages[0].dataUrl, lastPages[0].name);
      status.textContent = `Downloaded ${lastPages[0].name}`;
      return;
    }

    dlBtn.disabled = true;
    status.textContent = `Packing ${total} pages into ${zipName}…`;

    try {
      await new Promise((resolve) => requestAnimationFrame(resolve));
      const files = lastPages.map((page) => ({
        path: `${folderName}/${page.name}`,
        bytes: dataUrlToBytes(page.dataUrl),
      }));
      const blob = buildStoreZip(files);
      triggerBlobDownload(blob, zipName);
      status.textContent = `Downloaded ${zipName} (${total} pages).`;
    } catch (err) {
      status.textContent = `Page ZIP failed: ${err?.message || String(err)}`;
    } finally {
      dlBtn.disabled = lastPages.length === 0;
    }
  }

  }
